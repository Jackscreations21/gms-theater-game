
/* ============================================================================
   p4c — THE BUILD SYSTEM (spec docs/superpowers/specs/2026-08-07-build-
   system-design.md).  PR 1: the stock — wood, hardware and paint as bodies
   in the p4 detach registry, and the piece cap the order screen enforces.
   Later PRs add assemblies and nails, the saws, paint application, hinges,
   track, and the save system — all of it here, one part, one home.

   Wood is PARAMETRIC: one unit BoxGeometry serves every piece of every
   profile — a mesh is the shared box SCALED, and a cut never mints
   geometry, it re-scales one body and registers a second (the frame-rate
   invariant, spec §9).  Box geometry's six native material groups carry
   paint: mesh.material is an array of six entries from the shared color-
   keyed cache below (the LENSM lesson — never one material per piece).

   BUILD_KINDS is a var ON PURPOSE: p4's makeBodyMesh and canHang test it,
   and the Palace rig builds during p4's own parse-time run, before this
   part has executed — a hoisted var reads undefined there (falsy, correct:
   factory stock is all lanterns), where a const would throw (§5).
   ========================================================================== */
var BUILD_KINDS = {wood:1, hinge:1, track:1, carriage:1, paint:1, table:1};
const BUILD_CAP = 150;          // build pieces to a venue — enforced at the screen
const FT = 0.3048, INCH = 0.0254;
const WOOD_LEN = 8*FT;          // every stick and sheet is born 8ft long
const WOOD_PROF = {             // real sections, in metres
  sheet: {t:0.019, W:4*FT},     // 4x8 ply, 3/4" — dims {L, W}, t thick (local Z)
  s2x4:  {a:0.038, b:0.089},    // lumber: dims {L}, long axis local Y (§5 — own it)
  s4x4:  {a:0.089, b:0.089},
  s2x8:  {a:0.038, b:0.184}
};
const WOOD_COL = {ply:0xc8ab7a, pine:0xd8c193};
const WOODG = new T.BoxGeometry(1, 1, 1);
var WOODM = {};                 // color -> ONE MeshStandardMaterial, shared
function woodMat(color){
  return WOODM[color] || (WOODM[color] =
    new T.MeshStandardMaterial({color, roughness:.85}));
}
function makeWoodMesh(prof, dims){
  const P = WOOD_PROF[prof];
  const m = woodMat(prof === 'sheet' ? WOOD_COL.ply : WOOD_COL.pine);
  const mesh = new T.Mesh(WOODG, [m, m, m, m, m, m]);
  if(prof === 'sheet') mesh.scale.set(dims.L, dims.W, P.t);
  else mesh.scale.set(P.a, dims.L, P.b);
  mesh.castShadow = mesh.receiveShadow = true;
  mesh.userData.moves = true;
  return mesh;
}
function regWood(prof, dims){
  dims = dims || (prof === 'sheet' ? {L:WOOD_LEN, W:WOOD_PROF.sheet.W} : {L:WOOD_LEN});
  const b = regBody('wood', makeWoodMesh(prof, dims), null);
  b.prof = prof; b.dims = dims; b.state = 'loose';
  return b;
}

/* hardware and paint-can bodies — small cached geometries, shared materials */
const BUILDG = {};
function buildG(key, mk){ return BUILDG[key] || (BUILDG[key] = mk()); }
function makeBuildMesh(kind, color){
  const g = new T.Group();
  g.userData.moves = true;
  if(kind === 'hinge'){
    [-1, 1].forEach(s=>{
      const leaf = new T.Mesh(buildG('leaf', ()=>new T.BoxGeometry(0.1, 0.24, 0.012)), M.steel);
      leaf.position.x = s*0.06; g.add(leaf);
    });
    g.add(new T.Mesh(buildG('pin', ()=>new T.CylinderGeometry(0.014, 0.014, 0.28, 8)), M.steel));
  } else if(kind === 'track'){
    const rail = new T.Mesh(buildG('rail', ()=>new T.BoxGeometry(4*FT, 0.05, 0.12)), M.steel);
    rail.position.y = 0.025; g.add(rail);
  } else if(kind === 'carriage'){
    const car = new T.Mesh(buildG('car', ()=>new T.BoxGeometry(0.16, 0.08, 0.16)), M.steel);
    car.position.y = 0.04; g.add(car);
  } else if(kind === 'table'){
    /* the work table (build-usability §5, RULING K): orderable, carried by
       hand, stood anywhere — its top is a raised piece of deck */
    const top = new T.Mesh(buildG('tabletop', ()=>new T.BoxGeometry(1.6, 0.05, 0.8)), M.wood);
    top.position.y = 0.90; g.add(top);
    [-1, 1].forEach(sx=>[-1, 1].forEach(sz=>{
      const leg = new T.Mesh(buildG('tableleg', ()=>new T.BoxGeometry(0.06, 0.875, 0.06)), M.steel);
      leg.position.set(sx*0.72, 0.4375, sz*0.34); g.add(leg);
    }));
  } else {                                        // a paint can, banded its color
    const can = new T.Mesh(buildG('can', ()=>new T.CylinderGeometry(0.09, 0.09, 0.22, 12)), M.steel);
    can.position.y = 0.11; g.add(can);
    const band = new T.Mesh(buildG('band', ()=>new T.CylinderGeometry(0.092, 0.092, 0.1, 12)),
                            woodMat(color !== undefined ? color : 0x888888));
    band.position.y = 0.11; g.add(band);
  }
  g.traverse(o=>{ if(o.isMesh){ o.castShadow = o.receiveShadow = true; } });
  return g;
}

/* the piece cap's book: wood and hardware, wherever it stands (spec §3) */
function venueBuildCount(v){
  let n = 0;
  for(const b of BODIES) if(b.venue === v && BUILD_KINDS[b.kind]) n++;
  return n;
}

/* ============================================================================
   ASSEMBLIES & NAILS (PR 3, spec §5, RULING G).  An assembly is DATA: a
   root Group at the venue root, a list of member pieces, and a list of
   nails.  Per piece, the nail count decides the constraint — exactly one
   nail and the piece rides a PIVOT group at that nail (swing it by hand);
   two or more and it is a plain child of the root, rigid.  Nothing ever
   falls or detaches on its own: the hammer (removeNail) is the only way
   apart.  Anchors: a 'deck' nail pins the root where it stands; a 'pipe'
   nail hangs the root off a lineset's group so it FLIES — and tells p3's
   minTrimOf how deep it hangs, so nothing built goes through the deck.
   ========================================================================== */
var ASSEMBLIES = [];
var ASM_SEQ = 1;
const _av = new T.Vector3(), _av2 = new T.Vector3(), _av3 = new T.Vector3();
const _aq = new T.Quaternion(), _aq2 = new T.Quaternion();
const _ae = new T.Euler();
const _abox = new T.Box3();
const NAILG = new T.CylinderGeometry(0.016, 0.006, 0.05, 6);   // head down to point
function makeNailMesh(){
  const m = new T.Mesh(NAILG, M.steel);
  m.userData.nail = true;
  return m;
}
function newAsm(venue){
  const root = new T.Group();
  root.userData.moves = true;
  venueRoot(venue).add(root);
  const a = {id:ASM_SEQ++, venue, root, pieces:[], nails:[], anchor:null};
  ASSEMBLIES.push(a);
  return a;
}
function asmAdopt(a, b){
  if(b.state === 'slotted') unslotBody(b);
  delete b.frozen;   // nailed work is the assembly's story: hammering it
                     // apart later gives ordinary loose wood, not a float
  a.root.attach(b.mesh);
  b.state = 'fixed'; b.asm = a;
  if(a.pieces.indexOf(b) < 0) a.pieces.push(b);
}
function asmMerge(a, o){
  for(const p of o.pieces){ a.root.attach(p.mesh); p.asm = a; a.pieces.push(p); }
  for(const n of o.nails){ a.root.attach(n.mesh); n.asm = a; a.nails.push(n); }
  if(o.anchor && !a.anchor){
    a.anchor = o.anchor;
    if(o.anchor.type === 'pipe') o.anchor.ls.group.attach(a.root);
  }
  o.root.parent.remove(o.root);
  ASSEMBLIES.splice(ASSEMBLIES.indexOf(o), 1);
}
function nailCount(a, p){
  let n = 0;
  for(const x of a.nails) if(x.a === p || x.b === p) n++;
  return n;
}
/* target: {body} | {deck:true} | {ls}.  wp/wax in WORLD.  Returns the nail,
   or null where the shot is refused (deck AND pipe on one assembly, or a
   second pipe). */
function addNail(bA, target, wp, wax){
  let a = bA.asm || null;
  const tb = target.body || null;
  if(tb){
    const ta = tb.asm || null;
    if(!a && !ta){ a = newAsm(bA.venue); asmAdopt(a, bA); asmAdopt(a, tb); }
    else if(a && !ta) asmAdopt(a, tb);
    else if(!a && ta){ a = ta; asmAdopt(a, bA); }
    else if(a !== ta){
      if(a.anchor && ta.anchor) return null;   // two nailed-down things do not join
      if(ta.anchor) { const t = a; a = ta; asmMerge(a, t); }
      else asmMerge(a, ta);
    }
  } else {
    if(!a){ a = newAsm(bA.venue); asmAdopt(a, bA); }
    if(target.deck){
      if(a.anchor && a.anchor.type === 'pipe') return null;
      a.anchor = {type:'deck'};
    } else if(target.ls){
      if(a.anchor) return null;                // one anchor to an assembly
      a.anchor = {type:'pipe', ls:target.ls};
      target.ls.group.attach(a.root);
      asmHangDepth(target.ls);
    }
  }
  const nm = makeNailMesh();
  a.root.add(nm);
  nm.position.copy(a.root.worldToLocal(_av.copy(wp)));
  a.root.getWorldQuaternion(_aq).invert();
  const axL = _av2.copy(wax).applyQuaternion(_aq).normalize();
  nm.quaternion.setFromUnitVectors(_av3.set(0, 1, 0), axL);
  const n = {a:bA, b:(tb || (target.deck ? 'deck' : 'pipe')), asm:a,
             p:nm.position.clone(), axis:axL.clone(), mesh:nm};
  a.nails.push(n);
  asmJoints(a);
  if(typeof buildDirty === 'function') buildDirty();
  return n;
}
function detachPiece(a, p){
  asmUnpivot(a, p);
  venueRoot(a.venue).attach(p.mesh);
  p.state = 'loose'; p.asm = null; p.slider = null;
  if(typeof woodRestH === 'function') p.restH = woodRestH(p);
  a.pieces.splice(a.pieces.indexOf(p), 1);
}
function removeAsm(a){
  if(a.root.parent) a.root.parent.remove(a.root);
  ASSEMBLIES.splice(ASSEMBLIES.indexOf(a), 1);
}
function removeNail(n){
  const a = n.asm;
  a.nails.splice(a.nails.indexOf(n), 1);
  if(n.hinge){
    /* a pulled hinge is hardware again: the body respawns loose */
    venueRoot(a.venue).attach(n.mesh);
    const keep = BUILD_VENUE;
    BUILD_VENUE = a.venue;
    const hb = regBody('hinge', n.mesh, null);
    BUILD_VENUE = keep;
    hb.state = 'loose'; hb.restH = 0.14;
  }
  else if(n.mesh.parent) n.mesh.parent.remove(n.mesh);   // shared geometry, no dispose
  /* anchors live in their nails: pull the last one and the hold is gone */
  if(n.b === 'deck' && !a.nails.some(x=>x.b === 'deck')) a.anchor = null;
  if(n.b === 'pipe' && !a.nails.some(x=>x.b === 'pipe')){
    const ls = a.anchor && a.anchor.ls;
    a.anchor = null;
    venueRoot(a.venue).attach(a.root);
    if(ls) asmHangDepth(ls);
  }
  for(const p of a.pieces.slice()) if(nailCount(a, p) === 0) detachPiece(a, p);
  if(a.pieces.length === 0) removeAsm(a);
  else if(a.pieces.length === 1 && !a.anchor && a.nails.length === 0){
    detachPiece(a, a.pieces[0]);
    removeAsm(a);
  }
  else asmJoints(a);
  if(typeof buildDirty === 'function') buildDirty();
}
/* one nail = a pivot group AT the nail; the piece swings on it (p9) */
function asmPivot(a, p, n){
  const g = new T.Group();
  g.position.copy(n.p);
  g.userData.axis = n.axis.clone();
  a.root.add(g);
  g.updateMatrixWorld(true);
  g.attach(p.mesh);
  p.pivot = g; p.pivotN = n;
}
function asmUnpivot(a, p){
  if(!p.pivot) return;
  a.root.attach(p.mesh);
  if(p.pivot.parent) p.pivot.parent.remove(p.pivot);
  p.pivot = null; p.pivotN = null;
}
function asmJoints(a){
  for(const p of a.pieces){
    const k = nailCount(a, p);
    const n = k === 1 ? a.nails.find(x=>x.a === p || x.b === p) : null;
    /* one nail swings — UNLESS it is a world nail (deck/pipe pins where
       it stands) or a carriage nail (the slide is the carriage's dof) */
    if(n && (a.pieces.length > 1 || a.anchor) &&
       n.b !== 'deck' && n.b !== 'pipe' && !n.carriage){
      if(p.pivotN !== n){ asmUnpivot(a, p); asmPivot(a, p, n); }
    } else asmUnpivot(a, p);
  }
}
/* ============================================================================
   HINGES & TRACK (PR 6, spec §7).  A HINGE is a nail that swings: hold
   the hinge body at the seam between a LOOSE piece and its neighbour,
   fire the gun — the hinge body is consumed into a range-limited pivot
   (its own pin axis is the hinge axis: you place it the way it should
   swing), and the loose piece joins the neighbour's assembly on it.
   Nail the door shut later and it goes rigid; hammer that nail and it
   swings again; hammer the HINGE and the body respawns loose.

   TRACK: sections gun-nail to the deck and CHAIN end-to-end into one
   deck-anchored assembly (a.track.n sections along root-local X).  A
   carriage released onto a run RIDES it — a slider Group, the pivot
   pattern with translation for rotation.  Wood gun-nailed to a carriage
   attaches UNDER the slider, so pushing any of it slides the lot, hard
   stops at the run's ends (slideTo).
   ========================================================================== */
const TRACK_LEN = 4*FT;
function addHinge(hb, wp, wax){
  let sw = null, swd = 1e9, tg = null, tgd = 1e9;
  for(const b of BODIES){
    if(b.kind !== 'wood' || b.venue !== hb.venue) continue;
    b.mesh.getWorldPosition(_av2);
    const reach = Math.max(b.mesh.scale.x, b.mesh.scale.y, b.mesh.scale.z)/2 + 0.35;
    const d = _av2.distanceTo(wp);
    if(d > reach) continue;
    if(b.state === 'loose' && d < swd){ swd = d; sw = b; }
  }
  if(!sw) return null;
  for(const b of BODIES){
    if(b === sw || b.kind !== 'wood' || b.venue !== hb.venue) continue;
    if(b.state !== 'loose' && b.state !== 'fixed') continue;
    b.mesh.getWorldPosition(_av2);
    const reach = Math.max(b.mesh.scale.x, b.mesh.scale.y, b.mesh.scale.z)/2 + 0.35;
    const d = _av2.distanceTo(wp);
    if(d < reach && d < tgd){ tgd = d; tg = b; }
  }
  if(!tg) return null;
  let a = tg.asm || null;
  if(!a){ a = newAsm(tg.venue); asmAdopt(a, tg); }
  asmAdopt(a, sw);
  const nm = hb.mesh;                       // the hinge body IS the joint marker
  a.root.attach(nm);
  a.root.getWorldQuaternion(_aq).invert();
  const axL = _av3.copy(wax).applyQuaternion(_aq).normalize();
  const n = {a:sw, b:tg, asm:a, p:a.root.worldToLocal(_av.copy(wp)).clone(),
             axis:axL.clone(), mesh:nm, hinge:true, range:Math.PI/2};
  a.nails.push(n);
  BODIES.splice(BODIES.indexOf(hb), 1);     // consumed
  asmJoints(a);
  return n;
}
/* lay a POSED track section: into an existing run, or as a new one */
function layTrack(tb, asm){
  let a = asm || null;
  const keep = BUILD_VENUE;
  BUILD_VENUE = tb.venue;
  if(!a){
    a = newAsm(tb.venue);
    a.anchor = {type:'deck'};
    a.track = {n:0};
    tb.mesh.updateMatrixWorld(true);
    tb.mesh.getWorldPosition(_av);
    a.root.parent.worldToLocal(_av);
    a.root.position.copy(_av);
    _ae.setFromQuaternion(tb.mesh.getWorldQuaternion(_aq), 'YXZ');
    a.root.rotation.y = _ae.y;
    a.root.updateMatrixWorld(true);
  }
  asmAdopt(a, tb);
  tb.mesh.position.set(a.track.n * TRACK_LEN, 0.025, 0);
  tb.mesh.rotation.set(0, 0, 0);
  const nm = makeNailMesh();
  a.root.add(nm);
  nm.position.set(a.track.n * TRACK_LEN, 0.055, 0);
  a.nails.push({a:tb, b:'deck', asm:a, p:nm.position.clone(),
                axis:new T.Vector3(0, -1, 0), mesh:nm});
  a.track.n++;
  BUILD_VENUE = keep;
  asmJoints(a);
  return a;
}
/* the held-track offer: the end of an existing run first, then the deck */
function snapTrackHeld(b){
  b.mesh.updateMatrixWorld(true);
  b.mesh.getWorldPosition(_av);
  for(const a of ASSEMBLIES){
    if(!a.track || a.venue !== b.venue) continue;
    a.root.updateMatrixWorld(true);
    const endW = a.root.localToWorld(_av2.set(a.track.n * TRACK_LEN, 0.025, 0));
    if(endW.distanceTo(_av) < 0.45)
      return {lay:{asm:a}, pos:endW.clone(),
              quat:a.root.getWorldQuaternion(_aq).clone(), point:endW.clone()};
  }
  if(_av.y < 0.35){
    const relW = snapQuantize(b.mesh, _aq2.identity());
    const pos = _av.clone(); pos.y = 0.05;
    return {lay:{asm:null}, pos,
            quat:_aq2.identity().clone().multiply(relW).clone(), point:pos.clone()};
  }
  return null;
}
function rideTrack(cb, a){
  if(!a.track || cb.kind !== 'carriage') return false;
  const slider = new T.Group();
  a.root.add(slider);
  a.root.updateMatrixWorld(true);
  cb.mesh.getWorldPosition(_av);
  a.root.worldToLocal(_av);
  slider.position.set(clamp(_av.x, 0, (a.track.n - 1)*TRACK_LEN), 0.05, 0);
  slider.updateMatrixWorld(true);
  slider.add(cb.mesh);
  cb.mesh.position.set(0, 0.02, 0);
  cb.mesh.rotation.set(0, 0, 0);
  cb.state = 'riding'; cb.slider = slider; cb.tAsm = a;
  return true;
}
function slideTo(a, slider, x){
  slider.position.x = clamp(x, 0, (a.track.n - 1)*TRACK_LEN);
  return slider.position.x;
}
function nailToCarriage(wb, cb){
  const a = cb.tAsm;
  if(!a || wb.kind !== 'wood') return null;
  if(wb.state === 'slotted') unslotBody(wb);
  cb.slider.attach(wb.mesh);
  wb.state = 'fixed'; wb.asm = a; wb.slider = cb.slider;
  a.pieces.push(wb);
  const nm = makeNailMesh();
  cb.slider.add(nm);
  cb.mesh.getWorldPosition(_av);
  nm.position.copy(cb.slider.worldToLocal(_av));
  nm.position.y += 0.06;
  const n = {a:wb, b:cb, asm:a, p:nm.position.clone(),
             axis:new T.Vector3(0, 0, 1), mesh:nm, carriage:true};
  a.nails.push(n);
  asmJoints(a);
  return n;
}
/* pop a rider off its run — refuse while it carries work */
function unrideTrack(cb){
  if(!cb.slider) return false;
  const a = cb.tAsm;
  if(a.pieces.some(p=>p.slider === cb.slider)) return false;
  venueRoot(cb.venue).attach(cb.mesh);
  if(cb.slider.parent) cb.slider.parent.remove(cb.slider);
  cb.slider = null; cb.tAsm = null;
  cb.state = 'loose';
  return true;
}

/* how deep everything nailed to this pipe hangs below it — p3's minTrimOf
   reads ls.asmH so a flown flat stops at the deck, the #15 rule */
function asmHangDepth(ls){
  let h = 0;
  for(const a of ASSEMBLIES){
    if(!a.anchor || a.anchor.type !== 'pipe' || a.anchor.ls !== ls) continue;
    _abox.setFromObject(a.root);
    ls.group.getWorldPosition(_av);
    h = Math.max(h, _av.y - _abox.min.y);
  }
  ls.asmH = h;
}
/* the rest height of a loose piece: half its world box, so a dropped plank
   lies ON the floor instead of floating at a lantern's 0.25 */
function woodRestH(b){
  _abox.setFromObject(b.mesh);
  return Math.max(0.02, (_abox.max.y - _abox.min.y)/2);
}

/* ============================================================================
   THE GHOST SNAP (spec §5): a held piece of wood squares itself up to
   whatever it is offered against — yaw to the nearest 45°, pitch and roll
   to the nearest 90°, faces flush (OBB half-extents projected through the
   quantized relative rotation, |R|·h).  Returns where the piece WANTS to
   be plus the nail's point and axis; p9 applies the pose and the nail gun
   confirms it.  Targets: other wood, the deck (any y=0 floor), and the
   live stage's fly pipes.
   ========================================================================== */
const _sm = new T.Matrix4();
function snapQuantize(heldM, targetQ){
  heldM.getWorldQuaternion(_aq2);
  _aq.copy(targetQ).invert().multiply(_aq2);          // held relative to target
  _ae.setFromQuaternion(_aq, 'YXZ');
  _ae.x = Math.round(_ae.x/(Math.PI/2))*(Math.PI/2);
  _ae.z = Math.round(_ae.z/(Math.PI/2))*(Math.PI/2);
  _ae.y = Math.round(_ae.y/(Math.PI/4))*(Math.PI/4);
  return _aq.setFromEuler(_ae);                        // quantized RELATIVE quat
}
/* |R|·h — the held box's half extents seen along the target's axes */
function snapHalf(relQ, s, out){
  _sm.makeRotationFromQuaternion(relQ);
  const e = _sm.elements;
  out.set(
    Math.abs(e[0])*s.x/2 + Math.abs(e[4])*s.y/2 + Math.abs(e[8])*s.z/2,
    Math.abs(e[1])*s.x/2 + Math.abs(e[5])*s.y/2 + Math.abs(e[9])*s.z/2,
    Math.abs(e[2])*s.x/2 + Math.abs(e[6])*s.y/2 + Math.abs(e[10])*s.z/2);
  return out;
}
const _shalf = new T.Vector3(), _slp = new T.Vector3(), _sq = new T.Quaternion();
/* build-usability §3: candidates by SURFACE gap, the joint axis by real
   face overlap (daylight first), the winner by the smallest flush error */
const SNAP_SEEK = 0.35, SNAP_OFFER = 0.22, SNAP_SLACK = 0.08;
/* a REAL identity for snapQuantize's targetQ — passing _aq2 in aliased the
   temp snapQuantize writes first, so flat offers forced pieces upright */
const _IDQ = new T.Quaternion();
/* the work table's top surface (§5): legs 0.875, slab 0.875..0.925 */
const TABLE_TOP = 0.925;
const _tbV = new T.Vector3(), _tbV2 = new T.Vector3();
/* the highest tabletop under a world point, or null.  Own temps — this is
   called from inside snapWood and updateBodies, which own _av/_bodyV. */
function tableTopAt(wx, wy, wz, venue, self){
  let top = null;
  for(const t of BODIES){
    if(t.kind !== 'table' || t === self || t.venue !== venue) continue;
    if(t.state === 'held') continue;
    t.mesh.updateWorldMatrix(true, false);
    _tbV.set(wx, wy, wz);
    t.mesh.worldToLocal(_tbV);                    // a Group: local IS metric
    if(Math.abs(_tbV.x) > 0.8 || Math.abs(_tbV.z) > 0.4) continue;
    t.mesh.getWorldPosition(_tbV2);
    const ty = _tbV2.y + TABLE_TOP;
    if(wy > ty - 0.35 && (top === null || ty > top)) top = ty;
  }
  return top;
}
function snapWood(held){
  const hm = held.mesh;
  hm.updateMatrixWorld(true);
  hm.getWorldPosition(_av);
  /* 0 — a carriage riding a track: the piece mounts on it (PR 6) */
  for(const cb of BODIES){
    if(cb.kind !== 'carriage' || !cb.slider || cb.venue !== held.venue) continue;
    cb.mesh.getWorldPosition(_av2);
    if(_av2.distanceTo(_av) > 0.5) continue;
    const a = cb.tAsm;
    a.root.getWorldQuaternion(_sq);
    const rel = snapQuantize(hm, _sq);
    snapHalf(rel, hm.scale, _shalf);
    const pos = _av2.clone(); pos.y += 0.04 + _shalf.y;
    return {target:{carriage:cb}, pos, quat:_sq.clone().multiply(rel),
            point:_av2.clone(), axis:new T.Vector3(0, 0, 1)};
  }
  /* 1 — the best joint among the wood in reach.  Candidates are judged by
     the gap between SURFACES — centre distance is meaningless for 8ft
     stock, which is how end-to-end butts never offered — and a face is
     only a joint if the other two axes genuinely overlap (SNAP_SLACK).
     Faces with DAYLIGHT between them beat faces already overlapped (the
     daylight side is the side you are offering from); the smallest flush
     error wins.  All of it METRIC: the local frame is the UNIT box, so
     unit coordinates are converted through the target's scale — the old
     mix of the two could bury a snapped piece inside its target. */
  let best = null;
  for(const b of BODIES){
    if(b === held || b.kind !== 'wood' || b.venue !== held.venue) continue;
    if(b.state === 'held') continue;
    const tm = b.mesh;
    tm.updateWorldMatrix(true, false);
    tm.getWorldQuaternion(_sq);
    const rel = snapQuantize(hm, _sq).clone();
    snapHalf(rel, hm.scale, _shalf);                   // held halves on target axes
    _slp.copy(_av);
    tm.worldToLocal(_slp);                             // held centre, target UNIT local
    let seek = 0, dayAx = -1, dayG = 1e9, anyAx = -1, anyG = 1e9;
    for(let i = 0; i < 3; i++){
      const pm = Math.abs(_slp[_AXES[i]]) * tm.scale[_AXES[i]];
      const flush = tm.scale[_AXES[i]]/2 + _shalf.getComponent(i);
      const over = pm - flush;                         // daylight past flush
      if(over > 0) seek += over * over;
      let lap = true;
      for(let j = 0; j < 3; j++){
        if(j === i) continue;
        const pj = Math.abs(_slp[_AXES[j]]) * tm.scale[_AXES[j]];
        const fj = tm.scale[_AXES[j]]/2 + _shalf.getComponent(j);
        if(pj - fj > SNAP_SLACK){ lap = false; break; }
      }
      if(!lap) continue;
      const g = Math.abs(over);
      if(over > 0.001 && g < dayG){ dayG = g; dayAx = i; }
      if(g < anyG){ anyG = g; anyAx = i; }
    }
    if(Math.sqrt(seek) > SNAP_SEEK) continue;          // out of arm's offering
    const ax = dayAx >= 0 ? dayAx : anyAx, g = dayAx >= 0 ? dayG : anyG;
    if(ax < 0 || g > SNAP_OFFER) continue;             // nothing lines up
    if(!best || g < best.g) best = {tgt:b, ax, g, rel};
  }
  if(best){
    const tm = best.tgt.mesh;
    tm.getWorldQuaternion(_sq);
    snapHalf(best.rel, hm.scale, _shalf);
    _slp.copy(_av);
    tm.worldToLocal(_slp);
    const k = _AXES[best.ax];
    const sign = _slp[k] < 0 ? -1 : 1;
    /* flush is METRIC; the local frame is the unit box — divide back out */
    _slp[k] = sign * (tm.scale[k]/2 + _shalf.getComponent(best.ax)) / tm.scale[k];
    const pos = tm.localToWorld(_slp.clone());
    const quat = _sq.clone().multiply(best.rel);       // world = target · rel
    /* the nail: ON the touching face, under the held centre, clamped to it */
    _slp[k] = sign * 0.5;
    for(let j = 0; j < 3; j++){
      if(j === best.ax) continue;
      _slp[_AXES[j]] = clamp(_slp[_AXES[j]], -0.5, 0.5);
    }
    const point = tm.localToWorld(_slp.clone());
    const axis = _av3.set(best.ax===0?sign:0, best.ax===1?sign:0, best.ax===2?sign:0)
      .applyQuaternion(_sq).normalize();
    return {target:{body:best.tgt}, pos, quat, point, axis: axis.clone()};
  }
  /* 2 — flat surfaces: a work table's top first, then the deck */
  hm.getWorldQuaternion(_sq);
  const relW = snapQuantize(hm, _IDQ);
  snapHalf(relW, hm.scale, _shalf);
  /* 2a — a table under it: lie flat ON the top.  Never a nail — the gun
     refuses a table target (§5); you build on it and carry the work off */
  const tt = tableTopAt(_av.x, _av.y, _av.z, held.venue, held);
  if(tt !== null && _av.y - _shalf.y < tt + 0.35){
    const pos = _av.clone(); pos.y = tt + _shalf.y;
    return {target:{table:true}, pos, quat:relW.clone(),
            point:new T.Vector3(_av.x, tt, _av.z), axis:new T.Vector3(0, -1, 0)};
  }
  if(_av.y - _shalf.y < 0.3){
    const pos = _av.clone(); pos.y = _shalf.y;
    return {target:{deck:true}, pos, quat:relW.clone(),
            point:new T.Vector3(_av.x, 0, _av.z), axis:new T.Vector3(0, -1, 0)};
  }
  /* 3 — a live fly pipe: hang the piece under the batten */
  if(typeof FLY !== 'undefined'){
    for(const ls of FLY){
      if(!ls.group) continue;
      ls.group.getWorldPosition(_av2);
      if(Math.abs(_av.z - _av2.z) > 0.35) continue;
      if(Math.abs((_av.y + _shalf.y) - _av2.y) > 0.4) continue;
      if(Math.abs(_av.x - _av2.x) > (D.procW + 6)/2) continue;
      const pos = new T.Vector3(_av.x, _av2.y - 0.055 - _shalf.y, _av2.z);
      return {target:{ls}, pos, quat:relW.clone(),
              point:new T.Vector3(_av.x, _av2.y - 0.05, _av2.z),
              axis:new T.Vector3(0, 1, 0)};
    }
  }
  return null;
}
/* the same offer for a whole HELD ASSEMBLY: the deck under it, or a pipe
   over it.  No re-orientation — you carry it how you carry it. */
function snapAsm(a){
  _abox.setFromObject(a.root);
  _abox.getCenter(_av);
  if(_abox.min.y < 0.3){
    const lift = -_abox.min.y;
    let low = a.pieces[0], ly = 1e9;
    for(const p of a.pieces){
      p.mesh.getWorldPosition(_av2);
      if(_av2.y < ly){ ly = _av2.y; low = p; }
    }
    return {target:{deck:true}, dy:lift, piece:low,
            point:new T.Vector3(_av.x, 0, _av.z), axis:new T.Vector3(0, -1, 0)};
  }
  if(typeof FLY !== 'undefined'){
    for(const ls of FLY){
      if(!ls.group) continue;
      ls.group.getWorldPosition(_av2);
      if(Math.abs(_av.z - _av2.z) > 0.4) continue;
      if(Math.abs(_abox.max.y - _av2.y) > 0.45) continue;
      let high = a.pieces[0], hy = -1e9;
      for(const p of a.pieces){
        p.mesh.getWorldPosition(_av3);
        if(_av3.y > hy){ hy = _av3.y; high = p; }
      }
      return {target:{ls}, dy:(_av2.y - 0.055) - _abox.max.y, piece:high,
              point:new T.Vector3(_av.x, _av2.y - 0.05, _av2.z),
              axis:new T.Vector3(0, 1, 0)};
    }
  }
  return null;
}
/* ============================================================================
   THE SEAM (build-usability spec §4): the nail gun pointed at two pieces
   ALREADY touching joins them, no hand on either.  A pair "touches" when
   no axis of the target's frame separates them by more than SEAM_TOUCH
   (the OBB test run on one box's axes — exact for our squared-up world,
   honest enough for the rest).  The contact point is the held centre
   projected onto the touching face, clamped to it; the axis is that
   face's normal.  addNail already builds, joins and merges assemblies
   from any mix of loose and fixed pieces, and already refuses what must
   be refused (two anchored assemblies) — the seam adds NO new rules.
   ========================================================================== */
const SEAM_TOUCH = 0.05;         // faces within this are touching
const SEAM_REACH = 0.45;         // the muzzle must be this close to the seam
const _smA = new T.Vector3(), _smB = new T.Vector3(), _smH = new T.Vector3(),
      _smQ = new T.Quaternion(), _smQ2 = new T.Quaternion(), _smP = new T.Vector3();
const _AXES = ['x', 'y', 'z'];
/* the seam between two wood bodies: {point, axis} in WORLD, or null */
function seamPair(A, B){
  const am = A.mesh, bm = B.mesh;
  am.updateWorldMatrix(true, false);
  bm.updateWorldMatrix(true, false);
  bm.getWorldPosition(_smB);
  _smA.copy(_smB);
  am.worldToLocal(_smA);                        // B's centre, A-unit-local
  am.getWorldQuaternion(_smQ);
  bm.getWorldQuaternion(_smQ2);
  _aq.copy(_smQ).invert().multiply(_smQ2);      // B relative to A
  snapHalf(_aq, bm.scale, _smH);                // B's half extents on A's axes
  let axis = 0, worst = -1e9;
  for(let i = 0; i < 3; i++){
    const k = _AXES[i];
    const pm = _smA[k] * am.scale[k];           // metric offset on A's axis
    const gap = Math.abs(pm) - (am.scale[k]/2 + _smH.getComponent(i));
    if(gap > SEAM_TOUCH) return null;           // daylight on this axis: apart
    if(gap > worst){ worst = gap; axis = i; }   // the touching face is the
  }                                             // axis NEAREST to separating
  const K = _AXES[axis];
  const sgn = _smA[K] < 0 ? -1 : 1;
  _smP.copy(_smA);
  _smP[K] = sgn * 0.5;                          // onto A's touching face…
  for(let i = 0; i < 3; i++){
    if(i === axis) continue;
    const k = _AXES[i];
    _smP[k] = clamp(_smP[k], -0.5, 0.5);        // …and clamped to it
  }
  const point = am.localToWorld(_smP.clone());
  const ax = _av3.set(axis === 0 ? sgn : 0, axis === 1 ? sgn : 0, axis === 2 ? sgn : 0)
    .applyQuaternion(_smQ).normalize().clone();
  return {point, axis: ax};
}
/* the nearest nailable seam to the muzzle.  The 2m prefilter also keeps
   the pair walk to the local venue — the venues are 420m apart. */
function seamSeek(muzzle){
  let best = null, bd = SEAM_REACH;
  const near = [];
  for(const b of BODIES){
    if(b.kind !== 'wood') continue;
    if(b.state !== 'loose' && b.state !== 'fixed') continue;
    b.mesh.getWorldPosition(_av2);
    if(_av2.distanceTo(muzzle) < 2) near.push(b);
  }
  for(let i = 0; i < near.length; i++)
    for(let j = i + 1; j < near.length; j++){
      const s = seamPair(near[i], near[j]);
      if(!s) continue;
      const d = s.point.distanceTo(muzzle);
      if(d < bd){ bd = d; best = {a:near[i], b:near[j], point:s.point, axis:s.axis}; }
    }
  return best;
}

/* ============================================================================
   THE RAY NAIL (build-feel RULING L): the gun pointed AT a spot on the
   wood drives the nail there.  The ray (1.2m — arm's length and a bit)
   picks the piece and the spot; the partner is the piece TOUCHING it at
   that spot (its box, grown by SEAM_TOUCH, contains the point).  If
   nothing touches at the point itself, the nearest seam ON THE HIT
   PIECE within SEAM_REACH of the spot still takes the nail — pointing
   at a plank that lies against another means the pair, wherever along
   the plank the ray landed.  A hit with no partner anywhere is a
   refusal ({miss:'bite'}), never a cosmetic nail: a nail that joins
   nothing has no place in the joint, hammer or save bookkeeping.
   ========================================================================== */
const NAIL_RAY = 1.2;
const _nrRay = new T.Raycaster();
const _nrQ = new T.Quaternion(), _nrV = new T.Vector3();
function nailRay(origin, dir){
  _nrRay.set(origin, dir);
  _nrRay.near = 0; _nrRay.far = NAIL_RAY;
  const meshes = [];
  for(const b of BODIES){
    if(b.kind !== 'wood') continue;
    if(b.state !== 'loose' && b.state !== 'fixed') continue;
    b.mesh.updateWorldMatrix(true, false);
    meshes.push(b.mesh);
  }
  if(!meshes.length) return null;
  const hits = _nrRay.intersectObjects(meshes, false);
  if(!hits.length) return null;
  const h = hits[0];
  let A = null;
  for(const b of BODIES) if(b.mesh === h.object){ A = b; break; }
  if(!A) return null;
  /* the face normal in world: box normals ride the quaternion exactly —
     the scale is per-axis and the normals are axis-aligned */
  const ax = h.face.normal.clone()
    .applyQuaternion(h.object.getWorldQuaternion(_nrQ)).normalize();
  /* the partner: the piece whose box holds the hit point */
  let B = null, bg = 1e9;
  for(const b of BODIES){
    if(b === A || b.kind !== 'wood' || b.venue !== A.venue) continue;
    if(b.state !== 'loose' && b.state !== 'fixed') continue;
    b.mesh.updateWorldMatrix(true, false);
    _nrV.copy(h.point);
    b.mesh.worldToLocal(_nrV);
    const gx = Math.max(0, Math.abs(_nrV.x) - 0.5) * b.mesh.scale.x,
          gy = Math.max(0, Math.abs(_nrV.y) - 0.5) * b.mesh.scale.y,
          gz = Math.max(0, Math.abs(_nrV.z) - 0.5) * b.mesh.scale.z;
    const g = Math.sqrt(gx*gx + gy*gy + gz*gz);
    if(g <= SEAM_TOUCH && g < bg){ bg = g; B = b; }
  }
  if(B) return {a:A, b:B, point:h.point.clone(), axis:ax};
  /* nothing touching at the spot itself: the nearest seam on the HIT
     piece still within reach of the spot */
  let best = null, bd = SEAM_REACH;
  for(const b of BODIES){
    if(b === A || b.kind !== 'wood' || b.venue !== A.venue) continue;
    if(b.state !== 'loose' && b.state !== 'fixed') continue;
    b.mesh.getWorldPosition(_nrV);
    if(_nrV.distanceTo(h.point) > 3) continue;
    const s = seamPair(A, b);
    if(!s) continue;
    const d = s.point.distanceTo(h.point);
    if(d < bd){ bd = d; best = {a:A, b, point:s.point, axis:s.axis}; }
  }
  return best || {miss:'bite', a:A, point:h.point.clone()};
}

/* feet-and-inches for the tape, the saws and the labels: 2.4384 -> 8'0" */
function ftIn(m){
  const inches = Math.round(m/0.0254);
  return Math.floor(inches/12) + "'" + (inches%12) + '"';
}

/* ============================================================================
   THE SAWS (PR 4, spec §6).  One cut station pair per shed: a TRACK TABLE
   for sheets and a CHOP BENCH for lumber (90° only in v1).  One mechanic
   for both: release a held piece over the station and it SEATS (length
   along table X, kept in st.pieces — 'seated' pieces never settle);
   grab the CUTTER and slide it (snapped to the INCH, a pencil tick from
   the tape snapping FIRST); trigger with the cutter in hand cuts
   whichever piece lies under the blade.  A cut re-scales one body and
   registers a second — never new geometry; both sides keep the paint
   (shared material entries, copied array).  A side under six inches is
   scrap and vanishes.  Geometry lives in p2m with the sheds; the logic
   is here.
   ========================================================================== */
var SAWS = {palace:null, arc:null};
const SAW_MIN = 0.1524;          // six inches — under this is scrap
function seatLen(b){ return b.seatAxis === 'W' ? b.dims.W : b.dims.L; }
/* seat a held/loose piece on a station: sheets on the track table, lumber
   on the chop bench, length along table X.  Returns false if refused. */
function seatWood(st, b){
  if(!st || b.kind !== 'wood') return false;
  if(st.kind === 'track' ? b.prof !== 'sheet' : b.prof === 'sheet') return false;
  if(b.state === 'slotted') unslotBody(b);
  if(b.station) return false;
  /* which of the sheet's axes lies along the table: whichever the hand
     offered closer to table X */
  let axis = 'L';
  if(b.prof === 'sheet'){
    b.mesh.updateMatrixWorld(true);
    _av.set(1, 0, 0).applyQuaternion(b.mesh.getWorldQuaternion(_aq));
    st.group.getWorldQuaternion(_aq2);
    _av2.set(1, 0, 0).applyQuaternion(_aq2);
    axis = Math.abs(_av.dot(_av2)) >= 0.5 ? 'L' : 'W';
  }
  st.seat.add(b.mesh);
  if(b.prof === 'sheet')
    b.mesh.rotation.set(-Math.PI/2, axis === 'W' ? Math.PI/2 : 0, 0, 'YXZ');
  else
    b.mesh.rotation.set(0, 0, -Math.PI/2);       // lumber: long axis to X
  b.mesh.position.set(0, 0, 0);
  b.mesh.updateMatrixWorld(true);
  _abox.setFromObject(b.mesh);
  b.mesh.position.y = (_abox.max.y - _abox.min.y)/2 + 0.001;
  b.mesh.position.x = 0;
  b.state = 'seated'; b.station = st; b.seatAxis = axis;
  st.pieces.push(b);
  return true;
}
/* slide the cutter: snap to the inch, but a pencil tick on the piece under
   the blade snaps FIRST (the tape's mark, spec §6) */
function sawSetCut(st, rawX){
  let x = clamp(rawX, -st.span/2, st.span/2);
  x = Math.round(x/0.0254)*0.0254;
  for(const p of st.pieces){
    if(!p.tick) continue;
    p.mesh.updateMatrixWorld(true);
    _av.copy(p.tick.p).applyMatrix4(p.mesh.matrixWorld);
    st.group.worldToLocal(_av);
    if(Math.abs(rawX - _av.x) < 0.04){ x = _av.x; break; }
  }
  st.cut = x;
  st.cutter.position.x = x;
  return x;
}
function sawResize(b, newLen, cx){
  if(b.seatAxis === 'W'){ b.dims.W = newLen; b.mesh.scale.y = newLen; }
  else { b.dims.L = newLen; b.mesh.scale[b.prof === 'sheet' ? 'x' : 'y'] = newLen; }
  b.mesh.position.x = cx;
  delete b.tick;                 // the mark is consumed by the cut
}
/* ============================================================================
   PAINT (PR 5, spec §7).  A rack in each shed: open cans, one per color
   the shed OWNS (four stock colors; a delivered can released over the
   rack ADDS its color and is consumed), and one roller.  Dip the roller
   in a can, trigger against wood: sheets take the color on the touched
   FACE (box material group order px nx py ny pz nz), lumber whole-piece.
   Materials only ever come from the WOODM cache — repaint is a pointer
   swap, never a new material.
   ========================================================================== */
var RACKS = {palace:null, arc:null};
const RACK_STOCK = [0, 1, 2, 7];         // black, white, grey, brown to start
function rackCans(rack){
  for(const m of rack.canMeshes){ if(m.parent) m.parent.remove(m); }
  rack.canMeshes.length = 0;
  rack.colors.forEach((c, i)=>{
    const m = makeBuildMesh('paint', c);
    m.position.set(-rack.shelfW/2 + 0.16 + i*0.24, rack.shelfY, 0);
    m.userData.paintColor = c;
    rack.group.add(m);
    rack.canMeshes.push(m);
  });
}
function rackAddColor(rack, color){
  if(rack.colors.indexOf(color) >= 0) return false;
  rack.colors.push(color);
  rackCans(rack);
  return true;
}
/* a delivered can released over the rack: color in, body consumed */
function rackTakeCan(rack, b){
  if(!b || b.kind !== 'paint') return false;
  rackAddColor(rack, b.color);
  if(b.mesh.parent) b.mesh.parent.remove(b.mesh);
  if(b.slot) b.slot.userData.body = null;
  BODIES.splice(BODIES.indexOf(b), 1);
  return true;
}
function paintWood(b, wp, color){
  if(!b || b.kind !== 'wood' || color === undefined || color === null) return false;
  const m = woodMat(color);
  if(b.prof !== 'sheet'){
    for(let i = 0; i < 6; i++) b.mesh.material[i] = m;
    return true;
  }
  b.mesh.updateMatrixWorld(true);
  _av.copy(wp);
  b.mesh.worldToLocal(_av);                // unit-box local: extents are ±0.5
  const ax = Math.abs(_av.x), ay = Math.abs(_av.y), az = Math.abs(_av.z);
  let idx;
  if(ax >= ay && ax >= az) idx = _av.x >= 0 ? 0 : 1;
  else if(ay >= az)        idx = _av.y >= 0 ? 2 : 3;
  else                     idx = _av.z >= 0 ? 4 : 5;
  b.mesh.material[idx] = m;
  return true;
}

function sawCut(st){
  const x = st.cut;
  const b = st.pieces.find(p=>{
    const half = seatLen(p)/2;
    return x > p.mesh.position.x - half + 0.005 && x < p.mesh.position.x + half - 0.005;
  });
  if(!b) return null;
  const len = seatLen(b), lo = b.mesh.position.x - len/2;
  let s1 = x - lo, s2 = len - s1;
  if(s1 < SAW_MIN && s2 < SAW_MIN) return null;
  if(s1 < SAW_MIN){                       // the near side is scrap: keep the far
    sawResize(b, s2, x + s2/2);
    return {kept:b, off:null};
  }
  if(s2 < SAW_MIN){
    sawResize(b, s1, lo + s1/2);
    return {kept:b, off:null};
  }
  const dims2 = b.seatAxis === 'W' ? {L:b.dims.L, W:s2} : {L:s2, W:b.dims.W};
  sawResize(b, s1, lo + s1/2);
  const keep = BUILD_VENUE;
  BUILD_VENUE = st.venue;
  const nb = regWood(b.prof, dims2);
  BUILD_VENUE = keep;
  nb.mesh.material = b.mesh.material.slice();   // paint rides the cut
  st.seat.add(nb.mesh);
  nb.mesh.rotation.copy(b.mesh.rotation);
  nb.mesh.position.copy(b.mesh.position);
  nb.mesh.position.x = x + s2/2 + 0.01;
  nb.state = 'seated'; nb.station = st; nb.seatAxis = b.seatAxis;
  st.pieces.push(nb);
  if(typeof buildDirty === 'function') buildDirty();
  return {kept:b, off:nb};
}

/* ============================================================================
   THE SAVE SYSTEM (PR 7, spec §8) — the game's first.  Everything the
   build system owns — wood and hardware wherever they stand, paint,
   assemblies with their nails, hinges and pivot poses, track runs and
   their riders, pallets and their loads, pending orders with time left,
   rack colors, forklift poses — serialized to localStorage, versioned,
   and rebuilt at boot by REPLAYING the same functions the hands use
   (makeSerBody, asmAdopt, addNail, rideTrack), so the loaded world obeys
   every rule the built one did.  Deliberately NOT saved: shows, cues,
   fly positions, the lantern patch — a different feature.  Load is
   wrapped whole: any throw clears the key and the boot carries on empty
   (a bad save must never brick the game).
   ========================================================================== */
const BUILD_SAVE_KEY = 'house.build';
const BUILD_SAVE_VER = 1;
var _buildDirty = false, _buildFlushT = 0, _buildSlowT = 10, _buildLoading = false;
function buildDirty(){ if(!_buildLoading) _buildDirty = true; }
/* dirty writes flush within a second; a slow heartbeat saves every ten
   regardless, so a missed dirty call can never lose more than that */
function buildTick(dt){
  _buildFlushT -= dt;
  _buildSlowT -= dt;
  if(_buildDirty && _buildFlushT <= 0){
    _buildFlushT = 1;
    _buildSlowT = 10;
    _buildDirty = false;
    buildSave();
  } else if(_buildSlowT <= 0){
    _buildSlowT = 10;
    buildSave();
  }
}
function serPose(o){
  o.updateMatrixWorld(true);
  const p = o.getWorldPosition(new T.Vector3());
  const q = o.getWorldQuaternion(new T.Quaternion());
  return [p.x, p.y, p.z, q.x, q.y, q.z, q.w];
}
function setPose(o, s){
  _av.set(s[0], s[1], s[2]);
  _aq.set(s[3], s[4], s[5], s[6]);
  if(o.parent){
    o.parent.updateMatrixWorld(true);
    o.parent.worldToLocal(_av);
    o.parent.getWorldQuaternion(_aq2).invert();
    _aq.premultiply(_aq2);
  }
  o.position.copy(_av);
  o.quaternion.copy(_aq);
  o.updateMatrixWorld(true);
}
function serBody(b){
  const d = {k:b.kind};
  if(b.prof) d.prof = b.prof;
  if(b.dims) d.dims = {L:b.dims.L, W:b.dims.W};
  if(b.color !== undefined) d.c = b.color;
  if(b.kind === 'wood') d.paint = b.mesh.material.map(m=>m.color.getHex());
  if(b.tick) d.tick = [b.tick.p.x, b.tick.p.y, b.tick.p.z];
  if(b.frozen) d.fz = 1;                    // parked in place (RULING N)
  return d;
}
function makeSerBody(d, venue){
  const keep = BUILD_VENUE;
  BUILD_VENUE = venue;
  let b;
  if(d.k === 'wood'){
    b = regWood(d.prof, {L:d.dims.L, W:d.dims.W});
    if(d.paint) b.mesh.material = d.paint.map(c=>woodMat(c));
  } else {
    b = regBody(d.k, makeBodyMesh(d.k, d.c), null);
    if(d.c !== undefined) b.color = d.c;
    b.state = 'loose';
  }
  if(d.tick) b.tick = {p:new T.Vector3(d.tick[0], d.tick[1], d.tick[2])};
  if(d.fz) b.frozen = true;
  BUILD_VENUE = keep;
  return b;
}
function lsAddress(ls){
  const i0 = FLY.indexOf(ls);
  if(i0 >= 0) return {stage:STAGE, idx:i0};
  for(const k in STAGES){
    const i = STAGES[k].fly.indexOf(ls);
    if(i >= 0) return {stage:k, idx:i};
  }
  return null;
}
function lsResolve(ad){
  if(!ad) return null;
  if(ad.stage === STAGE) return FLY[ad.idx] || null;
  return (STAGES[ad.stage] && STAGES[ad.stage].fly[ad.idx]) || null;
}
function onPallet(b){
  const o = ORDERS[b.venue];
  if(!o || !b.slot) return false;
  return o.pallets.some(p=>p.slots.indexOf(b.slot) >= 0);
}
function buildSerialize(){
  const out = {v:BUILD_SAVE_VER, bodies:[], asms:[], pallets:[], pending:[],
               racks:{}, lifts:{}};
  for(const b of BODIES){
    if(!BUILD_KINDS[b.kind]) continue;
    if(b.state === 'fixed' || b.state === 'riding') continue;  // their assembly's story
    if(b.state === 'slotted' && onPallet(b)) continue;         // the pallet's story
    const d = serBody(b);
    d.venue = b.venue;
    d.pose = serPose(b.mesh);
    if(b.state === 'slotted'){
      const sh = SHEDS[b.venue] ? SHEDS[b.venue].slots.indexOf(b.slot) : -1;
      const ca = CARTS[b.venue] ? CARTS[b.venue].slots.indexOf(b.slot) : -1;
      if(sh >= 0) d.slot = {t:'shed', i:sh};
      else if(ca >= 0) d.slot = {t:'cart', i:ca};
      else continue;                       // an anchor we cannot address: skip it
    }
    if(b.state === 'seated' && b.station)
      d.seat = {kind:b.station.kind, x:b.mesh.position.x};
    out.bodies.push(d);
  }
  for(const a of ASSEMBLIES){
    const d = {venue:a.venue, rootPose:serPose(a.root), pieces:[], nails:[], cars:[]};
    if(a.track) d.track = a.track.n;
    if(a.anchor) d.anchor = a.anchor.type === 'pipe'
      ? {t:'pipe', ls:lsAddress(a.anchor.ls)} : {t:'deck'};
    for(const cb of BODIES)
      if(cb.kind === 'carriage' && cb.tAsm === a)
        d.cars.push({c:serBody(cb), x:cb.slider.position.x, _slider:cb.slider});
    for(const p of a.pieces){
      if(p.kind === 'track') continue;     // track runs re-lay their own sections
      const pd = serBody(p);
      pd.pose = serPose(p.mesh);
      if(p.slider){
        const ci = d.cars.findIndex(c=>c._slider === p.slider);
        if(ci >= 0) pd.car = ci;
      }
      pd._b = p;
      d.pieces.push(pd);
    }
    for(const n of a.nails){
      if(a.track && n.b === 'deck') continue;       // laid again by layTrack
      if(n.carriage) continue;                      // implied by pd.car
      a.root.updateMatrixWorld(true);
      const wp = a.root.localToWorld(n.p.clone());
      const wax = n.axis.clone()
        .applyQuaternion(a.root.getWorldQuaternion(new T.Quaternion()));
      d.nails.push({ai:d.pieces.findIndex(x=>x._b === n.a),
                    bi:n.b === 'deck' ? 'deck' : n.b === 'pipe' ? 'pipe'
                       : d.pieces.findIndex(x=>x._b === n.b),
                    p:[wp.x, wp.y, wp.z], ax:[wax.x, wax.y, wax.z],
                    h:n.hinge ? 1 : 0, r:n.range});
    }
    d.pieces.forEach(pd=>{ delete pd._b; });
    d.cars.forEach(cd=>{ delete cd._slider; });
    out.asms.push(d);
  }
  for(const v in ORDERS){
    const o = ORDERS[v];
    for(const pal of o.pallets){
      out.pallets.push({venue:v, spot:pal.spot, pose:serPose(pal.group),
        slots:pal.slots.map(an=>({p:[an.position.x, an.position.y, an.position.z],
          rx:an.rotation.x, rz:an.rotation.z,
          d:an.userData.body ? serBody(an.userData.body) : null}))});
    }
    for(const pd of o.pending) out.pending.push({venue:v, items:pd.items, t:pd.t});
  }
  for(const v in RACKS) if(RACKS[v]) out.racks[v] = RACKS[v].colors.slice();
  for(const v in LIFTS) if(LIFTS[v])
    out.lifts[v] = {x:LIFTS[v].x, z:LIFTS[v].z, yaw:LIFTS[v].yaw, forkY:LIFTS[v].forkY};
  return JSON.stringify(out);
}
function buildSave(){
  try{ localStorage.setItem(BUILD_SAVE_KEY, buildSerialize()); }catch(e){}
}
function buildClearSave(){
  try{ localStorage.removeItem(BUILD_SAVE_KEY); }catch(e){}
  if(typeof toast === 'function') toast('saved build cleared — the next visit starts empty');
}
function buildPalletFromSer(pd){
  const o = ORDERS[pd.venue];
  if(!o) return;
  const g = new T.Group();
  const top = new T.Mesh(new T.BoxGeometry(1.7, 0.05, 1.2), M.wood);
  top.position.y = 0.15; g.add(top);
  [-0.5, 0.5].forEach(bz=>{
    const br = new T.Mesh(new T.BoxGeometry(1.7, 0.12, 0.14), M.woodDk);
    br.position.set(0, 0.06, bz); g.add(br);
  });
  const slots = [];
  for(const sd of pd.slots){
    const an = new T.Object3D();
    an.position.set(sd.p[0], sd.p[1], sd.p[2]);
    an.rotation.x = sd.rx || 0;
    an.rotation.z = sd.rz || 0;
    g.add(an); slots.push(an);
  }
  venueRoot(pd.venue).add(g);
  setPose(g, pd.pose);
  o.pallets.push({group:g, slots, spot:pd.spot});
  pd.slots.forEach((sd, i)=>{
    if(sd.d) slotBody(makeSerBody(sd.d, pd.venue), slots[i]);
  });
}
function buildLoad(){
  let raw = null;
  try{ raw = localStorage.getItem(BUILD_SAVE_KEY); }catch(e){ return; }
  if(!raw) return;
  _buildLoading = true;
  try{
    const s = JSON.parse(raw);
    if(!s || s.v !== BUILD_SAVE_VER) throw new Error('unknown save version');
    for(const v in s.racks) if(RACKS[v])
      s.racks[v].forEach(c=>rackAddColor(RACKS[v], c));
    for(const d of s.bodies){
      const b = makeSerBody(d, d.venue);
      if(d.slot){
        const an = d.slot.t === 'shed' ? SHEDS[d.venue].slots[d.slot.i]
                                       : CARTS[d.venue].slots[d.slot.i];
        if(an && !an.userData.body){ slotBody(b, an); continue; }
      }
      venueRoot(d.venue).add(b.mesh);
      setPose(b.mesh, d.pose);
      if(d.seat && typeof SAWS !== 'undefined' && SAWS[d.venue] && SAWS[d.venue][d.seat.kind]){
        if(seatWood(SAWS[d.venue][d.seat.kind], b)){ b.mesh.position.x = d.seat.x; continue; }
      }
      b.restH = b.kind === 'wood' ? woodRestH(b) : 0.14;
    }
    for(const pd of s.pallets) buildPalletFromSer(pd);
    for(const ad of s.asms){
      if(ad.track){
        let a = null;
        for(let i = 0; i < ad.track; i++){
          const tb = makeSerBody({k:'track'}, ad.venue);
          if(i === 0){
            venueRoot(ad.venue).add(tb.mesh);
            setPose(tb.mesh, ad.rootPose);
          }
          a = layTrack(tb, a);
        }
        (ad.cars || []).forEach(cd=>{
          const cb = makeSerBody(cd.c, ad.venue);
          venueRoot(ad.venue).add(cb.mesh);
          a.root.updateMatrixWorld(true);
          cb.mesh.position.copy(a.root.localToWorld(new T.Vector3(cd.x, 0.1, 0)));
          cb.mesh.updateMatrixWorld(true);
          rideTrack(cb, a);
          cd._cb = cb;
        });
        (ad.pieces || []).forEach(pd=>{
          const b = makeSerBody(pd, ad.venue);
          venueRoot(ad.venue).add(b.mesh);
          setPose(b.mesh, pd.pose);
          if(pd.car !== undefined && ad.cars[pd.car] && ad.cars[pd.car]._cb){
            nailToCarriage(b, ad.cars[pd.car]._cb);
            setPose(b.mesh, pd.pose);
          }
        });
        continue;
      }
      const a = newAsm(ad.venue);
      setPose(a.root, ad.rootPose);
      const ps = ad.pieces.map(pd=>{
        const b = makeSerBody(pd, ad.venue);
        venueRoot(ad.venue).add(b.mesh);
        setPose(b.mesh, pd.pose);
        asmAdopt(a, b);
        setPose(b.mesh, pd.pose);
        return b;
      });
      for(const nd of ad.nails){
        const wp = new T.Vector3(nd.p[0], nd.p[1], nd.p[2]);
        const wax = new T.Vector3(nd.ax[0], nd.ax[1], nd.ax[2]);
        let n = null;
        if(nd.bi === 'deck') n = addNail(ps[nd.ai], {deck:true}, wp, wax);
        else if(nd.bi === 'pipe')
          n = addNail(ps[nd.ai], {ls:lsResolve(ad.anchor && ad.anchor.ls)}, wp, wax);
        else n = addNail(ps[nd.ai], {body:ps[nd.bi]}, wp, wax);
        if(n && nd.h){
          n.hinge = true;
          n.range = nd.r || Math.PI/2;
          if(n.mesh.parent) n.mesh.parent.remove(n.mesh);
          const hm = makeBuildMesh('hinge');
          a.root.add(hm);
          hm.position.copy(n.p);
          n.mesh = hm;
          asmJoints(a);
        }
      }
    }
    for(const od of s.pending)
      if(ORDERS[od.venue]) ORDERS[od.venue].pending.push({items:od.items, t:od.t});
    for(const v in s.lifts){
      const L = LIFTS[v];
      if(!L) continue;
      L.x = s.lifts[v].x; L.z = s.lifts[v].z; L.yaw = s.lifts[v].yaw;
      L.forkY = L.prevForkY = s.lifts[v].forkY;
      L.forks.position.y = L.forkY;
      cartPose(L);
    }
  }catch(e){
    try{ localStorage.removeItem(BUILD_SAVE_KEY); }catch(e2){}
    if(typeof toast === 'function') toast('the saved build would not load — cleared');
  }
  _buildLoading = false;
}
