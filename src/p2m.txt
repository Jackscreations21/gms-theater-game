
/* ============================================================================
   p2m — THE WAREHOUSES.  One shed behind each venue: racking for the rig's
   loose lanterns (the detach system is a later PR — this part builds the
   buildings, the racks and the carts), a pushable cart each, and a roller
   door into the venue's backstage.

   The Palace shed is plain world geometry behind the stage house — it
   builds at parse time like the rest of the Palace, and p2i (later in the
   build) files it into its own room.  The Arc shed needs ARC.group, which
   only exists once buildArc has run, so buildArcShed() is called from the
   tail of buildArc (p2j) — function declarations hoist across the
   concatenated file.
   ========================================================================== */
var SHEDS = {palace:null, arc:null};
var CARTS = {palace:null, arc:null};

/* the poured slab the Palace shed stands on (the Arc shed is built from
   buildArc's own concrete, passed in) */
const M_SHED = {
  conc: new T.MeshStandardMaterial({color:0x8e8b84, roughness:.9})
};

/* one rack bay: two shelves of slot anchors against a wall */
function shedRack(parent, cx, cy, cz, yaw, slots){
  const g = new T.Group(); g.position.set(cx, cy, cz); g.rotation.y = yaw;
  parent.add(g);
  const frame = new T.Mesh(new T.BoxGeometry(3.6, 2.2, 0.1), M.steel);
  frame.position.set(0, 1.1, -0.3); frame.castShadow = frame.receiveShadow = true;
  g.add(frame);
  [0.55, 1.45].forEach(sy=>{
    const shelf = new T.Mesh(new T.BoxGeometry(3.6, 0.06, 0.62), M.steel);
    shelf.position.set(0, sy, 0); shelf.castShadow = shelf.receiveShadow = true;
    g.add(shelf);
    for(let i=0;i<4;i++){
      const a = new T.Object3D();
      a.position.set(-1.35 + i*0.9, sy + 0.06, 0);
      g.add(a); slots.push(a);
    }
  });
  return g;
}

/* ---- the cart: a two-shelf pushcart with a handle bar on its -z end ---- */
function buildCart(venue, parent, x, z){
  const g = new T.Group();
  g.userData.moves = true;
  g.userData.cartInfo = true;
  const frame = new T.Mesh(new T.BoxGeometry(0.74, 0.06, 1.24), M.steel);
  frame.position.y = 0.26; g.add(frame);
  const top = new T.Mesh(new T.BoxGeometry(0.74, 0.06, 1.24), M.steel);
  top.position.y = 0.86; g.add(top);
  [[-0.35,-0.6],[0.35,-0.6],[-0.35,0.6],[0.35,0.6]].forEach(cn=>{
    const leg = new T.Mesh(new T.CylinderGeometry(0.02,0.02,0.86,6), M.steel);
    leg.position.set(cn[0], 0.55, cn[1]); g.add(leg);
    const wheel = new T.Mesh(new T.CylinderGeometry(0.08,0.08,0.05,10), M.fixture);
    wheel.rotation.z = Math.PI/2; wheel.position.set(cn[0], 0.08, cn[1]); g.add(wheel);
  });
  const hz = -0.72;
  const handle = new T.Mesh(new T.CylinderGeometry(0.022,0.022,0.7,8), M.pipe);
  handle.rotation.z = Math.PI/2; handle.position.set(0, 1.02, hz); g.add(handle);
  [-0.33, 0.33].forEach(hx=>{
    const up = new T.Mesh(new T.CylinderGeometry(0.018,0.018,0.34,6), M.steel);
    up.position.set(hx, 0.95, hz); g.add(up);
  });
  g.traverse(o=>{ if(o.isMesh){ o.castShadow = true; o.receiveShadow = true; } });
  const cart = {
    venue, group:g, x, z, yaw:0, yBase:0,
    handleH:1.02, handleZ:hz, handleHalf:0.33, grabR:0.30,
    slots:[]
  };
  [0.32, 0.92].forEach(sy=>{
    for(let i=0;i<3;i++){
      const a = new T.Object3D();
      a.position.set(0, sy, -0.42 + i*0.42);
      g.add(a); cart.slots.push(a);
    }
  });
  parent.add(g);
  CARTS[venue] = cart;
  cartPose(cart);
  if(typeof INTERACT !== 'undefined') INTERACT.push(g);
  return cart;
}
/* the cart's pose lives in {x, z, yaw, yBase} in WORLD coordinates; the Arc
   cart's parent hangs off ARC.group at x=420, so the offset comes back off */
function cartPose(c){
  const ox = (c.venue === 'arc' && typeof ARC !== 'undefined') ? ARC.X : 0;
  c.group.position.set(c.x - ox, c.yBase, c.z);
  c.group.rotation.y = c.yaw;
}

/* ============================================================================
   THE FORKLIFT (build spec §4, RULING H) — a walk-behind pallet jack with
   a forklift silhouette, one per shed.  Grab the handle and it follows,
   exactly the cart: the record lives in CARTS as well as LIFTS, so the p9
   grab, chase and wall code drive it WITHOUT A FORK of the machinery.
   While it is held, the RIGHT stick runs the forks (rx is smooth turn and
   the left stick walks — ry was the free axis).  Forks under a pallet's
   boards + raise = the pallet rides them; lower them off at the floor and
   it stands where it stops — or snaps to a painted PALLET SLOT within
   half a metre.  Slots: four to a shed, one in each stage wing.
   ========================================================================== */
var LIFTS = {palace:null, arc:null};
var PALLET_SLOTS = [];            // {venue, x, z, mesh} — x/z in WORLD coords
const LIFT_FORK_TOP = 1.2;        // metres of fork travel
const LIFT_FORK_RATE = 0.5;      // m/s under the stick
const M_SLOT = new T.MeshBasicMaterial({color:0xc9a03c, transparent:true, opacity:0.35});
const M_LIFT = new T.MeshStandardMaterial({color:0xc9611d, roughness:.6, metalness:.3});
function palletSlot(venue, wx, wz, parent){
  const m = new T.Mesh(new T.PlaneGeometry(1.9, 1.4), M_SLOT);
  m.rotation.x = -Math.PI/2;
  const ox = (venue === 'arc' && typeof ARC !== 'undefined') ? ARC.X : 0;
  m.position.set(wx - ox, 0.02, wz);
  m.raycast = ()=>{};             // paint on the floor, not a thing
  parent.add(m);
  PALLET_SLOTS.push({venue, x:wx, z:wz, mesh:m});
}
/* ---- the cut stations (build spec §6; the logic lives in p4c) ----------
   One pair per shed: the TRACK TABLE for sheets, the CHOP BENCH for
   lumber.  st.cutter is the one grabbable — a rail-and-head for the
   track saw, an arm-and-blade for the chop — sliding along table X. */
function buildSaw(venue, kind, parent, wx, wz){
  const g = new T.Group(); g.name = 'saw:' + venue + ':' + kind;
  const ox = (venue === 'arc' && typeof ARC !== 'undefined') ? ARC.X : 0;
  g.position.set(wx - ox, 0, wz);
  const W = kind === 'track' ? 2.9 : 2.8, D2 = kind === 'track' ? 1.6 : 0.62;
  const topY = 0.86;
  const top = new T.Mesh(new T.BoxGeometry(W, 0.07, D2), M.steel);
  top.position.y = topY; g.add(top);
  [[-W/2+0.15, -D2/2+0.1],[W/2-0.15, -D2/2+0.1],[-W/2+0.15, D2/2-0.1],[W/2-0.15, D2/2-0.1]]
    .forEach(cn=>{
      const leg = new T.Mesh(new T.CylinderGeometry(0.03, 0.03, topY, 6), M.steel);
      leg.position.set(cn[0], topY/2, cn[1]); g.add(leg);
    });
  const seat = new T.Object3D();
  seat.position.y = topY + 0.035;
  g.add(seat);
  const cutter = new T.Group();
  cutter.position.y = topY + 0.035;
  if(kind === 'track'){
    const rail = new T.Mesh(new T.BoxGeometry(0.09, 0.03, D2 + 0.3), M.pipe);
    rail.position.y = 0.08; cutter.add(rail);
    const head = new T.Mesh(new T.BoxGeometry(0.2, 0.16, 0.22), M.fixture);
    head.position.y = 0.16; cutter.add(head);
  } else {
    const arm = new T.Mesh(new T.BoxGeometry(0.07, 0.5, 0.1), M.steel);
    arm.position.set(0, 0.25, -D2/2 - 0.06); cutter.add(arm);
    const blade = new T.Mesh(new T.CylinderGeometry(0.16, 0.16, 0.012, 16), M.fixture);
    blade.rotation.z = Math.PI/2; blade.position.y = 0.3; cutter.add(blade);
  }
  cutter.userData.moves = true;
  g.add(cutter);
  g.traverse(o=>{ if(o.isMesh){ o.castShadow = true; o.receiveShadow = true; } });
  parent.add(g);
  const st = {venue, kind, group:g, seat, cutter, pieces:[], cut:0,
              span:W - 0.1, grabR:0.28, gripY:topY + 0.2};
  if(!SAWS[venue]) SAWS[venue] = {};
  SAWS[venue][kind] = st;
  return st;
}

/* ---- the paint rack (build spec §7; the logic lives in p4c) ------------
   A wall shelf of open cans — one per color the shed owns — and ONE
   roller, homed to the rack.  The roller is its own grab class in p9;
   release anywhere re-racks it, the belt-tool discipline. */
function buildRack(venue, parent, wx, wz, yaw){
  const g = new T.Group(); g.name = 'rack:' + venue;
  const ox = (venue === 'arc' && typeof ARC !== 'undefined') ? ARC.X : 0;
  g.position.set(wx - ox, 0, wz);
  g.rotation.y = yaw || 0;
  const shelfW = 2.4;
  [1.0, 1.5].forEach(sy=>{
    const shelf = new T.Mesh(new T.BoxGeometry(shelfW, 0.05, 0.36), M.steel);
    shelf.position.set(0, sy, 0);
    shelf.castShadow = shelf.receiveShadow = true;
    g.add(shelf);
  });
  parent.add(g);
  const rack = {venue, group:g, shelfW, shelfY:1.025, colors:RACK_STOCK.map(i=>PAINT_COLORS[i].c),
                canMeshes:[], roller:null};
  /* the roller: a handle and a head that shows the last dip */
  const rg = new T.Group();
  const handle = new T.Mesh(new T.CylinderGeometry(0.012, 0.012, 0.2, 8), M.pipe);
  handle.position.y = -0.1; rg.add(handle);
  const head = new T.Mesh(new T.CylinderGeometry(0.035, 0.035, 0.18, 10), M.steel);
  head.rotation.z = Math.PI/2; head.position.y = 0.03; rg.add(head);
  const home = new T.Object3D();
  home.position.set(shelfW/2 - 0.15, 1.55, 0.05);
  g.add(home); home.add(rg);
  rack.roller = {mesh:rg, head, color:null, home, grabR:0.22};
  RACKS[venue] = rack;
  rackCans(rack);
  return rack;
}

  [-0.24, 0.24].forEach(mx=>{      // the mast the carriage rides
    const up = new T.Mesh(new T.BoxGeometry(0.07, 1.62, 0.07), M.steel);
    up.position.set(mx, 0.81, 0.1); g.add(up);
  });
  const handle = new T.Mesh(new T.CylinderGeometry(0.022, 0.022, 0.56, 8), M.pipe);
  handle.rotation.z = Math.PI/2; handle.position.set(0, 1.0, -0.62); g.add(handle);
  const stem = new T.Mesh(new T.CylinderGeometry(0.02, 0.02, 0.5, 6), M.steel);
  stem.rotation.x = 0.55; stem.position.set(0, 0.82, -0.5); g.add(stem);
  [[-0.26,-0.4],[0.26,-0.4],[-0.28,1.2],[0.28,1.2]].forEach(cn=>{
    const wheel = new T.Mesh(new T.CylinderGeometry(0.09, 0.09, 0.06, 10), M.fixture);
    wheel.rotation.z = Math.PI/2; wheel.position.set(cn[0], 0.09, cn[1]); g.add(wheel);
  });
  const forks = new T.Group(); g.add(forks);
  [-0.28, 0.28].forEach(fx=>{
    const f = new T.Mesh(new T.BoxGeometry(0.15, 0.05, 1.1), M.steel);
    f.position.set(fx, 0.09, 0.85); forks.add(f);

  carr.position.set(0, 0.24, 0.33); forks.add(carr);
  g.traverse(o=>{ if(o.isMesh){ o.castShadow = true; o.receiveShadow = true; } });

    slots:[], lift:true, forks, forkY:0, prevForkY:0, riding:null
  };
  parent.add(g);
  LIFTS[venue] = lift;
  CARTS[venue + 'Lift'] = lift;   // the p9 cart machinery takes it whole
  cartPose(lift);
  if(typeof INTERACT !== 'undefined') INTERACT.push(g);
  return lift;
}
/* forks up under the boards takes a pallet; forks flat on the floor while
   one is riding sets it down — snapped to a slot if one is in reach */
const _liftV = new T.Vector3();
function updateLifts(dt){
  for(const k in LIFTS){
    const L = LIFTS[k];
    if(!L) continue;
    L.forks.position.y = L.forkY;
    const rising = L.forkY > L.prevForkY + 1e-5;
    const lowering = L.forkY < L.prevForkY - 1e-5;
    if(!L.riding && rising && L.prevForkY < 0.13 && L.forkY >= 0.13){
      const o = ORDERS[L.venue];
      L.group.updateMatrixWorld(true);
      for(const pal of o.pallets){
        pal.group.getWorldPosition(_liftV);
        const lp = L.group.worldToLocal(_liftV.clone());
        if(Math.abs(lp.x) < 0.5 && lp.z > 0.35 && lp.z < 1.45 && Math.abs(lp.y) < 0.5){
          L.forks.attach(pal.group);
          L.riding = pal;
          pal.spot = -1;            // the apron spot is free again
          if(typeof toast === 'function') toast('the pallet is on the forks');
          break;
        }
      }
    }
    if(L.riding && lowering && L.forkY <= 0.02) liftDrop(L);
    L.prevForkY = L.forkY;
  }
}
function liftDrop(L){
  const pal = L.riding;
  L.riding = null;
  const root = (typeof venueRoot === 'function') ? venueRoot(L.venue) : world;
  root.attach(pal.group);
  pal.group.updateMatrixWorld(true);
  pal.group.getWorldPosition(_liftV);
  let best = null, bd = 0.6;
  for(const s of PALLET_SLOTS){
    if(s.venue !== L.venue) continue;
    const d = Math.hypot(_liftV.x - s.x, _liftV.z - s.z);
    if(d < bd){ bd = d; best = s; }
  }
  const ox = (L.venue === 'arc' && typeof ARC !== 'undefined') ? ARC.X : 0;
  if(best){
    pal.group.position.set(best.x - ox, 0, best.z);
    pal.group.rotation.set(0, 0, 0);
  } else {
    const g = (typeof groundAt === 'function')
      ? groundAt(_liftV.x, _liftV.z, _liftV.y + 0.6) : 0;
    pal.group.position.y -= _liftV.y - (g === null ? 0 : g);
  }
  if(typeof toast === 'function') toast('pallet set down');
}

/* ============================================================================
   THE PALACE SHED — brick, behind the stage house, straight out the new
   doorway in the back wall.  World geometry at parse time; p2i pins the
   group to its own room ('shed') and the door station to the stage.
   ========================================================================== */
(function buildPalaceShed(){
  const x0 = -12, x1 = 12, z0 = -30.2, z1 = D.backWall - 0.7;   // interior faces
  const g = new T.Group(); g.name = 'shedPalace';
  world.add(g);

  /* the slab — top face exactly at deck height.  It ends at D.backWall,
     where the stage deck takes over: run it any further and the two floors
     share the y=0 plane and z-fight on the stage side of the wall */
  const fl = box(x1 - x0 + 0.8, 0.6, D.backWall - z0, M_SHED.conc,
                 (x0 + x1)/2, -0.3, (z0 + D.backWall)/2, g);
  WALKABLE.push(fl);

  // the shell: rear wall, side walls, a flat roof
  box(x1 - x0 + 1.6, 6.2, 0.7, M.brick, (x0 + x1)/2, 3.1, z0 - 0.35, g);
  box(0.7, 6.2, z1 - z0 + 1.0, M.brick, x0 - 0.35, 3.1, (z0 + z1)/2 + 0.1, g);
  box(0.7, 6.2, z1 - z0 + 1.0, M.brick, x1 + 0.35, 3.1, (z0 + z1)/2 + 0.1, g);
  box(x1 - x0 + 2.4, 0.5, z1 - z0 + 1.4, M.steel, (x0 + x1)/2, 6.45, (z0 + z1)/2, g);

  /* the roller leaf, hung on the shed side of the theatre's back wall */
  const dg = new T.Group();
  dg.position.set(0, 0, z1 - 0.12);
  dg.userData.moves = true;
  g.add(dg);
  const leaf = new T.Mesh(new T.BoxGeometry(3.3, 3.4, 0.14), M.steel);
  leaf.position.y = 1.7;
  leaf.castShadow = leaf.receiveShadow = true;
  dg.add(leaf);
  // the head box and guides it runs in
  box(4.0, 0.5, 0.4, M.steel, 0, 3.85, z1 - 0.2, g);
  box(0.12, 3.8, 0.3, M.steel, -1.76, 1.9, z1 - 0.15, g);
  box(0.12, 3.8, 0.3, M.steel,  1.76, 1.9, z1 - 0.15, g);

  SHEDS.palace = {
    x0, x1, z0, z1, group:g, slots:[],
    door:{group:dg, open:0, target:0, travel:3.2, H:3.4, halfW:1.6, x:0, y0:0}

     work light has no business reaching it through a brick wall.  It lives
     in the shed group; buildRooms (p2i) lifts the light itself into SHARED,

  box(0.5, 0.2, 0.36, M.fixture, 0, 4.15, z1 - 0.5, g);
  const bulb = new T.Mesh(new T.SphereGeometry(0.09, 8, 6), M.bulb);
  bulb.position.set(0, 4.0, z1 - 0.5); g.add(bulb);
  const pl = new T.PointLight(0xffd9a0, 0.7, 13, 1.5);
  pl.position.set(0, 3.9, z1 - 0.7); g.add(pl);

  for(const rx of [-8.4, -4.5, -0.6, 3.3])
    shedRack(g, rx, 0, z0 + 0.7, 0, SHEDS.palace.slots);

  /* The cart is a TOP-LEVEL world child, not a child of the shed group: a
     cart gets pushed out of its shed in ordinary play, and a cart filed
     with the shed room would vanish the moment the roller door shut behind
     it.  p2i pins it 'shared' — fifteen meshes drawn behind a shut door is

  buildCart('palace', world, 8.5, -27.5);

     side of the shed, and one waits in each stage wing (top-level world
     children like the cart — a slot on the stage is stage furniture) */
  buildLift('palace', world, 4.5, D.backWall - 2.5);
  for(let i = 0; i < 4; i++)

  palletSlot('palace', -16.5, -15.5, world);
  palletSlot('palace',  16.5, -15.5, world);

  buildSaw('palace', 'track', g, 2.0, -24.0);

  buildRack('palace', g, 11.3, -23.0, -Math.PI/2);

     top-level world child so p2i can pin it to the stage room, never the
     shed's (the shed may be culled while you are looking right at it) */
  const ctl = box(0.34, 0.5, 0.14, M.fixture, 2.4, 1.25, D.backWall + 0.08);
  station(ctl, 'shedP', 'WAREHOUSE DOOR  —  [E] open or shut it');
  SHEDS.palace.ctl = ctl;
})();

function shedDoorToggle(which){
  const S = SHEDS[which]; if(!S) return false;
  S.door.target = S.door.target > 0.5 ? 0 : 1;
  if(typeof toast === 'function')
    toast('warehouse door '+(S.door.target ? 'opening' : 'closing'));
  return true;
}
function updateSheds(dt){
  const d = SHEDS.palace && SHEDS.palace.door;
  if(d && Math.abs(d.target - d.open) > 0.001){
    d.open += clamp(d.target - d.open, -dt*0.55, dt*0.55);
    d.group.position.y = d.y0 + d.open * d.travel;
  }
  updateOrders(dt);
  updateLifts(dt);
}

/* ============================================================================
   THE ARC SHED — one long concrete lean-to across the back of BOTH stage
   houses, in ARC-local coordinates.  Called from the tail of buildArc with
   its material set; the rear roller doors themselves are cut into each
   house's back wall by arcBuildHouse (p2j).
   ========================================================================== */
function buildArcShed(MAT){
  if(typeof ARC === 'undefined' || !ARC.rooms.shed || SHEDS.arc) return;
  const R = ARC.rooms.shed;
  const conc = (MAT && MAT.conc) ? MAT.conc : M.steel;
  const x0 = -34, x1 = 34, z0 = -60, z1 = -47.0;

  const fl = aBox(x1 - x0, 0.6, 13.2, conc, 0, -0.3, -53.5, R);
  arcWalk(fl);
  aBox(x1 - x0 + 0.8, 6.4, 0.6, conc, 0, 3.2, z0 - 0.3, R);        // rear
  aBox(0.6, 6.4, 13.6, conc, x0 - 0.3, 3.2, -53.6, R);             // sides
  aBox(0.6, 6.4, 13.6, conc, x1 + 0.3, 3.2, -53.6, R);
  aBox(x1 - x0 + 1.4, 0.5, 14.4, conc, 0, 6.65, -53.7, R);         // roof
  /* the strip of back wall BETWEEN the two stage houses — neither house
     builds it, and without it the shed leaks out between them */
  aBox(8.6, 6.4, 0.6, conc, 0, 3.2, z1 - 0.3, R);
  ARC.walls.push({axis:'z', plane:z1 - 0.3, lo:-4.2, hi:4.2, gaps:[]});

  SHEDS.arc = {x0, x1, z0, z1, group:R, slots:[]};
  ARC.shed = SHEDS.arc;

  /* a keep fitting over EACH rear door — the ambient bed is the foyer's and
     goes out entirely inside a house (updateArc), which left this room
     pitch black.  Same pattern as the dock light: it never goes out. */
  [-26, 26].forEach(dx=>{
    const lamp = aBox(0.7, 0.24, 0.5, new T.MeshBasicMaterial({color:0xffd9a0}),
                      dx, 4.3, z1 - 0.6, R);
    aGlow(lamp, 0xffd9a0, 0.9, 'shed', true);
  });

  // racking along the rear wall, clear of both doorway lanes at x = ±26
  for(const rx of [-14, -6, 6, 14])
    shedRack(R, rx, 0, z0 + 0.7, 0, SHEDS.arc.slots);

  /* The cart hangs off ARC.group itself, NOT the shed's room group: it gets
     pushed onto a stage, and from the foyer neither house room is the shed,
     so a room-filed cart would vanish mid-shove.  ARC.group's frame is the
     same x=420 offset cartPose already removes (the room groups all sit at
     the origin inside it). */
  buildCart('arc', ARC.group, ARC.X + 0, -55);

  /* the arc's forklift and slots: four painted spots in the shared shed,
     and one in each wing of each house, just inside its rear door */
  buildLift('arc', ARC.group, ARC.X - 14, -50);
  [-31, -16, 16, 31].forEach(sx=> palletSlot('arc', ARC.X + sx, -52, ARC.group));
  buildSaw('arc', 'track', R, ARC.X + 6, -54);
  buildSaw('arc', 'chop',  R, ARC.X + 0, -51);
  buildRack('arc', R, ARC.X - 10, -59.2, 0);
  for(const key of ['main','studio']){
    const H = ARC.houses[key];
    if(!H) continue;
    palletSlot('arc', ARC.X + H.cx - 16.5, H.zBack + 2.6, ARC.group);
    palletSlot('arc', ARC.X + H.cx + 16.5, H.zBack + 2.6, ARC.group);
  }
}

/* ============================================================================
   PUSHING A CART — the cart obeys the same walls the player does, branched
   on the CART's venue (never the player's: you could in principle lean
   through a doorway).  Axis-separated by cartMoveTo, so it slides.
   ========================================================================== */
function cartBlocked(cart, nx, nz){
  const ox = cart.x, oz = cart.z;
  if(cart.venue === 'arc'){
    if(typeof arcBounds === 'function' && !arcBounds(nx, nz)) return true;
    if(typeof arcWallBlocks === 'function' && arcWallBlocks(nx, nz, ox, oz, 0.9)) return true;
  } else {
    const S = SHEDS.palace;
    if(nz < S.z0 + 0.5 || nz > FOH.z1 - 0.9) return true;
    if(nx < DOCK.x0 - 16 || nx > XL + 0.6) return true;
    /* throughWall is deliberately not consulted: the proscenium opening is
       legitimate cart territory */
    if(typeof doorBlocks === 'function' && doorBlocks(nx, nz, oz, 0.9)) return true;
    if(typeof backWallBlocks === 'function' && backWallBlocks(nx, nz, oz)) return true;
    if(typeof shedWallBlocks === 'function' && shedWallBlocks(nx, nz)) return true;
    if(typeof dockWallBlocks === 'function' && dockWallBlocks(nx, nz, ox)) return true;
    if(typeof dockDoorBlocks === 'function' && dockDoorBlocks(nx, nz, ox, 0.9)) return true;
  }
  const g = (typeof groundAt === 'function') ? groundAt(nx, nz, cart.yBase + 0.6) : 0;
  if(g === null) return true;
  if(Math.abs(g - cart.yBase) > 0.3) return true;
  return false;
}
function cartMoveTo(cart, wx, wz){
  if(!cartBlocked(cart, wx, cart.z)) cart.x = wx;
  if(!cartBlocked(cart, cart.x, wz)) cart.z = wz;
  const g = (typeof groundAt === 'function') ? groundAt(cart.x, cart.z, cart.yBase + 0.6) : 0;
  if(g !== null && Math.abs(g - cart.yBase) <= 0.3) cart.yBase = g;
  cartPose(cart);
}

/* ============================================================================
   ORDERING NEW STOCK (spec §6; grown by the build-system spec §3).  A wall
   screen in each shed (p9) writes an order; ~30s later a loaded pallet
   stands at the roller-door apron and its slots join the snap scan.
   RULING C: ordering is free.  RULING D': up to THREE orders pending per
   shed, each its own pallet, and an emptied pallet clears itself away.
   RULING E: at most 24 loose GEAR bodies to a venue — wood and hardware
   keep a separate book, BUILD_CAP (p4c), enforced here too.  Timers run on
   GAME time, never setTimeout — a timer that survives a stage swap GOes
   the wrong theatre (M7).
   ========================================================================== */
var ORDERS = {palace:{pending:[], pallets:[]}, arc:{pending:[], pallets:[]}};
const ORDER_DELIVERY = 30;      // seconds from ORDER to the pallet
const ORDER_MAX = 12;           // units to an order — one pallet's worth
const ORDER_PENDING_MAX = 3;    // RULING D': three orders out per shed at once
const STOCK_CAP = 24;           // RULING E, per venue — gear only
function venueLooseCount(v){
  if(typeof BODIES === 'undefined') return 0;
  let n = 0;
  for(const b of BODIES)
    if(b.venue === v && b.state !== 'hung' && !(BUILD_KINDS && BUILD_KINDS[b.kind])) n++;
  return n;
}
/* an order is a list of UNITS — {kind, prof?, color?}; plain strings are
   taken as gear kinds (the pre-wood callers, and half the tests) */
function orderPlace(which, units){
  const o = ORDERS[which];
  if(!o || !SHEDS[which]) return 'BUSY';
  if(o.pending.length >= ORDER_PENDING_MAX) return 'BUSY';
  units = units.map(u => typeof u === 'string' ? {kind:u} : u);
  if(!units.length || units.length > ORDER_MAX) return 'COUNT';
  let gear = 0, build = 0;
  for(const u of units){ if(BUILD_KINDS && BUILD_KINDS[u.kind]) build++; else gear++; }
  if(gear && venueLooseCount(which) + gear > STOCK_CAP) return 'STOCK FULL';
  if(build && typeof venueBuildCount === 'function' &&
     venueBuildCount(which) + build > BUILD_CAP) return 'PIECES FULL';
  o.pending.push({items:units, t:ORDER_DELIVERY});
  return 'OK';
}
/* the pallet: boards on two bearers, its seats laid out FROM the manifest —
   sheets flat-stack the middle, long stock lies in two columns, gear and
   smalls take the six seats (twice over).  An anchor carries the LIE of
   what sits on it, because slotBody zeroes the mesh against it.  Born a
   child of the shed group (it may cull with the shed while it stands on
   the apron); the forklift re-parents it by attach() — the fork group
   while riding, the venue root when set down anywhere else — so a pallet
   out on a stage can never vanish with the shed room (the cart's lesson). */
function spawnPallet(which, items){
  const S = SHEDS[which], o = ORDERS[which];
  const g = new T.Group();
  const used = o.pallets.map(p=>p.spot);
  let spot = 0; while(used.indexOf(spot) >= 0) spot++;
  const at = which === 'palace' ? {x:-2.4 - spot*2.6, z:S.z1 - 1.6}
                                : {x:-26 + spot*2.8, z:S.z1 - 1.8};
  g.position.set(at.x, 0, at.z);
  const top = new T.Mesh(new T.BoxGeometry(1.7, 0.05, 1.2), M.wood);
  top.position.y = 0.15; g.add(top);
  [-0.5, 0.5].forEach(bz=>{
    const b = new T.Mesh(new T.BoxGeometry(1.7, 0.12, 0.14), M.woodDk);
    b.position.set(0, 0.06, bz); g.add(b);
  });
  const slots = [];
  let nSheet = 0, nLong = 0, nSmall = 0;
  for(const u of (items || [])){
    const a = new T.Object3D();
    if(u.kind === 'wood' && u.prof === 'sheet'){
      a.position.set(0, 0.21 + nSheet*0.024, 0);
      a.rotation.x = -Math.PI/2;                    // thickness up: it lies flat
      nSheet++;
    } else if(u.kind === 'wood' || u.kind === 'track'){
      a.position.set(0, 0.24 + Math.floor(nLong/2)*0.1, nLong%2 ? 0.3 : -0.3);
      if(u.kind === 'wood') a.rotation.z = -Math.PI/2;   // long axis along the boards
      nLong++;
    } else {
      a.position.set(-0.55 + (nSmall%3)*0.55, 0.42 + Math.floor(nSmall/6)*0.34,
                     -0.3 + (Math.floor(nSmall/3)%2)*0.6);
      nSmall++;
    }
    g.add(a); slots.push(a);
  }
  S.group.add(g);
  return {group:g, slots, spot};
}
function deliverOrder(which, items){
  const o = ORDERS[which];
  const pal = spawnPallet(which, items);
  o.pallets.push(pal);
  /* delivered stock belongs to the shed's venue, wherever the board is */
  const keep = BUILD_VENUE;
  BUILD_VENUE = which;
  items.forEach((u, i)=>{
    let b;
    if(u.kind === 'wood') b = regWood(u.prof);
    else {
      b = regBody(u.kind, makeBodyMesh(u.kind, u.color), null);
      if(u.color !== undefined) b.color = u.color;
      b.state = 'loose';
    }
    slotBody(b, pal.slots[i]);
  });
  BUILD_VENUE = keep;
  if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
  if(typeof toast === 'function') toast('the warehouse — your order is on the apron');
}
let _orderUiT = 0;
function updateOrders(dt){
  let redraw = false;
  for(const k in ORDERS){
    const o = ORDERS[k];
    if(!o) continue;
    for(let i = o.pending.length - 1; i >= 0; i--){
      const pd = o.pending[i];
      pd.t -= dt;
      if(pd.t <= 0){
        o.pending.splice(i, 1);
        deliverOrder(k, pd.items);
        redraw = true;
      }
    }
    for(let i = o.pallets.length - 1; i >= 0; i--){
      const pal = o.pallets[i];
      const loaded = pal.slots.some(a=>a.userData.body);
      if(loaded){ pal.emptyT = undefined; continue; }
      /* RULING D: a few seconds after the last body comes off, it goes */
      if(pal.emptyT === undefined) pal.emptyT = 5;
      pal.emptyT -= dt;
      if(pal.emptyT <= 0){
        if(pal.group.parent) pal.group.parent.remove(pal.group);
        if(typeof disposeTree === 'function') disposeTree(pal.group);
        o.pallets.splice(i, 1);
        redraw = true;
      }
    }
  }
  /* keep the wall screens honest — countdowns tick on both boards' time */
  _orderUiT -= dt;
  if((redraw || _orderUiT <= 0) && typeof VR !== 'undefined' && VR && VR.orders){
    _orderUiT = 0.5;
    for(const k in VR.orders)
      if(VR.orders[k] && typeof vrDrawOrder === 'function') vrDrawOrder(VR.orders[k]);
  }
}
