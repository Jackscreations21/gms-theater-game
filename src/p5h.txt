
/* ============================================================================
   BEETLEJUICE — the fifth production, and the first taken off a recording

   Two different kinds of work wear one name here, and the line between them
   matters more than anything else in this file:

     MEASURED     cue times, fade durations, blackouts, the act break.  They
                  came off a recording of a real performance with
                  tools/video.js and they are quoted in the cue comments as
                  the times they actually are.
     INTERPRETED  every channel level, every focus, and all of this scenery.
                  We are rendering lit AREAS onto our own 25-channel rig,
                  which is not the rig they used.

   RULING AO: the scenery is the show's VOCABULARY, not its drawing.  A
   clapboard world, a cemetery under a moon, a hard portal round the whole
   picture — our proportions, our detailing, our words on any sign.  Nobody's
   ground plan or light plot went into this, which is the same thing p5c's
   header says about the other four.  A video makes tracing easy for the
   first time; that is exactly why the ruling is written down.

   The set changes are SCENES (p5c): one stage, and the sets take turns on
   it.  That machinery was built for this show, removed with it, and this is
   the first production to use it.  A scene that is off has its layers
   disabled, so it costs no draw call and no raycast while it waits.
   ========================================================================== */

const BJ = {
  opW: 13.6, opH: 9.2,        // the scenic portal's opening, INSET inside the
                              // house's own 15.0 x 10.4 proscenium — frame,
                              // cornice and trim all stay within it, because a
                              // false portal that fouls the architecture is
                              // just a mistake you cannot see from the front
  frame: 0.62,                // how heavy the portal reads
  pz: 0.45                    // it stands just upstage of the apron edge
};
const BJ_CLOTH_W = 17.2;      // the show curtain, both halves together
const BJ_SIGN_OUT = 9.0;      // how far the sign travels to clear the header:
                              // its bottom edge ends at 12.7 against a 10.4
                              // proscenium, so it is properly masked out

/* ---- the graveyard backdrop: a painted cloth, laid in the way a scenic
   artist lays one in — sky graded dark at the top, a moon with the halo it
   throws, a hill, and a silhouette along the bottom edge.  Ours, not
   theirs (RULING AO). ---- */
function bjBackdropTex(){
  const W = 1024, H = 512, c = cvs(W, H), g = c.getContext('2d');
  const sky = g.createLinearGradient(0, 0, 0, H);
  sky.addColorStop(0,    '#080d1c');
  sky.addColorStop(0.52, '#1a2a43');
  sky.addColorStop(1,    '#2b3c53');
  g.fillStyle = sky; g.fillRect(0, 0, W, H);
  /* stars, thinning towards the horizon the way they do */
  for(let i = 0; i < 300; i++){
    const y = Math.pow(Math.random(), 1.9) * H * 0.72;
    g.fillStyle = 'rgba(226,236,255,' + (0.12 + Math.random()*0.62).toFixed(2) + ')';
    g.fillRect(rnd(0, W), y, 1.4, 1.4);
  }
  /* the moon, low and to one side, and the halo on the cloth around it */
  const mx = W*0.72, my = H*0.24;
  const halo = g.createRadialGradient(mx, my, 8, mx, my, 150);
  halo.addColorStop(0, 'rgba(255,214,150,.42)');
  halo.addColorStop(1, 'rgba(255,214,150,0)');
  g.fillStyle = halo; g.beginPath(); g.arc(mx, my, 150, 0, 6.283); g.fill();
  g.fillStyle = '#ffe6bc'; g.beginPath(); g.arc(mx, my, 34, 0, 6.283); g.fill();
  /* the hill it all stands on */
  g.fillStyle = '#141b28';
  g.beginPath(); g.moveTo(0, H);
  g.lineTo(0, H*0.74);
  for(let x = 0; x <= W; x += 64)
    g.lineTo(x, H*0.74 - Math.sin(x/W*3.1)*38 - Math.sin(x/W*11)*9);
  g.lineTo(W, H); g.closePath(); g.fill();
  /* headstones and a bare tree or two, flat black against it */
  g.fillStyle = '#0a0e16';
  for(let i = 0; i < 26; i++){
    const x = rnd(10, W - 30), hgt = rnd(16, 44), y = H*0.80 + rnd(-10, 26);
    g.fillRect(x, y - hgt, rnd(10, 20), hgt);
    if(Math.random() < 0.3){ g.beginPath(); g.arc(x + 7, y - hgt, 7, Math.PI, 0); g.fill(); }
  }
  for(const tx of [W*0.14, W*0.88]){
    g.fillRect(tx, H*0.60, 9, H*0.34);
    for(let b = 0; b < 7; b++){
      g.save(); g.translate(tx + 4, H*0.60 + b*13);
      g.rotate((b % 2 ? -1 : 1) * (0.5 + b*0.06));
      g.fillRect(0, 0, rnd(26, 54), 4); g.restore();
    }
  }
  return tex(c, 1, 1);
}

/* ---- weathered vertical boarding: the portal, and later the house ---- */
function bjBoardTex(){
  const W = 512, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#4a4238'; g.fillRect(0, 0, W, H);
  for(let x = 0; x < W; x += 32){
    const w = 30 + Math.round(Math.random()*2);
    const gr = g.createLinearGradient(x, 0, x + w, 0);
    gr.addColorStop(0, 'rgba(0,0,0,.42)');
    gr.addColorStop(0.30, 'rgba(146,132,112,.20)');
    gr.addColorStop(0.70, 'rgba(96,86,72,.10)');
    gr.addColorStop(1, 'rgba(0,0,0,.42)');
    g.fillStyle = gr; g.fillRect(x, 0, w, H);
    /* the grain, running the length of each board */
    for(let i = 0; i < 14; i++){
      g.fillStyle = 'rgba(30,26,20,' + (0.06 + Math.random()*0.10).toFixed(2) + ')';
      g.fillRect(x + rnd(2, w - 3), rnd(0, H), 1, rnd(30, 190));
    }
  }
  /* nail heads down the board line, and the rust that runs from them */
  for(let y = 30; y < H; y += 124)
    for(let x = 16; x < W; x += 32){
      g.fillStyle = 'rgba(22,18,14,.7)';
      g.beginPath(); g.arc(x, y, 2, 0, 6.283); g.fill();
      if(Math.random() < 0.35){
        const gr = g.createLinearGradient(x, y, x, y + 56);
        gr.addColorStop(0, 'rgba(104,58,26,.30)'); gr.addColorStop(1, 'rgba(104,58,26,0)');
        g.fillStyle = gr; g.fillRect(x - 2.5, y, 5, 56);
      }
    }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}

/* ---- the show curtain: a hard vertical stripe, our own colours ---- */
function bjCurtainCanvas(){
  const W = 1024, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#17131f'; g.fillRect(0, 0, W, H);
  const band = W / 22;
  for(let i = 0; i < 22; i++){
    if(i % 2) continue;
    const gr = g.createLinearGradient(i*band, 0, (i+1)*band, 0);
    gr.addColorStop(0, 'rgba(214,206,190,.10)');
    gr.addColorStop(0.5, 'rgba(224,216,198,.52)');
    gr.addColorStop(1, 'rgba(214,206,190,.10)');
    g.fillStyle = gr; g.fillRect(i*band, 0, band, H);
  }
  /* a low arch of light across the cloth, so it is not a flat pattern */
  const gr2 = g.createRadialGradient(W/2, H*0.86, 40, W/2, H*0.86, W*0.52);
  gr2.addColorStop(0, 'rgba(150,190,120,.20)');
  gr2.addColorStop(1, 'rgba(150,190,120,0)');
  g.fillStyle = gr2; g.fillRect(0, 0, W, H);
  for(let i = 0; i < 200; i++){
    g.fillStyle = 'rgba(0,0,0,' + (0.04 + Math.random()*0.10).toFixed(2) + ')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(6, 30), rnd(30, 160));
  }
  const t = tex(c, 1, 1);
  return {tex: t};
}

SHOWS.beetlejuice = {
  name:  'BEETLEJUICE',
  blurb: 'A hard portal round the whole picture, a cemetery under a low moon, and more haze than air.',
  note:  'An interpretation in the show’s own vocabulary, not a copy of anyone’s design.',
  load(){
    const G = showRoot();

    const BOARD = bjBoardTex();
    const board  = new T.MeshStandardMaterial({map:BOARD, roughness:.86, metalness:.05});
    const boardD = new T.MeshStandardMaterial({color:0x2b2620, roughness:.9,  metalness:.04});
    const deckM  = new T.MeshStandardMaterial({color:0x24262b, roughness:.95});
    const stone  = new T.MeshStandardMaterial({color:0x6e7076, roughness:.88, metalness:.05});
    const iron   = new T.MeshStandardMaterial({color:0x1d2024, roughness:.55, metalness:.72});
    const bark   = new T.MeshStandardMaterial({color:0x241d18, roughness:.94, metalness:.03});
    const hillM  = new T.MeshStandardMaterial({color:0x191d26, roughness:.98});
    const hillMF = new T.MeshStandardMaterial({color:0x232936, roughness:.98});
    /* ONE emissive material shared by every trim strip and the moon's halo —
       a material per object is the draw-call trap in TRAPS.md */
    const glow   = new T.MeshStandardMaterial({color:0xbfe08a, emissive:0x9fd066,
                                               emissiveIntensity:1.5, roughness:.6});
    const moonM  = new T.MeshStandardMaterial({color:0xffd9a0, emissive:0xffb257,
                                               emissiveIntensity:1.25, roughness:.8});

    /* ---- the deck: dark, and the top stays at y = 0 ---- */
    {
      const sh = new T.Shape(), HW = D.stageW/2;
      /* shape y maps to -z after the rotate, so the back wall is +16.6 here */
      const usZ = -(D.backWall + 0.4);
      sh.moveTo(-HW + 0.4, usZ); sh.lineTo(HW - 0.4, usZ);
      sh.lineTo(HW - 0.4, -D.apron);
      sh.quadraticCurveTo(0, -(D.apron + 2*D.thrust*0.9), -HW + 0.4, -D.apron);
      sh.lineTo(-HW + 0.4, usZ);
      const geo = new T.ExtrudeGeometry(sh, {depth:0.1, bevelEnabled:false, curveSegments:36});
      geo.rotateX(-Math.PI/2);
      const m = new T.Mesh(geo, deckM);
      m.name = 'bj:deck';
      m.receiveShadow = true; G.add(m); showWalk(m);
    }

    /* ================================================================
       THE PORTAL — the one thing that never leaves, in every reference
       frame of the show.  It sits INSIDE the house proscenium, which is
       what a false portal is, so it can never foul the architecture.
       ================================================================ */
    {
      const hx = BJ.opW/2 + BJ.frame/2, hy = (BJ.opH + BJ.frame)/2;
      const legs = [
        {geo:new T.BoxGeometry(BJ.frame, BJ.opH + BJ.frame, 0.5), pos:[-hx, hy, 0]},
        {geo:new T.BoxGeometry(BJ.frame, BJ.opH + BJ.frame, 0.5), pos:[ hx, hy, 0]},
        {geo:new T.BoxGeometry(BJ.opW + 2*BJ.frame, BJ.frame, 0.5),
         pos:[0, BJ.opH + BJ.frame/2, 0]}
      ];
      const f = new T.Mesh(mergeParts(legs), board);
      f.name = 'bj:portal'; f.position.z = BJ.pz;
      f.castShadow = true; G.add(f);

      /* a stepped cornice, so the top edge is not one flat board */
      const corn = [];
      for(let i = 0; i < 3; i++)
        corn.push({geo:new T.BoxGeometry(BJ.opW + 2*BJ.frame + 0.10 - i*0.18, 0.16, 0.62 + i*0.12),
                   pos:[0, BJ.opH + BJ.frame + 0.09 + i*0.16, 0]});
      const cm = new T.Mesh(mergeParts(corn), boardD);
      cm.position.z = BJ.pz; cm.castShadow = true; G.add(cm);

      /* the cold line round the opening — three strips, ONE material */
      const tr = [
        {geo:new T.BoxGeometry(0.09, BJ.opH, 0.10), pos:[-BJ.opW/2, BJ.opH/2, 0]},
        {geo:new T.BoxGeometry(0.09, BJ.opH, 0.10), pos:[ BJ.opW/2, BJ.opH/2, 0]},
        {geo:new T.BoxGeometry(BJ.opW, 0.09, 0.10), pos:[0, BJ.opH, 0]}
      ];
      const tm = new T.Mesh(mergeParts(tr), glow);
      tm.name = 'bj:portalTrim'; tm.position.z = BJ.pz + 0.30; G.add(tm);
    }

    /* ================================================================
       SCENE — THE CEMETERY.  The show opens here; measured, the wide
       looks sit at 3:09, 6:11 and 9:31 and the first big blackout is at
       1:26 for 9.0s.
       ================================================================ */
    const cem = sceneAdd('cemetery', 'THE CEMETERY');
    {
      /* three hill layers, silhouetted against the moon.  A profile is a
         polygon, so the shape carries the read and the mesh stays flat. */
      const hill = (w, peak, wob, z, mat)=>{
        const sh = new T.Shape();
        sh.moveTo(-w, 0);
        const N = 9;
        for(let i = 0; i <= N; i++){
          const t = i/N, x = -w + 2*w*t;
          const y = peak * (0.42 + 0.58*Math.sin(Math.PI*t)) *
                    (0.78 + 0.22*Math.sin(t*wob*6.283));
          if(i === 0) sh.lineTo(x, y); else sh.quadraticCurveTo(x - w/N, y, x, y);
        }
        sh.lineTo(w, 0); sh.lineTo(-w, 0);
        const geo = new T.ExtrudeGeometry(sh, {depth:0.3, bevelEnabled:false, curveSegments:8});
        const m = new T.Mesh(geo, mat);
        m.name = "bj:hill";
        m.position.z = z; cem.group.add(m);
        return m;
      };
      hill(13.5, 4.4, 1.7, -15.4, hillMF);        // furthest, lightest
      hill(12.0, 3.2, 2.6, -13.8, hillM);
      hill(11.0, 2.1, 3.4, -12.2, hillM);         // nearest, darkest read

      /* the moon: a disc, and one faint halo on the SHARED glow material.
         Both sit upstage of every hill so the profiles read against them,
         and downstage of the deck edge at -16.6 so nothing clips the wall. */
      {
        const d = new T.Mesh(new T.CircleGeometry(1.75, 40), moonM);
        d.position.set(3.6, 5.8, -16.0); d.name = 'bj:moon'; cem.group.add(d);
        const h = new T.Mesh(new T.CircleGeometry(3.1, 36), glow);
        h.position.set(3.6, 5.8, -16.15); cem.group.add(h);
      }

      /* headstones — distinct objects, so distinct meshes, but each one is
         merged from its own slab, shoulders and base: detail without cost */
      const stones = [[-6.2,-7.2,0.10],[-5.0,-9.4,-0.14],[-3.4,-6.4,0.06],
                      [-1.5,-10.2,0.20],[1.9,-6.9,-0.09],[3.8,-9.8,0.15],
                      [5.4,-7.6,-0.18],[6.3,-10.4,0.08]];
      stones.forEach((s, i)=>{
        const h = 0.82 + (i % 3)*0.26, w = 0.58 + (i % 2)*0.12;
        const parts = [
          {geo:new T.BoxGeometry(w, h, 0.16), pos:[0, h/2 + 0.12, 0]},
          {geo:new T.CylinderGeometry(w/2, w/2, 0.16, 14, 1, false, 0, Math.PI),
           pos:[0, h + 0.12, 0], rot:[Math.PI/2, 0, 0]},
          {geo:new T.BoxGeometry(w + 0.26, 0.14, 0.30), pos:[0, 0.07, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(s[0], 0, s[1]); m.rotation.y = s[2];
        m.castShadow = true; cem.group.add(m);
      });

      /* two crosses, for the shape the slabs do not give */
      [[-6.1,-8.6,0.22],[6.1,-8.2,-0.26]].forEach(c=>{
        const parts = [
          {geo:new T.BoxGeometry(0.17, 1.55, 0.17), pos:[0, 0.9, 0]},
          {geo:new T.BoxGeometry(0.86, 0.17, 0.17), pos:[0, 1.32, 0]},
          {geo:new T.BoxGeometry(0.52, 0.13, 0.36), pos:[0, 0.065, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(c[0], 0, c[1]); m.rotation.y = c[2];
        m.castShadow = true; cem.group.add(m);
      });

      /* the railing: two runs, each merged from its own posts and rails */
      [-1, 1].forEach(sd=>{
        const parts = [];
        for(let i = 0; i < 4; i++){
          const x = sd*(3.2 + i*1.10);
          parts.push({geo:new T.BoxGeometry(0.08, 1.32, 0.08), pos:[x, 0.66, 0]});
          parts.push({geo:new T.ConeGeometry(0.07, 0.20, 6), pos:[x, 1.42, 0]});
        }
        parts.push({geo:new T.BoxGeometry(3.5, 0.07, 0.07), pos:[sd*4.95, 1.16, 0]});
        parts.push({geo:new T.BoxGeometry(3.5, 0.07, 0.07), pos:[sd*4.95, 0.34, 0]});
        const m = new T.Mesh(mergeParts(parts), iron);
        m.position.z = -5.4; cem.group.add(m);
      });

      /* the gate, shut, centre — one mesh, merged from its own bars */
      {
        const parts = [
          {geo:new T.BoxGeometry(0.11, 2.30, 0.11), pos:[-2.9, 1.15, 0]},
          {geo:new T.BoxGeometry(0.11, 2.30, 0.11), pos:[ 2.9, 1.15, 0]},
          {geo:new T.BoxGeometry(5.9, 0.09, 0.09),  pos:[0, 2.24, 0]}
        ];
        for(let i = 0; i < 9; i++)
          parts.push({geo:new T.BoxGeometry(0.06, 1.84, 0.06), pos:[-2.4 + i*0.6, 0.92, 0]});
        const m = new T.Mesh(mergeParts(parts), iron);
        m.position.z = -5.4; m.name = 'bj:gate'; cem.group.add(m);
      }

      /* one bare tree, upstage left, merged from trunk and limbs */
      {
        const parts = [{geo:new T.CylinderGeometry(0.16, 0.30, 4.6, 9), pos:[0, 2.3, 0]}];
        const limbs = [[0.6,3.4,0.5,-0.9],[-0.7,3.0,0.55,0.8],[0.35,4.1,0.42,-0.4],
                       [-0.4,4.3,0.38,0.5],[0.9,2.6,0.4,-1.2]];
        limbs.forEach(l=>parts.push({geo:new T.CylinderGeometry(0.05, 0.10, 1.5, 7),
                                     pos:[l[0], l[1], 0], rot:[0, 0, l[3]]}));
        const m = new T.Mesh(mergeParts(parts), bark);
        m.position.set(-9.6, 0, -11.2); m.castShadow = true; cem.group.add(m);
      }

      /* low mist across the boards: the existing machines, not new geometry */
      addShowSmoke({name:'CEMETERY SL', kind:'fog', x:-6.0, y:0.2, z:-7.0, yaw:1.571, tilt:0.05});
      addShowSmoke({name:'CEMETERY SR', kind:'fog', x: 6.0, y:0.2, z:-7.0, yaw:-1.571, tilt:0.05});
      addShowSmoke({name:'UPSTAGE HAZE', kind:'haze', x:0, y:4.0, z:-13.0, yaw:0, tilt:0.2});
    }

    /* ================================================================
       SCENE — THE HOUSE, from outside.  Measured, it carries 62:44 and
       72:58, either side of the act break: late act one and the top of
       act two.  Our own proportions and our own detailing (RULING AO) —
       a clapboard mass that leans, a porch you can stand on, and a
       chimney too big for it.
       ================================================================ */
    const hse = sceneAdd('house', 'THE HOUSE');
    {
      /* it has to read THROUGH the portal, so nothing may exceed the
         portal's own 13.6 x 9.2 opening */
      const MW = 6.6, MH = 5.4, MD = 4.2, cx = -1.2, cz = -8.6;

      /* the main mass, with a gable each end — one mesh */
      {
        const parts = [{geo:new T.BoxGeometry(MW, MH, MD), pos:[0, MH/2, 0]}];
        [-1, 1].forEach(sd=>{
          const g = new T.BufferGeometry();
          const h = 1.9;
          g.setAttribute('position', new T.Float32BufferAttribute([
            -MW/2, 0, 0,  MW/2, 0, 0,  0, h, 0
          ], 3));
          g.computeVertexNormals();
          parts.push({geo:g, pos:[0, MH, sd*MD/2]});
        });
        const m = new T.Mesh(mergeParts(parts), board);
        m.name = 'bj:house'; m.position.set(cx, 0, cz);
        m.castShadow = true; hse.group.add(m);
      }

      /* the roof: two slopes, leaning off true because the house does */
      {
        const parts = [];
        [-1, 1].forEach(sd=>{
          parts.push({geo:new T.BoxGeometry(MW + 0.5, 0.18, MD/2 + 0.55),
                      pos:[0, MH + 0.95, sd*(MD/4 + 0.12)],
                      rot:[sd*0.72, 0, 0]});
        });
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.position.set(cx, 0, cz); m.castShadow = true; hse.group.add(m);
      }

      /* a setback tower, which is where the lean reads from */
      {
        /* kept low on purpose: the whole house has to clear the portal's own
           9.2m opening or the portal crops its own set */
        const parts = [
          {geo:new T.BoxGeometry(2.5, 2.4, 2.5), pos:[0, 1.2, 0]},
          {geo:new T.BoxGeometry(2.8, 0.16, 2.8), pos:[0, 2.48, 0]}
        ];
        [-1, 1].forEach(sd=>parts.push({geo:new T.BoxGeometry(2.9, 0.16, 1.7),
          pos:[0, 2.95, sd*0.72], rot:[sd*0.80, 0, 0]}));
        const m = new T.Mesh(mergeParts(parts), board);
        m.position.set(cx + 1.5, MH, cz - 0.2);
        m.rotation.z = 0.045;                  // the lean, and the only one
        m.castShadow = true; hse.group.add(m);
      }

      /* the chimney, deliberately too big */
      {
        const parts = [
          {geo:new T.BoxGeometry(1.05, 3.4, 1.05), pos:[0, 1.7, 0]},
          {geo:new T.BoxGeometry(1.3, 0.24, 1.3),  pos:[0, 3.5, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(cx - 2.3, MH - 0.4, cz + 0.6);
        m.castShadow = true; hse.group.add(m);
      }

      /* the porch — and its floor is the one thing here you can stand on,
         so it goes on the SCENE's walk list, not the show's: it has to
         leave WALKABLE the moment this set goes off */
      {
        const parts = [
          {geo:new T.BoxGeometry(MW + 1.4, 0.16, 2.3), pos:[0, 0.52, 0]}
        ];
        for(let i = 0; i < 4; i++)
          parts.push({geo:new T.BoxGeometry(0.17, 2.5, 0.17),
                      pos:[-MW/2 - 0.5 + i*(MW + 1.0)/3, 1.85, 1.0]});
        parts.push({geo:new T.BoxGeometry(MW + 1.6, 0.16, 2.6), pos:[0, 3.18, 0.1]});
        parts.push({geo:new T.BoxGeometry(MW + 1.6, 0.34, 0.18), pos:[0, 3.34, 1.32]});
        const m = new T.Mesh(mergeParts(parts), board);
        m.position.set(cx, 0, cz + MD/2 + 1.0);
        m.castShadow = true; hse.group.add(m);

        /* the deck of the porch, as its own flat walkable at its own height */
        const f = new T.Mesh(new T.BoxGeometry(MW + 1.4, 0.12, 2.3), boardD);
        f.name = 'bj:porchFloor';
        f.position.set(cx, 0.60, cz + MD/2 + 1.0);
        f.receiveShadow = true; hse.group.add(f);
        /* sceneShow owns WALKABLE for scene pieces — it adds this on the way in

        {w: 5.2, h:3.5, z:-9.9,  c:'#5fd8e8'},
        {w: 3.4, h:2.3, z:-11.8, c:'#9fd066'}
      ];
      /* NOT neonTube: it runs a CatmullRom curve through the corner points,
         which overshoots a right angle badly — a 12.6m frame came out 14.5m
         wide and dipped 0.53m through the deck.  Four merged bars give a hard
         edge, and registering the mesh on SHOW.neon by hand keeps the whole
         machinery (fade off the cue, mains hum, one in five flickering). */
      rings.forEach(r=>{
        const y0 = 0.35, y1 = y0 + r.h, x = r.w/2, t = 0.15;
        const parts = [
          {geo:new T.BoxGeometry(r.w + t, t, t), pos:[0, y0, 0]},
          {geo:new T.BoxGeometry(r.w + t, t, t), pos:[0, y1, 0]},
          {geo:new T.BoxGeometry(t, r.h, t),     pos:[-x, (y0 + y1)/2, 0]},
          {geo:new T.BoxGeometry(t, r.h, t),     pos:[ x, (y0 + y1)/2, 0]}
        ];
        /* its own MeshBasicMaterial, because updateNeon writes a colour into
           every registered tube every frame */
        const m = new T.Mesh(mergeParts(parts), new T.MeshBasicMaterial({color:new T.Color(r.c)}));
        m.name = 'bj:ring'; m.position.z = r.z;
        m.raycast = ()=>{};                    // neon is scenery, not a target
        aft.group.add(m);
        SHOW.neon.push({mesh:m, base:new T.Color(r.c)});
      });

      /* the large angular structure measured at 121:54 — a skeletal chevron
         across the opening, same language as the frames but built as
         structure, so ONE shared material does for all of it */
      {
        const green = new T.MeshStandardMaterial({color:0x4f8f3a, emissive:0x2f6f22,
                                                  emissiveIntensity:0.9, roughness:.6});
        const parts = [];
        for(let i = 0; i < 7; i++){
          const t = i/6, x = -6.2 + 12.4*t;
          const y = 2.2 + Math.abs(Math.sin(t*3.1416))*2.6;
          parts.push({geo:new T.BoxGeometry(2.2, 0.22, 0.22), pos:[x, y, -13.4],
                      rot:[0, 0, (t - 0.5)*1.1]});
          if(i < 6)
            parts.push({geo:new T.BoxGeometry(0.18, 1.9, 0.18), pos:[x + 1.0, y - 0.95, -13.4]});
        }
        parts.push({geo:new T.BoxGeometry(12.8, 0.26, 0.26), pos:[0, 1.1, -13.4]});
        const m = new T.Mesh(mergeParts(parts), green);
        m.name = 'bj:chevron'; aft.group.add(m);
      }
    }

    /* ================================================================
       SCENE — THE CRYPT.  Measured 56:46, a stone structure with a peaked
       roof and a doorway somebody stands in.  Our own proportions and our
       own masonry (RULING AO).
       ================================================================ */
    const cry = sceneAdd('crypt', 'THE CRYPT');
    {
      const CW = 4.6, CH = 3.6, CD = 3.4, cz = -8.0;
      /* the mass, with the doorway left as a gap between two piers */
      {
        const parts = [
          {geo:new T.BoxGeometry(CW, CH, 0.34), pos:[0, CH/2, -CD/2]},        // back
          {geo:new T.BoxGeometry(0.34, CH, CD), pos:[-CW/2, CH/2, 0]},        // sides
          {geo:new T.BoxGeometry(0.34, CH, CD), pos:[ CW/2, CH/2, 0]},
          {geo:new T.BoxGeometry(1.35, CH, 0.34), pos:[-1.6, CH/2, CD/2]},    // front piers
          {geo:new T.BoxGeometry(1.35, CH, 0.34), pos:[ 1.6, CH/2, CD/2]},
          {geo:new T.BoxGeometry(CW, 0.9, 0.34),  pos:[0, CH - 0.45, CD/2]}   // over the door
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.name = 'bj:crypt'; m.position.set(0, 0, cz);
        m.castShadow = true; cry.group.add(m);
      }
      /* the roof: two slopes and a coping course */
      {
        const parts = [];
        [-1, 1].forEach(sd=>parts.push({geo:new T.BoxGeometry(CW + 0.6, 0.24, CD*0.78),
          pos:[sd*(CW/4 + 0.1), CH + 0.62, 0], rot:[0, 0, sd*0.62]}));
        parts.push({geo:new T.BoxGeometry(CW + 0.8, 0.22, CD + 0.5), pos:[0, CH + 0.11, 0]});
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(0, 0, cz); m.castShadow = true; cry.group.add(m);
      }
      /* three steps up to the door, and the top one is standable */
      {
        const parts = [];
        for(let i = 0; i < 3; i++)
          parts.push({geo:new T.BoxGeometry(2.9 + i*0.4, 0.2, 0.44), pos:[0, 0.1 + (2-i)*0.2, i*0.44]});
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(0, 0, cz + CD/2 + 0.3); cry.group.add(m);

        const top = new T.Mesh(new T.BoxGeometry(2.9, 0.12, 0.44), stone);
        top.name = 'bj:cryptStep';
        top.position.set(0, 0.56, cz + CD/2 + 0.3);
        cry.group.add(top); sceneWalk(cry, top);
      }
      /* two urns either side, and a low wall running off each way */
      [-1, 1].forEach(sd=>{
        const parts = [
          {geo:new T.CylinderGeometry(0.34, 0.24, 0.28, 12), pos:[0, 0.9, 0]},
          {geo:new T.CylinderGeometry(0.18, 0.30, 0.7, 12),  pos:[0, 0.45, 0]},
          {geo:new T.BoxGeometry(0.56, 0.16, 0.56),          pos:[0, 0.08, 0]}
        ];
        const u = new T.Mesh(mergeParts(parts), stone);
        u.position.set(sd*3.3, 0, cz + CD/2 + 0.4); cry.group.add(u);

        const w = [];
        /* stops short of the opening edge: a wall half behind a portal leg
           shows from the side seats */
        for(let i = 0; i < 3; i++)
          w.push({geo:new T.BoxGeometry(0.9, 0.7 - (i%2)*0.08, 0.4), pos:[sd*(4.4 + i*0.92), 0.35, 0]});
        const wm = new T.Mesh(mergeParts(w), stone);
        wm.position.z = cz - 0.4; cry.group.add(wm);
      });
    }

    /* ================================================================
       SCENE — THE SIGN.  Measured 118:04: an illuminated sign over a
       crowd.  RULING AO puts OUR OWN WORDS on it — the real production's
       wording is exactly the kind of authored detail the ruling excludes.
       ================================================================ */
    const sgn = sceneAdd('signset', 'THE SIGN');
    {
      /* the lit panel, its wording drawn into a canvas the house way */
      const c = cvs(1024, 256), g = c.getContext('2d');
      g.fillStyle = '#1b1206'; g.fillRect(0, 0, 1024, 256);
      g.fillStyle = '#ffcb4a';
      g.font = 'bold 150px sans-serif';
      g.textAlign = 'center'; g.textBaseline = 'middle';
      g.fillText('YOUR TURN', 512, 132);          // ours, not theirs
      const signTex = tex(c, 1, 1);
      const panel = new T.MeshStandardMaterial({map:signTex, emissive:0xffcb4a,
                                                emissiveIntensity:0.85, emissiveMap:signTex,
                                                roughness:.7});
      {
        const p = new T.Mesh(new T.BoxGeometry(7.2, 1.8, 0.22), panel);
        p.name = 'bj:sign'; p.position.set(0, 5.4, -7.6); sgn.group.add(p);
      }
      /* the bulb surround, one mesh on the shared glow material */
      {
        const parts = [];
        for(let i = 0; i < 22; i++){
          const t = i/21;
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[-3.8 + 7.6*t, 6.45, -7.5]});
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[-3.8 + 7.6*t, 4.35, -7.5]});
        }
        for(let i = 0; i < 5; i++){
          const y = 4.35 + i*0.525;
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[-3.8, y, -7.5]});
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[ 3.8, y, -7.5]});
        }
        sgn.group.add(new T.Mesh(mergeParts(parts), glow));
      }
      /* the frame it hangs in, and two posts to the deck */
      {
        const parts = [
          {geo:new T.BoxGeometry(8.2, 0.28, 0.4), pos:[0, 6.7, -7.7]},
          {geo:new T.BoxGeometry(8.2, 0.28, 0.4), pos:[0, 4.1, -7.7]},
          {geo:new T.BoxGeometry(0.3, 2.9, 0.4),  pos:[-4.0, 5.4, -7.7]},
          {geo:new T.BoxGeometry(0.3, 2.9, 0.4),  pos:[ 4.0, 5.4, -7.7]},
          {geo:new T.BoxGeometry(0.28, 4.1, 0.28), pos:[-4.0, 2.05, -7.7]},
          {geo:new T.BoxGeometry(0.28, 4.1, 0.28), pos:[ 4.0, 2.05, -7.7]}
        ];
        const m = new T.Mesh(mergeParts(parts), iron);
        m.name = 'bj:signFrame'; m.castShadow = true; sgn.group.add(m);
      }
      /* a low rostrum for whoever is called forward, standable */
      {
        const r = new T.Mesh(new T.BoxGeometry(4.2, 0.4, 2.2), boardD);
        r.name = 'bj:rostrum'; r.position.set(0, 0.2, -4.6);
        r.receiveShadow = true; sgn.group.add(r); sceneWalk(sgn, r);
      }
      /* two banks of seats facing it, merged, for the crowd to be in */
      [-1, 1].forEach(sd=>{
        const parts = [];
        for(let i = 0; i < 3; i++)
          for(let k = 0; k < 3; k++)
            parts.push({geo:new T.BoxGeometry(0.5, 0.5, 0.5),
                        pos:[sd*(2.4 + k*0.62), 0.25 + i*0.22, -6.4 + i*0.66]});
        const m = new T.Mesh(mergeParts(parts), boardD);
        sgn.group.add(m);
      });
    }

    /* ================================================================
       SCENE — THE BARE STAGE.  Measured 137:57: a line of the company
       across an empty stage, four minutes before the call.  A bare stage
       in a real house is not nothing — it is the masking, the ghost light
       and the boards, so that is what this is.
       ================================================================ */
    const bare = sceneAdd('bare', 'THE BARE STAGE');
    {
      /* four legs of masking, two a side, merged per side */
      [-1, 1].forEach(sd=>{
        const parts = [];
        for(let i = 0; i < 2; i++)
          parts.push({geo:new T.BoxGeometry(0.16, 8.6, 2.6),
                      pos:[sd*(6.2 - i*0.5), 4.3, -4.0 - i*4.0]});
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.name = 'bj:masking'; bare.group.add(m);
      });
      /* a border across the top of the opening */
      {
        const b = new T.Mesh(new T.BoxGeometry(13.2, 1.6, 0.16), boardD);
        b.position.set(0, 8.3, -3.4); bare.group.add(b);
      }
      /* the ghost light: a bulb on a stand, centre stage, on the shared glow */
      {
        const parts = [
          {geo:new T.CylinderGeometry(0.34, 0.40, 0.06, 14), pos:[0, 0.03, 0]},
          {geo:new T.CylinderGeometry(0.045, 0.045, 1.5, 8), pos:[0, 0.78, 0]},
          {geo:new T.CylinderGeometry(0.14, 0.10, 0.12, 10), pos:[0, 1.58, 0]}
        ];
        const s = new T.Mesh(mergeParts(parts), iron);
        s.position.set(-0.8, 0, -5.2); bare.group.add(s);
        const bulb = new T.Mesh(new T.SphereGeometry(0.17, 12, 9), glow);
        bulb.name = 'bj:ghostLight'; bulb.position.set(-0.8, 1.78, -5.2);
        bare.group.add(bulb);
      }
      /* the back wall of the theatre, seen because nothing is masking it */
      {
        const w = new T.Mesh(new T.BoxGeometry(15.4, 9.0, 0.2),
          new T.MeshStandardMaterial({color:0x22242a, roughness:.97}));
        w.position.set(0, 4.5, -13.6); bare.group.add(w);
      }
    }

    /* ================================================================
       THE SIGN — the show's own, and it hangs DOWNSTAGE of the house
       curtain so the audience has it in front of them while the curtain
       is still in.  Every lineset lives upstage of the plaster line
       (makeLineset: z = -0.50 - i*0.80), so it cannot be a good on a
       batten however much it behaves like one: it travels on the mover
       instead (RULINGS AS, AT).  In at pre-show, out at the end of the
       opening light sequence, before the curtain lifts.

       It is an ALWAYS scene — on the stage whatever set is up, and never
       swept off by sceneShow.  The wording is the production's own title
       in our lettering, which is the whole of what RULING AO allows.
       ================================================================ */
    const sgnFly = sceneAdd('bjSign', 'THE SIGN (flown)');
    {
      const SZ = 0.10;                    // downstage of the curtain at -0.50
      const c = cvs(1024, 256), g = c.getContext('2d');
      g.fillStyle = '#140a18'; g.fillRect(0, 0, 1024, 256);
      g.fillStyle = '#bfe08a';
      g.font = 'bold 128px sans-serif';
      g.textAlign = 'center'; g.textBaseline = 'middle';
      g.fillText('BEETLEJUICE', 512, 132);
      const signTex = tex(c, 1, 1);
      const panel = new T.MeshStandardMaterial({map:signTex, emissive:0xbfe08a,
                                                emissiveIntensity:0.9, emissiveMap:signTex,
                                                roughness:.7});
      const p = new T.Mesh(new T.BoxGeometry(8.6, 2.0, 0.22), panel);
      p.name = 'bj:flySign'; p.position.set(0, 5.2, SZ); sgnFly.group.add(p);

      /* the bulb surround — one mesh on the shared glow material */
      {
        const parts = [];
        for(let i = 0; i < 26; i++){
          const t = i/25;
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[-4.5 + 9.0*t, 6.42, SZ - 0.1]});
          parts.push({geo:new T.SphereGeometry(0.11, 7, 5), pos:[-4.5 + 9.0*t, 3.98, SZ - 0.1]});
        }
        sgnFly.group.add(new T.Mesh(mergeParts(parts), glow));
      }
      /* the frame, and the two lift lines that carry it */
      {
        const parts = [
          {geo:new T.BoxGeometry(9.6, 0.3, 0.42), pos:[0, 6.62, SZ - 0.14]},
          {geo:new T.BoxGeometry(9.6, 0.3, 0.42), pos:[0, 3.78, SZ - 0.14]},
          {geo:new T.BoxGeometry(0.32, 3.1, 0.42), pos:[-4.7, 5.2, SZ - 0.14]},
          {geo:new T.BoxGeometry(0.32, 3.1, 0.42), pos:[ 4.7, 5.2, SZ - 0.14]}
          /* no lift lines modelled: they would run to the grid at y≈15, and
             the beetlejuice suite's portal check rightly refuses anything
             scenic above the 10.4 opening.  Nothing above the header is ever
             seen from the house, so the wires would cost a guard rail and
             buy nothing. */
        ];
        const m = new T.Mesh(mergeParts(parts), iron);
        m.name = 'bj:flySignFrame'; m.castShadow = true; sgnFly.group.add(m);
      }
      sceneAlways(sgnFly);                       // on whatever set is up
      sceneTravel(sgnFly, 'y', 0, 2.6);          // 0 is IN; BJ_SIGN_OUT is out
    }

    /* ---- the show curtain ---- */
    {
      const c = bjCurtainCanvas();
      const key = 'bjCurtain';
      const halfW = D.procW/2 + 3.0;
      GOODS[key] = {label:'BEETLEJUICE show curtain', h:13.0, wt:420,
        traveler:true, made:true, build:()=>{
          const g = new T.Group();
          [-1, 1].forEach(sd=>{
            const t2 = c.tex.clone();
            t2.needsUpdate = true;
            t2.wrapS = t2.wrapT = T.ClampToEdgeWrapping;
            t2.repeat.set(halfW / BJ_CLOTH_W, 1);
            t2.offset.set(sd < 0 ? 0 : (BJ_CLOTH_W - halfW) / BJ_CLOTH_W, 0);
            const mat = new T.MeshStandardMaterial({map:t2, roughness:.95,
              side:T.DoubleSide, color:0xffffff});
            const d = drape(halfW, 13.0, mat, 9, .22);
            d.position.set(sd*(D.procW/4 - 0.4), -6.5, 0);
            d.userData.side = sd;
            g.add(d);
          });
          return g; }};
      TRIMS[key] = TRIMS.house;
      SHOW.goods.push(key);
      SHOW.curtainKey = key;
    }

    /* ---- the backdrop, and the sky behind act two.  Both are CLOTHS, so
       both are goods on real linesets and every "lifts" / "drops" in the
       plot is a trim in the fly snapshot the cue already carries — no new
       machinery on this half at all (RULING AT). ---- */
    {
      const bd = bjBackdropTex();
      GOODS.bjBackdrop = {label:'BEETLEJUICE backdrop', h:11.5, wt:240, made:true,
        build:()=>{
          const m = new T.MeshStandardMaterial({map:bd, roughness:1, side:T.DoubleSide});
          const g2 = new T.Group();
          const d = drape(D.procW + 6, 11.5, m, 2, .05);
          d.position.y = -5.75; d.receiveShadow = true; g2.add(d);
          return g2; }};
      TRIMS.bjBackdrop = 11.6;          // bottom edge to the deck, near enough
      /* only the backdrop goes on SHOW.goods.  That list is what the strike
         DELETES out of the catalogue (p5c: delete GOODS[k]), so a stock good
         put on it is destroyed for every show that follows — `sky` is stock
         and is already in the palette, so it is simply hung. */
      SHOW.goods.push('bjBackdrop');
      buildGoodsPalette();
    }

    /* ---- the hang ---- */
    hangGoods(FLY[0], SHOW.curtainKey);
    FLY[0].target = TRIMS[SHOW.curtainKey]; FLY[0].travTarget = 0; FLY[0].open = 0;
    hangGoods(FLY[1], 'border');    FLY[1].target = TRIMS.border;
    hangGoods(FLY[2], 'legs');      FLY[2].target = inTrimOf(FLY[2]);
    hangGoods(FLY[3], 'border');    FLY[3].target = TRIMS.border;
    hangGoods(FLY[4], 'electric');  FLY[4].target = 9.2;
    hangGoods(FLY[5], 'legs');      FLY[5].target = inTrimOf(FLY[5]);
    hangGoods(FLY[6], 'truss');     FLY[6].target = 11.4;
    hangGoods(FLY[8], 'electric');  FLY[8].target = 10.4;
    /* the graveyard's own backdrop is IN at the top of the show; the sky
       belongs to act two and waits out until then */
    hangGoods(FLY[7], 'bjBackdrop'); FLY[7].target = TRIMS.bjBackdrop;
    hangGoods(FLY[9], 'sky');        FLY[9].target = OUT_TRIM;
    FLY.forEach(ls=>{ ls.pos = ls.target; ls.group.position.y = ls.pos; });

    sceneShow('cemetery');
    RIG.haze = 0.55;                 // the beam fan is in every reference frame
    plotBeetlejuice();
    standByAtTheTop();
  }
};

/* ============================================================================
   THE PLOT — the opening sequence.

   Every TIME in these comments is measured off the recording; every LEVEL and
   every FOCUS is interpreted onto our 25 channels.  Fade durations are the
   measured ones where a slope was clean inside a single held shot.  The rest
   of the show's cues arrive with the full measured list.
   ========================================================================== */
function plotBeetlejuice(){
  const FRONT = [1,2,3,4,5,6], BOOM = [7,8], WASH = [9,10,11,12,13,14],
        BACK = [15,16,17,18], CYC = [19,20,21,22], MOVE = [23,24], SPOT = 25;
  const all = v => FIXTURES.forEach(f=>{ f.level = v; });
  const set = (list, v, col)=>list.forEach(n=>{
    const f = chan(n); if(!f) return; f.level = v; if(col) f.color.set(col); });
  const one = (n, v, col)=>set([n], v, col);

  /* the front of house, thrown UP onto the cloth for the curtain cues */
  const onTheCloth = ()=>aimFrontAt(FRONT, 0, 7.4, -1.0);

  const looks = [
    /* ---- the top of the night ---- */
    {at:33, n:0.5, label:'preset — house open, curtain in', fade:0, house:0.9, haze:0.15,
     curtain:'shut', scene:'cemetery',
     look(){ all(0); onTheCloth(); set(FRONT, 0.5, '#e8d2a8'); }},

    {at:70, n:1, label:'house to half', fade:5, house:0.5, haze:0.2,
     curtain:'shut', scene:'cemetery',
     look(){ all(0); onTheCloth(); set(FRONT, 0.5, '#e8d2a8'); }},

    /* ============ ACT ONE, i — THE CEMETERY ============
       measured: the first big blackout is at 1:26 and holds 9.0s; the wide
       cemetery look is held from 3:09 for 125s, the longest single held shot
       in the act. */
    {at:95, n:2, label:'act one — the cemetery, low and blue', fade:6, house:0, haze:0.6,
     curtain:'out', scene:'cemetery',
     look(){ all(0);
       set(CYC, 0.55, '#2f5f9c'); set(BACK, 0.42, '#4a78b4');
       set(WASH, 0.18, '#5f7fa8'); set(BOOM, 0.30, '#3f6ea6'); }},

    {at:189, n:3, label:'the moon takes the upstage', fade:4, house:0, haze:0.62,
     curtain:'out', scene:'cemetery',
     look(){ all(0);
       set(CYC, 0.68, '#2b5896'); set(BACK, 0.30, '#6f8fc0');
       set(BOOM, 0.22, '#3a689e'); }},

    /* measured: a fade UP of 1.6s — one of the fastest clean slopes in the file */
    {at:371, n:4, label:'the mourners — a hard front, fast', fade:1.6, house:0, haze:0.55,
     curtain:'out', scene:'cemetery',
     look(){ all(0);
       aimFrontAt(FRONT, 0, 1.7, -6.5);
       set(FRONT, 0.62, '#cfd8e8'); set(CYC, 0.34, '#2b5896');
       set(WASH, 0.20, '#7f92ad'); }},

    {at:480, n:5, label:'down to the one of them', fade:3, house:0, haze:0.5,
     curtain:'out', scene:'cemetery',
     look(){ all(0);
       one(SPOT, 0.85, '#f2ecd8'); set(CYC, 0.18, '#22467c');
       set(BOOM, 0.12, '#33608f'); }},

    /* measured: 9:31 is the widest, brightest look of the cemetery sequence —
       centre/edge ratio 4.31, the strongest wide shot in the whole file */
    {at:571, n:6, label:'the beam fan opens over the gate', fade:5, house:0, haze:0.75,
     curtain:'out', scene:'cemetery',
     look(){ all(0);
       set(BACK, 0.72, '#4f86c8'); set(MOVE, 0.60, '#78c8e8');
       set(CYC, 0.30, '#25497e'); one(SPOT, 0.40, '#f2ecd8'); }},

    /* INTERPRETED, and worth being explicit about: the interior first appears
       at 15:57, but there is NO measured blackout anywhere between 9:00 and
       18:00. The recording does not show how that change was covered, so
       covering it with a blackout of our own is a DECISION, not a measurement. */
    {at:920, n:7, label:'blackout — the walls come in', fade:1.2, house:0, haze:0.5,
     curtain:'out', scene:'cemetery',
     look(){ all(0); }},

    /* ============ ACT ONE, ii — INSIDE THE HOUSE ============
       measured: 15:57 to 31:34 under five looks, which is why the wall curves */
    {at:957, n:8, label:'inside — the walls, cold', fade:3, house:0, haze:0.35,
     curtain:'out', scene:'interior',
     look(){ all(0);
