function fixturesOnLineset(i){ return FIXTURES.filter(f=>f.ls===i); }
function buildElectric(ls){
  const idx = ls.id-1;
  fixturesOnLineset(idx).forEach(f=>{ f.group.visible = true; f.hung = true; });
  // cable drape + junction box
  const g = ls.goods || new T.Group();
  const jb = new T.Mesh(new T.BoxGeometry(.42,.34,.24), M.fixture);
  jb.position.set(D.procW/2+1.2,-.2,0); g.add(jb);
  const cable = new T.Mesh(new T.TorusGeometry(.5,.035,5,16,Math.PI*1.4), M.rope);
  cable.position.set(D.procW/2+1.8,-.6,0); cable.rotation.y=Math.PI/2; g.add(cable);
  if(!ls.goods){ ls.goods=g; ls.group.add(g); }
}
function stripElectric(ls){
  fixturesOnLineset(ls.id-1).forEach(f=>{ f.group.visible = false; f.hung = false; });
}

function placeScenic(key, x, z){
  const spec = SCENIC[key]; if(!spec) return null;
  const g = spec.build();
  g.traverse(o=>{ if(o.isMesh){ o.castShadow=true; o.receiveShadow=true; } });
  g.position.set(x, 0, z);
  setGroup.add(g);
  const p = {key, group:g, label:spec.label, rot:0};
  SET.push(p); stagePieces.push(g); g.userData.piece = p;
  refreshSetUI();
  return p;
}
function strikePiece(p){
  if(typeof clearMachinesFor === 'function') clearMachinesFor(p.group);
  const i = SET.indexOf(p); if(i<0) return;
  SET.splice(i,1); setGroup.remove(p.group); disposeTree(p.group);
  const j = stagePieces.indexOf(p.group); if(j>=0) stagePieces.splice(j,1);
  if(selPiece===p) selPiece=null;
  refreshSetUI();
}
function strikeAll(){ SET.slice().forEach(strikePiece); }
/* strike only what stands on the LIVE stage's deck.  SET is one world-space
   store for all three stages, and a show load must not clear scenery parked
   on another building's deck.  ±24 m around the stage origin is well past
   the stage box and nowhere near the next house (they sit 52 m apart). */
function strikeStageSet(){
  const o = (typeof stageOrigin === 'function') ? stageOrigin() : {x:0, z:0};
  SET.slice().forEach(p=>{
    const g = p.group.position;
    if(Math.abs(g.x - o.x) <= 24 && Math.abs(g.z - o.z) <= 24) strikePiece(p);
  });
}

/* ============================================================================
   SOUND — everything synthesised in WebAudio, no assets
   ========================================================================== */
const Snd = (()=>{
  let ac=null, master=null, bus={}, started=false;
  const loops = {};
  const BUSSES = ['fx','music','amb','rail'];
  const level = {master:.8, fx:.85, music:.6, amb:.5, rail:.6};

  function init(){
    if(started) return; started=true;
    try{
      const AC = window.AudioContext||window.webkitAudioContext;
      if(!AC) return;
      ac = new AC();
      master = ac.createGain(); master.gain.value = level.master; master.connect(ac.destination);
      BUSSES.forEach(b=>{ const g=ac.createGain(); g.gain.value=level[b]; g.connect(master); bus[b]=g; });
    }catch(e){ ac=null; console.warn('audio unavailable', e); }
    if(ac && ac.state==='suspended') ac.resume().catch(()=>{});
  }
  function noiseBuf(sec){
    const n = ac.sampleRate*sec, b = ac.createBuffer(1,n,ac.sampleRate), d=b.getChannelData(0);
    for(let i=0;i<n;i++) d[i]=Math.random()*2-1;
    return b;
  }
  function env(node, b, a, d, s, r, dur){
    const t=ac.currentTime; const g=ac.createGain();
    g.gain.setValueAtTime(0,t);
    g.gain.linearRampToValueAtTime(1,t+a);
    g.gain.linearRampToValueAtTime(s,t+a+d);
    g.gain.setValueAtTime(s,t+dur);
    g.gain.linearRampToValueAtTime(0,t+dur+r);
    node.connect(g); g.connect(bus[b]);
    return g;
  }
  function tone(freq, b, dur, type, gain, detune){
    const o=ac.createOscillator(); o.type=type||'sine'; o.frequency.value=freq;
    if(detune) o.detune.value=detune;
    const g=env(o,b||'fx',.01,.08,gain===undefined?.3:gain,.25,dur);
    o.start(); o.stop(ac.currentTime+dur+.4); return o;
  }
  function noiseHit(b, dur, f0, f1, q, gain){
    const s=ac.createBufferSource(); s.buffer=noiseBuf(Math.max(dur,.3));
    const fl=ac.createBiquadFilter(); fl.type='bandpass'; fl.Q.value=q||1;
    fl.frequency.setValueAtTime(f0,ac.currentTime);
    fl.frequency.exponentialRampToValueAtTime(Math.max(40,f1),ac.currentTime+dur);
    s.connect(fl);
    const g=env(fl,b,.005,.05,gain===undefined?.5:gain,.3,dur);
    s.start(); s.stop(ac.currentTime+dur+.5); return s;
  }
  /* --- one shots --- */
  const SFX = {
    applause: ()=>{ const s=ac.createBufferSource(); s.buffer=noiseBuf(6); s.loop=false;
      const f=ac.createBiquadFilter(); f.type='bandpass'; f.frequency.value=1900; f.Q.value=.7;
      const g=ac.createGain(); g.gain.setValueAtTime(0,ac.currentTime);
      g.gain.linearRampToValueAtTime(.55,ac.currentTime+.35);
      g.gain.setValueAtTime(.5,ac.currentTime+3.4);
      g.gain.linearRampToValueAtTime(0,ac.currentTime+6);
      s.connect(f); f.connect(g); g.connect(bus.fx); s.start();
      // clap transients
      for(let i=0;i<120;i++) setTimeout(()=>{ if(ac) noiseHit('fx',.05,rnd(1400,4200),700,3,.16); }, rnd(0,4200));
    },
    ovation: ()=>{ SFX.applause(); for(let i=0;i<8;i++) setTimeout(()=>{
      tone(rnd(300,700),'fx',.4,'sawtooth',.06); }, rnd(200,3000)); },
    thunder: ()=>{ noiseHit('fx',2.6,120,32,.6,.9);
      setTimeout(()=>noiseHit('fx',3.4,70,25,.5,.7),260); },
    doorSlam: ()=>{ noiseHit('fx',.22,320,60,1.4,.9); tone(72,'fx',.28,'sine',.5); },
    glass: ()=>{ for(let i=0;i<14;i++) setTimeout(()=>tone(rnd(1800,5200),'fx',.14,'triangle',.14),rnd(0,220)); },
    gunshot: ()=>{ noiseHit('fx',.14,2600,180,.8,1); tone(60,'fx',.2,'square',.5); },
    bell: ()=>{ [523.25,659.25,783.99].forEach((f,i)=>{ tone(f,'fx',2.6,'sine',.2 - i*.05); tone(f*2,'fx',1.6,'sine',.06); }); },
    clock: ()=>{ for(let i=0;i<12;i++) setTimeout(()=>{ tone(180,'fx',1.6,'sine',.28); tone(361,'fx',1.2,'sine',.12); }, i*1400); },
    phone: ()=>{ for(let i=0;i<6;i++) setTimeout(()=>{ tone(440,'fx',.4,'sine',.22); tone(480,'fx',.4,'sine',.22); }, i*900); },
    crash: ()=>{ noiseHit('fx',1.2,5200,600,.5,.8); tone(90,'fx',.6,'square',.35); },
    whoosh: ()=>{ noiseHit('fx',.9,180,2800,1.2,.5); },
    sting: ()=>{ [0,4,7,11].forEach((s,i)=>setTimeout(()=>tone(220*Math.pow(2,s/12),'music',1.6,'sawtooth',.16),i*40)); },
    fanfare: ()=>{ const n=[392,392,392,523,659,523,659,784];
      n.forEach((f,i)=>setTimeout(()=>{ tone(f,'music',.42,'square',.13); tone(f/2,'music',.42,'triangle',.1); },i*230)); },
    chord: ()=>{ [130.8,164.8,196,261.6].forEach(f=>tone(f,'music',4.5,'sawtooth',.09)); },
    minorChord: ()=>{ [130.8,155.6,196,261.6].forEach(f=>tone(f,'music',4.5,'sawtooth',.09)); },
    heartbeat: ()=>{ for(let i=0;i<10;i++){ setTimeout(()=>{ tone(52,'fx',.16,'sine',.6);
      setTimeout(()=>tone(46,'fx',.2,'sine',.45),170); }, i*1100); } },
    footsteps: ()=>{ for(let i=0;i<8;i++) setTimeout(()=>noiseHit('fx',.09,420,140,2,.35), i*430); },
    wind: ()=>toggleLoop('wind'), rain: ()=>toggleLoop('rain'),
    crowd: ()=>toggleLoop('crowd'), sea: ()=>toggleLoop('sea'),
    hum: ()=>toggleLoop('hum'), preshow: ()=>toggleLoop('preshow')
  };
  function makeLoop(name){
    if(name==='wind'){
      const s=ac.createBufferSource(); s.buffer=noiseBuf(4); s.loop=true;
      const f=ac.createBiquadFilter(); f.type='lowpass'; f.frequency.value=420; f.Q.value=3;
      const g=ac.createGain(); g.gain.value=.34; s.connect(f); f.connect(g); g.connect(bus.amb); s.start();
      const lfo=setInterval(()=>{ if(!ac)return; f.frequency.linearRampToValueAtTime(rnd(180,760),ac.currentTime+2.2);
        g.gain.linearRampToValueAtTime(rnd(.16,.42),ac.currentTime+2.2); },2200);
      return {stop:()=>{s.stop();clearInterval(lfo);}};
    }
    if(name==='rain'){
      const s=ac.createBufferSource(); s.buffer=noiseBuf(4); s.loop=true;
      const f=ac.createBiquadFilter(); f.type='highpass'; f.frequency.value=1400;
      const g=ac.createGain(); g.gain.value=.28; s.connect(f); f.connect(g); g.connect(bus.amb); s.start();
      return {stop:()=>s.stop()};
    }
    if(name==='sea'){
      const s=ac.createBufferSource(); s.buffer=noiseBuf(5); s.loop=true;
      const f=ac.createBiquadFilter(); f.type='lowpass'; f.frequency.value=700;
      const g=ac.createGain(); g.gain.value=.2; s.connect(f); f.connect(g); g.connect(bus.amb); s.start();
      const lfo=setInterval(()=>{ if(!ac)return; g.gain.linearRampToValueAtTime(rnd(.08,.36),ac.currentTime+3); },3000);
      return {stop:()=>{s.stop();clearInterval(lfo);}};
    }
    if(name==='crowd'){
      const s=ac.createBufferSource(); s.buffer=noiseBuf(5); s.loop=true;
      const f=ac.createBiquadFilter(); f.type='bandpass'; f.frequency.value=620; f.Q.value=.9;
      const g=ac.createGain(); g.gain.value=.22; s.connect(f); f.connect(g); g.connect(bus.amb); s.start();
      const iv=setInterval(()=>{ if(!ac)return; if(Math.random()>.75) noiseHit('amb',.3,rnd(700,2000),400,2,.1); },900);
      return {stop:()=>{s.stop();clearInterval(iv);}};
    }
    if(name==='hum'){
      const o=ac.createOscillator(); o.type='sine'; o.frequency.value=52;
      const o2=ac.createOscillator(); o2.type='sine'; o2.frequency.value=104.5;
      const g=ac.createGain(); g.gain.value=.055; o.connect(g); o2.connect(g); g.connect(bus.amb);
      o.start(); o2.start(); return {stop:()=>{o.stop();o2.stop();}};
    }
    if(name==='preshow'){
      const g=ac.createGain(); g.gain.value=.12; g.connect(bus.music);
      const scale=[261.6,293.7,329.6,392,440,523.3,587.3,659.3];
      const iv=setInterval(()=>{ if(!ac)return;
        const f=scale[Math.floor(Math.random()*scale.length)];
        const o=ac.createOscillator(); o.type='triangle'; o.frequency.value=f;
        const e=ac.createGain(); e.gain.setValueAtTime(0,ac.currentTime);
        e.gain.linearRampToValueAtTime(.5,ac.currentTime+.4);
        e.gain.linearRampToValueAtTime(0,ac.currentTime+2.4);
        o.connect(e); e.connect(g); o.start(); o.stop(ac.currentTime+2.6);
      },700);
      return {stop:()=>clearInterval(iv)};
    }
    return {stop:()=>{}};
  }
  function toggleLoop(name, force){
    init(); if(!ac) return false;
    const on = loops[name]!==undefined;
    if(force===true && on) return true;
    if(force===false && !on) return false;
    if(on){ loops[name].stop(); delete loops[name]; return false; }
    loops[name] = makeLoop(name); return true;
  }
  /* rail motor noise while a lineset travels */
  const railNodes = {};
  function railStart(ls){
    if(!ac || railNodes[ls.id]) return;
    const s=ac.createBufferSource(); s.buffer=noiseBuf(2); s.loop=true;
    const f=ac.createBiquadFilter(); f.type='bandpass'; f.frequency.value=380+ls.weight*0.4; f.Q.value=6;
    const g=ac.createGain(); g.gain.value=0;
    g.gain.linearRampToValueAtTime(.09+Math.min(.14,ls.weight/2600),ac.currentTime+.3);
    s.connect(f); f.connect(g); g.connect(bus.rail); s.start();
    railNodes[ls.id]={s,g};
  }
  function railStop(ls){
    const n = railNodes[ls.id]; if(!n) return;
    n.g.gain.linearRampToValueAtTime(0,ac.currentTime+.35);
    setTimeout(()=>{ try{n.s.stop();}catch(e){} },500);
    delete railNodes[ls.id];
    if(ac) noiseHit('rail',.1,220,90,2,.16);   // batten settling
  }
  /* every motor at once, without the settling thud — for the stage swap,
     where a parked lineset never "arrives" and its id collides with the
     next stage's lineset of the same number */
  function railStopAll(){
    for(const id of Object.keys(railNodes)){
      const n = railNodes[id];
      n.g.gain.linearRampToValueAtTime(0,ac.currentTime+.35);
      setTimeout(()=>{ try{n.s.stop();}catch(e){} },500);
      delete railNodes[id];
    }
  }
  function play(name){ init(); if(ac && SFX[name]) try{ SFX[name](); }catch(e){} }
  function setLevel(b,v){ level[b]=v; init(); if(!ac) return;
    if(b==='master') master.gain.value=v; else if(bus[b]) bus[b].gain.value=v; }
  function isLooping(n){ return loops[n]!==undefined; }
  return {init, play, setLevel, level, railStart, railStop, railStopAll, toggleLoop, isLooping,
          names:()=>Object.keys(SFX), busses:()=>BUSSES};
