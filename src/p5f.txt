   HAMILTON
   Brick and timber: a two-storey structure across the back with a gallery you
   can walk, staircases either side, rope and rigging overhead, and two
   concentric revolves cut into the deck that actually turn.

   An interpretation in the show's vocabulary.  Nobody's drawings were used.
   ========================================================================== */

/* ---- the revolves ---- */
function setRevolve(which, v){
  if(!SHOW.revolve) return 0;
  const r = SHOW.revolve[which];
  if(!r) return 0;
  r.speed = clamp(v, -1, 1);
  return r.speed;
}
function updateRevolve(dt){
  const R = SHOW.revolve;
  if(!R) return;
  for(const k of ['outer','inner']){
    const r = R[k]; if(!r) continue;
    r.rate += clamp(r.speed - r.rate, -dt*0.9, dt*0.9);   // they take a moment to get going
    r.angle += r.rate * dt * 0.42;
    r.group.rotation.y = r.angle;
  }
}

/* ---- old brick, laid by hand ---- */
function hamiltonBrickTex(){
  const W = 512, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#3a2a22'; g.fillRect(0,0,W,H);
  const bw = 64, bh = 26;
  for(let row = 0; row*bh < H; row++){
    const off = (row % 2) * bw/2;
    for(let x = -bw; x < W; x += bw){
      const r = 96 + Math.random()*46, gg = 52 + Math.random()*26, b = 38 + Math.random()*20;
      g.fillStyle = 'rgb('+(r|0)+','+(gg|0)+','+(b|0)+')';
      g.fillRect(x + off + 2, row*bh + 2, bw - 4, bh - 4);
      if(Math.random() < 0.22){
        g.fillStyle = 'rgba(0,0,0,'+(0.06+Math.random()*0.16).toFixed(2)+')';
        g.fillRect(x + off + 2, row*bh + 2, bw - 4, bh - 4);
      }
    }
  }
  // mortar shadow and general soot
  g.fillStyle = 'rgba(20,14,10,.28)';
  for(let row = 0; row*bh < H; row++) g.fillRect(0, row*bh, W, 2);
  for(let i=0;i<300;i++){
    g.fillStyle = 'rgba(0,0,0,'+(0.03+Math.random()*0.10).toFixed(2)+')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(6,60), rnd(10,90));
  }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}
/* ---- the show cloth: parchment and gold, with the title ---- */
function hamiltonCurtainCanvas(){
  const W = 2048, H = 1024, c = cvs(W, H), g = c.getContext('2d');
  const base = g.createLinearGradient(0, 0, 0, H);
  base.addColorStop(0.00, '#1a1006');
  base.addColorStop(0.30, '#4a2f10');
  base.addColorStop(0.55, '#6d4715');
  base.addColorStop(0.80, '#3d260d');
  base.addColorStop(1.00, '#120b04');
  g.fillStyle = base; g.fillRect(0, 0, W, H);
  // pleats
  for(let x=0; x<W; x+=56){
    const gr = g.createLinearGradient(x,0,x+56,0);
    gr.addColorStop(0,'rgba(0,0,0,.62)');
    gr.addColorStop(0.5,'rgba(255,206,130,.16)');
    gr.addColorStop(1,'rgba(0,0,0,.62)');
    g.fillStyle = gr; g.fillRect(x,0,56,H);
  }
  const cx = W/2, cy = H*0.47;
  g.textAlign = 'center'; g.textBaseline = 'middle';
  g.font = '700 190px Georgia, "Times New Roman", serif';
  g.fillStyle = 'rgba(246,226,176,.95)';
  g.fillText('HAMILTON', cx, cy);
  g.lineWidth = 4;
  g.strokeStyle = 'rgba(120,80,20,.8)';
  g.strokeText('HAMILTON', cx, cy);
  g.font = '600 54px Georgia, serif';
  g.fillStyle = 'rgba(214,186,138,.8)';
  g.fillText('A N   A M E R I C A N   M U S I C A L', cx, cy + 150);
  // a rule above and below
  g.fillStyle = 'rgba(214,178,110,.55)';
  g.fillRect(W*0.24, cy - 150, W*0.52, 4);
  g.fillRect(W*0.24, cy + 200, W*0.52, 4);
  for(let i=0;i<700;i++){
    g.fillStyle = 'rgba(0,0,0,'+(0.03+Math.random()*0.10).toFixed(2)+')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(2,24), rnd(10,80));
  }
  const vig = g.createLinearGradient(0,0,0,H);
  vig.addColorStop(0,'rgba(0,0,0,.78)'); vig.addColorStop(0.4,'rgba(0,0,0,.06)');
  vig.addColorStop(1,'rgba(0,0,0,.74)');
  g.fillStyle = vig; g.fillRect(0,0,W,H);
  return {canvas:c, tex:tex(c,1,1)};
}
