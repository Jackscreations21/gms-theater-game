
/* ============================================================================
   THREE STAGES, ONE BOARD

   There is one lighting console, one fly rail and one cue stack in this game,
   and they belong to whichever theatre you are standing in.  Walk from the
   Palace's stalls to the Arc's Main House and the board in front of you is
   the Main House's board: its channels, its linesets, its cue stack, its
   house lights, its production.

   That is done by SWAPPING THE CONTENTS of the globals rather than by
   handing every function a stage argument.  FIXTURES, FLY, CUES, SHOW and
   HOUSE stay exactly where they were, and every one of the two hundred
   functions that reads them carries on working — they simply describe a
   different room now.  The alternative was threading a stage through the
   whole codebase, which would have been a fortnight of work and a great
   many places to get it wrong.

   The Arc's two stages are the same box as the Palace's, so their rigs and
   their fly systems are built by the SAME code, just translated: one group
   moved to the house's centre line and its proscenium, and everything inside
   it lands where it should.
   ========================================================================== */
const STAGES = {};
let STAGE = 'palace';

function makeStage(o){
  const st = {
    key:o.key, label:o.label, venue:o.venue, room:o.room,
    cx:o.cx || 0, zPros:o.zPros || 0,
    rigGroup:o.rigGroup, poolGroup:o.poolGroup, flyGroup:o.flyGroup,
    showParent:o.showParent,
    fixtures:[], fly:[], cues:[], nextCue:0, selCue:-1,
    show:null, house:null, rig:null, smokeRack:null, fohBar:null, spkBars:null, live:false
  };
  STAGES[o.key] = st;
  return st;
}

/* ---- what a stage owns, lifted out of the globals and put back again ---- */
function stageCapture(st){
  /* wall-clock WebAudio does not park.  Stop every rail motor (a parked
     lineset never arrives, so its hum would follow you out — and its id
     collides with the next stage's lineset of the same number) and the rain
     rumble, and clear the flags so both re-arm when this stage is live. */
  if(typeof Snd !== 'undefined' && Snd.railStopAll) Snd.railStopAll();
  FLY.forEach(ls=>{ ls.moving = false; });
  if(typeof Snd !== 'undefined' && Snd.toggleLoop && SHOW.rainSound){
    Snd.toggleLoop('rain', false); SHOW.rainSound = false;
  }
  st.fixtures = FIXTURES.splice(0, FIXTURES.length);
  st.fly      = FLY.splice(0, FLY.length);
  st.cues     = CUES.splice(0, CUES.length);
  st.nextCue  = nextCue;
  st.selCue   = (typeof selCue !== 'undefined') ? selCue : -1;
  st.show     = Object.assign({}, SHOW);
  st.house    = Object.assign({}, HOUSE);
  st.rig      = {haze:RIG.haze, grand:RIG.grand, blackout:RIG.blackout};
  st.smokeRack = (typeof smokeCapture === 'function') ? smokeCapture() : null;
  /* the FOH bar is one object per stage, meshes and state together — its
     meshes are already parked inside this stage's rigGroup, so capturing is
     just remembering which bar the board was driving */
  st.fohBar   = FOHBAR;
  st.spkBars  = SPKBARS;
  st.live     = false;
}
function stageRestore(st){
  FIXTURES.push.apply(FIXTURES, st.fixtures); st.fixtures.length = 0;
  FLY.push.apply(FLY, st.fly);                st.fly.length = 0;
  CUES.push.apply(CUES, st.cues);             st.cues.length = 0;
  nextCue = st.nextCue;
  if(typeof selCue !== 'undefined') selCue = (st.selCue === undefined) ? -1 : st.selCue;
  /* replace, don't merge: Object.assign can overwrite but never remove, so
     the other stage's ad-hoc keys (wall, dropKey, …) survived onto the live
     SHOW.  Wipe first, then lay the parked copy down whole. */
  if(st.show){
    for(const k of Object.keys(SHOW)) delete SHOW[k];
    Object.assign(SHOW, st.show);
  }
  if(st.house) Object.assign(HOUSE, st.house);
  if(st.rig){ RIG.haze = st.rig.haze; RIG.grand = st.rig.grand; RIG.blackout = st.rig.blackout; }
  /* a stage that has never had the board builds its house smoke kit here */
  if(typeof smokeRestore === 'function'){ smokeRestore(st.smokeRack); st.smokeRack = null; }
  rigGroup  = st.rigGroup;
  poolGroup = st.poolGroup;
  flyGroup  = st.flyGroup;
  if(st.fohBar) FOHBAR = st.fohBar;
  if(st.spkBars) SPKBARS = st.spkBars;
  st.live = true;
}
function stageOf(key){ return STAGES[key] || null; }
function stageLive(){ return STAGES[STAGE]; }
/* the live stage's deck slab — what the scenic palette and FOCUS raycast.
   Raycasting the Palace's `deck` from the Arc found it INVISIBLY through
   every wall (§5: visible is only a drawing flag) and aimed FOH across town */
function stageDeck(){
  const st = STAGES[STAGE];
  if(st && st.venue === 'arc' && typeof ARC !== 'undefined'){
    const H = ARC.houses[st.arcHouse];
    if(H && H.deck) return H.deck;
  }
  return (typeof deck !== 'undefined') ? deck : null;
}
/* where the live stage's origin sits in the world — loose scenery and drag
   clamps are worked in world space, so the stage box has to come to them */
function stageOrigin(){
  const st = STAGES[STAGE];
  if(st && st.venue === 'arc' && typeof ARC !== 'undefined')
    return {x: ARC.X + st.cx, z: st.zPros};
  return {x: 0, z: 0};
}
/* the house-light level of any stage, live or not */
function stageHouseLevel(key){
  if(key === STAGE) return HOUSE.house;
  const st = STAGES[key];
  return st && st.house ? st.house.house : 0;
}

function stageSwitch(key, quiet){
  if(key === STAGE || !STAGES[key]) return STAGES[STAGE];
  /* the code console drives the LIVE board — carried across a swap, the rest
     of a running script would fire cues and fly goods in the other theatre.
     A show you walked out on halts; it does not haunt the next room. */
  if(typeof Prog !== 'undefined' && Prog.running && typeof haltProgram === 'function'){
    haltProgram();
    Prog.log('■ halted — the board patched to ' + STAGES[key].label);
  }
  /* same rule for the one-shot cue follow: cancelled, not carried */
  if(typeof cancelFollow === 'function') cancelFollow();
  /* and for the sound, which is the cue clock while it runs (BB).  TRAPS lists
     audio among the things that have driven the wrong stage's rig after a
     walk: a show you walked out on goes quiet, it does not play on. */
  if(typeof showSoundStop === 'function') showSoundStop();
  /* the crew work the stage they were called on.  Rather than let a live job
     queue mix this stage's plan with the next stage's rail and doors (H2) or
     restore the work-light snapshot onto the wrong board when they finish
     (H3), they stand down at the boundary — and the show look comes home
     first, while their own stage is still the live one. */
  if(typeof CREW !== 'undefined' && CREW.running && typeof crewStop === 'function'){
    const was = CREW.running;
    crewStop(true);
    CREW.onDone = null;
    if(CREW.savedLook && typeof crewWorkLight === 'function') crewWorkLight(false);
    if(typeof crewLog === 'function')
      crewLog('stood down — the board patched away mid ' +
              (was === 'in' ? 'load in' : 'load out'));
  }
  stageCapture(STAGES[STAGE]);
  STAGE = key;
  stageRestore(STAGES[key]);
  /* the console, the rail, the cue stack and the show card are all looking
     at different objects now — every one of them has to be redrawn */
  if(typeof refreshChannelStrip === 'function') refreshChannelStrip();
  if(typeof refreshFlyUI === 'function') refreshFlyUI();
  if(typeof refreshCues === 'function') refreshCues();
  if(typeof refreshShowUI === 'function') refreshShowUI();
  if(typeof refreshHangUI === 'function') refreshHangUI();
  if(typeof refreshWrongUI === 'function') refreshWrongUI();
  if(typeof buildSmokeUI === 'function') buildSmokeUI();
  if(typeof syncSmokeUI === 'function') syncSmokeUI();
  if(typeof syncMasters === 'function') syncMasters();
  if(typeof refreshVenueUI === 'function') refreshVenueUI();
  if(typeof refreshStageChip === 'function') refreshStageChip();
  if(typeof refreshStageNames === 'function') refreshStageNames();
  if(typeof buildGoodsPalette === 'function') buildGoodsPalette();
  if(typeof vrOnStageChange === 'function') vrOnStageChange();
  if(!quiet && typeof toast === 'function')
    toast('The board is patched to ' + STAGES[key].label);
  return STAGES[key];
}

/* ---- which stage are you standing in ---- */
function stageAt(x, z){
  if(typeof ARC === 'undefined' || !ARC.built) return 'palace';
  if(venueAt(x) !== 'arc') return 'palace';
  const room = arcRoomAt(x, z);
  if(room === 'main')   return 'arcMain';
  if(room === 'studio') return 'arcStudio';
  return null;              // the foyer belongs to neither: keep what you had
}
/* called every frame; the board follows you about the building */
function updateStageFromPosition(){
  if(typeof Player === 'undefined') return STAGE;
  const x = (Player.mode === 'walk') ? Player.pos.x : camera.position.x;
  const z = (Player.mode === 'walk') ? Player.pos.z : camera.position.z;
  const want = stageAt(x, z);
  if(want && want !== STAGE) stageSwitch(want);
  return STAGE;
}

/* ============================================================================
   BUILDING A STAGE
   The Palace was built first, in the globals, so it is simply adopted.  The
   Arc's two are built by pointing the globals at empty groups, running the
   very same builders, and lifting the result out again.
   ========================================================================== */
function adoptPalaceStage(){
  const st = makeStage({
    key:'palace', label:'THE PALACE', venue:'palace', room:'stage',
    cx:0, zPros:0,
    rigGroup, poolGroup, flyGroup,
    showParent:()=>(typeof ROOM_GROUP !== 'undefined' && ROOM_GROUP.stage)
                    ? ROOM_GROUP.stage : world
  });
  st.live = true;
  STAGE = 'palace';
  return st;
}

function buildArcStage(key, label, houseKey){
  const H = ARC.houses[houseKey];
  if(!H) return null;
  const room = ARC.rooms[H.room];

  /* one group per stage, set on the house's centre line and its proscenium.
     Everything inside is built in the Palace's own stage coordinates, which
     is exactly what makes it come out identical.                         */
  const mk = ()=>{
    const g = new T.Group();
    g.position.set(H.cx, 0, H.zPros);
    room.add(g);
    return g;
  };
  const rg = mk(), fg = mk(), sg = mk();
  /* The floor-pool decals are worked out in WORLD space — where the beam
     actually meets the deck — so the group that holds them has to sit at the
     world origin, not on this house's centre line, or every pool lands one
     building over. */
  const pg = new T.Group();
  pg.position.set(-ARC.X, 0, 0);
  room.add(pg);

  /* The crew walk in WORLD coordinates, but this room sits four hundred
     metres out along x — so they get a root that cancels the offset back
     out again.  Parent them straight to the room and every hand is one
     building's width to the right of where it thinks it is. */
  const crewRootG = new T.Group();
  crewRootG.name = 'crew:' + key;
  crewRootG.position.x = -ARC.X;
  room.add(crewRootG);

  const st = makeStage({
    key, label, venue:'arc', room:H.room,
    cx:H.cx, zPros:H.zPros,
    rigGroup:rg, poolGroup:pg, flyGroup:fg,
    showParent:()=>sg
  });
  st.arcHouse = houseKey;
  H.stage = key;

  /* everything the crew need to know about working in this house */
  const dockKey = houseKey + 'Dock';
  st.crew = {
    label: label,
    root: ()=>crewRootG,
    gate: {x: ARC.X + H.dock.inner - H.dock.dir*2.2, z: H.dock.z},
    off:  {x: ARC.X + H.dock.inner + H.dock.dir*4.5, z: H.dock.z},
    rail: ARC.X + H.cx + AS.W/2 - 2.6,          // the fly floor down the side
    zOff: H.zPros,
    xMin: ARC.X + H.cx - 12, xMax: ARC.X + H.cx + 12,
    zMin: H.zPros - 15,      zMax: H.zPros - 1,
    doors:(open)=>{ if(typeof arcDoorSet === 'function') arcDoorSet(dockKey, open); }
  };

  /* stand the globals aside, build into the new groups, take the result */
  const keepFix = FIXTURES.splice(0, FIXTURES.length);
  const keepFly = FLY.splice(0, FLY.length);
  const keepCue = CUES.splice(0, CUES.length);
  const keepRig = rigGroup, keepPool = poolGroup, keepFlyG = flyGroup;
  const keepBar = FOHBAR;
  const keepSpk = SPKBARS;
  /* bodies built while the Arc goes up are Arc property (the detach
     system's venue tag, p4) */
  const keepVen = BUILD_VENUE;
  BUILD_VENUE = 'arc';
  rigGroup = rg; poolGroup = pg; flyGroup = fg;

  for(let i=0;i<14;i++) makeLineset(i);
  buildRig();
  FIXTURES.forEach(f=>{ f.homeAim = f.aim.clone(); });
  /* the FOH bar's floor clamp is against THIS house's stalls, not the
     Palace's — the Arc's rake climbs a good metre higher under the bar */
  {
    const rk = H.rake;
    const row = Math.max(0, (H.zPros + FOHBAR.z - rk.zFirst)/rk.RUN);
    FOHBAR.min = Math.min(FOHBAR.max - 0.5, rk.Y0 + row*rk.RISE + 0.2 + 2.0);
    /* the Arc's auditorium ceiling is a slab with its soffit at 15.95
       (p2j) — anchor the drop wires just under it, not at the Palace's
       24.6m plaster the buildRig default assumes */
    FOHBAR.wireTop = 15.8;
    fohBarPose(FOHBAR);
  }
  /* speaker bars: their floor is the flat front of the rake, and the
     wires anchor under the same 15.95m soffit as the FOH bar's */
  if(SPKBARS){
    const rk = H.rake;
    for(const k in SPKBARS){
      const b = SPKBARS[k];
      b.min = Math.min(b.max - 0.5, rk.Y0 + 0.2 + 1.55 + SPK_DROP);
      b.wireTop = 15.8;
      fohBarPose(b);
    }
  }

  /* the standing hang every receiving house keeps up: house curtain in,
     borders and legs in, electrics at their trims, the rest out */
  hangGoods(FLY[0], 'house');   FLY[0].target = TRIMS.house;
  hangGoods(FLY[1], 'border');  FLY[1].target = TRIMS.border;
  hangGoods(FLY[2], 'legs');    FLY[2].target = inTrimOf(FLY[2]);
  hangGoods(FLY[3], 'border');  FLY[3].target = TRIMS.border;
  hangGoods(FLY[4], 'electric');FLY[4].target = TRIMS.electric;
  hangGoods(FLY[5], 'legs');    FLY[5].target = inTrimOf(FLY[5]);
  hangGoods(FLY[6], 'truss');   FLY[6].target = TRIMS.truss;
  hangGoods(FLY[8], 'electric');FLY[8].target = TRIMS.electric;
  hangGoods(FLY[10],'legs');    FLY[10].target = inTrimOf(FLY[10]);
  hangGoods(FLY[12],'electric');FLY[12].target = TRIMS.electric;
  hangGoods(FLY[13],'cyc');     FLY[13].target = TRIMS.cyc;
  FLY.forEach(ls=>{ ls.pos = ls.target; ls.group.position.y = ls.pos; });

  st.fixtures = FIXTURES.splice(0, FIXTURES.length);
  st.fly      = FLY.splice(0, FLY.length);
  st.cues     = CUES.splice(0, CUES.length);
  st.fohBar   = FOHBAR;
  st.spkBars  = SPKBARS;
  st.nextCue  = 0;
  st.show     = showBlank();     // the one template — see p5c
  st.house    = {house:0.7, work:0, practical:0, lobby:0.9, backstage:0.7};
  st.rig      = {haze:0.35, grand:1, blackout:false};

  FIXTURES.push.apply(FIXTURES, keepFix);
  FLY.push.apply(FLY, keepFly);
  CUES.push.apply(CUES, keepCue);
  rigGroup = keepRig; poolGroup = keepPool; flyGroup = keepFlyG;
  FOHBAR = keepBar;
  SPKBARS = keepSpk;
  BUILD_VENUE = keepVen;
  return st;
}

function buildStages(){
  if(STAGES.palace) return STAGES;
  adoptPalaceStage();
  if(typeof ARC !== 'undefined' && ARC.built){
    buildArcStage('arcMain',   'THE ARC — MAIN HOUSE',    'main');
    buildArcStage('arcStudio', 'THE ARC — STUDIO THEATRE','studio');
  }
  return STAGES;
}

/* ---- the Arc's auditorium lamps follow that stage's own house master ---- */
function updateStageHouseLights(){
  if(typeof ARC === 'undefined' || !ARC.built) return;
  for(const k of ['main','studio']){
    const H = ARC.houses[k];
    if(!H || !H.lamps) continue;
    const v = stageHouseLevel(H.stage);
    for(const l of H.lamps) l.light.intensity = v * l.power;
  }
}
