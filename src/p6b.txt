
/* ============================================================================
   THE CREW — load in and load out

   A small crew of stagehands that actually do the work: they come in through
   the dock, carry the scenery on in armfuls, hang the goods off the pin rail,
   set the props, and then at the end of the night take it all out again in
   reverse.  Nothing here is a canned animation — there is a queue of jobs and
   a handful of people who pick them up as they come free, so the timing is
   different every run and it looks like a crew rather than a cutscene.
   ========================================================================== */
const CREW = {
  people: [], jobs: [], running:null, t:0, group:null,
  onDone:null, log:[], speed:1
};

const CREW_COLOURS = [0x2a3140, 0x3d3a35, 0x24303a, 0x35292b, 0x2e3630, 0x3a3340];
const CREW_HATS    = [0xffb340, 0xe8e2d4, 0x3fd0e0, 0xff6a4a, 0xffb340, 0xe8e2d4];
const M_CREW_SKIN  = new T.MeshStandardMaterial({color:0xb98a68, roughness:.85});
const M_CREW_CASE  = new T.MeshStandardMaterial({color:0x1b1d22, roughness:.85});

/* ============================================================================
   WHERE THE CREW ARE WORKING
   Every number the crew used to know — which door they come in by, where the
   pin rail is, how far downstage they may set a flat — was the Palace's.
   There are three stages now, so the geometry comes from whichever one the
   board is patched to.
   ========================================================================== */
function crewPalaceFrame(){
  const big = (typeof DOCKDOORS !== 'undefined') && DOCKDOORS.find(d=>d.big);
  const gz = big ? big.z : -6;
  return {
    label:'THE PALACE',
    root: ()=>(typeof ROOM_GROUP !== 'undefined' && ROOM_GROUP.stage) ? ROOM_GROUP.stage : world,
    gate: {x: DOCK.inner + 2.0, z: gz},
    off:  {x: DOCK.inner + DOCK.dir*5, z: gz},
    rail: (typeof XR !== 'undefined' ? XR : -D.stageW/2) + 2.8,
    zOff: 0, xMin:-12, xMax:12, zMin:-13, zMax:-1,
    doors:(open)=>{ if(typeof setAllDockDoors === 'function') setAllDockDoors(open); }
  };
}
function crewFrame(){
  const st = (typeof STAGES !== 'undefined' && typeof STAGE !== 'undefined') ? STAGES[STAGE] : null;
  if(st && st.crew) return st.crew;
  return crewPalaceFrame();
}
function crewRoot(){
  if(CREW.group) return CREW.group;
  const g = new T.Group(); g.name = 'crew';
  crewFrame().root().add(g);
  CREW.group = g;
  return g;
}
/* the hands belong to the building they are working in, or they are drawn
   into a room that is switched off and nobody sees the get-in at all */
function crewAttach(){
  const room = crewFrame().root();
  const g = crewRoot();
  if(g.parent !== room) room.add(g);
  return g;
}

/* ---- one stagehand ---- */
function makeHand(i){
  const g = new T.Group();
  const cloth = new T.MeshStandardMaterial({color:CREW_COLOURS[i % CREW_COLOURS.length], roughness:.9});
  const hat   = new T.MeshStandardMaterial({color:CREW_HATS[i % CREW_HATS.length], roughness:.5});

  const torso = new T.Mesh(new T.BoxGeometry(0.46, 0.62, 0.26), cloth);
  torso.position.y = 1.16; g.add(torso);
  const hips = new T.Mesh(new T.BoxGeometry(0.42, 0.22, 0.25), cloth);
  hips.position.y = 0.84; g.add(hips);
  const head = new T.Mesh(new T.SphereGeometry(0.13, 12, 10), M_CREW_SKIN);
  head.position.y = 1.60; g.add(head);
  const cap = new T.Mesh(new T.SphereGeometry(0.145, 12, 8, 0, 6.283, 0, 1.1), hat);
  cap.position.y = 1.62; g.add(cap);

  const limb = (w,h,d,mat)=>{
    const p = new T.Group();
    const m = new T.Mesh(new T.BoxGeometry(w,h,d), mat);
    m.position.y = -h/2; p.add(m);
    return p;
  };
  const armL = limb(0.13, 0.62, 0.15, cloth); armL.position.set(-0.30, 1.44, 0);
  const armR = limb(0.13, 0.62, 0.15, cloth); armR.position.set( 0.30, 1.44, 0);
  const legL = limb(0.16, 0.84, 0.18, cloth); legL.position.set(-0.12, 0.84, 0);
  const legR = limb(0.16, 0.84, 0.18, cloth); legR.position.set( 0.12, 0.84, 0);
  g.add(armL, armR, legL, legR);

  const hands = new T.Group();          // where whatever they are carrying rides
  hands.position.set(0, 1.02, 0.42);
  g.add(hands);

  g.traverse(o=>{ if(o.isMesh){ o.castShadow = true; o.receiveShadow = true; } });
  g.visible = false;
  crewRoot().add(g);
  /* RULING DK — EVERY figure is minted lazily, not just the lead: crewSpawn(6)
     runs on the first call that needs hands, which may be a carpenter call long
     after boot.  Hooked HERE rather than in crewSpawn or carpLead because this
     is the one place a figure's cloth and hat are made, and both paths come
     through it.  (carpLead registers again after adding its apron — a Set, so
     saying it twice is free.) */
  if(typeof envRegister === 'function') envRegister(g);

  return {
    group:g, armL, armR, legL, legR, hands, torso, cap,
    x:0, z:0, ax:0, az:0, yaw:0, phase:Math.random()*6.283,
    state:'off', job:null, wait:0, carry:null, speed:rnd(2.0, 2.7), id:i+1
  };
}
function crewSpawn(n){
  while(CREW.people.length < n) CREW.people.push(makeHand(CREW.people.length));
  return CREW.people.slice(0, n);
}

/* ---- the lead carpenter (carpenters RULING AA) --------------------------
   A SEVENTH figure, appended lazily on the first carpenter call.  The six
   show hands are minted FIRST so the lead is always CREW.people[6] and
   crewSpawn(6)'s slice can never hand them to a show plan.  makeHand
   anatomy in their own dress — apron, no cap, own cloth colour — and
   trade:'carpenter'; crewAssign guards both directions on that flag. */
function carpLead(){
  let lead = CREW.people.find(h=>h.trade === 'carpenter');
  if(lead) return lead;
  crewSpawn(6);
  lead = makeHand(CREW.people.length);
  lead.trade = 'carpenter';
  lead.cap.visible = false;                          // no cap under the shop lights
  lead.torso.material.color.setHex(0x4a3b2c);        // one cloth material per hand: this recolours the whole outfit
  const apron = new T.Mesh(new T.BoxGeometry(0.4, 0.62, 0.04),
    new T.MeshStandardMaterial({color:0xb08d57, roughness:.95}));
  apron.position.set(0, 1.06, 0.16);
  apron.castShadow = apron.receiveShadow = true;
  lead.group.add(apron);
  CREW.people.push(lead);
  /* RULING DK — the seventh figure is built LAZILY, the first time a carpenter
     call needs a lead, so the boot walk never saw his cloth, his hat or this
     apron.  Undriven that is a man standing in the shed at 7.27x the room's
     ambience through a blackout. */
  if(typeof envRegister === 'function') envRegister(lead.group);
  return lead;
}

/* where they come from and go back to */
function crewGate(){ return crewFrame().gate; }
function crewOff(){  return crewFrame().off; }

/* ---- walking ---- */
function handGoTo(h, x, z, then){
  h.ax = x; h.az = z; h.state = 'walk'; h.then = then || null;
}
function updateHand(h, dt){
  if(h.state === 'off'){ h.group.visible = false; return; }
  h.group.visible = true;
  if(h.state === 'walk'){
    const dx = h.ax - h.x, dz = h.az - h.z;
    const d = Math.hypot(dx, dz);
    if(d < 0.22){
      h.state = 'idle';
      const t = h.then; h.then = null;
      if(t) t(h);
    } else {
      const s = h.speed * CREW.speed * dt;
      h.x += dx/d * Math.min(s, d);
      h.z += dz/d * Math.min(s, d);
      const want = Math.atan2(dx, dz);
      let diff = want - h.yaw;
      while(diff >  Math.PI) diff -= 6.283;
      while(diff < -Math.PI) diff += 6.283;
      h.yaw += clamp(diff, -dt*7, dt*7);
      h.phase += dt * 8.5 * CREW.speed;
    }
  } else if(h.state === 'work'){
    h.wait -= dt;
    h.phase += dt * 3.2;
    if(h.wait <= 0){
      h.state = 'idle';
      const t = h.then; h.then = null;
      if(t) t(h);
    }
  }
  // pose
  const moving = h.state === 'walk';
  const sw = Math.sin(h.phase) * (moving ? 0.62 : 0.04);
  h.legL.rotation.x =  sw;
  h.legR.rotation.x = -sw;
  if(h.carry || h.carryB){
    h.armL.rotation.x = h.armR.rotation.x = -1.32;   // both arms out, holding it
    h.armL.rotation.z =  0.16; h.armR.rotation.z = -0.16;
  } else if(h.state === 'work'){
    const r = Math.sin(h.phase*2.2)*0.5 - 0.9;       // working overhead
    h.armL.rotation.x = r; h.armR.rotation.x = r - 0.3;
    h.armL.rotation.z = h.armR.rotation.z = 0;
  } else {
    h.armL.rotation.x = -sw*0.7; h.armR.rotation.x = sw*0.7;
    h.armL.rotation.z = h.armR.rotation.z = 0;
  }
  h.torso.position.y = 1.16 + (moving ? Math.abs(Math.sin(h.phase))*0.035 : 0);
  h.group.position.set(h.x, 0, h.z);
  h.group.rotation.y = h.yaw;
}

/* ---- what they are carrying ---- */
function crewPickUp(h, kind){
  if(h.carry) crewPutDown(h);
  const g = new T.Group();
  if(kind === 'case'){
    const b = new T.Mesh(new T.BoxGeometry(1.1, 0.7, 0.62), M_CREW_CASE);
    b.position.y = 0.1; g.add(b);
    for(const sx of [-1,1]) for(const sy of [-1,1]){
      const c = new T.Mesh(new T.BoxGeometry(0.09,0.09,0.66), M.steel);
      c.position.set(sx*0.5, 0.1 + sy*0.3, 0); g.add(c);
    }
  } else if(kind === 'flat'){
    const b = new T.Mesh(new T.BoxGeometry(1.3, 2.4, 0.08),
      new T.MeshStandardMaterial({color:0x8a7a62, roughness:.92}));
    b.position.y = 0.7; b.rotation.z = 0.1; g.add(b);
  } else if(kind === 'prop'){
    const b = new T.Mesh(new T.BoxGeometry(0.55, 0.5, 0.45),
      new T.MeshStandardMaterial({color:0x6d4326, roughness:.9}));
    b.position.y = 0.1; g.add(b);
  } else {
    const b = new T.Mesh(new T.CylinderGeometry(0.06,0.06,2.2,7), M.steel);
    b.rotation.z = Math.PI/2; b.position.y = 0.1; g.add(b);
  }
  g.traverse(o=>{ if(o.isMesh) o.castShadow = true; });
  h.hands.add(g);
  /* RULING DK — a 'flat' and a 'prop' each mint their own standard material,
     and this runs once PER PICKUP: a changeover is a stream of them, and a
     changeover is exactly when the stage is dark.  Registering a dummy is safe
     even though crewPutDown disposes it — the drive only writes a number, and
     the next recollect drops it with the rest of what has left the scene. */
  if(typeof envRegister === 'function') envRegister(g);
  h.carry = g;
  return g;
}
function crewPutDown(h){
  if(!h.carry) return;
  h.hands.remove(h.carry);
  disposeTree(h.carry);
  h.carry = null;
}

/* ---- the REAL carry (carpenters RULING AB) -------------------------------
   Carpenter loads are real p4 bodies, never the dummy prop above: pickup is
   h.hands.attach(b.mesh), set-down is venueRoot(b.venue).attach(b.mesh) —
   the forklift's exact shape (p2m) — because crewPutDown DISPOSES what it
   holds and must never see a real plank.  The body rides under h.carryB
   (h.carry stays the dummy path's own field).

   While it rides, b.state is 'carried' — a dedicated string, chosen over
   'held' because updateBodies (p4) demotes any 'held' body that is not the
   player's to 'loose' EVERY frame and then settles 'loose' bodies toward
   the floor while the mesh is still parented to the walking hand.
   'carried' passes every switch on b.state untouched: updateBodies ignores
   it (only 'held' demotes, only 'loose' settles), the snap/nail/target
   scans want 'loose'/'fixed' (p4c), carpSurvey wants 'loose'/'slotted',
   buildSerialize's generic arm writes it out like loose wood at its world
   pose (honest: a reload drops it where it was carried), and the VR grab
   loop skips it explicitly (p9) so the player cannot pull a plank out of
   the crew's arms mid-walk.  Own temps — the aliasing trap (TRAPS.md). */
const _crewV = new T.Vector3(), _crewQ = new T.Quaternion(), _crewQ2 = new T.Quaternion();
function carpPickUp(h, b){
  if(b.state === 'slotted') unslotBody(b);
  else if(b.state === 'seated' && b.station){
    b.station.pieces.splice(b.station.pieces.indexOf(b), 1);
    b.station = null;
  }
  h.hands.attach(b.mesh);
  /* ride in the arms: lumber laid across them, a sheet up on its edge */
  if(b.kind === 'wood'){
    if(b.prof === 'sheet'){ b.mesh.rotation.set(0, 0, 0); b.mesh.position.set(0, 0.35, 0.28); }
    else { b.mesh.rotation.set(0, 0, Math.PI/2); b.mesh.position.set(0, 0.08, 0.25); }
  }
  b.state = 'carried';
  h.carryB = b;
  return b;
}
/* set the real load down loose exactly where the carrier stands — the
   stand-down path (RULING AB: never dispose, never teleport) */
function carpSetDown(h){
  const b = h.carryB;
  if(!b) return null;
  h.carryB = null;
  venueRoot(b.venue).attach(b.mesh);
  b.state = 'loose';
  if(b.kind === 'wood' && typeof woodRestH === 'function') b.restH = woodRestH(b);
  return b;
}
/* take a finished piece or an off-cut OFF a station, loose beside it in a
   neat row — grabBody's seat-release shape (p4), then honest stock again */
function carpAside(st, b, run){
  const i = st.pieces.indexOf(b);
  if(i >= 0) st.pieces.splice(i, 1);
  b.station = null;
  venueRoot(b.venue).attach(b.mesh);
  b.state = 'loose';
  b.mesh.rotation.set(b.prof === 'sheet' ? -Math.PI/2 : 0, 0,
                      b.prof === 'sheet' ? 0 : Math.PI/2);   // lying flat on the deck
  b.mesh.updateMatrixWorld(true);
  b.restH = (typeof woodRestH === 'function') ? woodRestH(b) : 0.05;
  const n = run ? (run.aside = (run.aside || 0) + 1) : 1;
  st.group.getWorldPosition(_crewV);
  _crewV.x += -0.9 + (n - 1) * 0.35;                 // a row, not a pile
  _crewV.z += (st.kind === 'track' ? 1.6 : 1.2);     // clear of the table top
  _crewV.y = b.restH;                                // venue roots stand at y=0 unrotated (p4)
  b.mesh.parent.worldToLocal(_crewV);
  b.mesh.position.copy(_crewV);
}

/* ---- mark-local -> world (PR 4, RULINGS AB/AC) ----------------------------
   The mark (CARP_RUN.mark, stamped by the crayon in p9) is WORLD {x, z, yaw}
   on a deck at y = 0; the blueprint's piece poses and nail points are LOCAL
   to it.  The frame is one yaw about Y standing at the mark — built here,
   fresh, every time it is needed, and from nothing but the mark itself: any
   other source is the wrong-deck trap (the Arc offset belongs to attach()
   and worldToLocal, NEVER added by hand).  The out arguments are WRITTEN —
   callers pass scratch THEY own (the aliasing trap, TRAPS.md). */
const _carpMV = new T.Vector3(), _carpMV2 = new T.Vector3();
const _carpMQ = new T.Quaternion(), _carpYQ = new T.Quaternion();
const _carpME = new T.Euler();
function carpMarkPoint(mark, p, out){
  _carpYQ.setFromEuler(_carpME.set(0, mark.yaw || 0, 0));
  out.set(p[0], p[1], p[2]).applyQuaternion(_carpYQ);
  out.x += mark.x; out.z += mark.z;
  return out;
}
function carpMarkAxis(mark, ax, out){
  _carpYQ.setFromEuler(_carpME.set(0, mark.yaw || 0, 0));
  return out.set(ax[0], ax[1], ax[2]).applyQuaternion(_carpYQ);
}
function carpMarkPose(mark, pos, rot, outP, outQ){
  carpMarkPoint(mark, pos, outP);          // leaves the mark's yaw in _carpYQ
  outQ.setFromEuler(_carpME.set(rot[0], rot[1], rot[2]));
  outQ.premultiply(_carpYQ);
  return outP;
}

/* ============================================================================
   THE JOB QUEUE
   ========================================================================== */
function crewLog(s){ CREW.log.unshift(s); if(CREW.log.length > 12) CREW.log.pop(); }

/* what there is to move.  With a show loaded that is the show's own scenery,
   taken in armfuls; without one, the crew build a set out of stock pieces. */
/* The same load list has to serve the way in and the way out, or the load out
   builds a fresh set of loads that have never been placed and strikes nothing.
   It is rebuilt only when what is on the stage actually changes.           */
/* What the crew can pick up.  Effects — rain, fire — and anything carrying a
   light are theirs to leave alone; a hidden light is a dark stage and a
   hidden rain slab is a broken cue.  Everything else is scenery, including
   the pieces that can be repositioned on a slider.                        */
function crewHandleable(o){
  if(o.userData.effect) return false;
  let light = false;
  o.traverse(c=>{ if(c.isLight) light = true; });
  return !light;
}
function crewLoads(){
  /* keyed by stage AND show: the same production can stand on two stages at
     once, and a load list cached next door strikes the wrong building's set */
  const key = ((typeof STAGE !== 'undefined') ? STAGE : 'palace') + ':' +
              ((typeof SHOW !== 'undefined' && SHOW.key) ? SHOW.key : '(stock)');
  if(CREW.loads && CREW.loadsFor === key) return CREW.loads;
  const loads = [];
  if(typeof SHOW !== 'undefined' && SHOW.group && SHOW.group.children.length){
    const kids = SHOW.group.children.filter(crewHandleable);
    const per = Math.max(1, Math.ceil(kids.length / 9));
    for(let i=0;i<kids.length;i+=per) loads.push({parts:kids.slice(i, i+per), kind:'flat'});
  } else {
    const plan = [['platform',-4,-5],['platform',-1.4,-5],['riser',3.2,-6],
                  ['stairs',-6.6,-3],['flat',-8,-8],['flat',6.4,-8],
                  ['wall',0,-9],['table',2.2,-3],['chair',-2.4,-2.6]];
    plan.forEach(p=>{ if(SCENIC[p[0]]) loads.push({scenic:p, kind: p[0]==='flat'||p[0]==='wall'?'flat':'prop'}); });
  }
  CREW.loads = loads;
  CREW.loadsFor = key;
  return loads;
}
function crewForgetLoads(){ CREW.loads = null; CREW.loadsFor = null; }
function crewLinesets(){
  /* locked lines included: the crew's flyIn/flyOut go through flyTo, and the
     flyman works the locks himself */
  return FLY.filter(ls=>ls.goodsKey && ls.goodsKey !== 'none');
}

function crewPlan(dir){
  const jobs = [];
  const loads = crewLoads();
  const lines = crewLinesets();
  /* The rail stays out until the deck is finished.  A 'hold' job blocks the
     queue until every hand is free, so nothing comes in over the heads of
     people still carrying scenery.                                        */
  if(dir === 'in'){
    jobs.push({kind:'doors', open:true});
    loads.forEach(l=>jobs.push({kind:'on', load:l}));
    jobs.push({kind:'hold', say:'deck is set — take the goods in'});
    lines.forEach(ls=>jobs.push({kind:'fly', ls, in:true}));
    jobs.push({kind:'doors', open:false});
  } else {
    jobs.push({kind:'doors', open:true});
    lines.slice().reverse().forEach(ls=>jobs.push({kind:'fly', ls, in:false}));
    jobs.push({kind:'hold', say:'rail is out — strike the deck'});
    loads.slice().reverse().forEach(l=>jobs.push({kind:'off', load:l}));
    jobs.push({kind:'doors', open:false});
  }
  return jobs;
}

/* ============================================================================
   CALLING A SHOW IN
   Pick a production and the crew bring it in from the lorry: the set goes up
   piece by piece, the goods go out and are hauled back in on the rail, and it
   all happens under work light.  When they are done the show's own opening
   look comes up and the stage is yours.
   ========================================================================== */
function crewLoadShow(key){
  if(typeof SHOWS === 'undefined' || !SHOWS[key]) return false;
  showLoad(key);                       // build it, then take it all off again
  crewForgetLoads();
  crewWorkLight(true);
  crewStart('in');
  CREW.onDone = dir=>{ if(dir === 'in') crewWorkLight(false); };
  crewLog('called in: ' + SHOWS[key].name + ' at ' + crewFrame().label);
  toast(SHOWS[key].name + ' — the lorry is at ' + crewFrame().label);
  return true;
}
/* a get-in happens under work light, and the show look comes back afterwards */
function crewWorkLight(on){
  if(on){
    if(!CREW.savedLook){
      CREW.savedLook  = (typeof snapshotLX === 'function') ? snapshotLX() : null;
      CREW.savedHouse = {house:HOUSE.house, work:HOUSE.work,
                         practical:HOUSE.practical, haze:RIG.haze};
    }
    FIXTURES.forEach(f=>{ f.level = 0; f.lvlDur = 0; f.lvlTo = 0; });
    HOUSE.work = 0.9; HOUSE.house = 0.3; HOUSE.practical = 0; RIG.haze = 0;
  } else {
    if(CREW.savedLook){
      CREW.savedLook.forEach((r,i)=>{ setLevel(i+1, r.lvl, 4); setColorCh(i+1, r.col, 4); });
      const hs = CREW.savedHouse;
      HOUSE.house = hs.house; HOUSE.work = hs.work;
      HOUSE.practical = hs.practical; RIG.haze = hs.haze;
      CREW.savedLook = null; CREW.savedHouse = null;
      crewLog('work light out, show look up');
    } else {
      HOUSE.work = 0;
    }
  }
  if(typeof syncMasters === 'function') syncMasters();
  if(typeof refreshChannelStrip === 'function') refreshChannelStrip();
}

function crewStart(dir){
  if(CREW.running) crewStop(true);
  crewAttach();
  CREW.onDone = null;
  CREW.running = dir;
  CREW.t = 0;
  CREW.log = [];
  const jobs = crewPlan(dir);
  CREW.jobs = jobs;
  const hands = crewSpawn(6);
  const gate = crewGate(), off = crewOff();
  hands.forEach((h, i)=>{
    h.state = 'idle'; h.job = null; h.wait = 0;
    crewPutDown(h);
    h.x = off.x + rnd(-1.5, 1.5); h.z = off.z + rnd(-2.2, 2.2) + i*0.4;
    h.yaw = Math.PI/2;
    h.group.visible = true;
  });
  if(dir === 'in'){
    // on the way in the stage starts empty and the rail starts out —
    // otherwise the crew would be hauling in things that are already in
    crewHideLoads(true);
    crewClearStock();
    crewLinesets().forEach(ls=>{
      ls.target = ls.pos = OUT_TRIM;
      ls.group.position.y = ls.pos;
    });
    if(typeof refreshFlyUI === 'function') refreshFlyUI();
  }
  crewLog((dir === 'in' ? 'Load in' : 'Load out')+' called — '+jobs.length+' jobs');
  toast(dir === 'in' ? 'Load in: the crew are on the dock' : 'Load out: strike it all');
  if(typeof refreshCrewUI === 'function') refreshCrewUI();
  return jobs.length;
}
function crewHideLoads(hide){
  if(typeof SHOW === 'undefined' || !SHOW.group) return;
  SHOW.group.children.forEach(o=>{
    if(!crewHandleable(o)) return;               // effects and lights stay put
    setPieceVisible(o, hide);
  });
}
/* the stock set the crew build when there is no show: take it off again */
function crewClearStock(){
  const loads = CREW.loads;
  if(!loads) return;
  loads.forEach(l=>{
    if(l.scenic && l.placed && SET.indexOf(l.placed) !== -1) strikePiece(l.placed);
    if(l.scenic) l.placed = null;
  });
}
function crewStop(silent){
  CREW.running = null;
  CREW.jobs = [];
  CREW.people.forEach(h=>{
    /* a carpenter load is REAL: down loose where they stand, never through
       the dummy-dispose path below (RULING AB) */
    if(h.carryB) carpSetDown(h);
    crewPutDown(h);
    h.state = 'off'; h.job = null; h.group.visible = false;
  });
  CARP_RUN = null;               // a stood-down build does not resume — call again (RULING AC)
  crewHideLoads(false);
  if(!silent) toast('Crew stood down');
  if(typeof refreshCrewUI === 'function') refreshCrewUI();
}

/* hand the next job to whoever is free */
function crewAssign(h){
  const next = CREW.jobs[0];
  if(!next) return false;
  /* the trade guards (RULING AA), checked BEFORE the shift so the head of
     the queue waits for the right person instead of being consumed by the
     wrong one: the saw jobs are the lead's alone, and the lead never works
     anything else — not shows, and not the hands' haul/nail jobs either */
  const sawJob = next.kind === 'carpFetch' || next.kind === 'carpCut';
  if(sawJob !== (h.trade === 'carpenter')) return false;
  const job = CREW.jobs.shift();
  h.job = job;
  const gate = crewGate();

  if(job.kind === 'carpFetch'){
    /* lead: walk to the surveyed stick, REAL-carry it to its saw, seat it.
       All of it in world coordinates off live objects at EXECUTION time —
       the crew root cancels the Arc offset (p2k), so h.x/h.z ARE world. */
    /* body null means RE-SEAT what this run already has on that bench,
       turned the other way (a schedule that rips after crosscutting):
       the lead lifts it — carpPickUp unfiles a seated piece — and offers
       it again on the axis this entry asks for */
    const b = job.body ||
      ((typeof CARP_RUN !== 'undefined' && CARP_RUN) ? CARP_RUN.bench[job.saw] : null);
    const set = (typeof SAWS !== 'undefined' && SAWS) ? SAWS[job.venue] : null;
    const st = set && set[job.saw];
    const takeable = b && (b.state === 'loose' || b.state === 'slotted' ||
                           (!job.body && b.state === 'seated'));
    if(!takeable || !st){
      h.job = null; return true;           // the stick was taken since the plan: skip
    }
    b.mesh.getWorldPosition(_crewV);
    handGoTo(h, _crewV.x, _crewV.z, ()=>{
      h.state = 'work'; h.wait = rnd(0.5, 0.9);        // stoop and take it up
      h.then = ()=>{
        if(b.state !== 'loose' && b.state !== 'slotted' &&
           !(!job.body && b.state === 'seated')){ h.job = null; return; }
        carpPickUp(h, b);
        st.group.getWorldPosition(_crewV);
        handGoTo(h, _crewV.x, _crewV.z + 1.0, ()=>{
          h.state = 'work'; h.wait = rnd(0.8, 1.4);    // offer it up to the bench
          h.then = ()=>{
            /* a sheet seats on the axis the SCHEDULE asks for, because
               seatWood reads whichever axis the hand offers: 'L' puts its
               length along table X (a crosscut — the default, so the
               schedule's lengths mean what they say), 'W' turns it a
               quarter round so the cut rips a strip off its width. */
            if(b.prof === 'sheet'){
              st.group.getWorldQuaternion(_crewQ);
              h.hands.getWorldQuaternion(_crewQ2).invert();
              b.mesh.quaternion.copy(_crewQ2.multiply(_crewQ));
              if(job.axis === 'W') b.mesh.rotateY(Math.PI/2);
              b.mesh.updateMatrixWorld(true);
            }
            if(seatWood(st, b)){
              h.carryB = null;
              if(typeof CARP_RUN !== 'undefined' && CARP_RUN) CARP_RUN.bench[job.saw] = b;
              crewLog(b.prof + ' on the ' + (job.saw === 'track' ? 'table' : 'bench'));
            } else carpSetDown(h);          // refused seat: down loose, never disposed
            h.job = null;
          };
        });
      };
    });
    return true;
  }

  if(job.kind === 'carpCut'){
    /* lead: blade at job.len off the LOW end of the piece this run has on
       that bench (the fetched stick, then the registered off side after
       each cut — p6c's record).  The kept low side IS the scheduled
       product; the lead pencils the exact mark first so the tape tick
       wins over the inch snap (sawSetCut, p4c). */
    const set = (typeof SAWS !== 'undefined' && SAWS) ? SAWS[job.venue] : null;
    const st = set && set[job.saw];
    if(!st){ h.job = null; return true; }
    st.group.getWorldPosition(_crewV);
    handGoTo(h, _crewV.x, _crewV.z + 1.0, ()=>{
      h.state = 'work'; h.wait = rnd(1.4, 2.4);        // line it up and cut
      h.then = ()=>{
        const run = (typeof CARP_RUN !== 'undefined') ? CARP_RUN : null;
        const cur = run && run.bench && run.bench[job.saw];
        if(cur && cur.state === 'seated' && cur.station === st){
          const lo = cur.mesh.position.x - seatLen(cur)/2;
          _crewV.set(lo + job.len, 0, 0);
          st.group.localToWorld(_crewV);
          cur.mesh.updateMatrixWorld(true);
          cur.tick = {p: cur.mesh.worldToLocal(_crewV.clone())};   // the pencil mark
          sawSetCut(st, lo + job.len);
          const r = sawCut(st);
          if(r){
            if(run && job.piece >= 0) run.made[job.piece] = r.kept;
            /* the product leaves the bench so the next cut meets ONLY the
               remainder; it waits loose beside the saw for PR 4's haul */
            carpAside(st, r.kept, run);
            /* the remainder stays on the bench if this run still wants it:
               another cut, or a re-seat (a fetch with no body of its own)
               that means to turn it round and rip it (RULING AE) */
            const nx = CREW.jobs[0];
            const more = !!nx && nx.saw === job.saw &&
              (nx.kind === 'carpCut' || (nx.kind === 'carpFetch' && !nx.body));
            if(run) run.bench[job.saw] = more ? r.off : null;
            /* a usable final off side goes back to the shed as stock,
               loose by the bench (RULING AC) */
            if(r.off && !more) carpAside(st, r.off, run);
            crewLog('cut ' + ftIn(job.len) + ' on the ' + (job.saw === 'track' ? 'table' : 'bench'));
          }
        }
        h.job = null;
      };
    });
    return true;
  }

  if(job.kind === 'carpHaul'){
    /* hand: real-carry ONE piece to its blueprint pose at the mark (RULING
       AB — one piece per trip, the same mesh end to end, never the dummy
       path).  job.body is a stick used uncut; body:null means the piece
       came off a saw and CARP_RUN.made knows which body the cut kept. */
    const run = (typeof CARP_RUN !== 'undefined') ? CARP_RUN : null;
    const b = job.body || (run && run.made && run.made[job.piece]) || null;
    if(!run || !run.mark || !b || (b.state !== 'loose' && b.state !== 'slotted')){
      h.job = null; return true;   // no mark to build at, or the piece was taken
    }
    b.mesh.getWorldPosition(_crewV);
    handGoTo(h, _crewV.x, _crewV.z, ()=>{
      h.state = 'work'; h.wait = rnd(0.5, 0.9);          // stoop and take it up
      h.then = ()=>{
        if(b.state !== 'loose' && b.state !== 'slotted'){ h.job = null; return; }
        carpPickUp(h, b);
        carpMarkPoint(run.mark, job.pos, _crewV);
        handGoTo(h, _crewV.x, _crewV.z + 0.6, ()=>{      // stand just downstage of the lines
          h.state = 'work'; h.wait = rnd(0.8, 1.4);      // lower it onto its lines
          h.then = ()=>{
            /* set-down at the blueprint pose: venueRoot attach — the
               forklift's shape, three owns the Arc offset — then the WORLD
               pose written through the parent's frame */
            const root = venueRoot(b.venue);
            root.attach(b.mesh);
            root.updateWorldMatrix(true, false);
            carpMarkPose(run.mark, job.pos, job.rot, _carpMV, _carpMQ);
            b.mesh.position.copy(root.worldToLocal(_carpMV));
            root.getWorldQuaternion(_crewQ).invert();
            b.mesh.quaternion.copy(_crewQ.multiply(_carpMQ));
            b.state = 'loose';
            /* the POSE height is the rest height.  updateBodies (p4) settles
               every loose body toward groundAt + restH, and groundAt sees
               architecture, not the wood already placed under this piece —
               so a skin posed on its frame (flat: 0.038 + half 0.0095) would
               sink through it under a woodRestH value.  For a piece lying
               directly on the deck the pose y IS woodRestH's half-box; for a
               piece bearing on wood beneath, restH carries the stack exactly
               the way woodRestH means it: this is where the piece rests.
               Nothing new rides the save (restH is recomputed on load), and
               a later hammer blow re-derives it via detachPiece. */
            b.restH = job.pos[1];
            b.mesh.updateMatrixWorld(true);
            h.carryB = null;
            if(run.placed) run.placed[job.piece] = b;
            crewLog(b.prof + ' set at the mark');
            h.job = null;
          };
        });
      };
    });
    return true;
  }

  if(job.kind === 'carpNail'){
    /* hand: work-wait at the joint, then BOTH of the blueprint's nails for
       it (RULING G: two is rigid).  No deck nail ever — the finished piece
       lies un-anchored at the mark (RULING AC).  addNail reads CURRENT
       world transforms, so the scene is brought current before each shot;
       a refused nail is logged and the run carries on. */
    const run = (typeof CARP_RUN !== 'undefined') ? CARP_RUN : null;
    const bi = run && run.placed && run.placed[job.i];
    const bj = run && run.placed && run.placed[job.j];
    if(!run || !run.mark || !bi || !bj){ h.job = null; return true; }
    carpMarkPoint(run.mark, job.nails[0].p, _crewV);
    handGoTo(h, _crewV.x + 0.3, _crewV.z + 0.45, ()=>{   // kneel beside the seam
      h.state = 'work'; h.wait = rnd(1.0, 1.8);          // gun up, two shots
      h.then = ()=>{
        for(const nd of job.nails){
          scene.updateMatrixWorld(true);
          carpMarkPoint(run.mark, nd.p, _carpMV);
          carpMarkAxis(run.mark, nd.ax, _carpMV2);
          if(!addNail(bi, {body:bj}, _carpMV, _carpMV2))
            crewLog('a nail refused at ' + job.i + '-' + job.j);
        }
        crewLog('nailed ' + job.i + ' to ' + job.j);
        h.job = null;
      };
    });
    return true;
  }

  if(job.kind === 'doors'){
    handGoTo(h, gate.x, gate.z, ()=>{
      h.state = 'work'; h.wait = 1.0;
      h.then = ()=>{
        crewFrame().doors(job.open);
        crewLog(job.open ? 'dock doors up' : 'dock doors down');
        h.job = null;
      };
    });
    return true;
  }

  if(job.kind === 'fly'){
    // to the pin rail, haul, then back out of the way
    const fr = crewFrame();
    handGoTo(h, fr.rail, clamp(job.ls.z + fr.zOff, fr.zMin, fr.zMax), ()=>{
      h.state = 'work'; h.wait = rnd(1.4, 2.4);
      h.then = ()=>{
        if(job.in) flyIn(job.ls); else flyOut(job.ls);
        crewLog('lineset '+job.ls.id+' '+(job.in ? 'in' : 'out')+
                ' — '+(GOODS[job.ls.goodsKey] ? GOODS[job.ls.goodsKey].label : ''));
        if(typeof syncFlyUI === 'function') syncFlyUI();
        h.job = null;
      };
    });
    return true;
  }

  if(job.kind === 'on'){
    // out to the dock, pick it up, carry it on, set it, walk back
    const off = crewOff();
    handGoTo(h, off.x, off.z + rnd(-1.5,1.5), ()=>{
      crewPickUp(h, job.load.kind);
      const spot = crewSpot(job.load);
      handGoTo(h, spot.x, spot.z, ()=>{
        h.state = 'work'; h.wait = rnd(0.8, 1.6);
        h.then = ()=>{
          crewPutDown(h);
          crewDeliver(job.load);
          handGoTo(h, gate.x, gate.z + rnd(-1.5,1.5), ()=>{ h.job = null; });
        };
      });
    });
    return true;
  }

  if(job.kind === 'off'){
    const spot = crewSpot(job.load);
    handGoTo(h, spot.x, spot.z, ()=>{
      h.state = 'work'; h.wait = rnd(0.8, 1.6);
      h.then = ()=>{
        crewStrike(job.load);
        crewPickUp(h, job.load.kind);
        const off = crewOff();
        handGoTo(h, off.x, off.z + rnd(-1.5,1.5), ()=>{
          crewPutDown(h);
          handGoTo(h, gate.x, gate.z + rnd(-1.5,1.5), ()=>{ h.job = null; });
        });
      };
    });
    return true;
  }
  h.job = null;
  return true;
}

/* where on the deck a load belongs */
function crewSpot(load){
  const fr = crewFrame();
  /* the stock plan is written in stage coordinates; the hands walk in world.
     The frame's bounds are centred on the stage it serves, so the middle of
     them IS the stage origin — at the Palace that comes out unchanged. */
  if(load.scenic) return {x:(fr.xMin + fr.xMax)/2 + load.scenic[1],
                          z:fr.zOff + load.scenic[2]};
  const b = worldBox2(load.parts);
  if(b) return {x:clamp(b.x, fr.xMin, fr.xMax), z:clamp(b.z, fr.zMin, fr.zMax)};
  return {x:rnd(fr.xMin*0.5, fr.xMax*0.5),
          z:rnd(fr.zMin + 3, fr.zMax - 1)};
}
function worldBox2(parts){
  if(!parts || !parts.length) return null;
  let n = 0, x = 0, z = 0;
  for(const p of parts){
    const b = (typeof worldBox === 'function') ? worldBox(p) : null;
    if(!b) continue;
    const c = b.getCenter(new T.Vector3());
    x += c.x; z += c.z; n++;
  }
  return n ? {x:x/n, z:z/n} : null;
}
function crewDeliver(load){
  if(load.scenic){
    // if it was struck by hand since, put it back rather than leaving a ghost
    if(!load.placed || SET.indexOf(load.placed) === -1){
      const spot = crewSpot(load);            // world, not the plan's stage coords
      load.placed = placeScenic(load.scenic[0], spot.x, spot.z);
    }
    crewLog('set '+load.scenic[0]);
  } else {
    load.parts.forEach(o=>setPieceVisible(o, false));
    crewLog('brought on '+load.parts.length+' pieces');
  }
}
function crewStrike(load){
  if(load.scenic){
    if(load.placed && SET.indexOf(load.placed) !== -1) strikePiece(load.placed);
    load.placed = null;
    crewLog('struck '+load.scenic[0]);
  } else {
    load.parts.forEach(o=>setPieceVisible(o, true));
    crewLog('took off '+load.parts.length+' pieces');
  }
}

/* ---- the director, run every frame ---- */
function updateCrew(dt){
  if(!CREW.people.length) return;
  for(const h of CREW.people) updateHand(h, dt);
  if(!CREW.running) return;
  CREW.t += dt;

  let busy = CREW.people.filter(h=>h.job || h.state === 'walk' || h.state === 'work').length;
  // a hold sits at the head of the queue until everyone has finished
  while(CREW.jobs.length && CREW.jobs[0].kind === 'hold' && busy === 0){
    const held = CREW.jobs.shift();
    if(held.say) crewLog(held.say);
  }
  for(const h of CREW.people){
    if(h.job || h.state === 'walk' || h.state === 'work') continue;
    if(CREW.jobs.length && CREW.jobs[0].kind !== 'hold') crewAssign(h);
    if(h.job) busy++;
  }
  if(!CREW.jobs.length && !busy){
    // everyone off, and we are done
    const off = crewOff();
    let clear = true;
    for(const h of CREW.people){
      /* whoever sat this run out stays sat out: the lead through a show,
         the six hands through a carpenter call — walking an 'off' figure
         to the door would parade them across the stage first */
      if(h.state === 'off') continue;
      if(Math.abs(h.x - off.x) > 3){ handGoTo(h, off.x + rnd(-1.5,1.5), off.z + rnd(-2,2)); clear = false; }
    }
    if(clear){
      const was = CREW.running;
      CREW.running = null;
      CREW.people.forEach(h=>{ h.state = 'off'; h.group.visible = false; });
      /* a finished catalogue build earns its line in the log (PR 4) */
      if(was === 'carp' && typeof CARP_RUN !== 'undefined' && CARP_RUN && CARP_RUN.key &&
         typeof CARP_CAT !== 'undefined' && CARP_CAT[CARP_RUN.key])
        crewLog('built: ' + CARP_CAT[CARP_RUN.key].label + ' — lying at the mark');
      crewLog(was === 'carp' ? 'Carpenter work complete'
            : (was === 'in' ? 'Load in' : 'Load out')+' complete');
      toast(was === 'carp' ? 'The carpenters are done'
          : was === 'in' ? 'Load in complete — the stage is yours' : 'Load out complete');
      if(CREW.onDone) CREW.onDone(was);
      if(typeof refreshCrewUI === 'function') refreshCrewUI();
    }
  }
}
function crewBusy(){
  return CREW.people.filter(h=>h.job || h.state === 'walk' || h.state === 'work').length;
}
