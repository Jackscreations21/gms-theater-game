const RIG = { haze:.55, grand:1, blackout:false, shadowBudget:4 };
/* what the beams actually read through: the hazer on the board plus whatever
   the smoke machines have put in the air */
function hazeNow(){
  const s = (typeof SMOKE !== 'undefined' && SMOKE.haze) ? SMOKE.haze : 0;
  return Math.min(1, RIG.haze + s*0.7);
}

/* ---- gobo + pool textures ---- */
function goboTex(kind){
  const c=cvs(256,256), x=c.getContext('2d');
  x.fillStyle='#000'; x.fillRect(0,0,256,256);
  x.fillStyle='#fff';
  if(kind===0){ // soft open circle
    const g=x.createRadialGradient(128,128,10,128,128,128);
    g.addColorStop(0,'#fff'); g.addColorStop(.62,'rgba(255,255,255,.92)');
    g.addColorStop(1,'rgba(255,255,255,0)');
    x.fillStyle=g; x.fillRect(0,0,256,256);
  } else if(kind===1){ // breakup
    for(let k=0;k<180;k++){ x.save(); x.translate(Math.random()*256,Math.random()*256);
      x.rotate(Math.random()*6.283); x.globalAlpha=rnd(.35,1);
      x.fillRect(0,0,rnd(6,44),rnd(3,16)); x.restore(); }
  } else if(kind===2){ // leaves
    for(let k=0;k<90;k++){ x.save(); x.translate(Math.random()*256,Math.random()*256);
      x.rotate(Math.random()*6.283); x.globalAlpha=rnd(.4,1); x.beginPath();
      x.ellipse(0,0,rnd(6,22),rnd(3,10),0,0,6.283); x.fill(); x.restore(); }
  } else if(kind===3){ // window
    x.fillRect(24,24,208,208); x.fillStyle='#000';
    for(let i=0;i<3;i++){ x.fillRect(24+ i*70+ 60, 24, 10, 208); x.fillRect(24, 24+i*70+60, 208, 10); }
  } else if(kind===4){ // stars
    for(let k=0;k<160;k++){ const r=rnd(1,3.4); x.globalAlpha=rnd(.4,1);
      x.beginPath(); x.arc(Math.random()*256,Math.random()*256,r,0,6.283); x.fill(); }
  } else { // bars
    for(let i=0;i<9;i++) x.fillRect(0, i*28, 256, 15);
  }
  // vignette so the gobo stays inside the beam
  const v=x.createRadialGradient(128,128,60,128,128,130);
  v.addColorStop(0,'rgba(0,0,0,0)'); v.addColorStop(1,'rgba(0,0,0,1)');
  x.globalAlpha=1; x.globalCompositeOperation='destination-out'; x.fillStyle=v; x.fillRect(0,0,256,256);
  const t=new T.CanvasTexture(c); t.wrapS=t.wrapT=T.ClampToEdgeWrapping; return t;
}
const GOBOS = [0,1,2,3,4,5].map(goboTex);

/* ---- volumetric beam shader ---- */
const BEAM_VS = `
varying vec3 vWpos; varying vec3 vWnorm; varying vec2 vUv;
void main(){
  vUv = uv;
  vec4 wp = modelMatrix * vec4(position,1.0);
  vWpos = wp.xyz;
  vWnorm = normalize(mat3(modelMatrix) * normal);
  gl_Position = projectionMatrix * viewMatrix * wp;
}`;
const BEAM_FS = `
uniform vec3 uColor; uniform float uInt; uniform float uHaze; uniform float uTime;
uniform float uSoft; uniform sampler2D uGobo; uniform float uUseGobo;
varying vec3 vWpos; varying vec3 vWnorm; varying vec2 vUv;
float hash(vec2 p){ return fract(sin(dot(p,vec2(41.3,289.1)))*43758.5453); }
void main(){
  if(uInt <= 0.002) discard;
  vec3 V = normalize(cameraPosition - vWpos);
  float edge = 1.0 - abs(dot(normalize(vWnorm), V));
  edge = pow(clamp(edge,0.0,1.0), uSoft);
  float along = vUv.y;                       // 0 = lens, 1 = far end
  float atten = mix(1.0, 0.06, pow(along,1.35));
  float tip   = smoothstep(0.0,0.07,along);  // hide the hard apex
  float fade  = 1.0 - smoothstep(0.72,1.0,along);
  float dust  = 0.85 + 0.15*hash(floor(vUv*vec2(60.0,140.0)) + floor(uTime*7.0));
  float g = 1.0;
  if(uUseGobo > 0.5){
    float a = atan(vWnorm.z, vWnorm.x);
    g = texture2D(uGobo, vec2(fract(vUv.x*3.0), fract(along*2.0 + uTime*0.0))).r*0.75 + 0.35;
  }
  float i = uInt * uHaze * edge * atten * tip * fade * dust * g;
  gl_FragColor = vec4(uColor * i, i);
}`;

function makeBeam(len, rad){
  const geo = new T.CylinderGeometry(rad, 0.055, len, 12, 1, true);
  geo.rotateX(Math.PI/2);
  geo.translate(0,0,len/2);
  const mat = new T.ShaderMaterial({
    vertexShader:BEAM_VS, fragmentShader:BEAM_FS,
    uniforms:{ uColor:{value:new T.Color(0xffffff)}, uInt:{value:0}, uHaze:{value:.55},
               uTime:{value:0}, uSoft:{value:2.4}, uGobo:{value:GOBOS[0]}, uUseGobo:{value:0} },
    transparent:true, blending:T.AdditiveBlending, depthWrite:false, side:T.DoubleSide
  });
  /* RULING DM — the beam is the light itself, not a surface the light falls on,
     so it stays out of the grade (see gradeExempt in p2).  A raw ShaderMaterial
     never includes the tonemapping chunk in any case; saying it here makes the
     intent explicit rather than incidental to which material class was used. */
  gradeExempt(mat);
  const m = new T.Mesh(geo, mat);
  m.frustumCulled = true;          // a beam behind you costs nothing
  m.raycast = ()=>{};              // beams are air — never pickable
  return m;
}

/* ---- fixture bodies ------------------------------------------------------
   Real lanterns, not primitives.  Every geometry is built ONCE via fixG and
   shared across all instances — three stages hang ~25 channels each, so
   per-body geometry would be ~900 duplicate buffers.  Contracts the rest of
   p4 reads and must survive any edit here: userData.lens on every body THAT HAS
   ONE, userData.base/yoke/head on movers.  New for the detach system (PR 5):
   userData.clamp is the C-clamp that meets the pipe.  Budget: each body
   stays well under 700 triangles — these are drawn ~75x in a headset.

   RULING CY MADE ONE BODY EMPTY.  The blinder has no geometry at all — no
   shell, no cells, no lens — because he asked for the light to come out of the
   neon rather than out of a box hanging in his picture.  So "userData.lens on
   every body" is now "on every body that has a mesh", which is the same
   contract for everything with something to look at.  A test in
   tests/stages.js names the exception and refuses a second one.

   RULING DY MERGED THE SHELL, so the piece list below is a DESCRIPTION of the
   geometry and no longer a list of objects.  mergeShell (just under lensMat)
   runs at the bottom of every builder and collapses a frame's static mesh
   children into one Mesh per material.  What that changes for a reader of the
   builders: after the merge a profile is three drawables (fixture shell, steel
   shell, lens) where it was seventeen, and userData.clamp names the STEEL
   SHELL the C-clamp is now part of rather than a jaw of its own.  The clamp is
   a declaration that a body meets a pipe, and nothing reads its transform;
   if something ever needs to, it needs its own unmerged mesh and the draw
   that costs.  Everything else in the contract is untouched: the body ROOT is
   still the Group that BODIES holds, the mover's yoke and head are still live
   Groups merged separately from the shell above them, and the lens is still
   its own mesh on its own keyed material.                                    */
const FIXG = {};
function fixG(k, make){ return FIXG[k] || (FIXG[k] = make()); }
/* lens materials are cached by colour and SHARED — never tint one per
   fixture (userData.lens is write-only today; detach/order work, take note) */
const LENSM = {};
function lensMat(c){ return LENSM[c] || (LENSM[c] = new T.MeshBasicMaterial({color:c})); }

/* ---- RULING DY — A BODY'S STATIC SHELL IS ONE DRAW PER MATERIAL -----------
   MEASURED, not guessed (tools/draws.js, on a55bfcd): the empty Palace submits
   350 draws an eye at the boot view — 700 a frame, because there is no
   multiview and each eye is a full projectObject pass — against a Quest
   comfort band of roughly 100-200.  The architecture is not the problem; it is
   already merged and runs at ~1.1 draws a block.  The LIGHTING RIG is 476 of
   the scene's 878 visible drawables, and 423 of those are the lantern BODIES
   at 10.8 draws each: a barrel, a gate, a lamp housing, four shutter handles,
   four colour-frame bars and five pieces of hanging iron, every one of them its
   own Mesh and therefore its own draw call, thirty-nine times over.

   None of those pieces is addressed by anything.  Three things in a body ARE,
   and the merge stops at all three:

     THE BODY ROOT.  It is a BODIES.mesh — detachable (unhangBody), carried,
       slotted, hovered, raycast through pickAll.  The merge happens INSIDE one
       body and produces 39 merged bodies, never one; the root Group keeps its
       identity, its children live where they always did, and hangBody /
       unhangBody / slotBody move exactly what they moved before.
     THE YOKE AND THE HEAD.  updateRig writes userData.yoke.rotation.y and
       userData.head.rotation.x every frame, so a mover has THREE frames that
       move independently and each is merged on its own.  Baking a head into
       the shell above it would weld a moving light solid.
     THE LENS.  It is painted through the LENSM keyed cache, which is how a
       thing gets REPAINTED in this codebase (INVARIANTS).  Merging it freezes
       its colour with nothing thrown — the paint-roller-head failure mode in
       TRAPS.  So no mesh wearing a keyed-cache material is ever merged, which
       also spares the speaker grille (same cache, same reason, and it is what
       tests/stages.js counts to prove one grille material across twelve boxes).

   MATERIALS ARE NOT TOUCHED.  A merged shell wears the same shared M.fixture /
   M.steel object its pieces wore, so the envTrack / envCarrier / atmTrack
   registrations behind INVARIANTS are unaffected: there is nothing new to
   register and nothing cloned per fixture to drift.

   AND THE GEOMETRY IS CACHED LIKE EVERY OTHER PIECE.  The merged result goes
   into FIXG under the body's own key, so all 117 lantern bodies across three
   stages share one buffer per material per type — the same argument the
   per-piece cache was built on, and mergeParts CLONES its inputs, so the
   cached pieces it reads are not mutated (the shared-cache trap, TRAPS).

   WHAT IT COSTS, stated rather than discovered: mergeParts emits NON-INDEXED
   geometry, so a merged shell processes about three vertices per triangle
   where an indexed cylinder shared them.  Triangles are unchanged per
   geometry (that is asserted) — but the wrist meter's tri line counts
   SUBMITTED triangles, and one merged bounding sphere culls less finely than
   seventeen small ones, so a body half out of frustum now submits whole:
   measured +0.5% on the boot view.  The buffers are per TYPE rather than per
   body, and the budget this ruling is about is draw calls.

   BODY_MERGE is the one line that puts the pieces back, and it is what the
   draw-ceiling assertion in tests/stages.js is negative-checked against.     */
const BODY_MERGE = true;
/* a material minted by the LENSM keyed cache.  Four entries, so a scan is
   cheaper and harder to get wrong than a parallel Set.

   AND IT IS A GUARD WITH NO LIVE CASE IN THE RIG TODAY, which is worth saying
   rather than leaving to look load-bearing (TRAPS).  A merge is per material,
   and every body carries exactly ONE keyed-cache piece — its lens, or the PA
   box's grille — so a lens is already alone in its group and a group of one is
   never merged.  What this line protects against is the SECOND one: a body
   with two lens-cache pieces (two cells, a lens and a filter) would otherwise
   merge them into one frozen mesh and every repaint after that would be
   invisible with nothing thrown, which is the paint-roller-head trap exactly.
   tests/stages.js builds the two-piece frame the rig does not have, because a
   guard nothing approaches cannot be negative-checked (TRAPS again). */
function lensMatted(m){ for(const k in LENSM) if(LENSM[k] === m) return true; return false; }
/* Merge ONE frame: the static mesh children of `frame`, grouped by material,
   each group becoming a single Mesh whose geometry is cached under `key`.
   Child Groups are left alone — that is how the mover's yoke and head survive,
   and they are merged by their own calls.  `root` is the node whose userData
   publishes handles; any handle that named a swallowed mesh is re-pointed at
   the shell that swallowed it, so a contract cannot silently start naming an
   orphan. */
function mergeShell(key, frame, root){
  if(!BODY_MERGE) return frame;
  root = root || frame;
  const groups = [], byMat = new Map();
  for(const c of frame.children){
    if(!c.isMesh || !c.geometry) continue;        // a live sub-frame is a Group, and stays one
    if(Array.isArray(c.material)) continue;       // nothing in a body has one; not ours to guess at
    /* noMerge is an opt-out with NO LIVE CASE today, same disclosure as the
       lens guard above: it exists so a future builder can keep a piece its
       own draw without learning this function's internals. */
    if(lensMatted(c.material) || c.userData.noMerge) continue;
    /* mergeParts composes an XYZ Euler, which is a Mesh's default and what
       every body piece uses.  A piece that ever carries another order costs a
       DRAW here rather than being baked with the wrong rotation — err toward a
       draw, never toward wrong geometry.  Nothing takes this branch today. */
    if(c.rotation.order !== 'XYZ') continue;
    /* and the same refusal for a negative scale: mergeParts' own header
       forbids the mirrored part (flipped winding vanishes under FrontSide),
       and a flipped piece is the ONE failure the test net cannot see — the
       tri count and the box both survive a winding flip.  Nothing sets any
       scale on a body piece today; if it ever does, it costs a draw. */
    if(c.scale.x < 0 || c.scale.y < 0 || c.scale.z < 0) continue;
    let e = byMat.get(c.material);
    if(!e){ e = {mat:c.material, parts:[], meshes:[]}; byMat.set(c.material, e); groups.push(e); }
    e.parts.push({geo:c.geometry,
                  pos:[c.position.x, c.position.y, c.position.z],
                  rot:[c.rotation.x, c.rotation.y, c.rotation.z],
                  scale:[c.scale.x, c.scale.y, c.scale.z]});
    e.meshes.push(c);
  }
  groups.forEach((e, i)=>{
    /* one piece is already one draw, and merging it would trade a shared
       cached buffer for a non-indexed copy of itself */
    if(e.parts.length < 2) return;
    const shell = new T.Mesh(fixG(key + '#' + i, ()=>mergeParts(e.parts)), e.mat);
    for(const m of e.meshes){
      frame.remove(m);
      for(const k in root.userData) if(root.userData[k] === m) root.userData[k] = shell;
    }
    frame.add(shell);
  });
  return frame;
}

/* yoke + C-clamp + safety loop, shared by every hanging body */
function addHang(g, yokeR){
  const yoke = new T.Mesh(fixG('yoke'+yokeR, ()=>new T.TorusGeometry(yokeR,.022,5,14,Math.PI)), M.steel);
  yoke.rotation.y = Math.PI/2; g.add(yoke);
  const stem = new T.Mesh(fixG('stem', ()=>new T.CylinderGeometry(.02,.02,.14,6)), M.steel);
  stem.position.y = yokeR + .06; g.add(stem);
  const jaw = new T.Mesh(fixG('jaw', ()=>new T.BoxGeometry(.1,.13,.08)), M.steel);
  jaw.position.y = yokeR + .18; g.add(jaw);
  const bolt = new T.Mesh(fixG('bolt', ()=>new T.CylinderGeometry(.013,.013,.14,6)), M.steel);
  bolt.rotation.x = Math.PI/2; bolt.position.set(0, yokeR + .11, .06); g.add(bolt);
  const safety = new T.Mesh(fixG('safety', ()=>new T.TorusGeometry(.085,.008,4,10)), M.steel);
  safety.position.set(.06, yokeR + .1, 0); safety.rotation.x = .6; g.add(safety);
  /* RULING DY re-points this at the merged steel shell the jaw becomes part of
     (mergeShell does the re-pointing, so the handle can never name an orphan).
     It is a declaration that this body meets a pipe; nothing reads its
     transform, and something that needed to would have to keep its own mesh. */
  g.userData.clamp = jaw;
  return g;
}
/* a square gel-frame at the nose: four thin bars, not a plate (a plate
   would occlude the lens) */
function addFrame(g, half, z){
  [half, -half].forEach(y=>{
    const b = new T.Mesh(fixG('frH'+half, ()=>new T.BoxGeometry(half*2+.04, .028, .02)), M.steel);
    b.position.set(0, y, z); g.add(b);
  });
  [half, -half].forEach(x=>{
    const b = new T.Mesh(fixG('frV'+half, ()=>new T.BoxGeometry(.028, half*2+.04, .02)), M.steel);
    b.position.set(x, 0, z); g.add(b);
  });
}

/* ---- RULING EG — THE STUDIO KIT ------------------------------------------
   Three kinds a television studio and a film stage light with, and they go in
   the ONE catalogue: a Palace that can order a softlight is not a problem worth
   a second list.  They are built to the letter of bodyFresnel and bodyPar above
   — every geometry through fixG so four hundred bodies share one buffer per
   TYPE, addHang for the yoke / C-clamp / safety, addFrame where a gel frame
   belongs (the panel has none: a bi-colour LED is gelled by dialling it), and
   exactly ONE lensMat piece each so the lens stays repaintable through the
   keyed cache.

   EACH CLOSES WITH mergeShell, WHICH IS NOT OPTIONAL FOR A NEW BODY.  423
   lantern bodies were 54% of the scene's visible drawables at 10.8 draws each
   before RULING DY; three new kinds that skipped the merge would put that
   straight back, one grid at a time.  Merged, each of these is THREE draws —
   one M.fixture shell, one M.steel shell, one lens — which is exactly what a
   par can costs, and tests/workshop.js holds the budget.

   AND NOT ONE NEW MATERIAL BETWEEN THEM.  Every piece wears M.fixture, M.steel
   or a LENSM colour the rig already mints, so RULING EI (every material reaches
   envTrack) is satisfied by CONSTRUCTION rather than by a registration someone
   has to remember — and a missed registration renders with the fog and the
   grade bypassed rather than broken, which is the silent failure INVARIANTS
   names.  The same assertion in tests/workshop.js is what keeps it true.      */

}

/* ============================================================================
   THE STEPLADDER (RULING EK) — the one body you CLIMB.

   There is no climbing code in this game and this adds none.  What the game
   has is `groundAt` (p7): a downward raycast over WALKABLE, called
   `intersectObjects(WALKABLE, true)` — RECURSIVE — and a player who steps up
   onto whatever it finds, up to 0.62m at a time (`tryMove`).  So the ladder
   climbs by BEING walkable, the way the Arc's feature stair does, and its
   0.44m treads clear that step-up with 0.18m in hand.

   IT IS THE ROOT GROUP THAT GOES ON `WALKABLE`, NOT THE TREADS, and that is
   the load-bearing choice:
     - the raycast is recursive, so one entry tests every tread;
     - the children move with the group for free, so a carried ladder carries
       its footholds;
     - it survives `mergeShell` below, which REMOVES each tread mesh from the
       group and puts a merged shell back.  Tread meshes registered
       individually would still be in WALKABLE afterwards while no longer
       being children of anything — a set of frozen footholds hanging at the
       world origin, and no ladder to climb;
     - and it is ONE splice to unpick when the body is destroyed (removeBody).

   THE GOING IS WHAT MAKES IT CLIMBABLE, AND IT IS NOT A LOOK.  `groundAt`
   answers with the HIGHEST surface in a column, from 1.40m above the feet
   down — and the ground-follow in p7 snaps the player straight onto whatever
   it says.  So if two treads overlap in PLAN, standing on the lower one puts
   the higher one in your own column and you are teleported up it; three of
   them overlap and the step is 1.32m, which `tryMove` then refuses in every
   direction, and the ladder is a wall you cannot get on or off.  Measured
   that way on the first build of this body: a near-vertical rake put treads
   2, 3 and 4 all over tread 1.

   The fix is the shape the Arc's feature stair already has — its 0.62m treads
   go 0.62m per step, so the columns TILE — and it is why the going here is
   exactly the tread depth: 0.16m of z per 0.44m of rise, no overlap to
   teleport up and no gap to fall through, from the deck to the platform.
   A 70-degree pitch, which is what a platform stepladder is.

   THE LADDER MUST NOT STAND ON ITSELF.  `updateBodies` settles this body with
   `groundAt(x, z, y + 0.5)`, and groundAt starts its ray another 1.4 up — so
   any of the ladder's own geometry over its OWN ORIGIN COLUMN below local
   y = 1.90 would be found as this ladder's floor, and it would climb itself
   until it ran out of rungs.  Nothing is, and the numbers are named so the
   next person to re-rake it has something to check: the lowest tread that
   reaches back toward the origin is the fourth, whose near edge is z +0.2255,
   and the rear legs and their spreader stand at x +/- 0.30 and never cross
   the x = 0 column at all.  The two pieces that DO cross it — the platform at
   3.05 and the cap at 3.90 — are both above 1.90.  A test drops one from six
   metres and watches it land on the deck.
   ========================================================================== */
const LADDER = {
  H:3.90,             // overall — RULING EK, and it is what sets SS.GRID
  PLAT:3.05,          // the top platform: the surface you stand on
  TREADS:6, RISE:0.44,// six treads, 0.44 apart (tryMove steps up 0.62)
  GOING:0.16,         // z per tread — AND the tread depth, so the plan tiles
  W:0.68,             // tread width; the stiles stand at +/- 0.34
  ZTOP:-0.46,         // the front frame's z where it meets the cap
  BZ0:-0.85, BZ1:-0.20// rear legs, foot to platform
};
/* one raked member: a box whose LOCAL Y runs from (y0,z0) to (y1,z1).  The
   sign is the one TRAPS warns about — rotating (0, L/2, 0) by rx puts the top
   at z = (L/2) sin(rx), so rx is atan2(dz, dy) and NOT its negative. */
function ladMember(g, key, x, y0, z0, y1, z1, w, t, mat){
  const L = Math.hypot(y1 - y0, z1 - z0);
  const m = new T.Mesh(fixG(key, ()=>new T.BoxGeometry(w, L, t)), mat);
  m.position.set(x, (y0 + y1)/2, (z0 + z1)/2);
  m.rotation.x = Math.atan2(z1 - z0, y1 - y0);
  g.add(m);
  return m;
}
/* the front frame's z at a height y, and the rear legs' — the rake is the
   going over the rise, so fz() IS the tiling rule the header describes */
const LAD_TOPY = LADDER.H - 0.035;                    // where the stiles end
function ladFZ(y){
  return LADDER.ZTOP + (LADDER.GOING / LADDER.RISE) * (LAD_TOPY - y);
}
function ladBZ(y){
  return LADDER.BZ0 + (LADDER.BZ1 - LADDER.BZ0) * (y / LADDER.PLAT);
}
function bodyLadder(){
  const g = new T.Group();
  const G = LADDER.GOING;
  [-1, 1].forEach(s=>{
    ladMember(g, 'ladStile', s*0.34, 0, ladFZ(0), LAD_TOPY, LADDER.ZTOP,
              0.06, 0.04, M.galv);
    ladMember(g, 'ladRear',  s*0.30, 0, LADDER.BZ0, LADDER.PLAT, LADDER.BZ1,
              0.05, 0.05, M.galv);
  });
  /* the treads: the TOP of each is the step height, so a tread meant to be
     stood on at 0.44 sits its own thickness below it */
  for(let i = 1; i <= LADDER.TREADS; i++){
    const top = i * LADDER.RISE;
    const tr = new T.Mesh(fixG('ladTread',
      ()=>new T.BoxGeometry(LADDER.W, 0.035, G)), M.galv);
    tr.position.set(0, top - 0.0175, ladFZ(top));
    g.add(tr);
  }
  /* the spreader sits ABOVE 1.40 on purpose: lower, and a floor-level ground
     ray finds it and walls off the space under the ladder */
  const spr = new T.Mesh(fixG('ladSpreader',
    ()=>new T.BoxGeometry(0.60, 0.05, 0.05)), M.galv);
  spr.position.set(0, 1.55, ladBZ(1.55)); g.add(spr);
  /* the platform's TOP is 3.05, and in plan it takes up exactly where the
     sixth tread leaves off and runs back over the legs — tiled, like the
     treads, so stepping onto it is one 0.41m rise and nothing else */
  const pz1 = ladFZ(LADDER.TREADS * LADDER.RISE) - G/2;   // the sixth tread's back edge
  const pz0 = LADDER.BZ1 - 0.10;
  const plat = new T.Mesh(fixG('ladPlatform',
    ()=>new T.BoxGeometry(LADDER.W + 0.04, 0.05, pz1 - pz0)), M.hazard);
  plat.position.set(0, LADDER.PLAT - 0.025, (pz0 + pz1)/2); g.add(plat);
  /* the cap is the tool tray at the very top, and it stands clear of the
     platform's own column so nobody is snapped up onto it */
  const cap = new T.Mesh(fixG('ladCap',
    ()=>new T.BoxGeometry(LADDER.W + 0.06, 0.06, 0.26)), M.hazard);
  cap.position.set(0, LADDER.H - 0.03, LADDER.ZTOP); g.add(cap);
  g.traverse(o=>{ if(o.isMesh){ o.castShadow = o.receiveShadow = true; } });
  /* two merged shells, one per material — the galv frame and the marked
     surfaces — where twelve meshes would have been twelve draws (RULING DY's
     rule, applied to a body that is not a lantern) */
  mergeShell('ladder', g);
  /* and THIS is the climb: the root group, for every reason in the header */
  WALKABLE.push(g);
  return g;
}

/* ---- bodies off the pipe -------------------------------------------------
   The detach system.  Every lantern and PA box is a BODY that can come off
   its hanging POINT.  The point is per-stage state and parks with the board
   (a lantern point IS its FIXTURES record; a speaker point rides SPKBARS);
   the body is VENUE-level and never parks — carry it into the other house
   and it is still wherever you left it.  `state` is the one word for where
   a body is (§5: one flag, one meaning):
     'hung'     on a point (mesh a child of the point's group)
     'held'     in a VR hand (mesh at the venue root, following the grip
                kinematically — so a hold record dropped without a release,
                session end or stage swap, leaves a correctly-parented mesh)
     'slotted'  in a rack, cart or pallet slot (mesh a child of the anchor)
     'loose'    standing wherever it was dropped (mesh at the venue root)   */
var BODIES = [];
var BUILD_VENUE = 'palace';   // buildArcStage flips this while it builds (p2k)
const BODY_LABEL = {profile:'PROFILE SPOT', fresnel:'FRESNEL', par:'PAR CAN',
                    cyc:'CYC UNIT', mover:'MOVING HEAD', blinder:'BLINDER',
                    speaker:'PA BOX',
                    soft:'SOFTLIGHT', panel:'LED PANEL', hmi:'HMI FRESNEL',
                    wood:'WOOD', hinge:'HINGE', track:'TRACK', carriage:'CARRIAGE',
                    paint:'PAINT CAN', table:'WORK TABLE', ladder:'STEPLADDER'};
function makeBodyMesh(kind, color){
  /* BUILD_KINDS is p4c's hoisted var — undefined while the Palace rig builds
     during THIS part's parse-time run, so factory stock never takes this
     branch (and a typeof guard is unnecessary: var, not const) */
  if(BUILD_KINDS && BUILD_KINDS[kind]) return makeBuildMesh(kind, color);
  /* the stepladder is gear, not build stock: it never nails into an assembly
     and never counts against BUILD_CAP, so it is NOT in BUILD_KINDS and it is
     named here.  Without this line it would fall through to the lantern chain
     and every delivered ladder — and every ladder off the SAVE — would come
     back a profile spot, silently. */
  if(kind === 'ladder') return bodyLadder();
  return kind==='fresnel'?bodyFresnel(): kind==='par'?bodyPar():
         kind==='cyc'?bodyCyc(): kind==='mover'?bodyMover():
         kind==='blinder'?bodyBlinder():
         kind==='soft'?bodySoft(): kind==='panel'?bodyPanel(): kind==='hmi'?bodyHmi():
         kind==='speaker'?bodySpeaker(): bodyProfile();
}
/* register a body that is already sitting hung on its point */
function regBody(kind, mesh, point){
  mesh.userData.moves = true;
  const b = {kind, venue:BUILD_VENUE, mesh, state:'hung', point, slot:null};
  BODIES.push(b);
  return b;
}
/* where an un-hung body lives: the venue root, never a room or stage group —
   the cart's rule (p2m), so it neither culls with a shut door nor parks with
   a stage.  Runtime children of `world` sit outside the room groups
   (buildRooms sorted world.children once, at boot), which is the point. */
function venueRoot(v){
  return (v === 'arc' && typeof ARC !== 'undefined' && ARC.group) ? ARC.group : world;
}
/* take a body off its point: the point stays put, and its channel goes dead
   at the one gate in updateRig — the level path, never `visible` (§5) */
function unhangBody(b){
  if(b.state !== 'hung' || !b.point) return b;
  const p = b.point;
  if(!p.spk && p.beam && p.beam.parent !== p.group){
    /* a mover's beam and glow are parented to the body's head; they are
       point equipment, so they stay behind when the body goes */
    p.group.add(p.beam); p.beam.position.set(0,0,0); p.beam.rotation.set(0,0,0);
    p.group.add(p.glow); p.glow.position.set(0,0,.4); p.glow.rotation.set(0,0,0);
  }
  p.body = null;
  venueRoot(b.venue).attach(b.mesh);
  b.point = null; b.state = 'loose';
  return b;
}
/* hang a body on an empty point.  RULING A: the circuit lives in the pipe —
   any lantern body answers whatever the point is patched as.  Speaker points
   are rigging and take only PA boxes (and vice versa). */
/* KIT THAT IS CARRIED BUT NEVER RIGGED.  The last line of canHang is
   PERMISSIVE — "anything that is not a speaker answers a lighting point" —
   which was true while every non-build body was a lantern.  It stopped being
   true with the stepladder (RULING EK): left alone, canHang would happily
   clamp a 3.9m ladder to a lighting bar and fly it out.  A new gear kind that
   is not a lantern goes on this list, deliberately, by name. */
const NO_HANG = {ladder:1};
function canHang(b, p){
  if(p.body) return false;
  /* build stock (p4c) nails to assemblies, never hangs on a patch point */
  if(BUILD_KINDS && BUILD_KINDS[b.kind]) return false;
  if(NO_HANG[b.kind]) return false;
  return p.spk ? b.kind === 'speaker' : b.kind !== 'speaker';
}
function hangBody(b, p){
  if(b.state === 'hung' || !canHang(b, p)) return false;
  if(b.state === 'slotted') unslotBody(b);
  if(p.spk){
    p.bar.group.add(b.mesh);
    b.mesh.position.set(0, p.y, 0);
    b.mesh.rotation.set(p.tilt || 0, 0, 0);
  } else {
    p.group.add(b.mesh);
    b.mesh.position.set(0,0,0); b.mesh.rotation.set(0,0,0);
    if(p.mover && b.mesh.userData.head){
      const h = b.mesh.userData.head;
      h.add(p.beam); p.beam.position.set(0,0,0); p.beam.rotation.set(0,0,0);
      h.add(p.glow); p.glow.position.set(0,0,.4); p.glow.rotation.set(0,0,0);
    }
  }
  p.body = b.mesh;
  b.point = p; b.slot = null; b.state = 'hung';
  return true;
}
function unslotBody(b){
  if(b.state !== 'slotted') return b;
  venueRoot(b.venue).attach(b.mesh);
  if(b.slot) b.slot.userData.body = null;
  b.slot = null; b.state = 'loose';
  return b;
}
function slotBody(b, anchor){
  if(anchor.userData.body || b.state === 'hung') return false;
  anchor.add(b.mesh);
  b.mesh.position.set(0,0,0); b.mesh.rotation.set(0,0,0);
  anchor.userData.body = b;
  b.slot = anchor; b.state = 'slotted';
  return true;
}
/* the squeeze that takes a body, whatever it was doing */
function grabBody(b){
  delete b.frozen;               // the grab takes the park off (RULING N)
  if(b.state === 'hung') unhangBody(b);
  else if(b.state === 'slotted') unslotBody(b);
  else if(b.state === 'seated' && b.station){        // off the saw table (p4c)
    b.station.pieces.splice(b.station.pieces.indexOf(b), 1);
    venueRoot(b.venue).attach(b.mesh);
    b.station = null;
  }
  b.state = 'held';
  /* whatever was resting ON this, or on what this was standing on, may have
     just lost its floor — wake the venue so the rota never has to notice */
  wakeBodies(b.venue);
  return b;
}
/* A settled body is only re-tested on a rota (see updateBodies), so anything
   that can pull the ground out from under one calls this.  The rota is the
   backstop for whatever nobody thought to wake; this is the fast path, and
   it is what keeps the saving invisible. */
function wakeBodies(venue){
  for(const b of BODIES) if(venue === undefined || b.venue === venue) b.rest = false;
}
/* loose bodies settle to the floor under them.  A body still marked 'held'
   that no live hand names has had its hold record dropped without a release
   (session end, stage swap — the p9 discipline): demote it here, which is
   what makes those drops safe. */
const _bodyV = new T.Vector3();
/* A piece that has come to rest is not re-tested every frame.  Every loose
   body used to pay a recursive raycast AND a scan of the whole registry each
   frame just to conclude it had not moved: at BUILD_CAP that was 150 of each,
   ~11% of a 72Hz budget on a desktop and several times that on a headset.
   Resting pieces are now checked on a rota instead, spread by body index so
   the work is level across frames rather than spiking every REST_ROTA-th one.
   grabBody wakes a venue outright, so the rota is only the backstop. */
const REST_ROTA = 12;
var BODY_TICK = 0;
function updateBodies(dt){
  if(!BODIES.length) return;
  BODY_TICK++;
  const inHand = (typeof VR !== 'undefined' && VR && VR.held && VR.held.kind === 'body')
    ? VR.held.body : null;
  let bi = 0;
  for(const b of BODIES){
    bi++;
    if(b.state === 'held' && b !== inHand){
      b.state = 'loose';
      /* a cold drop (session end, stage walk): wood still lies flat */
      if(b.kind === 'wood' && b.restH === undefined && typeof woodRestH === 'function')
        b.restH = woodRestH(b);
    }
    if(b.state !== 'loose') continue;
    /* a parked piece (Y, RULING N) hangs exactly where it was left */
    if(b.frozen) continue;
    if(b.rest && (BODY_TICK + bi) % REST_ROTA) continue;
    b.mesh.getWorldPosition(_bodyV);
    let g = (typeof groundAt === 'function') ? groundAt(_bodyV.x, _bodyV.z, _bodyV.y + 0.5) : 0;
    if(g === null) continue;
    /* a work table's top is a raised piece of deck (p4c, build-usability §5) */
    if(b.kind !== 'table' && typeof tableTopAt === 'function'){
      const tt = tableTopAt(_bodyV.x, _bodyV.y, _bodyV.z, b.venue, b);
      if(tt !== null && tt > g) g = tt;
    }
    /* a table's group origin is its feet: it stands, never floats at 0.25 —
       and a stepladder's origin is its feet too (RULING EK), so it joins the
       table here rather than inventing a second rule.  Doing it in the DEFAULT
       means every route that mints one — the delivery, the save, a test —
       stands it on the deck without having to remember to set restH. */
    const want = g + (b.restH !== undefined ? b.restH
                      : ((b.kind === 'table' || b.kind === 'ladder') ? 0 : 0.25));
    /* both venue roots stand at y=0 unrotated, so local y IS world y */
    if(Math.abs(_bodyV.y - want) > 0.005){
      b.mesh.position.y += clamp(want - _bodyV.y, -3*dt, 3*dt);
      b.rest = false;
    } else b.rest = true;
  }
}

/* ---- the real-light pool ----------------------------------------------
   Every GPU has a hard ceiling on shader uniforms, and a scene with two
   dozen shadow-casting spotlights blows straight through it — every material
   then fails to compile and you get a black screen. So we keep a small,
   FIXED number of actual lights and hand them to whichever fixtures are
   currently doing the most work. Beams, gobo pools and lens glows are drawn
   for all 25 channels regardless, so the rig still reads as fully lit.      */
const LIGHT_POOL = [];
const POOL_SIZE = 8, SHADOW_LIGHTS = 3;
for(let i=0;i<POOL_SIZE;i++){
  const l = new T.SpotLight(0xffffff, 0, 60, 0.3, 0.5, 1);
  l.target = new T.Object3D();
  scene.add(l); scene.add(l.target);
  if(i < SHADOW_LIGHTS){
    l.userData.canShadow = true;
    l.castShadow = true;
    l.shadow.mapSize.set(1024,1024);
    l.shadow.camera.near = 1; l.shadow.camera.far = 62;
    l.shadow.bias = -0.0015; l.shadow.radius = 2;
  }
  LIGHT_POOL.push(l);
}

/* ============================================================================
   RULING DN — GLOW PLANES INSTEAD OF SCREEN-SPACE BLOOM (route 2).

   True bloom needs an EffectComposer; a composer needs its own render target;
   and rendering into our own target forfeits driver FIXED-FOVEATED RENDERING,
   which is the only governor vrPerf has mid-session (p9).  His first headset
   readings came back with foveation already pegged at 1.00 and no travel left,
   so it cannot be spent: bloom would be bought by removing the safety net.  So
   this is an additive camera-facing plane per lit emitter instead — the idiom
   this file already speaks in the beam shader and the two gobo flares below.

   THREE THINGS MAKE IT CHEAP AND ALL THREE ARE LOAD-BEARING:

     - ONE InstancedMesh, one material, one geometry.  ONE DRAW CALL.
     - `count` is TRUNCATED to the lit fixtures.  An additive quad at opacity
       zero still rasterises and still blends, so fading one to nothing saves
       nothing at all; only not drawing it does.  Across a 94-cue plot most of
       the rig is out, so this is most of the saving.
     - A SCREEN-SPACE CLAMP.  World size alone is unbounded: a source a metre
       from the face fills the display, and that is exactly where he stands to
       look at the neon.  §3 of the spec sizes the uncapped case at 1.5–3ms,
       which is the whole cost of route 1 for one lantern.

   Capped again in a session by VR.glowCap, beside VR.beamCap (p9), because
   additive geometry in haze is overdraw and a mobile tile renderer hates
   overdraw most.  Same knob family, same reason — and, like vrCapBeams, the
   cap has to decide WHICH ones and not merely how many.

   TWO THINGS IN THIS FILE ARE CALLED A GLOW, AND THEY ARE NOT THE SAME THING.
   `f.glow` (in addFixture, below) is the hot LENS on the lantern: a 0.55m quad
   parented to the fixture, sitting at local z 0.4 so it clears the housing,
   one mesh per fixture.  `GLOW` here is the HALO: one batch, drawn in front of
   every lit lens, sized in screen space.  Core and halo.  The halo carries the
   same 0.45m offset down the beam for the same reason the lens carries its
   local z — see GLOW_LENS_OUT.

   FOG YES, GRADE NO, AND BOTH DELIBERATELY (INVARIANTS).  The material goes
   through envTrack like every other, which is what hands it the shared
   atmosphere uniforms: a glow seen across a hazy stage should be eaten by the
   haze the same way the lantern behind it is, and that is what the fog chunk
   does.  It is then gradeExempt — a glow plane IS the light, and the surfaces
   it falls on are graded already, so grading the source tints the same photon
   twice (RULING DM, the note beside the two gobo flares says the same).
   ========================================================================== */
const GLOW_MAX_FRAC = 0.22;   // most of the view HEIGHT one glow may ever cover
const GLOW_MIN_LVL  = 0.04;   // below this the fixture is not drawn at all
const GLOW_SIZE     = 1.9;    // world metres at full level, before the clamp
/* OUT OF THE HOUSING, DOWN THE BEAM.  f._org is f.group's world position and
   that point is INSIDE the lantern body — measured, for the first eight
   profiles the body AABB is 0.45-0.89 x 0.79 x 0.70-0.93m and _org is inside
   every one of them.  depthTest is left ON (an additive quad that ignores
   depth hangs in front of the scenery), so a halo centred on _org is occluded
   by the housing it belongs to, and the brightest part of a radial gradient is
   its core — exactly the part that would be buried.  It bites hardest in the
   case this ruling exists for: at a metre the clamp takes the quad down to
   0.30m, well inside a body whose half-diagonal is 0.65m, and the halo would
   simply disappear behind the lantern.  0.45m down f._dir clears it.  This is
   the same fix, for the same reason, as f.glow's local z of 0.4 below. */
const GLOW_LENS_OUT = 0.45;
/* the penalty that sends anything outside the view cone to the back of the
   session cap's queue.  It has to beat any squared distance this world can
   produce: the two buildings together span under 500m, so 250,000 is the whole
   of it and 1e9 is comfortably past. */
const GLOW_BEHIND = 1e9;
/* how much wider than the camera's own vertical half-angle that cone runs —
   see the note in glowUpdate for why it is deliberately generous. */
const GLOW_CONE = 2.2;
/* the allocation, once.  NOT FIXTURES.length: this runs at load, before
   buildRig has put a single fixture in the array, and the array is swapped
   whole by p2k when you walk to another stage — the three rigs are 39 points
   each today and a dropped light (DQ) will make that grow.  An instance costs
   19 floats of buffer whether it is drawn or not; only `count` costs a pixel. */
const GLOW_CAP      = 64;
const GLOW = { mesh:null, max:0 };
function glowBuild(){
  const c = cvs(64, 64), g = c.getContext('2d');
  const rg = g.createRadialGradient(32, 32, 0, 32, 32, 32);
  rg.addColorStop(0.0, 'rgba(255,255,255,1)');
  rg.addColorStop(0.35, 'rgba(255,255,255,0.42)');
  rg.addColorStop(1.0, 'rgba(255,255,255,0)');
  g.fillStyle = rg; g.fillRect(0, 0, 64, 64);
  const mat = new T.MeshBasicMaterial({
    map: new T.CanvasTexture(c), transparent: true,
    blending: T.AdditiveBlending, depthWrite: false, side: T.FrontSide });
  envTrack(mat);                   // RULING DL — the haze reaches the glow too
  gradeExempt(mat);                // RULING DM — additive light, graded once
  GLOW.max = GLOW_CAP;
  const mesh = new T.InstancedMesh(new T.PlaneGeometry(1, 1), mat, GLOW.max);
  mesh.name = 'glow';
  /* r128's InstancedMesh CONSTRUCTOR already sets frustumCulled = false
     (three.js :19632), so this line changes nothing today and no test can make
     it fail.  It is written out anyway because the reason has to survive: r128
     sizes an instanced bounding sphere from the BASE geometry, so a batch whose
     instances move every frame would look like one quad at the origin and be
     culled wholesale (TRAPS, first entry), and later three.js versions dropped
     that constructor default.  Stated here so a future frustumCulled = true is
     a deliberate act rather than an inherited one.  The raycast opt-out is the
     other half of the same rule and IS load-bearing. */
  mesh.frustumCulled = false;
  mesh.raycast = ()=>{};           // never a grab, never a ground hit
  /* PER-INSTANCE COLOUR WORKS ON r128, BUT ONLY BECAUSE THE TWO SHADER
     PREFIXES DISAGREE, AND IT IS WORTH WRITING DOWN.  color_pars_fragment in
     r128 declares vColor for USE_COLOR only — it has never heard of
     USE_INSTANCING_COLOR — so on the chunk text alone this could not reach a
     fragment.  What saves it is WebGLProgram's FRAGMENT prefix, which defines
     USE_COLOR when EITHER vertexColors or instancingColor is set, while the
     VERTEX prefix defines it for vertexColors alone.  Net effect: leave
     vertexColors OFF.  Turning it on would declare the `color` attribute in
     the vertex shader, the geometry has none, WebGL supplies (0,0,0,1) for a
     missing generic attribute, and every glow would render BLACK. */
  mesh.instanceColor = new T.InstancedBufferAttribute(
    new Float32Array(GLOW.max*3), 3);
  /* both buffers are rewritten every frame, so both say so.  r128 does not set
     a usage hint on instanceMatrix itself and the driver hint costs nothing
     either way — but declaring it on one buffer and not the other invites the
     reader to conclude the other one is static, which it is not. */
  mesh.instanceMatrix.setUsage(T.DynamicDrawUsage);
  mesh.instanceColor.setUsage(T.DynamicDrawUsage);
  mesh.renderOrder = 20;
  mesh.count = 0;                  // nothing is lit until updateRig says so
  GLOW.mesh = mesh;
  scene.add(mesh);
}
/* THE VERTICAL HALF-ANGLE, READ OFF THE PROJECTION AND NOT OFF `fov`.
   element 5 of a perspective projection is 1/tan(fovY/2), which for the desk
   camera is exactly Math.tan(camera.fov*DEG/2) and for an ArrayCamera is the
   only honest answer there is — a session's ArrayCamera inherits a `fov` field
   that nothing ever writes.  The fallback is for a camera with no projection
   set up at all, which is a boot-order fault rather than a state to serve. */
function glowTanHalf(cam){
  const e = cam.projectionMatrix && cam.projectionMatrix.elements;
  const p = e ? e[5] : 0;
  return (p > 0.0001 && isFinite(p)) ? 1/p : Math.tan((cam.fov || 60)*Math.PI/360);
}
const _gwm = new T.Matrix4(), _gwp = new T.Vector3(), _gwq = new T.Quaternion(),
      _gws = new T.Vector3(), _gwc = new T.Vector3(), _gwf = new T.Vector3(),
      _gwd = new T.Vector3(), _glit = [];
function glowUpdate(cam){
  const mesh = GLOW.mesh;
  if(!mesh || !cam) return;
  /* the session cap, read the way vrCapBeams reads VR.beamCap — 0 or absent
     means no cap, which is what vrQualityOff leaves behind.  `VR` is a var and
     this guard is safe before p9 has run, the same as the light pool's. */
  const cap = (VR && VR.active && VR.glowCap)
    ? Math.min(VR.glowCap, GLOW.max) : GLOW.max;
  /* WORLD pose, both of them.  camera.position stops being a world point the
     moment VR.rig holds the camera (TRAPS), and camera.quaternion is then the
     head's rotation INSIDE the rig — turning with the right stick spins the
     rig, so a plane built from the local quaternion would swing away from the
     face by exactly the player's yaw.  Asked once, outside the loop. */
  cam.getWorldPosition(_gwc);
  cam.getWorldQuaternion(_gwq);
  const tanHalf = glowTanHalf(cam);
  /* who is lit at all.  One reused array — this runs every frame and a fresh
     one here is thousands of throwaway objects a second (TRAPS, per-frame
     allocation is a lag source rather than a tidiness question). */
  _glit.length = 0;
  for(const f of FIXTURES)
    if((f._lvl || 0) >= GLOW_MIN_LVL) _glit.push(f);
  /* WHICH ONES THE CAP KEEPS, AND NOT MERELY HOW MANY.  vrCapBeams learned
     this the hard way and its comment is the whole lesson: truncating in rig
     order keeps the wrong N.  FIXTURES is in rig order, which starts at the
     FOH bar — so a session cap of 12 taken off the top keeps twelve lanterns
     that are BEHIND a player at downstage centre facing upstage, and the halo
     delivers nothing at all while still paying twelve matrix composes and a
     buffer upload.  Measured with tools/glow.js: 2.6% of screen uncapped,
     0.0% capped, from the one position this ruling exists to serve.

     AND NEAREST ALONE IS NOT THE ANSWER EITHER — it moves the blind spot
     rather than closing it.  From a stalls seat the twelve NEAREST lanterns
     are that same FOH bar, directly overhead and behind the head: pure
     proximity scored 0.0% there where rig order had scored 0.7%.  A halo is a
     screen effect and the cap bounds FILL, so a slot spent on something the
     eye cannot see is spent twice over.

     NOR IS "IN FRONT OF THE EYE PLANE" ENOUGH, which is what this was first
     written as.  A half-space test keeps a lantern 80 degrees off the axis —
     ahead, and comfortably off the screen.  At downstage centre looking up at
     the electrics that is most of the near rig, and the capped session kept
     4% of the glow the eye could actually see.  The test that caught it is in
     tests/vr.js and it is a RATIO against the same seat uncapped, because
     "the cap draws something" is not the claim; "the cap keeps most of what
     you can see" is.

     So the rank is: inside a generous view cone first, nearest of those first,
     everything else last however close.  Nearest is right INSIDE the cone
     because a near halo is clamped and covers GLOW_MAX_FRAC of the height flat
     out, while a far one falls away with distance.  Behind the head is a
     subset of outside-the-cone and needs no separate test.  Two dot products
     and a square root per lit fixture, and only when something has to be
     dropped at all — the desktop path has 39 lit against an allocation of 64
     and never sorts. */
  if(_glit.length > cap){
    _gwf.set(0, 0, -1).applyQuaternion(_gwq);
    /* THE CONE IS DELIBERATELY WIDER THAN THE FRUSTUM WE CAN SEE, for the two
       reasons this file cannot measure its way out of.  The screen's DIAGONAL
       half-angle is about 2.04x the vertical one at 16:9, and a Quest 3's
       vertical is about 2.06x the desk camera's 60 degrees — which is the only
       projection reachable in a session (see the note at the end of updateRig).
       2.2 covers both.  Erring wide costs a slot; erring narrow drops a halo
       the headset can see, and that is the failure with no tell. */
    const cosCone = 1/Math.sqrt(1 + tanHalf*tanHalf*GLOW_CONE*GLOW_CONE);
    for(const f of _glit){
      _gwd.copy(f._org).sub(_gwc);
      const d2 = _gwd.lengthSq();
      const inCone = d2 < 0.0001 ||
        _gwd.dot(_gwf) > cosCone*Math.sqrt(d2);
      f._glowRank = d2 + (inCone ? 0 : GLOW_BEHIND);
    }
    _glit.sort((a,b)=>a._glowRank - b._glowRank);
  }
  const n = Math.min(_glit.length, cap);
  for(let i = 0; i < n; i++){
    const f = _glit[i], lvl = f._lvl;
    /* out of the housing and down the beam — see GLOW_LENS_OUT.  _org and
       _dir are both world and both refreshed at the top of this function. */
    _gwp.copy(f._org).addScaledVector(f._dir, GLOW_LENS_OUT);
    const dist = _gwc.distanceTo(_gwp);
    /* the clamp: GLOW_MAX_FRAC of the view height at THIS distance */
    const lim = dist * tanHalf * 2 * GLOW_MAX_FRAC;
    const sz  = Math.min(GLOW_SIZE * (0.45 + 0.55*lvl), lim);
    _gws.set(sz, sz, 1);
    _gwm.compose(_gwp, _gwq, _gws);
    mesh.setMatrixAt(i, _gwm);
    mesh.instanceColor.setXYZ(i, f.color.r*lvl, f.color.g*lvl, f.color.b*lvl);
  }
  mesh.count = n;
  mesh.instanceMatrix.needsUpdate = true;
  mesh.instanceColor.needsUpdate = true;
}
glowBuild();

/* ============================================================================
   A fixture's AIM is stored in WORLD space.  It has to be: lookAt takes a
   world point, the light pool lives on the scene, and the cue stack snapshots
   the aim and puts it back.  A light plot, on the other hand, is written in
   STAGE coordinates — "point at 0, 7.2, -1" means the middle of the opening
   of whatever stage you are talking about.

   With one theatre the two were the same thing and nobody noticed.  With
   three, a plot written for the Palace aimed every lantern in the Arc back
   across town — which from the audience reads as the whole rig pointing at
   the side wall.
   ========================================================================== */
function stageToWorld(x, y, z, out){
  const v = (out || new T.Vector3()).set(x, y, z);
  if(typeof rigGroup !== 'undefined' && rigGroup){
    rigGroup.updateWorldMatrix(true, false);
    v.applyMatrix4(rigGroup.matrixWorld);
  }
  return v;
}
const _w2s = new T.Matrix4();
function worldToStage(v, out){
  const o = (out || new T.Vector3()).copy(v);
  if(typeof rigGroup !== 'undefined' && rigGroup){
    rigGroup.updateWorldMatrix(true, false);
    o.applyMatrix4(_w2s.copy(rigGroup.matrixWorld).invert());
  }
  return o;
}

/* ---- RULING EG — WHAT THE THREE NEW KINDS ARE, OPTICALLY ------------------
   Every addFixture call in this file spells its five numbers out, because a
   theatre plot is written unit by unit.  The studio kit is ordered from a
   catalogue instead, so the numbers belong to the KIND and are stated here once
   rather than copied into four studio rigs — and RANK AND POWER ARE TWO
   DIFFERENT NUMBERS (RULING BF), so both are written even where they agree.
   Neither is left to default: `hmi`'s rank 3.6 sits deliberately ABOVE the
   profile's 3.1-4.2 band's middle and above its own power 3.4, so a film
   stage's key light wins one of the eight real lights off a background unit,
   which is the whole reason a film stage looks like a film stage.
   A caller still wins: these are defaults under `o`, not an override of it.  */
const STUDIO_OPTICS = {
  soft:  {angle:70, beamRad:5.0, beamLen:12, power:2.2, rank:2.2},
  panel: {angle:60, beamRad:4.2, beamLen:11, power:1.8, rank:1.8},
  hmi:   {angle:24, beamRad:2.0, beamLen:26, power:3.4, rank:3.6}
};
/* A DIFFUSE SOURCE IS A DIFFERENT LIGHT, and the cyc unit has always been the
   one exception in the two lines below: a flood has no hard edge to give, and
   its beam in haze is a wash rather than a shaft.  A softbox and an LED panel
   are the same kind of source, so they join the exception rather than each
   growing a ternary of its own.  The HMI is a FRESNEL — hard daylight, a
   shadow you can read — and is deliberately not here. */
const DIFFUSE_FIX = {cyc:1, soft:1, panel:1};
/* the hot-lens glow sits AT the lens, and the lens is not in the same place on
   every kind: a cyc unit's face is at .24 and the flat studio sources are
   nearer still (a panel is six centimetres deep, so a glow at .4 would hang in
   the air a hand's width in front of it).  Anything not named here keeps the
   .4 the deep-barrelled lanterns were built with. */
const GLOW_Z = {cyc:.24, soft:.16, panel:.09, hmi:.40};
function addFixture(o){
  const f = Object.assign({
    ch:FIXTURES.length+1, name:'', type:'profile', pos:new T.Vector3(), aim:new T.Vector3(),
    angle:20, level:0, color:new T.Color(0xffffff), gobo:0, ls:-1, lsOffset:-0.4, lsX:0,
    beamLen:22, beamRad:2.2, power:2.6, rank:null, audience:false,
    mover:false, pan:0, tilt:-45, panT:0, tiltT:-45, follow:false
  }, STUDIO_OPTICS[o && o.type], o);
  /* RANK AND BRIGHTNESS ARE TWO DIFFERENT NUMBERS (RULING BF).  `power` used to
     do both jobs: it decided WHO got one of the eight real lights and HOW
     BRIGHT that light burned.  RULING BC then pushed the blinders down to 0.9
     for the first job — rightly, so an audience unit can never pull a light off
     the stage — and capped their brightness at 0.9 as a side effect nobody
     could see, because nothing in this repo has eyes.  `rank` is the sort key;
     `power` is the intensity.  It defaults to `power`, so every stage channel
     in all three houses is unchanged to the byte. */
  if(f.rank === null || f.rank === undefined) f.rank = f.power;

  const grp = new T.Group(); grp.position.copy(f.pos); rigGroup.add(grp);
  // the plot gave the aim in stage coordinates; from here on it is world
  stageToWorld(f.aim.x, f.aim.y, f.aim.z, f.aim);
  const body = makeBodyMesh(f.type);
  grp.add(body);
  f.group = grp; f.body = body;
  /* the point is this record; the lantern on it is a detachable body */
  regBody(f.type, body, f);

  const beam = makeBeam(f.beamLen, f.beamRad);
  (f.type==='mover' ? body.userData.head : grp).add(beam);
  f.beam = beam;

  // NOTE: fixtures do not own a WebGL light. A small fixed pool of real
  // SpotLights (below) is re-assigned every frame to whichever fixtures are
  // brightest. A fixed light count means the shaders compile once and stay
  // well inside every GPU's uniform budget.
  f._org = new T.Vector3(); f._dir = new T.Vector3(0,0,1); f._lvl = 0; f._live = false;

  // floor pool decal
  const pool = new T.Mesh(new T.PlaneGeometry(1,1),
    new T.MeshBasicMaterial({map:GOBOS[0], transparent:true, blending:T.AdditiveBlending,
      depthWrite:false, opacity:0}));
  pool.rotation.x = -Math.PI/2; pool.position.y = 0.012;
  pool.raycast = ()=>{};
  gradeExempt(pool.material);      // RULING DM — additive light, graded once not twice
  poolGroup.add(pool); f.pool = pool;

  // hot lens glow
  const glow = new T.Mesh(new T.PlaneGeometry(.55,.55), new T.MeshBasicMaterial({
    map:GOBOS[0], transparent:true, blending:T.AdditiveBlending, depthWrite:false, opacity:0}));
  glow.position.z = GLOW_Z[f.type] !== undefined ? GLOW_Z[f.type] : .4;
  glow.raycast = ()=>{};
  gradeExempt(glow.material);      // RULING DM — additive light, graded once not twice
  (f.type==='mover'?body.userData.head:grp).add(glow); f.glow = glow;

  FIXTURES.push(f);
  return f;
}

/* ============================================================================
   RULING DQ — DROPPABLE LIGHTS, AND THE SLOT COUNT IS TOLD THE TRUTH.

   Roblox lets you drop unlimited PointLight / SpotLight / SurfaceLight.  This
   engine has POOL_SIZE 8 real lights, SHADOW_LIGHTS 3 of them shadow-capable,
   fewer still in a headset (VR.lightCap), and RANK decides who gets one
   (RULING BC, load-bearing).  So a dropped light is a REAL FIXTURE that joins
   the rank queue like any lantern — not a second lighting system with its own
   rules — and every promise this facade makes is one the pool can keep:

     * `shadows` IS A REQUEST.  shadowWanted is what was asked for;
       shadowGranted is what the pool did, written where the pool actually
       hands out (updateRig) and cleared with the slot.  A checkbox that
       silently does nothing is worse than no checkbox, so it is answered in
       BOTH directions: a light that asks for no shadow stops the slot casting
       while it holds it, and a light that asked for one is TOLD when it did
       not get it.
     * THE CONE ANGLES ARE WHAT ACTUALLY RENDERS.  updateRig clamps a pool
       light's cone at 1.15 rad (65.9 degrees), so a table quoting Roblox's
       360-degree point light would be the same silent lie in another place.
       A SurfaceLight is a wide-angle spot for a related reason: r128 has no
       area light in the standard material, and faking one with four spots
       would eat half the pool.
     * NO NEW SORT KEY.  RULING BF's default stands — rank falls out of power
       — so a dropped light competes on how bright it is, exactly like every
       lantern in the rig.  DROP_POWER 2.0 is chosen for that: it sits at the
       bottom of the stage rig's range (the cyc units burn 2.0) and well over
       the audience rig's 0.9 / 0.8, so a default drop can never pull a real
       light off a lit stage — BC untouched — while turning it up is a
       deliberate act with a cost lightSlots() reports.
       (The plan proposed per-kind ranks of 0.45-0.6.  That is BELOW the
       audience rig, which would have made a dropped light the last thing in
       the building to get a real light — a checkbox that does nothing, by
       another road.  Rejected for that reason.)

   POSITION AND AIM ARE IN STAGE COORDINATES, like a light plot and like every
   addFixture call in buildRig: the group goes into rigGroup and addFixture
   runs the aim through stageToWorld itself.  With three stages that is the
   difference between lighting the Arc and aiming back across town (INVARIANTS).

   AND NOTHING PARKS IN p2k.  A dropped light rides FIXTURES, its group rides
   rigGroup and its floor pool rides poolGroup — all three already swap per
   stage — so this facade deliberately keeps NO registry of its own.  `dropped`
   on the record is what identifies one; never a count and never an index
   (TRAPS).  tests/stages.js holds the swap to it.
   ========================================================================== */
const DROP_POWER = 2.0;                    // see the rank note above
const LIGHT_KINDS = {
  PointLight:   {type:'par',     angle:65, beamRad:5.2, beamLen:12},
  SpotLight:    {type:'profile', angle:32, beamRad:2.4, beamLen:20},
  SurfaceLight: {type:'fresnel', angle:58, beamRad:4.6, beamLen:14}
};
function lightAdd(o){
  o = o || {};
  const kind = LIGHT_KINDS[o.kind] ? o.kind : 'SpotLight';
  const k = LIGHT_KINDS[kind];
  /* BOTH VECTORS ARE MINTED FRESH, and it is not tidiness: addFixture TAKES
     OWNERSHIP of the pos and aim it is handed — Object.assign copies the
     reference and stageToWorld then rewrites the aim IN PLACE — so a caller
     that passes a Vector3 it still holds finds its own vector silently turned
     into world coordinates.  (A negative check that copied the local aim back
     over f.aim proved to be a no-op for exactly this reason: they were the
     same object all along.) */
  const p = o.pos || {}, a = o.aim || {};
  const pos = new T.Vector3(p.x || 0, p.y === undefined ? 6 : p.y, p.z || 0);
  /* with no aim given it points at the deck under itself, which is where a
     light dropped in mid-air is looking */
  const aim = new T.Vector3(a.x === undefined ? pos.x : a.x, a.y || 0,
                            a.z === undefined ? pos.z : a.z);
  // lookAt of a zero-length direction is a coin toss; never hand it one
  if(aim.distanceToSquared(pos) < 1e-6) aim.z -= 1;
  /* a body built while the board stands in the Arc is Arc property — the same
     flip buildArcStage does (p2k), because the venue tag is what decides where
     the lantern stands if it is ever taken off its point */
  const keepVen = BUILD_VENUE;
  const st = (typeof STAGES !== 'undefined') ? STAGES[STAGE] : null;
  if(st && st.venue) BUILD_VENUE = st.venue;
  const f = addFixture({
    name: o.name || (kind + ' ' + (FIXTURES.length + 1)),
    type: k.type, pos, aim,
    angle:   o.angle   === undefined ? k.angle   : o.angle,
    beamLen: o.beamLen === undefined ? k.beamLen : o.beamLen,
    beamRad: o.beamRad === undefined ? k.beamRad : o.beamRad,
    power:   o.power   === undefined ? DROP_POWER : o.power,
    rank:    o.rank    === undefined ? null : o.rank,      // null => power (BF)
    /* dropped ON, the way a Roblox light is Enabled when you make one */
    level:   o.level   === undefined ? 1 : clamp(o.level, 0, 1),
    color:   o.color   === undefined ? new T.Color(0xffffff) : new T.Color(o.color)
  });
  BUILD_VENUE = keepVen;
  f.dropped = true;
  f.kind = kind;
  f.shadowWanted  = !!o.shadows;
  f.shadowGranted = false;          // only the hand-out in updateRig can grant it
  f.homeAim = f.aim.clone();        // the focus it was rigged with, as buildRig gives
  /* this mints geometry and materials after boot, so it registers itself —
     INVARIANTS, and the failure would be silent (atmMix and gradeMix are 0 on
     a material that missed the walk, so it renders as if nothing had changed) */
  if(typeof envRegister === 'function'){ envRegister(f.group); envRegister(f.pool); }
  return f;
}
/* the facade strikes what the facade dropped: the rig is the building's, and
   a UI that can delete channel 1 is a UI that will */
function lightRemove(f){
  if(!f || !f.dropped) return false;
  const i = FIXTURES.indexOf(f);
  if(i === -1) return false;
  FIXTURES.splice(i, 1);
  f._live = false; f.shadowGranted = false;
  /* its lantern is filed in BODIES like every other; removeBody unpicks the
     filing with the machinery that owns it (p4c).  A body already taken OFF
     the point is loose gear standing in the room by then — it stays. */
  const b = BODIES.find(x=>x.point === f);
  if(b) removeBody(b);
  if(f.group && f.group.parent) f.group.parent.remove(f.group);
  if(f.pool && f.pool.parent) f.pool.parent.remove(f.pool);
  /* beam, glow and floor pool are minted per fixture, not shared like the
     body geometry — so these are ours to let go of */
  [f.beam, f.pool, f.glow].forEach(m=>{
    if(!m) return;
    if(m.geometry) m.geometry.dispose();
    if(m.material){ ENV_MATS.delete(m.material); m.material.dispose(); }
  });
  /* chan(n) is FIXTURES[n-1], so a hole in the middle would leave every drop
     after it answering to a number that is not its own */
  FIXTURES.forEach((x, n)=>{ x.ch = n + 1; });
  return true;
}
/* how many of the pool are in play THIS FRAME — the headset caps it (p9).
   updateRig hands out against this same function, so the report cannot drift
   away from the hand-out. */
function lightCapNow(){
  return (VR && VR.active && VR.lightCap)
    ? Math.min(VR.lightCap, LIGHT_POOL.length) : LIGHT_POOL.length;
}
/* the pool, counted rather than promised */
function lightSlots(){
  let granted = 0, shadowed = 0, dropped = 0;
  for(const f of FIXTURES){
    if(f._live) granted++;
    if(f.shadowGranted) shadowed++;
    if(f.dropped) dropped++;
  }
  return {total: LIGHT_POOL.length, shadow: SHADOW_LIGHTS, inSession: lightCapNow(),
          granted, shadowed, dropped};
}

/* ---- the FOH bar --------------------------------------------------------
   The six FOH lanterns hang on a bar the board can raise and lower, not on
   air.  One bar per stage: buildRig puts it up inside whichever rigGroup is
   being built, so its whole state is a height in STAGE coordinates and
   nothing about it can leak a building over at the Arc.  FOHBAR names the
   LIVE stage's bar the way FIXTURES holds the live rig — p2k swaps it with
   the rest of the board.  A var on purpose: typeof does not protect a const
   declared later in the file (§5), and other parts reach for this. */
var FOHBAR = null;
const FOH_BAR_SPEED = 0.4;      // m/s — a sedate motor, not a fly cue
const FOH_BAR_STEP  = 1.2;      // one press on either board moves it this far
function fohBarPose(b){
  b.group.position.y = b.y;
  // the drop wires stretch up to their anchor, the way lift lines do
  const len = Math.max(.05, b.wireTop - b.y);
  b.wires.forEach(wr=>{ wr.scale.y = len; wr.position.y = len/2; });
}
function fohBarTo(y){
  if(!FOHBAR) return false;
  FOHBAR.target = clamp(y, FOHBAR.min, FOHBAR.max);
  return true;
}
function fohBarStep(s){
  if(!FOHBAR) return false;
  return fohBarTo(FOHBAR.target + s*FOH_BAR_STEP);
}

/* ---- the speaker bars ----------------------------------------------------
   L+R flown PA beside the arch, one pair per stage — the FOH bar pattern
   run twice: same pose math (fohBarPose is shape-generic), same park in
   p2k, same two boards.  A var for the same reason FOHBAR is.            */
var SPKBARS = null;
const SPK_BAR_SPEED = 0.4;
const SPK_BAR_STEP  = 1.2;
/* six boxes to an array (the owner sized them up from three, 2026-08-07),
   half a metre apart under the pipe; the bottom of the hang toes in the way
   a real J-array does */
const SPK_N = 6, SPK_TOP = -.34, SPK_STEP = .5;
const SPK_TILT = [0, 0, 0, .07, .15, .26];
const SPK_DROP = -SPK_TOP + (SPK_N-1)*SPK_STEP;   // pipe to the lowest box centre
function spkBarTo(side, y){
  if(!SPKBARS || !SPKBARS[side]) return false;
  const b = SPKBARS[side];
  b.target = clamp(y, b.min, b.max);
  return true;
}
function spkBarStep(side, s){
  if(!SPKBARS || !SPKBARS[side]) return false;
  return spkBarTo(side, SPKBARS[side].target + s*SPK_BAR_STEP);
}

function updateRig(dt, t){
  const master = RIG.blackout ? 0 : RIG.grand;
  _active.length = 0;
  /* where this stage's deck actually is, so the floor pools land on it */
  let _sx = 0, _sz = 0;
  if(typeof STAGES !== 'undefined' && STAGES[STAGE]){
    const st = STAGES[STAGE];
    _sx = (st.venue === 'arc' && typeof ARC !== 'undefined' ? ARC.X : 0) + st.cx;
    _sz = st.zPros;
  }
  /* the FOH bar rides toward its target at motor speed.  Only the live
     stage's bar is here to tick — a parked stage's waits, like its rail */
  if(FOHBAR && FOHBAR.y !== FOHBAR.target){
    FOHBAR.y += clamp(FOHBAR.target - FOHBAR.y, -FOH_BAR_SPEED*dt, FOH_BAR_SPEED*dt);
    fohBarPose(FOHBAR);
  }
  if(SPKBARS) for(const k in SPKBARS){
    const b = SPKBARS[k];
    if(b.y !== b.target){
      b.y += clamp(b.target - b.y, -SPK_BAR_SPEED*dt, SPK_BAR_SPEED*dt);
      fohBarPose(b);
    }
  }
  for(const f of FIXTURES){
    // electrics carry their fixtures
    if(f.ls >= 0){
      const ls = FLY[f.ls];
      f.group.position.set(f.lsX, ls.pos + f.lsOffset, ls.z);
      if(!f.mover) f.group.lookAt(f.aim);
    } else if(!f.mover){
      f.group.lookAt(f.aim);
    }
    if(f.mover){
      f.pan  += (f.panT  - f.pan ) * Math.min(1, dt*3.2);
      f.tilt += (f.tiltT - f.tilt) * Math.min(1, dt*3.2);
      /* RULING A allows a headless body in a mover's clamp (it lights, it
         just can't articulate) — and an empty point has no body at all */
      if(f.body && f.body.userData.head){
        f.body.userData.yoke.rotation.y = f.pan*DEG;
        f.body.userData.head.rotation.x = (f.tilt+90)*DEG;
      }
    }
    if(f.follow && followTarget){
      _v.copy(followTarget); _v.y = clamp(_v.y,0.2,3.2);
      f.aim.copy(_v); f.group.lookAt(f.aim);
    }
    /* THE gate.  An empty point's channel drives nothing: cues, subs and
       plots still write f.level, they just land on air, like a real
       unplugged circuit.  Everything downstream — beam, glow, pool, the
       real-light pool, the ambient bed — keys off lvl/_active, so this one
       line is the whole of channel-dead.  Never gate with `visible` (§5). */
    const lvl = f.body ? clamp(f.level,0,1) * master : 0;
    f._lvl = lvl;

    // where this fixture emits from, and where it's pointing
    const headed = f.mover && f.body && f.body.userData.head;
    const src = headed ? f.body.userData.head : f.group;
    src.updateMatrixWorld(true);
    src.getWorldPosition(f._org);
    if(headed){ src.getWorldDirection(_v2); f._dir.copy(_v2).normalize(); }
    else { f._dir.copy(f.aim).sub(f._org).normalize(); }
    if(lvl > 0.004) _active.push(f);

    // beam
    const u = f.beam.material.uniforms;
    u.uColor.value.copy(f.color);
    u.uInt.value = lvl * (DIFFUSE_FIX[f.type] ? 0.30 : 0.62);
    u.uHaze.value = 0.25 + hazeNow()*1.15;
    u.uTime.value = t;
    u.uUseGobo.value = f.gobo>0 ? 1 : 0;
    u.uGobo.value = GOBOS[f.gobo];
    f.beam.visible = lvl > 0.004 && hazeNow() > 0.005;
    // lens glow — camera.position stops being a world point once the VR rig
    // holds the camera, so ask for the world pose
    /* RULING DR — AND A LENS QUAD THAT IS DARK DOES NOT DRAW.  Opacity was the
       only thing this quad was ever told: an additive plane at opacity 0 still
       rasterises and still costs a draw call, so only `visible` skips it, and
       the whole rig's lenses were being submitted through every blackout — 39
       draws an eye to blend nothing.  RULING DN made the same point about the
       halo, which is why that batch truncates `count` rather than fading
       instances away.  Same threshold and same idiom as the beam above and the
       floor pool below; one gate, so desk and VR both get it (updateRig is the
       only place either one runs).  This is a DRAWING flag on a decal that is
       already raycast-opted-out, not a channel gate — §5's rule stands. */
    f.glow.visible = lvl > 0.004;
    f.glow.material.opacity = Math.min(1, lvl*1.3);
    f.glow.material.color.copy(f.color);
    if(lvl > 0.004) f.glow.lookAt(camera.getWorldPosition(_glowCam));
    // floor pool
    const pool = f.pool;
    if(lvl > 0.004){
      const origin = f._org, dir = f._dir;
      if(dir.y < -0.06){
        const dist = (0 - origin.y)/dir.y;
        const hit = origin.clone().addScaledVector(dir, dist);
        if(Math.abs(hit.x - _sx)<16 && hit.z < _sz + D.apron+1 && hit.z > _sz + D.backWall){
          const r = Math.max(.6, Math.tan(f.angle*DEG)*dist*2.1);
          pool.visible = true;
          pool.position.set(hit.x, 0.014, hit.z);
          pool.scale.set(r, r/Math.max(.25,-dir.y), 1);
          pool.rotation.z = Math.atan2(dir.x, dir.z);
          pool.material.map = GOBOS[f.gobo];
          pool.material.color.copy(f.color);
          pool.material.opacity = lvl*0.30;
        } else pool.visible=false;
      } else pool.visible = false;
    } else pool.visible = false;
  }

  /* hand the real lights to the fixtures currently doing the most work.
     In a VR session only VR.lightCap of them (p9): every live SpotLight is
     per-pixel work in every material, and a headset draws the room twice.
     `VR` is a var and this guard is safe before p9 has run.

     The sort key is RANK, not power (RULING BF) — who deserves a light and how
     bright it burns are different questions now. */
  _active.sort((a,b)=> (b._lvl*b.rank) - (a._lvl*a.rank));
  /* RULING DQ — the shadow flag goes out with the slot it was granted for */
  FIXTURES.forEach(f=>{ f._live = false; f.shadowGranted = false; });
  const nLights = lightCapNow();

  /* THE AUDIENCE RIG IS GUARANTEED A SHARE (RULING BG).  Rank alone leaves the
     audience units last by design, and last means nothing: six movers against
     six FOH lanterns took 2 of 8 in the pre-show and NONE of the 4 a headset
     hands out, so the owner's purple sweep ran perfectly and lit nobody.

     So up to AUD_LIGHT_RESERVE of the pool go to the best-ranked lit fixtures
     aimed into the house.  It is a ceiling and not a floor — two, never three,
     however many are lit — and it claims nothing at all when the audience rig
     is dark, which across the 94-cue plot is almost every cue.  No new lights:
     the pool is the size it always was, this is only who stands in it. */
  /* AND IT DIVIDES THE RIG RATHER THAN CAPPING IT (RULING BL).  BG handed the
     reserve out by rank, first come — and the blinders ARE audience units, at
     rank 0.9 against the movers' 0.8.  So at 1:03, where eight blinders sit red
     and six movers sweep purple, the two reserved slots went to blinders, the
     other six went to blinders, and the purple sweep the owner asked about
     rendered on ZERO real lights.  In a headset, four blinders and no movers.
     Measured; the probe is tools/audience-balcony.js.

     So the share goes round the audience GROUPS a lap at a time — the best-lit
     blinder, then the best-lit mover, then round again — instead of down one
     rank list.  Grouped by `type`, which is the only thing that distinguishes
     an audience unit from another kind of audience unit.  With one group lit it
     is exactly what it was; with two it can no longer starve one of them.

     BG's own invariant is untouched: still at most AUD_LIGHT_RESERVE, still a
     ceiling and never a floor, still nothing at all claimed when the rig is
     dark.  This decides WHICH of the two, not how many.                      */
  _handout.length = 0;
  {
    let nAud = 0;
    const _lap = [];
    while(nAud < AUD_LIGHT_RESERVE && _handout.length < nLights){
      let took = false;
      for(const f of _active){
        if(!f.audience || _handout.indexOf(f) !== -1) continue;
        if(_lap.indexOf(f.type) !== -1) continue;   // this group has had its turn
        _lap.push(f.type);
        _handout.push(f); nAud++; took = true;
        if(nAud >= AUD_LIGHT_RESERVE || _handout.length >= nLights) break;
      }
      if(!took) break;                              // no audience unit left lit
      _lap.length = 0;                              // next lap round the groups
    }
    for(const f of _active){
      if(_handout.length >= nLights) break;
      if(_handout.indexOf(f) === -1) _handout.push(f);
    }
  }
  /* RULING DS — THE SHADOW-CAPABLE SLOTS GO TO THE FIXTURES THAT ASKED.
     In r128 the VALUE of a light's castShadow is part of the lights-state hash,
     so flipping one re-acquires a program for every lit material and the first
     flip COMPILES the new NUM_SPOT_LIGHT_SHADOWS variant — a multi-ms hitch.
     RULING DQ's drops can ask for no shadow, so a refuser churning in and out
     of a shadow-capable slot flips that hash over and over.  What defuses it is
     THIS REORDER, keeping refusers out of the three casting slots in the first
     place; the change-only write below is not the defusal (a same-value
     assignment was always engine-free) — it exists so "a steady rig writes
     nothing" is a pinnable invariant, and survives castShadow ever growing a
     setter.

     IT IS A PERMUTATION AND NOTHING MORE.  The hand-out list is untouched in
     length and membership — RANK still decides who gets a light at all (RULING
     BC, load-bearing), and the reserve above still decides how many go to the
     audience rig (BG/BL) — so exactly the same slot INDICES stay occupied and
     only who sits in them changes.  It deliberately does not push a refuser out
     into a slot the hand-out was not already using: that would leave an idle
     shadow slot casting for nobody, and DQ's own case (one refusing drop alone
     in a dark house, holding slot 0 and stopping it casting) is pinned by a
     test.  So the hash still moves when it genuinely must — more askers than
     shadow slots, or nothing but shadow slots in play — and not otherwise.

     Three classes, in this order: asked for one (shadowWanted true), no opinion
     at all (a rigged lantern, undefined — the quality preset owns its casting),
     asked for none (false).  Stable inside each class, so rank order survives.
     The whole thing is skipped unless somebody in the hand-out has an opinion,
     which is to say unless a dropped light is holding a slot: the rig on its own
     runs exactly the code it ran before.                                     */
  if(_handout.length > 1){
    let opinion = false;
    for(let i=0;i<_handout.length;i++)
      if(_handout[i].shadowWanted !== undefined){ opinion = true; break; }
    if(opinion){
      _shadowSort.length = 0;
      for(let pass=0; pass<3; pass++)
        for(let i=0;i<_handout.length;i++){
          const f = _handout[i];
          const cls = f.shadowWanted === true ? 0 : f.shadowWanted === undefined ? 1 : 2;
          if(cls === pass) _shadowSort.push(f);
        }
      for(let i=0;i<_handout.length;i++) _handout[i] = _shadowSort[i];
    }
  }
  for(let i=0;i<LIGHT_POOL.length;i++){
    const l = LIGHT_POOL[i], f = _handout[i] || null;
    if(!f){ l.intensity = 0; continue; }
    f._live = true;
    /* RULING DQ — THE SHADOW IS ANSWERED HERE, where the pool actually hands
       out, and nowhere else.  A rigged lantern has no shadowWanted at all and
       keeps the pool's own casting (the quality preset owns that, p2); a
       dropped light that asked for none gets none while it holds this slot.
       shadowGranted is then the plain truth — is this fixture casting? — which
       is false for every slot past the third, and false for the whole rig at a
       quality level with shadows switched off. */
    /* RULING DS — AND IT IS WRITTEN ONLY WHEN IT CHANGES.  In r128 a CHANGE is
       what costs (the lights-state hash carries castShadow, so a flip
       re-acquires a program for every lit material) while a same-value
       assignment is free — the guard is here so "a steady rig writes nothing"
       can be pinned by a test, not because the old write cost anything.
       Coerced to a boolean because the stored value is one and QUALITY.shadows
       need not be. */
    if(l.userData.canShadow){
      const cast = !!(QUALITY.shadows && f.shadowWanted !== false);
      if(l.castShadow !== cast) l.castShadow = cast;
    }
    f.shadowGranted = !!l.castShadow;
    l.position.copy(f._org);
    l.target.position.copy(f._org).addScaledVector(f._dir, Math.max(4, f.beamLen));
    l.target.updateMatrixWorld();
    l.color.copy(f.color);
    l.angle = clamp(f.angle*DEG, 0.06, 1.15);
    l.penumbra = DIFFUSE_FIX[f.type] ? 0.85 : 0.45;
    l.distance = f.beamLen*2.4;
    l.intensity = f._lvl * f.power * (0.7 + 0.3*(1-hazeNow()*0.5));
  }

  // house / work / practicals
  const hl = HOUSE.house * (RIG.blackout?0:1);
  houseLights.forEach(l=>{ l.intensity = hl*2.2; });
  if(CHANDELIER){
    CHANDELIER.light.intensity = hl*3.4;
    M.bulb.color.setRGB(0.10+hl*0.90, 0.07+hl*0.78, 0.04+hl*0.56);
  }
  workLights.forEach(l=>{ l.intensity = HOUSE.work*2.6; });

  /* front of house and back of house are their own circuits */
  /* RULING DO — ClockTime is kept from Roblox's Lighting because the ARC CENTRE
     has a GLAZED FOYER, a room you can stand in where the hour means something.
     The Palace is a windowless Broadway house and is deliberately untouched.
     It MULTIPLIES the foyer's own level rather than replacing it, so a cue that
     sets HOUSE.lobby still wins the argument about how bright the room is and
     the clock only says how much of that is daylight.  It is a LEVEL, not a sun
     position: there is no sun in this engine to angle, and the panel says so
     rather than implying one. */
  const fo = HOUSE.lobby * (VENUE === 'arc' ? houseDaylight(HOUSE.clock) : 1),
        bo = HOUSE.backstage;
  if(FOH.lamps) FOH.lamps.forEach(l=>l.intensity = fo*2.4);
  M.bulbFoh.color.setRGB(0.10+fo*0.90, 0.07+fo*0.74, 0.04+fo*0.55);
  if(BOH.light){ BOH.light.intensity = bo*2.0; BOH.light2.intensity = bo*1.7; }
  M.bulbBoh.color.setRGB(0.08+bo*0.79, 0.10+bo*0.81, 0.12+bo*0.83);
  M.bulbRun.color.setRGB(0.02+bo*0.14, 0.04+bo*0.28, 0.08+bo*0.92);

  /* how much light is in the room right now? */
  /* The ambient bed is what stops a dark room reading as a void — but it has
     to come only from light that is actually in this room.  The lobby and the
     dock have their own lamps and used to feed the bed as well, which meant
     the auditorium never went properly black with everything off.          */
  /* The ambient bed used to be fed by the rig as well as the house, which
     meant a bright stage lit the whole auditorium — every seat and every
     inch of plaster came up with it.  Stage light belongs on the stage: the
     spotlights do that job on their own.  So the bed comes almost entirely
     from the house, the work light and the practicals, and the rig only
     lifts it a hair off the floor.                                        */
  let rigOut = 0;
  for(const f of FIXTURES) rigOut += f._lvl * f.power;
  const houseOut = (hl + HOUSE.work + HOUSE.practical) * 3;
  RIG.output = rigOut + houseOut;
  const bedHouse = Math.min(1, houseOut / 3.2);
  const bedRig   = Math.min(1, rigOut / 16);
  const bed      = Math.max(bedHouse, bedRig * 0.18);   // for the fog and the walls
  ambient.intensity = bedHouse * 0.34 + bedRig * 0.05;
  hemi.intensity    = bedHouse * 0.28 + bedRig * 0.04;
  M.exit.color.setRGB(0.02+bed*0.10, 0.10+bed*0.76, 0.05+bed*0.30);
  scene.fog.color.setRGB(bed*0.02, bed*0.024, bed*0.034);
  /* RULING DK — the environment is ambience, so it follows the light bed.  ONE
     bed, for the whole world: `bed` is deliberately the AUDITORIUM's, excluding
     the lobby and the dock (RULING BH again, a few lines up), while the PMREM
     is a single texture shared by every metal in both buildings — so the shed
     and the lobby ride the auditorium's level.  Never worse than before DK,
     when there was no environment at all, but it is not a per-room claim and
     must not be written as one.  A fixed environment lights a blackout as
     brightly as a full stage, which is the exact fault RULING BH fixed for the
     ambient.  r128 has no one number that scales an environment, so the drive
     is per material and envDrive walks the ONE registry p2 collected.
     RULING DT narrowed that registry to the METALS — the texture is
     material.envMap now rather than scene.environment, so a deck and a drape
     have no environment to scale and are not on the list.  This write is
     unchanged: it is still one number for the whole world, and it still has to
     follow the bed, because a blackout must not leave a gold proscenium bright
     (BH's principle, on the surfaces that still sample). */
  ENV_LIVE = ENV_INTENSITY * (0.25 + 0.75*bed);
  envDrive();
  /* RULING DL — the haze thickens as the room darkens, and the glare rides the
     RIG rather than the bed, so a full stage in smoke blooms at its edges and a
     blackout does not.  The uniform objects are shared by every fogged material
     (p2), so these are three writes for the whole world, not three per
     material — which is why this sits beside envDrive rather than inside it. */
  ATM.u.atmHaze.value  = ATM.haze * (1.35 - 0.35*bed);
  ATM.u.atmGlare.value = ATM.glare * Math.min(1, bedRig*1.4);
  ATM.u.atmHeight.value = ATM.height;
  /* RULING DM — nothing animates the grade today; it is his to tune from the
     LIGHTING panel (DO).  The write lives HERE anyway so that when a cue does
     reach for it there is one place it is applied, not a second one to find.
     RULING DU — the four writes are now one call that composes the matrix
     (p2).  Once per FRAME rather than on change, deliberately: the LIGHTING
     rows (p7) and their headset twins (DP) set GRADE.sat and GRADE.contrast
     straight onto the object with no notification of any kind, so change-driven
     recomposition would mean a hook on every surface that can move a grade
     value — three today and one more each time a cue or a page reaches for it,
     with a silently stale picture as the failure.  Per frame it is one 3x3 for
     the whole world, ~30 flops, against the ~22 ALU PER PIXEL it removes. */
  gradeCompose();
  scene.background.setRGB(bed*0.014, bed*0.017, bed*0.028);
  FLY.forEach(ls=>{ if(ls.practical){ ls.practical.light.intensity = HOUSE.practical*3.2;
    ls.practical.bulbs.forEach(b=>b.material.color.setRGB(
      0.2+HOUSE.practical*0.8, 0.12+HOUSE.practical*0.65, 0.05+HOUSE.practical*0.45)); } });
  /* RULING DN — the glow planes, LAST, off the _org and _lvl this function has
     just finished refreshing for every fixture.

     `camera` and not some session camera, and that is a decision rather than a
     default.  In an XR frame r128's WebXRManager writes the head pose into the
     game camera's matrix (getCamera copies cameraVR.matrix into it and
     decomposes) but it never touches camera.fov and never touches
     camera.projectionMatrix — only the ArrayCamera's, and the ArrayCamera is
     built inside the manager and is not reachable from here without calling
     into it mid-frame.  So the clamp is computed against the DESK frustum in a
     session too, and the size of that is worth writing down rather than waving
     at: the desk camera is 60 degrees, tanHalf 0.577, against a Quest 3 per-eye
     vertical of roughly 90-96 degrees, tanHalf 1.00-1.19.  The applied limit is
     therefore 48-58% of the intended one and a clamped halo is 42-52% SMALLER
     in the headset than on the monitor.  Direction is safe — it binds harder in
     the place it exists to protect, and VR.glowCap is the session's real
     governor — but it is a visible difference at close range, not a rounding
     error, and it is on the headset checklist as one. */
  glowUpdate(camera);
}
const HOUSE = { house:0.7, work:0.0, practical:0.0, lobby:0.9, backstage:0.7,
                clock:19.33 };   // RULING DO — 19:20, a curtain-up evening
/* RULING DO — a daylight LEVEL from the hour, for the Arc's glazed foyer.  Full
   between 08:00 and 16:00, out by 20:00 and before 06:00, and it never reaches
   zero because the foyer's own lamps are what HOUSE.lobby already means — this
   only says how much daylight is added to them. */
function houseDaylight(h){
  const t = ((h === undefined ? 19.33 : h) % 24 + 24) % 24;
  if(t >= 8 && t <= 16) return 1;
  if(t < 6 || t > 20) return 0.35;
  return t < 8 ? 0.35 + 0.65*(t-6)/2 : 0.35 + 0.65*(20-t)/4;
}
