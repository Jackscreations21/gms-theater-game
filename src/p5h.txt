
/* ============================================================================
   BEETLEJUICE — the fifth production, and the first taken off a recording

   Two different kinds of work wear one name here, and the line between them
   matters more than anything else in this file:

     MEASURED     cue times, fade durations, blackouts, the act break.  They
                  came off a recording of a real performance with
                  tools/video.js and they are quoted in the cue comments as
                  the times they actually are.
     INTERPRETED  every channel level, every focus, and all of this scenery.
                  We are rendering lit AREAS onto our own 25-channel rig,
                  which is not the rig they used.

   RULING AO: the scenery is the show's VOCABULARY, not its drawing.  A
   clapboard world, a cemetery under a moon, a hard portal round the whole
   picture — our proportions, our detailing, our words on any sign.  Nobody's
   ground plan or light plot went into this, which is the same thing p5c's
   header says about the other four.  A video makes tracing easy for the
   first time; that is exactly why the ruling is written down.

   The set changes are SCENES (p5c): one stage, and the sets take turns on
   it.  That machinery was built for this show, removed with it, and this is
   the first production to use it.  A scene that is off has its layers
   disabled, so it costs no draw call and no raycast while it waits.
   ========================================================================== */

const BJ = {
  opW: 13.6, opH: 9.2,        // the scenic portal's opening, INSET inside the
                              // house's own 15.0 x 10.4 proscenium — frame,
                              // cornice and trim all stay within it, because a
                              // false portal that fouls the architecture is
                              // just a mistake you cannot see from the front
  frame: 0.62,                // how heavy the portal reads
  pz: 0.45                    // it stands just upstage of the apron edge
};
const BJ_CLOTH_W = 17.2;      // the show curtain, both halves together

/* ---- weathered vertical boarding: the portal, and later the house ---- */
function bjBoardTex(){
  const W = 512, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#4a4238'; g.fillRect(0, 0, W, H);
  for(let x = 0; x < W; x += 32){
    const w = 30 + Math.round(Math.random()*2);
    const gr = g.createLinearGradient(x, 0, x + w, 0);
    gr.addColorStop(0, 'rgba(0,0,0,.42)');
    gr.addColorStop(0.30, 'rgba(146,132,112,.20)');
    gr.addColorStop(0.70, 'rgba(96,86,72,.10)');
    gr.addColorStop(1, 'rgba(0,0,0,.42)');
    g.fillStyle = gr; g.fillRect(x, 0, w, H);
    /* the grain, running the length of each board */
    for(let i = 0; i < 14; i++){
      g.fillStyle = 'rgba(30,26,20,' + (0.06 + Math.random()*0.10).toFixed(2) + ')';
      g.fillRect(x + rnd(2, w - 3), rnd(0, H), 1, rnd(30, 190));
    }
  }
  /* nail heads down the board line, and the rust that runs from them */
  for(let y = 30; y < H; y += 124)
    for(let x = 16; x < W; x += 32){
      g.fillStyle = 'rgba(22,18,14,.7)';
      g.beginPath(); g.arc(x, y, 2, 0, 6.283); g.fill();
      if(Math.random() < 0.35){
        const gr = g.createLinearGradient(x, y, x, y + 56);
        gr.addColorStop(0, 'rgba(104,58,26,.30)'); gr.addColorStop(1, 'rgba(104,58,26,0)');
        g.fillStyle = gr; g.fillRect(x - 2.5, y, 5, 56);
      }
    }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}

/* ---- the show curtain: a hard vertical stripe, our own colours ---- */
function bjCurtainCanvas(){
  const W = 1024, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#17131f'; g.fillRect(0, 0, W, H);
  const band = W / 22;
  for(let i = 0; i < 22; i++){
    if(i % 2) continue;
    const gr = g.createLinearGradient(i*band, 0, (i+1)*band, 0);
    gr.addColorStop(0, 'rgba(214,206,190,.10)');
    gr.addColorStop(0.5, 'rgba(224,216,198,.52)');
    gr.addColorStop(1, 'rgba(214,206,190,.10)');
    g.fillStyle = gr; g.fillRect(i*band, 0, band, H);
  }
  /* a low arch of light across the cloth, so it is not a flat pattern */
  const gr2 = g.createRadialGradient(W/2, H*0.86, 40, W/2, H*0.86, W*0.52);
  gr2.addColorStop(0, 'rgba(150,190,120,.20)');
  gr2.addColorStop(1, 'rgba(150,190,120,0)');
  g.fillStyle = gr2; g.fillRect(0, 0, W, H);
  for(let i = 0; i < 200; i++){
    g.fillStyle = 'rgba(0,0,0,' + (0.04 + Math.random()*0.10).toFixed(2) + ')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(6, 30), rnd(30, 160));
  }
  const t = tex(c, 1, 1);
  return {tex: t};
}

SHOWS.beetlejuice = {
  name:  'BEETLEJUICE',
  blurb: 'A hard portal round the whole picture, a cemetery under a low moon, and more haze than air.',
  note:  'An interpretation in the show’s own vocabulary, not a copy of anyone’s design.',
  load(){
    const G = showRoot();

    const BOARD = bjBoardTex();
    const board  = new T.MeshStandardMaterial({map:BOARD, roughness:.86, metalness:.05});
    const boardD = new T.MeshStandardMaterial({color:0x2b2620, roughness:.9,  metalness:.04});
    const deckM  = new T.MeshStandardMaterial({color:0x24262b, roughness:.95});
    const stone  = new T.MeshStandardMaterial({color:0x6e7076, roughness:.88, metalness:.05});
    const iron   = new T.MeshStandardMaterial({color:0x1d2024, roughness:.55, metalness:.72});
    const bark   = new T.MeshStandardMaterial({color:0x241d18, roughness:.94, metalness:.03});
    const hillM  = new T.MeshStandardMaterial({color:0x191d26, roughness:.98});
    const hillMF = new T.MeshStandardMaterial({color:0x232936, roughness:.98});
    /* ONE emissive material shared by every trim strip and the moon's halo —
       a material per object is the draw-call trap in TRAPS.md */
    const glow   = new T.MeshStandardMaterial({color:0xbfe08a, emissive:0x9fd066,
                                               emissiveIntensity:1.5, roughness:.6});
    const moonM  = new T.MeshStandardMaterial({color:0xffd9a0, emissive:0xffb257,
                                               emissiveIntensity:1.25, roughness:.8});

    /* ---- the deck: dark, and the top stays at y = 0 ---- */
    {
      const sh = new T.Shape(), HW = D.stageW/2;
      /* shape y maps to -z after the rotate, so the back wall is +16.6 here */
      const usZ = -(D.backWall + 0.4);
      sh.moveTo(-HW + 0.4, usZ); sh.lineTo(HW - 0.4, usZ);
      sh.lineTo(HW - 0.4, -D.apron);
      sh.quadraticCurveTo(0, -(D.apron + 2*D.thrust*0.9), -HW + 0.4, -D.apron);
      sh.lineTo(-HW + 0.4, usZ);
      const geo = new T.ExtrudeGeometry(sh, {depth:0.1, bevelEnabled:false, curveSegments:36});
      geo.rotateX(-Math.PI/2);
      const m = new T.Mesh(geo, deckM);
      m.name = 'bj:deck';
      m.receiveShadow = true; G.add(m); showWalk(m);
    }

    /* ================================================================
       THE PORTAL — the one thing that never leaves, in every reference
       frame of the show.  It sits INSIDE the house proscenium, which is
       what a false portal is, so it can never foul the architecture.
       ================================================================ */
    {
      const hx = BJ.opW/2 + BJ.frame/2, hy = (BJ.opH + BJ.frame)/2;
      const legs = [
        {geo:new T.BoxGeometry(BJ.frame, BJ.opH + BJ.frame, 0.5), pos:[-hx, hy, 0]},
        {geo:new T.BoxGeometry(BJ.frame, BJ.opH + BJ.frame, 0.5), pos:[ hx, hy, 0]},
        {geo:new T.BoxGeometry(BJ.opW + 2*BJ.frame, BJ.frame, 0.5),
         pos:[0, BJ.opH + BJ.frame/2, 0]}
      ];
      const f = new T.Mesh(mergeParts(legs), board);
      f.name = 'bj:portal'; f.position.z = BJ.pz;
      f.castShadow = true; G.add(f);

      /* a stepped cornice, so the top edge is not one flat board */
      const corn = [];
      for(let i = 0; i < 3; i++)
        corn.push({geo:new T.BoxGeometry(BJ.opW + 2*BJ.frame + 0.10 - i*0.18, 0.16, 0.62 + i*0.12),
                   pos:[0, BJ.opH + BJ.frame + 0.09 + i*0.16, 0]});
      const cm = new T.Mesh(mergeParts(corn), boardD);
      cm.position.z = BJ.pz; cm.castShadow = true; G.add(cm);

      /* the cold line round the opening — three strips, ONE material */
      const tr = [
        {geo:new T.BoxGeometry(0.09, BJ.opH, 0.10), pos:[-BJ.opW/2, BJ.opH/2, 0]},
        {geo:new T.BoxGeometry(0.09, BJ.opH, 0.10), pos:[ BJ.opW/2, BJ.opH/2, 0]},
        {geo:new T.BoxGeometry(BJ.opW, 0.09, 0.10), pos:[0, BJ.opH, 0]}
      ];
      const tm = new T.Mesh(mergeParts(tr), glow);
      tm.name = 'bj:portalTrim'; tm.position.z = BJ.pz + 0.30; G.add(tm);
    }

    /* ================================================================
       SCENE — THE CEMETERY.  The show opens here; measured, the wide
       looks sit at 3:09, 6:11 and 9:31 and the first big blackout is at
       1:26 for 9.0s.
       ================================================================ */
    const cem = sceneAdd('cemetery', 'THE CEMETERY');
    {
      /* three hill layers, silhouetted against the moon.  A profile is a
         polygon, so the shape carries the read and the mesh stays flat. */
      const hill = (w, peak, wob, z, mat)=>{
        const sh = new T.Shape();
        sh.moveTo(-w, 0);
        const N = 9;
        for(let i = 0; i <= N; i++){
          const t = i/N, x = -w + 2*w*t;
          const y = peak * (0.42 + 0.58*Math.sin(Math.PI*t)) *
                    (0.78 + 0.22*Math.sin(t*wob*6.283));
          if(i === 0) sh.lineTo(x, y); else sh.quadraticCurveTo(x - w/N, y, x, y);
        }
        sh.lineTo(w, 0); sh.lineTo(-w, 0);
        const geo = new T.ExtrudeGeometry(sh, {depth:0.3, bevelEnabled:false, curveSegments:8});
        const m = new T.Mesh(geo, mat);
        m.position.z = z; cem.group.add(m);
        return m;
      };
      hill(13.5, 4.4, 1.7, -15.4, hillMF);        // furthest, lightest
      hill(12.0, 3.2, 2.6, -13.8, hillM);
      hill(11.0, 2.1, 3.4, -12.2, hillM);         // nearest, darkest read

      /* the moon: a disc, and one faint halo on the SHARED glow material.
         Both sit upstage of every hill so the profiles read against them,
         and downstage of the deck edge at -16.6 so nothing clips the wall. */
      {
        const d = new T.Mesh(new T.CircleGeometry(1.75, 40), moonM);
        d.position.set(3.6, 6.6, -16.0); d.name = 'bj:moon'; cem.group.add(d);
        const h = new T.Mesh(new T.CircleGeometry(3.1, 36), glow);
        h.position.set(3.6, 6.6, -16.15); cem.group.add(h);
      }

      /* headstones — distinct objects, so distinct meshes, but each one is
         merged from its own slab, shoulders and base: detail without cost */
      const stones = [[-7.4,-7.2,0.10],[-5.6,-9.4,-0.14],[-3.4,-6.4,0.06],
                      [-1.5,-10.2,0.20],[1.9,-6.9,-0.09],[3.8,-9.8,0.15],
                      [6.2,-7.6,-0.18],[8.0,-10.4,0.08]];
      stones.forEach((s, i)=>{
        const h = 0.82 + (i % 3)*0.26, w = 0.58 + (i % 2)*0.12;
        const parts = [
          {geo:new T.BoxGeometry(w, h, 0.16), pos:[0, h/2 + 0.12, 0]},
          {geo:new T.CylinderGeometry(w/2, w/2, 0.16, 14, 1, false, 0, Math.PI),
           pos:[0, h + 0.12, 0], rot:[Math.PI/2, 0, 0]},
          {geo:new T.BoxGeometry(w + 0.26, 0.14, 0.30), pos:[0, 0.07, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(s[0], 0, s[1]); m.rotation.y = s[2];
        m.castShadow = true; cem.group.add(m);
      });

      /* two crosses, for the shape the slabs do not give */
      [[-9.0,-8.6,0.22],[9.4,-8.2,-0.26]].forEach(c=>{
        const parts = [
          {geo:new T.BoxGeometry(0.17, 1.55, 0.17), pos:[0, 0.9, 0]},
          {geo:new T.BoxGeometry(0.86, 0.17, 0.17), pos:[0, 1.32, 0]},
          {geo:new T.BoxGeometry(0.52, 0.13, 0.36), pos:[0, 0.065, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(c[0], 0, c[1]); m.rotation.y = c[2];
        m.castShadow = true; cem.group.add(m);
      });

      /* the railing: two runs, each merged from its own posts and rails */
      [-1, 1].forEach(sd=>{
        const parts = [];
        for(let i = 0; i < 7; i++){
          const x = sd*(3.2 + i*1.15);
          parts.push({geo:new T.BoxGeometry(0.08, 1.32, 0.08), pos:[x, 0.66, 0]});
          parts.push({geo:new T.ConeGeometry(0.07, 0.20, 6), pos:[x, 1.42, 0]});
        }
        parts.push({geo:new T.BoxGeometry(6.9, 0.07, 0.07), pos:[sd*6.65, 1.16, 0]});
        parts.push({geo:new T.BoxGeometry(6.9, 0.07, 0.07), pos:[sd*6.65, 0.34, 0]});
        const m = new T.Mesh(mergeParts(parts), iron);
        m.position.z = -5.4; cem.group.add(m);
      });

      /* the gate, shut, centre — one mesh, merged from its own bars */
      {
        const parts = [
          {geo:new T.BoxGeometry(0.11, 2.30, 0.11), pos:[-2.9, 1.15, 0]},
          {geo:new T.BoxGeometry(0.11, 2.30, 0.11), pos:[ 2.9, 1.15, 0]},
          {geo:new T.BoxGeometry(5.9, 0.09, 0.09),  pos:[0, 2.24, 0]}
        ];
        for(let i = 0; i < 9; i++)
          parts.push({geo:new T.BoxGeometry(0.06, 1.84, 0.06), pos:[-2.4 + i*0.6, 0.92, 0]});
        const m = new T.Mesh(mergeParts(parts), iron);
        m.position.z = -5.4; m.name = 'bj:gate'; cem.group.add(m);
      }

      /* one bare tree, upstage left, merged from trunk and limbs */
      {
        const parts = [{geo:new T.CylinderGeometry(0.16, 0.30, 4.6, 9), pos:[0, 2.3, 0]}];
        const limbs = [[0.6,3.4,0.5,-0.9],[-0.7,3.0,0.55,0.8],[0.35,4.1,0.42,-0.4],
                       [-0.4,4.3,0.38,0.5],[0.9,2.6,0.4,-1.2]];
        limbs.forEach(l=>parts.push({geo:new T.CylinderGeometry(0.05, 0.10, 1.5, 7),
                                     pos:[l[0], l[1], 0], rot:[0, 0, l[3]]}));
        const m = new T.Mesh(mergeParts(parts), bark);
        m.position.set(-9.6, 0, -11.2); m.castShadow = true; cem.group.add(m);
      }

      /* low mist across the boards: the existing machines, not new geometry */
      addShowSmoke({name:'CEMETERY SL', kind:'fog', x:-6.0, y:0.2, z:-7.0, yaw:1.571, tilt:0.05});
      addShowSmoke({name:'CEMETERY SR', kind:'fog', x: 6.0, y:0.2, z:-7.0, yaw:-1.571, tilt:0.05});
      addShowSmoke({name:'UPSTAGE HAZE', kind:'haze', x:0, y:4.0, z:-13.0, yaw:0, tilt:0.2});
    }

    /* ================================================================
       SCENE — THE HOUSE, from outside.  Measured, it carries 62:44 and
       72:58, either side of the act break: late act one and the top of
       act two.  Our own proportions and our own detailing (RULING AO) —
       a clapboard mass that leans, a porch you can stand on, and a
       chimney too big for it.
       ================================================================ */
    const hse = sceneAdd('house', 'THE HOUSE');
    {
      /* it has to read THROUGH the portal, so nothing may exceed the
         portal's own 13.6 x 9.2 opening */
      const MW = 6.6, MH = 5.4, MD = 4.2, cx = -1.2, cz = -8.6;

      /* the main mass, with a gable each end — one mesh */
      {
        const parts = [{geo:new T.BoxGeometry(MW, MH, MD), pos:[0, MH/2, 0]}];
        [-1, 1].forEach(sd=>{
          const g = new T.BufferGeometry();
          const h = 1.9;
          g.setAttribute('position', new T.Float32BufferAttribute([
            -MW/2, 0, 0,  MW/2, 0, 0,  0, h, 0
          ], 3));
          g.computeVertexNormals();
          parts.push({geo:g, pos:[0, MH, sd*MD/2]});
        });
        const m = new T.Mesh(mergeParts(parts), board);
        m.name = 'bj:house'; m.position.set(cx, 0, cz);
        m.castShadow = true; hse.group.add(m);
      }

      /* the roof: two slopes, leaning off true because the house does */
      {
        const parts = [];
        [-1, 1].forEach(sd=>{
          parts.push({geo:new T.BoxGeometry(MW + 0.5, 0.18, MD/2 + 0.55),
                      pos:[0, MH + 0.95, sd*(MD/4 + 0.12)],
                      rot:[sd*0.72, 0, 0]});
        });
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.position.set(cx, 0, cz); m.castShadow = true; hse.group.add(m);
      }

      /* a setback tower, which is where the lean reads from */
      {
        /* kept low on purpose: the whole house has to clear the portal's own
           9.2m opening or the portal crops its own set */
        const parts = [
          {geo:new T.BoxGeometry(2.5, 2.4, 2.5), pos:[0, 1.2, 0]},
          {geo:new T.BoxGeometry(2.8, 0.16, 2.8), pos:[0, 2.48, 0]}
        ];
        [-1, 1].forEach(sd=>parts.push({geo:new T.BoxGeometry(2.9, 0.16, 1.7),
          pos:[0, 2.95, sd*0.72], rot:[sd*0.80, 0, 0]}));
        const m = new T.Mesh(mergeParts(parts), board);
        m.position.set(cx + 1.5, MH, cz - 0.2);
        m.rotation.z = 0.045;                  // the lean, and the only one
        m.castShadow = true; hse.group.add(m);
      }

      /* the chimney, deliberately too big */
      {
        const parts = [
          {geo:new T.BoxGeometry(1.05, 3.4, 1.05), pos:[0, 1.7, 0]},
          {geo:new T.BoxGeometry(1.3, 0.24, 1.3),  pos:[0, 3.5, 0]}
        ];
        const m = new T.Mesh(mergeParts(parts), stone);
        m.position.set(cx - 2.3, MH - 0.4, cz + 0.6);
        m.castShadow = true; hse.group.add(m);
      }

      /* the porch — and its floor is the one thing here you can stand on,
         so it goes on the SCENE's walk list, not the show's: it has to
         leave WALKABLE the moment this set goes off */
      {
        const parts = [
          {geo:new T.BoxGeometry(MW + 1.4, 0.16, 2.3), pos:[0, 0.52, 0]}
        ];
        for(let i = 0; i < 4; i++)
          parts.push({geo:new T.BoxGeometry(0.17, 2.5, 0.17),
                      pos:[-MW/2 - 0.5 + i*(MW + 1.0)/3, 1.85, 1.0]});
        parts.push({geo:new T.BoxGeometry(MW + 1.6, 0.16, 2.6), pos:[0, 3.18, 0.1]});
        parts.push({geo:new T.BoxGeometry(MW + 1.6, 0.34, 0.18), pos:[0, 3.34, 1.32]});
        const m = new T.Mesh(mergeParts(parts), board);
        m.position.set(cx, 0, cz + MD/2 + 1.0);
        m.castShadow = true; hse.group.add(m);

        /* the deck of the porch, as its own flat walkable at its own height */
        const f = new T.Mesh(new T.BoxGeometry(MW + 1.4, 0.12, 2.3), boardD);
        f.name = 'bj:porchFloor';
        f.position.set(cx, 0.60, cz + MD/2 + 1.0);
        f.receiveShadow = true; hse.group.add(f);
        /* sceneShow owns WALKABLE for scene pieces — it adds this on the way in
           and splices it out on the way off.  Pushing it by hand here would
           leave a porch you can stand on while the cemetery is up. */
        sceneWalk(hse, f);
      }

      /* the steps up to it */
      {
        const parts = [];
        for(let i = 0; i < 3; i++)
          parts.push({geo:new T.BoxGeometry(2.4, 0.2, 0.42), pos:[0, 0.1 + i*0.2, -i*0.42]});
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.position.set(cx, 0, cz + MD/2 + 2.5); hse.group.add(m);
      }

      /* windows: two clusters, each merged, each a frame and a sill */
      [[-1.9, 1.5], [1.7, 1.6], [-1.6, 3.9], [1.9, 4.0]].forEach((wv, i)=>{
        const parts = [
          {geo:new T.BoxGeometry(1.05, 1.35, 0.10), pos:[0, 0, 0]},
          {geo:new T.BoxGeometry(0.10, 1.35, 0.16), pos:[0, 0, 0.04]},
          {geo:new T.BoxGeometry(1.05, 0.10, 0.16), pos:[0, 0, 0.04]},
          {geo:new T.BoxGeometry(1.28, 0.14, 0.26), pos:[0, -0.74, 0.04]}
        ];
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.position.set(cx + wv[0], wv[1], cz + MD/2 + 0.02);
        hse.group.add(m);
      });

      /* the door, shut */
      {
        const m = new T.Mesh(new T.BoxGeometry(1.15, 2.25, 0.12), boardD);
        m.name = 'bj:houseDoor';
        m.position.set(cx, 1.72, cz + MD/2 + 0.02);
        hse.group.add(m);
      }

      /* a picket run either side, leaning where the ground has moved */
      [-1, 1].forEach(sd=>{
        const parts = [];
        /* the run stops short of the portal edge — everything in this scene has
           to sit inside the 13.6m opening it is seen through */
        for(let i = 0; i < 6; i++)
          parts.push({geo:new T.BoxGeometry(0.11, 1.15, 0.06),
                      pos:[sd*(3.4 + i*0.52), 0.58, 0], rot:[0, 0, (i%3-1)*0.05]});
        parts.push({geo:new T.BoxGeometry(2.6, 0.08, 0.05), pos:[sd*4.7, 0.86, 0]});
        parts.push({geo:new T.BoxGeometry(2.6, 0.08, 0.05), pos:[sd*4.7, 0.34, 0]});
        const m = new T.Mesh(mergeParts(parts), boardD);
        m.position.z = cz + MD/2 + 2.9; hse.group.add(m);
      });

      /* one dead shrub each side, so the ground is not bare board */
      [[-5.2, -6.0], [4.4, -6.4]].forEach(s=>{
        const parts = [{geo:new T.CylinderGeometry(0.07, 0.13, 1.0, 7), pos:[0, 0.5, 0]}];
        for(let i = 0; i < 5; i++)
          parts.push({geo:new T.CylinderGeometry(0.03, 0.06, 0.8, 5),
                      pos:[(i%3-1)*0.22, 1.05, 0], rot:[0, 0, (i%3-1)*0.7]});
        const m = new T.Mesh(mergeParts(parts), bark);
        m.position.set(s[0], 0, s[1]); hse.group.add(m);
      });
    }

    /* ---- the show curtain ---- */
    {
      const c = bjCurtainCanvas();
      const key = 'bjCurtain';
      const halfW = D.procW/2 + 3.0;
      GOODS[key] = {label:'BEETLEJUICE show curtain', h:13.0, wt:420,
        traveler:true, made:true, build:()=>{
          const g = new T.Group();
          [-1, 1].forEach(sd=>{
            const t2 = c.tex.clone();
            t2.needsUpdate = true;
            t2.wrapS = t2.wrapT = T.ClampToEdgeWrapping;
            t2.repeat.set(halfW / BJ_CLOTH_W, 1);
            t2.offset.set(sd < 0 ? 0 : (BJ_CLOTH_W - halfW) / BJ_CLOTH_W, 0);
            const mat = new T.MeshStandardMaterial({map:t2, roughness:.95,
              side:T.DoubleSide, color:0xffffff});
            const d = drape(halfW, 13.0, mat, 9, .22);
            d.position.set(sd*(D.procW/4 - 0.4), -6.5, 0);
            d.userData.side = sd;
            g.add(d);
          });
          return g; }};
      TRIMS[key] = TRIMS.house;
      SHOW.goods.push(key);
      SHOW.curtainKey = key;
      buildGoodsPalette();
    }

    /* ---- the hang ---- */
    hangGoods(FLY[0], SHOW.curtainKey);
    FLY[0].target = TRIMS[SHOW.curtainKey]; FLY[0].travTarget = 0; FLY[0].open = 0;
    hangGoods(FLY[1], 'border');    FLY[1].target = TRIMS.border;
    hangGoods(FLY[2], 'legs');      FLY[2].target = inTrimOf(FLY[2]);
    hangGoods(FLY[3], 'border');    FLY[3].target = TRIMS.border;
    hangGoods(FLY[4], 'electric');  FLY[4].target = 9.2;
    hangGoods(FLY[5], 'legs');      FLY[5].target = inTrimOf(FLY[5]);
    hangGoods(FLY[6], 'truss');     FLY[6].target = 11.4;
    hangGoods(FLY[8], 'electric');  FLY[8].target = 10.4;
    FLY.forEach(ls=>{ ls.pos = ls.target; ls.group.position.y = ls.pos; });

    sceneShow('cemetery');
    RIG.haze = 0.55;                 // the beam fan is in every reference frame
    plotBeetlejuice();
    standByAtTheTop();
  }
};
