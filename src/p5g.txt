
/* ============================================================================
   THE PLAY THAT GOES WRONG
   The Cornley Polytechnic Drama Society present "The Murder at Haversham
   Manor": a green damask study with a fireplace, a grandfather clock, a
   gothic window and a second storey carried on one post.

   The point of the set is that it does not stay up.  Every piece that is
   meant to fail is built on its own hinge with its own weight, so it goes
   over under gravity, lands, rocks and settles — and can be stood back up
   again for the next performance.
   ========================================================================== */

/* ============================================================================
   THINGS THAT GO WRONG
   A hinged piece of scenery with its pivot at the bottom, left to fall over.
   The torque falls away as it approaches flat, it bounces when it lands and
   it rocks itself to a stop, so a wall going over reads as weight rather
   than as an animation being played at you.
   ========================================================================== */
const WRONG_G = 7.4;
function makeWrong(o){
  const p = {
    key:o.key, label:o.label, group:o.group,
    axis:o.axis || 'z',
    dir:o.dir === undefined ? 1 : o.dir,
    limit:o.limit === undefined ? 1.5 : o.limit,
    drop:o.drop || 0,
    rate:o.rate === undefined ? 1 : o.rate,
    base:{x:o.group.rotation.x, y:o.group.rotation.y, z:o.group.rotation.z},
    y0:o.group.position.y,
    walk:o.walk || [], sound:o.sound || 'crash',
    ang:0, vel:0, fall:0, fvel:0, bounces:0, state:'ok'
  };
  o.group.userData.moves = true;
  SHOW.wrong.push(p);
  return p;
}
function wrongFind(key){ return (SHOW.wrong || []).find(w=>w.key === key) || null; }
function wrongTrigger(key){
  const p = wrongFind(key);
  if(!p || p.state !== 'ok') return null;
  p.state = 'going';
  p.vel = rnd(0.25, 0.6);              // the shove that starts it
  p.fvel = 0;
  /* nobody stands on a floor that is on its way down */
  for(const o of p.walk){
    const i = WALKABLE.indexOf(o);
    if(i >= 0) WALKABLE.splice(i, 1);
  }
  if(typeof Snd !== 'undefined' && Snd.play) Snd.play(p.sound);
  if(typeof toast === 'function') toast(p.label + ' goes');
  if(typeof refreshWrongUI === 'function') refreshWrongUI();
  return p;
}
function wrongAll(){ (SHOW.wrong || []).forEach(p=>wrongTrigger(p.key)); }
function wrongReset(){
  for(const p of (SHOW.wrong || [])){
    p.ang = 0; p.vel = 0; p.fall = 0; p.fvel = 0; p.bounces = 0; p.state = 'ok';
    p.group.rotation.set(p.base.x, p.base.y, p.base.z);
    p.group.position.y = p.y0;
    for(const o of p.walk) if(WALKABLE.indexOf(o) < 0) WALKABLE.push(o);
  }
  if(typeof refreshWrongUI === 'function') refreshWrongUI();
}
function wrongStanding(){ return (SHOW.wrong || []).filter(p=>p.state === 'ok').length; }
function updateWrong(dt){
  const list = SHOW.wrong;
  if(!list || !list.length) return;
  for(const p of list){
    if(p.state === 'ok' || p.state === 'down') continue;
    /* it goes over faster the further it goes: the weight of it is further
       out from the hinge every degree, until it is flat and there is none */
    p.vel += dt * WRONG_G * Math.max(0.1, Math.cos(p.ang)) * p.rate;
    p.ang += p.vel * dt;
    if(p.drop){
      p.fvel += dt * 9.2;
      p.fall += p.fvel * dt;
      if(p.fall >= p.drop){
        p.fall = p.drop;
        p.fvel = -p.fvel * 0.22;
        if(Math.abs(p.fvel) < 0.6) p.fvel = 0;
      }
    }
    if(p.ang >= p.limit){
      p.ang = p.limit;
      p.vel = -Math.abs(p.vel) * 0.24;                 // it lands and rocks
      p.bounces++;
      /* It is only down when it has finished BOTH halves of the fall.  The
         chandelier tips over faster than it drops, and settling on the tilt
         alone left it hanging in mid-air with the ceiling gone.          */
      const landed = !p.drop || p.fall >= p.drop - 1e-4;
      if(landed && (p.bounces > 3 || Math.abs(p.vel) < 0.4)){
        p.vel = 0; p.state = 'down';
        if(typeof refreshWrongUI === 'function') refreshWrongUI();
      }
    }
    p.group.rotation[p.axis] = p.base[p.axis] + p.dir * p.ang;
    p.group.position.y = p.y0 - p.fall;
  }
}

/* ---- green damask, hung by people who have never hung paper ----
   NOT damaskTex: p2 already declares that for the Palace crimson, and two
   hoisted declarations in one script means the later one silently wins. */
function cornleyDamaskTex(){
  const W = 512, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#1f5a4e'; g.fillRect(0,0,W,H);
  const motif = (x, y)=>{
    g.save(); g.translate(x, y);
    g.fillStyle = 'rgba(96,168,142,.4)';
    g.beginPath();
    g.moveTo(0,-16); g.quadraticCurveTo(13,-8, 0,14); g.quadraticCurveTo(-13,-8, 0,-16);
    g.fill();
    g.beginPath();
    g.moveTo(-15,2); g.quadraticCurveTo(0,-4, 15,2); g.quadraticCurveTo(0,10, -15,2);
    g.fill();
    g.restore();
  };
  for(let y=0; y<H+40; y+=58)
    for(let x=0; x<W+40; x+=52)
      motif(x + ((y/58)%2)*26, y);
  // vertical seams where the lengths meet, and a general dinginess
  for(let x=0; x<W; x+=128){
    g.fillStyle = 'rgba(0,0,0,.22)'; g.fillRect(x, 0, 2, H);
    g.fillStyle = 'rgba(180,220,200,.06)'; g.fillRect(x+2, 0, 3, H);
  }
  for(let i=0;i<260;i++){
    g.fillStyle = 'rgba(0,0,0,'+(0.03+Math.random()*0.10).toFixed(2)+')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(4,40), rnd(10,90));
  }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}
/* ---- a shelf of books, seen from the front ---- */
function booksTex(){
  const W = 256, H = 256, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#22160e'; g.fillRect(0,0,W,H);
  const cols = ['#6b2418','#3d4a22','#1e3450','#5a3a12','#472046','#22423a','#7a5a20'];
  for(let shelf=0; shelf<4; shelf++){
    const y = shelf*64, h = 56;
    let x = 2;
    while(x < W - 6){
      const w = rnd(7, 17), lean = Math.random() < 0.12;
      g.save();
      g.translate(x, y + h);
      if(lean) g.rotate(rnd(-0.16, 0.16));
      g.fillStyle = cols[(Math.random()*cols.length)|0];
      g.fillRect(0, -h*rnd(0.78, 1), w, h);
      g.fillStyle = 'rgba(214,182,110,.5)';
      g.fillRect(1, -h*0.66, w-2, 2);
      g.fillRect(1, -h*0.5, w-2, 1.4);
      g.restore();
      x += w + 1.4;
    }
    g.fillStyle = '#2f2013'; g.fillRect(0, y + h, W, 8);
    g.fillStyle = 'rgba(0,0,0,.5)'; g.fillRect(0, y + h - 4, W, 4);
  }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}
/* ---- leaded glass, with a wet night behind it ---- */
function leadedTex(){
  const W = 256, H = 512, c = cvs(W, H), g = c.getContext('2d');
  const sky = g.createLinearGradient(0,0,0,H);
  sky.addColorStop(0, '#2b4f8a'); sky.addColorStop(0.55, '#1b3560'); sky.addColorStop(1, '#101d38');
  g.fillStyle = sky; g.fillRect(0,0,W,H);
  for(let i=0;i<40;i++){
    g.fillStyle = 'rgba(150,190,240,'+(0.04+Math.random()*0.10).toFixed(2)+')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(10,50), rnd(10,60));
  }
  g.strokeStyle = 'rgba(20,22,26,.85)'; g.lineWidth = 3;
  for(let y=-W; y<H+W; y+=40){
    g.beginPath(); g.moveTo(-10, y); g.lineTo(W+10, y + W); g.stroke();
    g.beginPath(); g.moveTo(-10, y + W); g.lineTo(W+10, y); g.stroke();
  }
  g.strokeStyle = 'rgba(16,18,22,.95)'; g.lineWidth = 7;
  for(let x=0; x<=W; x+=W/2){ g.beginPath(); g.moveTo(x,0); g.lineTo(x,H); g.stroke(); }
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}
