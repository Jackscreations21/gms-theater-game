function cvs(w,h){ const c=document.createElement('canvas'); c.width=w; c.height=h; return c; }
function tex(c, rx, ry, srgb){
  const t = new T.CanvasTexture(c);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  t.repeat.set(rx||1, ry||1);
  t.anisotropy = 4;
  if(srgb !== false) t.encoding = T.sRGBEncoding;
  return t;
}
function grain(x,w,h,amt){
  const img=x.getImageData(0,0,w,h), d=img.data;
  for(let i=0;i<d.length;i+=4){ const n=(Math.random()-0.5)*amt;
    d[i]=clamp(d[i]+n,0,255); d[i+1]=clamp(d[i+1]+n,0,255); d[i+2]=clamp(d[i+2]+n,0,255); }
  x.putImageData(img,0,0);
}

/* ----------------------------------------------------------------------------
   mergeParts — bake a cluster of static detail into ONE geometry.
   r128's core ships no BufferGeometryUtils, so this is the house version.
   Each part is {geo, pos:[x,y,z], rot:[x,y,z], scale:[x,y,z]}; the transform
   is baked into the vertices and the parts are concatenated, so a detailed
   object costs one draw call instead of one per part (RULING AK).
   EVERY part must be drawn with the material the merged mesh is given.
   Anything that moves, is grabbed, or is recoloured at runtime must stay its
   own mesh — see RULING AL for the list that must never come through here.
   No negative scale: mirroring reverses triangle winding relative to the
   shading normals, and a live mesh hides that with the matrix determinant,
   but baked vertices carry no determinant — a mirrored part vanishes under
   the default FrontSide. Mirrored parts must flip winding instead.
   De-indexing costs vertices — the right trade on a Quest, but visible: the
   planned nail-gun shell goes from 160 verts / 5 draw calls to 264 verts /
   1 draw call, 1.65x the vertices for the same triangles.
   -------------------------------------------------------------------------- */
const _mpM = new T.Matrix4(), _mpQ = new T.Quaternion(), _mpE = new T.Euler();
const _mpP = new T.Vector3(), _mpS = new T.Vector3();
function mergeParts(parts){
  const pos = [], nor = [], uvs = [];
  parts.forEach(p=>{
    const src = p.geo.index ? p.geo.toNonIndexed() : p.geo.clone();
    _mpP.set(p.pos ? p.pos[0] : 0, p.pos ? p.pos[1] : 0, p.pos ? p.pos[2] : 0);
    _mpE.set(p.rot ? p.rot[0] : 0, p.rot ? p.rot[1] : 0, p.rot ? p.rot[2] : 0);
    _mpS.set(p.scale ? p.scale[0] : 1, p.scale ? p.scale[1] : 1, p.scale ? p.scale[2] : 1);
    src.applyMatrix4(_mpM.compose(_mpP, _mpQ.setFromEuler(_mpE), _mpS));
    if(!src.attributes.normal) src.computeVertexNormals();   // per-part, so smooth normals elsewhere survive
    const pa = src.attributes.position, na = src.attributes.normal, ua = src.attributes.uv;
    for(let i=0;i<pa.count;i++){
      pos.push(pa.getX(i), pa.getY(i), pa.getZ(i));
      nor.push(na.getX(i), na.getY(i), na.getZ(i));
      ua ? uvs.push(ua.getX(i), ua.getY(i)) : uvs.push(0, 0);
    }
    src.dispose();
  });
  const out = new T.BufferGeometry();
  out.setAttribute('position', new T.Float32BufferAttribute(pos, 3));
  if(nor.length) out.setAttribute('normal', new T.Float32BufferAttribute(nor, 3));
  if(uvs.length) out.setAttribute('uv',     new T.Float32BufferAttribute(uvs, 2));
  out.computeBoundingSphere();
  return out;
}

/* silk damask wall covering — crimson ground, gold ornament */
function damaskTex(){
  const c=cvs(256,256), x=c.getContext('2d');
  x.fillStyle='#7d1b23'; x.fillRect(0,0,256,256);
  // woven ground
  for(let i=0;i<256;i+=2){ x.fillStyle='rgba(0,0,0,.08)'; x.fillRect(i,0,1,256); }
  for(let i=0;i<256;i+=2){ x.fillStyle='rgba(255,255,255,.04)'; x.fillRect(0,i,256,1); }
  // a damask motif, mirrored into a half-drop repeat
  function motif(cx,cy,s){
    x.save(); x.translate(cx,cy); x.scale(s,s);
    x.strokeStyle='rgba(206,170,90,.85)'; x.fillStyle='rgba(196,158,74,.42)';
    x.lineWidth=2.2;
    for(const dir of [-1,1]){
      x.save(); x.scale(dir,1);
      x.beginPath();                                   // the ogee lobe
      x.moveTo(0,-46); x.bezierCurveTo(30,-40, 44,-14, 30,6);
      x.bezierCurveTo(20,22, 6,30, 0,46);
      x.stroke();
      x.beginPath();                                   // an acanthus curl
      x.moveTo(0,-16); x.bezierCurveTo(20,-20, 26,-4, 14,6);
      x.bezierCurveTo(8,11, 3,10, 0,4);
      x.fill(); x.stroke();

      x.moveTo(2,26); x.quadraticCurveTo(20,22, 22,36);

      x.restore();

    x.beginPath(); x.arc(0,0,5,0,6.283); x.fill();     // the centre boss
    x.restore();

  motif(64,64,1); motif(192,64,1); motif(0,192,1); motif(128,192,1); motif(256,192,1);
  grain(x,256,256,10);
  return tex(c,7,5);

/* stencilled labels — cached by text and colour so a shed full of the same
   label is ONE texture (RULING AM) */
const STENCILS = {};
function stencilTex(text, bg){
  const key = text + '|' + bg;
  if(STENCILS[key]) return STENCILS[key];

function box(w,h,d,mat,x,y,z,parent){
  const m = new T.Mesh(new T.BoxGeometry(w,h,d), mat);
  m.position.set(x,y,z); m.castShadow=true; m.receiveShadow=true;
  (parent||world).add(m); STATIC.push(m); return m;
}
const _dummy = new T.Object3D();
/* one draw call for hundreds of identical ornaments */
function instanced(geo, mat, list, opt){
  opt = opt || {};
  if(!list.length) return null;
  const m = new T.InstancedMesh(geo, mat, list.length);
  for(let i=0;i<list.length;i++){
    const t = list[i];
    _dummy.position.set(t.x||0, t.y||0, t.z||0);
    _dummy.rotation.set(t.rx||0, t.ry||0, t.rz||0);
    _dummy.scale.set(t.s||t.sx||1, t.s||t.sy||1, t.s||t.sz||1);
    _dummy.updateMatrix();
    m.setMatrixAt(i, _dummy.matrix);
  }
  m.instanceMatrix.needsUpdate = true;
  /* r128 sizes an InstancedMesh's bounding sphere from ONE instance at the
     origin, so a spread-out batch vanishes the moment the origin leaves the
     frustum. Give it a sphere that actually contains every instance.        */
  geo.computeBoundingSphere();
  const r0 = geo.boundingSphere ? geo.boundingSphere.radius : 1;
  let cx=0, cy=0, cz=0, sMax=1;
  for(const t of list){ cx+=t.x||0; cy+=t.y||0; cz+=t.z||0;
    sMax = Math.max(sMax, t.s||t.sx||1, t.s||t.sy||1, t.s||t.sz||1); }
  cx/=list.length; cy/=list.length; cz/=list.length;
  let rad = 0;
  for(const t of list) rad = Math.max(rad,
    Math.hypot((t.x||0)-cx, (t.y||0)-cy, (t.z||0)-cz));
  geo.boundingSphere = new T.Sphere(new T.Vector3(cx,cy,cz), rad + r0*sMax + 0.5);
  m.castShadow = opt.cast !== false;
  m.receiveShadow = opt.receive !== false;
  m.frustumCulled = opt.cull !== false;
  (opt.parent||world).add(m);
  STATIC.push(m);
  return m;
}
/* a run of ornament along a straight line */
function runAlong(x0,y0,z0, x1,y1,z1, n, extra){
  const out=[];
  for(let i=0;i<n;i++){
    const t = n===1?0.5:i/(n-1);
    out.push(Object.assign({x:lerp(x0,x1,t), y:lerp(y0,y1,t), z:lerp(z0,z1,t)}, extra||{}));
  }
  return out;
}
/* a run of ornament along the curved front of a balcony */
function runCurve(fn, n, extra){
  const out=[];
  for(let i=0;i<n;i++){ const p = fn(i/(n-1));
    out.push(Object.assign({x:p.x, y:p.y, z:p.z, ry:p.ry||0}, extra||{})); }
  return out;
}

/* the curved sweep both balconies share: a shallow arc bowing toward the stage */
const BOW = 3.4;                               // how far the centre reaches forward
function balconyFront(t, zFront, halfW){       // t: -1 (house left) .. 1 (house right)
  const x = t*halfW;
  const z = zFront + BOW*(t*t);
  return {x, z};
}
