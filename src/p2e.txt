function shopLog(s){ SHOP.log.unshift(s); if(SHOP.log.length > 16) SHOP.log.pop(); }

/* The shed itself is gone — the workshop is a bench you open from the console,
   not a room you walk to.  The paint frame material stays, because the bench
   paints onto it, and the stock model below is unchanged.                   */
SHOP.frameMat = new T.MeshStandardMaterial({color:0xd9d2c4, roughness:1, side:T.DoubleSide,
  transparent:true, alphaTest:0.35});

/* ============================================================================
   SHOP OPERATIONS — pure model, driven by the bench UI
   ========================================================================== */
function newCanvas(w, h, fill){
  const c = cvs(w,h), x = c.getContext('2d');
  x.fillStyle = fill; x.fillRect(0,0,w,h);
  const t = tex(c,1,1); t.wrapS = t.wrapT = T.ClampToEdgeWrapping;
  return {canvas:c, tex:t};
}
/* --- backdrops: no cutting, you just take a cloth off the shelf --- */
function shopNewCloth(boltIdx){
  const spec = BOLTS[boltIdx] || BOLTS[0];
  const c = newCanvas(1024, 576, spec.col);
  const item = {id:++SHOP.seq, spec, len:D.procW+6, canvas:c.canvas, tex:c.tex, w:1024, h:576};
  SHOP.cloth.push(item);
  shopLog('took a '+spec.name+' backdrop off the shelf');
  Snd.play('whoosh');
  return item;
}
/* --- delivery --- */
function placeBuilt(group, label){
  group.traverse(o=>{ if(o.isMesh){ o.castShadow = true; o.receiveShadow = true; } });
  setGroup.add(group);
  const p = {key:'built', group, label:label||'built piece', rot:0};
  SET.push(p); stagePieces.push(group); group.userData.piece = p;
  refreshSetUI();
  return p;
}
function shopDeliverCloth(i){
  const cl = SHOP.cloth[i]; if(!cl) return false;
  const key = 'shop'+cl.id;
  const t = cl.tex;
  GOODS[key] = {label:'shop cloth #'+cl.id, h:11.5, wt:230, made:true,
    build:()=>{ const g = new T.Group();
      const m = new T.MeshStandardMaterial({map:t, roughness:1, side:T.DoubleSide,
        transparent:true, alphaTest:0.35});
      const d = drape(D.procW+6, 11.5, m, 2, .05); d.position.y = -5.75; g.add(d); return g; }};
  TRIMS[key] = 11.6;
  SHOP.cloth.splice(i,1);
  buildGoodsPalette();
  shopLog('sent a cloth to the fly rail');
  toast('Cloth is on the fly rail goods list');
  return true;
}
