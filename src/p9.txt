
/* ============================================================================
   VR  —  WebXR, built for a Quest 3 at 90Hz

   The same building, the same board, the same shows.  What changes is that
   you are standing in it: you walk on the sticks, and instead of a web page
   along the bottom of a monitor there is a lighting console in each house
   that you walk up to and point at.

   Nothing here touches the desktop.  The game detects what it is running on
   and configures itself; on a monitor every line below is inert.

   Two notes on how WebXR wants to be driven:
   · the camera is moved BY THE HEADSET, so the player has to be a rig that
     the camera sits inside — write to camera.position in a session and the
     next pose overwrites you;
   · the frame loop has to come from renderer.setAnimationLoop, because in a
     session requestAnimationFrame is the page's clock, not the headset's.
   ========================================================================== */
/* `var`, deliberately.  Function declarations hoist to the top of the whole
   concatenated script, so updatePlayer can be called before this file has been
   evaluated — and `typeof VR` on a `const` in that window does not return
   'undefined', it throws.  A `var` is simply undefined until we get here, which
   is what every guard below is written against. */
var VR = {
  supported: false, active: false,
  rig: null, camHome: null,
  controllers: [], grips: [], ray: null, cursor: null,
  axes: {lx:0, ly:0, rx:0, ry:0},
  speed: 3.2, turn: 2.1,
  desks: [], deskMeshes: [], hits: [], page: 'cues',
  canvas: null, ctx: null, tex: null,
  hover: null, held: null, heldLever: null, ropes: [], goButtons: [],
  saved: null, drawT: 0, session: null
};

/* ---------------------------------------------------------------- detect */
function vrDetect(){
  const btn = $('#vrBtn');
  if(!navigator.xr || !navigator.xr.isSessionSupported){
    if(btn) btn.style.display = 'none';
    return Promise.resolve(false);
  }
  return navigator.xr.isSessionSupported('immersive-vr').then(ok=>{
    VR.supported = !!ok;
    if(btn) btn.style.display = ok ? '' : 'none';
    return VR.supported;
  }).catch(()=>{ if(btn) btn.style.display = 'none'; return false; });
}

/* ============================================================================
   THE QUALITY TIER
   A Quest 3 at 90Hz has eleven milliseconds to draw the room TWICE.  The
   expensive things here are the ones this game is made of — additive beams in
   haze are overdraw, and overdraw is what a mobile tile renderer hates most.
   So a session turns the count down rather than the quality: the beams stay,
   there are simply fewer of them alight at once.
   ========================================================================== */
function vrQualityOn(){
  if(VR.saved) return;
  VR.saved = {
    shadows: renderer.shadowMap.enabled,
    budget: RIG.shadowBudget,
    smoke: SMOKE.n,
    pixelRatio: renderer.getPixelRatio ? renderer.getPixelRatio() : 1,
    far: camera.far
  };
  renderer.shadowMap.enabled = false;         // the single biggest win
  RIG.shadowBudget = 0;
  VR.beamCap = 10;                            /* beams alight at once — was 14;
                                                 the first Quest 3 run reported
                                                 the frame rate a little low */
  VR.glowCap = 12;                            /* RULING DN — glow planes drawn
                                                 at once.  Additive geometry in
                                                 haze is overdraw, exactly like
                                                 the beams above it, and this is
                                                 the same knob for the same
                                                 reason.  A shade above the beam
                                                 count because a glow is one
                                                 screen-clamped quad against a
                                                 volumetric cone, and it is the
                                                 thing that reads as brightness
                                                 when the beams are capped out */
  VR.lightCap = 4;                            /* real SpotLights handed out by
                                                 p4 — half the desktop pool.
                                                 Every material pays for every
                                                 live light on every pixel, and
                                                 a Quest pays it twice.  Beams,
                                                 gobo pools and lens glows still
                                                 draw for all 25 channels, so
                                                 the rig still reads fully lit */
  camera.far = 160;                           // the far building is switched off anyway
  camera.updateProjectionMatrix();
  if(renderer.xr && renderer.xr.setFoveation) renderer.xr.setFoveation(0.4);
  if(typeof toast === 'function')
    toast('VR: shadows off, beams capped, ' + (VR.targetHz || 90) + 'Hz');
}
function vrQualityOff(){
  if(!VR.saved) return;
  renderer.shadowMap.enabled = VR.saved.shadows;
  RIG.shadowBudget = VR.saved.budget;
  camera.far = VR.saved.far;
  camera.updateProjectionMatrix();
  VR.beamCap = 0;
  VR.lightCap = 0;
  VR.glowCap = 0;                             // RULING DN — 0 lifts it, as above
  VR.saved = null;
}
/* called from the rig update: keep only the nearest N beams alive */
const _bp = new T.Vector3();
function vrCapBeams(){
  if(!VR || !VR.active || !VR.beamCap) return;
  const live = [];
  for(const f of FIXTURES) if(f.beam && f.beam.visible) live.push(f);
  if(live.length <= VR.beamCap) return;
  camera.getWorldPosition(_bp);
  /* f._org is the WORLD emit position (p4 refreshes it every rig update).
     f.group.position is local to the rig group, which at the Arc sits 420m
     out — sorting that against a world camera kept the wrong 14 beams. */
  live.sort((a,b)=>a._org.distanceToSquared(_bp) -
                   b._org.distanceToSquared(_bp));
  for(let i = VR.beamCap; i < live.length; i++) live[i].beam.visible = false;
}

/* ============================================================================
   THE METER
   Nothing in the game reported frame rate before this — the only datum the
   first headset run produced was "a little low", no number.  So: a ring
   buffer of frame times, averaged twice a second, drawn on a small tag on
   the left wrist.  Glance at your watch.

   And a hand on the one knob WebXR lets us turn MID-session: foveation.
   (Framebuffer scale is read once at session start and never again.)  Over
   budget, peripheral resolution goes first — 0.4 stepping up toward 1.0;
   comfortably under, it relaxes back down, never below the 0.4 base the
   quality tier asks for.  Climbing is fast and relaxing is slow on purpose:
   a dropped frame is felt, a soft edge is not.

   RULING DJ — THE METER DOES NOT READ THE GAME'S CLAMPED dt.
   p7 clamps dt to 50ms so a model load cannot teleport the show, and for
   two rounds this meter read that clamped figure — so it could not report a
   frame worse than 50ms however bad the frame was.  The first headset
   readings ever taken came back at 25ms in an empty house and 48ms in
   Beetlejuice, TWO OFF A CEILING NOBODY KNEW WAS THERE: every frame past 50
   had been recorded as exactly 50, so the average was a floor and not a
   measurement.  p7 now hands the raw figure over beside the clamped one.

   PERF_CEIL is the meter's OWN ceiling and it is deliberately far higher.
   A 20ms frame is a frame rate and must be told the truth; a three-second
   model load is a hitch, and left uncapped it would sit in the 120-frame
   window poisoning the average for two seconds after it ended.
   ========================================================================== */
const PERF_CEIL = 200;
function vrPerf(dt, rawDt){
  let P = VR.perf;
  if(!P) P = VR.perf = {buf:new Float32Array(120), i:0, n:0, avg:0, worst:0, t:0, fov:0.4,
                        calls:0, tris:0};
  /* the unclamped frame time where p7 offers one, falling back to dt for any
     caller that has none — the suites drive vrUpdate directly */
  P.buf[P.i] = Math.min(PERF_CEIL, (rawDt > 0 ? rawDt : dt) * 1000);
  P.i = (P.i+1) % P.buf.length;
  if(P.n < P.buf.length) P.n++;
  P.t += dt;
  if(P.t < 0.5 || P.n < 30) return;     // let the window fill before judging
  P.t = 0;
  let sum = 0, worst = 0;
  for(let i=0;i<P.n;i++){ const v = P.buf[i]; sum += v; if(v > worst) worst = v; }
  P.avg = sum/P.n; P.worst = worst;
  /* RULING DJ — three.js has NO MULTIVIEW at any version (checked against
     the published r128, r160, r162 and r170 builds, not against a
     changelog), so both eyes cost two passes and the only lever left on
     draw calls is our own batching.  Nothing reported the count where it
     could be read.  renderer.info costs nothing to ask.
     These are the PREVIOUS frame's counts — renderer.render runs at the end
     of tick (p7) and vrUpdate runs before it — which for a number averaged
     twice a second is not worth a frame of latency to correct. */
  P.calls = renderer.info.render.calls;
  P.tris  = renderer.info.render.triangles;
  const budget = 1000/(VR.targetHz || 90);
  let f = P.fov;
  if(P.avg > budget*1.06)      f = Math.min(1.0, f + 0.15);
  else if(P.avg < budget*0.82) f = Math.max(0.4, f - 0.05);
  if(f !== P.fov){
    P.fov = f;
    if(renderer.xr.setFoveation) renderer.xr.setFoveation(f);
  }
  vrDrawMeter();
}

function vrDrawMeter(){
  const P = VR.perf;
  if(!VR.meter){
    if(!VR.grips.length) return;        // no hands yet, no wrist to sit on
    /* 256x128 against a 0.08 x 0.04 plane — RULING DJ's two numbers need a
       third line, and the canvas and the plane must keep the same aspect or
       the wrist plate skews */
    const c = cvs(256, 128);
    const tex = new T.CanvasTexture(c);
    const m = new T.Mesh(new T.PlaneGeometry(0.08, 0.04),
      new T.MeshBasicMaterial({map:tex, transparent:true, depthTest:false}));
    m.name = 'vr:meter';
    m.renderOrder = 30;
    m.position.set(0, 0.035, 0.10);     // just above the wrist, tilted to the eye
    m.rotation.x = -1.05;
    VR.grips[0].add(m);
    VR.meter = {mesh:m, g:c.getContext('2d'), tex:tex};
  }
  const g = VR.meter.g, budget = 1000/(VR.targetHz || 90);
  g.clearRect(0,0,256,128);
  g.fillStyle = 'rgba(8,10,13,0.78)'; g.fillRect(0,0,256,128);
  g.textBaseline = 'middle'; g.textAlign = 'left';
  g.font = '600 40px Helvetica, Arial, sans-serif';
  g.fillStyle = P.avg > budget ? '#ff7a6b' : '#9fe8a9';
  g.fillText(P.avg.toFixed(1), 10, 28);
  g.font = '500 24px Helvetica, Arial, sans-serif';
  g.fillStyle = '#8b95a3';
  g.fillText('ms', 108, 32);
  g.fillText('pk ' + P.worst.toFixed(0) + ' · fov ' + P.fov.toFixed(2), 10, 72);
  /* RULING DJ — the batching numbers, on their own line under the timing */
  g.font = '500 22px Helvetica, Arial, sans-serif';
  g.fillStyle = '#8b95a3';
  g.fillText((P.calls|0) + ' calls · ' + Math.round((P.tris|0)/1000) + 'k tri', 10, 104);
  VR.meter.tex.needsUpdate = true;
}

/* ============================================================================
   THE SESSION
   ========================================================================== */
function vrEnter(){
  if(!VR.supported || VR.active) return;
  navigator.xr.requestSession('immersive-vr', {
    optionalFeatures: ['local-floor', 'bounded-floor', 'hand-tracking']
  }).then(s=>{
    renderer.xr.setReferenceSpaceType('local-floor');
    return renderer.xr.setSession(s).then(()=>s);
  }).then(s=>{ VR.session = s; }).catch(e=>{
    if(typeof toast === 'function') toast('VR would not start: ' + e.message);
  });
}
function vrOnStart(){
  VR.active = true;
  VR.fly = false;                    // every session starts on its feet
  /* ask the runtime to pace the session at 72Hz.  The browser defaults a
     Quest to 90, and the first headset run could not hold it; a held 72
     beats a 90 that drops, and it hands every frame 13.9ms of budget
     instead of 11.1.  The rates on offer vary by device, so take the
     LOWEST one at or above 72 — never below: under 72 the display
     flickers.  Where the API is missing this leaves everything alone. */
  VR.targetHz = 90;
  const s = renderer.xr.getSession && renderer.xr.getSession();
  if(s && s.supportedFrameRates && s.updateTargetFrameRate){
    let want = 0;
    for(const r of s.supportedFrameRates) if(r >= 71.9 && (!want || r < want)) want = r;
    if(want){
      VR.targetHz = want;
      const p = s.updateTargetFrameRate(want);
      if(p && p.catch) p.catch(()=>{});   // refusal is fine — the default stands
    }
  }
  /* the camera has to live inside a rig, because in a session the headset
     writes camera.position every frame and would throw ours away */
  if(!VR.rig){
    VR.rig = new T.Group();
    VR.rig.name = 'vr:rig';
    scene.add(VR.rig);
  }
  VR.camHome = camera.parent;
  VR.rig.add(camera);
  Player.mode = 'walk';
  Player.pitch = 0;
  vrQualityOn();
  vrBuildControllers();
  vrBuildDesks();
  vrBuildRopes();
  vrBuildBelt();
  vrDrawConsole(true);
  /* RULING DK — AND THE SESSION'S OWN GEOMETRY SAMPLES THE ENVIRONMENT.  The
     five builders above mint ~142 standard materials long after init() ran, and
     the biggest block of them is the hemp rope rail you stand at.  Left
     unregistered they sit at the default 1: 1.8x the driven value at full bed
     and 7.27x in a blackout, in the one place the spec says it has to match
     ("It has to match in VR").  The normal order is to load a show on the desk
     and then put the headset on, so nothing downstream would have healed it. */
  if(typeof envRecollect === 'function') envRecollect();
  const app = $('#app'); if(app) app.classList.add('inVR');
  if(typeof toast === 'function') toast('Walk with the left stick, turn with the right');
}
function vrOnEnd(){
  VR.active = false;
  VR.session = null;
  if(VR.camHome) VR.camHome.add(camera); else scene.add(camera);
  vrQualityOff();
  /* the roller re-racks BEFORE the hold record drops — it is parented to
     the controller, the one carried thing that is (release always homes it) */
  if(VR.held && VR.held.kind === 'roller') vrReRack(VR.held.rack);
  VR.held = null;
  VR.heldLever = null;
  /* every tool back on the belt, no line, no stale joint offer */
  if(VR.tools){ vrHolster(0); vrHolster(1); }
  VR.snap = null;
  if(VR.tapeLn) VR.tapeLn.visible = false;
  vrLabel(null);
  /* the meter's window is this session's story — start the next one clean,
     and put the foveation base back where vrQualityOn will set it */
  if(VR.perf){ VR.perf.n = 0; VR.perf.i = 0; VR.perf.t = 0; VR.perf.fov = 0.4; }
  const app = $('#app'); if(app) app.classList.remove('inVR');
}

/* ---- where the rig stands.  Called instead of writing camera.position ---- */
function vrPlaceRig(){
  if(!VR || !VR.rig) return;
  VR.rig.position.set(Player.pos.x, Player.pos.y, Player.pos.z);
  VR.rig.rotation.set(0, Player.yaw, 0);
}

/* ============================================================================
   CONTROLLERS
   ========================================================================== */
function vrBuildControllers(){
  if(VR.controllers.length) return;
  const rayMat = new T.LineBasicMaterial({color:0x8fd0ff, transparent:true, opacity:0.7});
  for(let i=0;i<2;i++){
    const c = renderer.xr.getController(i);
    c.userData.hand = i;
    VR.rig.add(c);
    VR.controllers.push(c);

    /* both hands point: the right one runs the desks, and either trigger
       is the desktop E key on whatever the ray lands on in the room */
    const g = new T.BufferGeometry().setFromPoints(
      [new T.Vector3(0,0,0), new T.Vector3(0,0,-4)]);
    const line = new T.Line(g, rayMat);
    line.name = 'vr:ray' + i;
    c.add(line);
    if(i === 1){
      VR.ray = line;
      const dot = new T.Mesh(new T.SphereGeometry(0.014, 8, 6),
        new T.MeshBasicMaterial({color:0xffd08a}));
      dot.visible = false;
      VR.rig.add(dot);
      VR.cursor = dot;
    }
    c.addEventListener('selectstart', ()=>vrSelect(i, true));
    c.addEventListener('selectend',   ()=>vrSelect(i, false));
    c.addEventListener('squeezestart',()=>vrSqueeze(i, true));
    c.addEventListener('squeezeend',  ()=>vrSqueeze(i, false));

    const grip = renderer.xr.getControllerGrip(i);
    const body = new T.Mesh(new T.BoxGeometry(0.045, 0.03, 0.12),
      new T.MeshStandardMaterial({color:0x1a1c22, roughness:.6}));
    body.position.z = 0.02;
    const ring = new T.Mesh(new T.TorusGeometry(0.032, 0.008, 6, 16),
      new T.MeshStandardMaterial({color:0x3a3f48, roughness:.5, metalness:.6}));
    ring.rotation.x = 0.4; ring.position.set(0, 0.01, -0.03);
    grip.add(body, ring);
    VR.rig.add(grip);
    VR.grips.push(grip);
  }
}

/* read the sticks straight off the input sources */
function vrReadSticks(){
  VR.axes.lx = VR.axes.ly = VR.axes.rx = VR.axes.ry = 0;
  const s = renderer.xr.getSession && renderer.xr.getSession();
  if(!s || !s.inputSources) return;
  for(const src of s.inputSources){
    const gp = src.gamepad;
    if(!gp || !gp.axes) continue;
    // 0,1 are the touchpad on some runtimes; 2,3 the stick.  Take whichever moves.
    const ax = (Math.abs(gp.axes[2]||0) + Math.abs(gp.axes[3]||0) > 0.001)
      ? [gp.axes[2]||0, gp.axes[3]||0] : [gp.axes[0]||0, gp.axes[1]||0];
    const dead = v => Math.abs(v) < 0.12 ? 0 : v;
    if(src.handedness === 'left'){
      VR.axes.lx = dead(ax[0]); VR.axes.ly = dead(ax[1]);
      /* xr-standard: left 4 is X — held, it frees a carried piece from the
         45-degree grid (RULING M).  Guarded: some pads stop at 4 buttons. */
      const bt = gp.buttons;
      VR.btnX = !!(bt && bt[4] && bt[4].pressed);
    }
    else                         { VR.axes.rx = dead(ax[0]); VR.axes.ry = dead(ax[1]); }
  }
}

/* the LEFT controller is the compass (RULING O): stick walking follows
   where it POINTS, not where the headset happens to face — point it right,
   push forward, walk right.  Returns a world yaw for updatePlayer (p7), or
   null when there is no session to read.  A controller aimed within ~14°
   of straight up or down leaves no floor direction: fall back to the
   HEADSET's yaw, which is what a hand at rest used to mean anyway. */
const _cmpV = new T.Vector3();
function vrMoveYaw(){
  if(!VR || !VR.active || !VR.controllers[0]) return null;
  const c = VR.controllers[0];
  c.updateMatrixWorld(true);
  _m.identity().extractRotation(c.matrixWorld);
  _cmpV.set(0, 0, -1).applyMatrix4(_m);
  if(_cmpV.x*_cmpV.x + _cmpV.z*_cmpV.z < 0.06){
    camera.updateMatrixWorld(true);
    _m.identity().extractRotation(camera.matrixWorld);
    _cmpV.set(0, 0, -1).applyMatrix4(_m);
    if(_cmpV.x*_cmpV.x + _cmpV.z*_cmpV.z < 1e-6) return Player.yaw;
  }
  /* p7's convention: forward = (-sin yaw, -cos yaw) */
  return Math.atan2(-_cmpV.x, -_cmpV.z);
}

/* the A button on the right hand: tap to jump, two quick taps to fly.
   Time comes off the frame clock (dt), not the wall, so a test can drive it. */
function vrButtonJump(dt){
  VR.clk = (VR.clk || 0) + dt;
  const s = renderer.xr.getSession && renderer.xr.getSession();
  let a = false;
  if(s && s.inputSources) for(const src of s.inputSources){
    if(src.handedness !== 'right') continue;
    const b = src.gamepad && src.gamepad.buttons;
    /* xr-standard mapping: 0 trigger, 1 squeeze, 3 stick, 4 = A */
    if(b && b[4] && b[4].pressed) a = true;
  }
  if(a && !VR.btnA){                              // the rising edge of a press
    if(VR.clk - (VR.tapA === undefined ? -1e9 : VR.tapA) < 0.35){
      VR.fly = !VR.fly;
      Player.vel.y = 0;
      VR.tapA = -1e9;                             // a third press starts over
      if(typeof toast === 'function')
        toast(VR.fly ? 'FLYING — push the stick where you look; double-tap A or land to walk'
                     : 'Walking');
    } else {
      VR.tapA = VR.clk;
      if(!VR.fly && Player.onGround){ Player.vel.y = 4.6; Player.onGround = false; }
    }
  }
  VR.btnA = a;
}

/* Y on the left hand: park the held piece of wood exactly where it is —
   position, angle, mid-air included (RULING N).  The park is b.frozen:
   updateBodies never settles it, a grab takes it back (grabBody clears the
   flag), and the flag rides the save.  Edge-triggered off the frame, the
   vrButtonJump pattern. */
function vrButtonFreeze(){
  const s = renderer.xr.getSession && renderer.xr.getSession();
  let y = false;
  if(s && s.inputSources) for(const src of s.inputSources){
    if(src.handedness !== 'left') continue;
    const bt = src.gamepad && src.gamepad.buttons;
    /* xr-standard: left 5 is Y.  Guarded: some pads stop at 4 buttons. */
    if(bt && bt[5] && bt[5].pressed) y = true;
  }
  if(y && !VR.btnY && VR.held && VR.held.kind === 'body' &&
     VR.held.body.kind === 'wood'){
    const b = VR.held.body;
    VR.held = null;
    VR.snap = null;
    b.state = 'loose';
    b.frozen = true;
    vrLabel(null);
    if(typeof buildDirty === 'function') buildDirty();
    if(typeof toast === 'function')
      toast('parked — it stays put exactly there; grab it to move it');
  }
  VR.btnY = y;
}

/* ============================================================================
   THE CONSOLE FACE
   Drawn onto a canvas and mapped onto the desk.  It cannot be the page you
   see on a monitor — a browser will not hand you HTML as a texture — so this
   is a console drawn for the purpose, with the same things on it.
   ========================================================================== */
const VRC = {W:1200, H:760};
function vrCanvas(){
  if(VR.canvas) return VR.canvas;
  const c = cvs(VRC.W, VRC.H);
  VR.canvas = c;
  VR.ctx = c.getContext('2d');
  VR.tex = tex(c, 1, 1);
  return c;
}

/* meta rides the hit record: the goods picker tags its regions so the row
   that opens it and the entry that hangs are findable by what they MEAN,
   not by their pixel coordinates (vr.js pins the fly rows by pixel and
   that has been a trap twice) */
function vrHit(x, y, w, h, fn, meta){
  const o = {x, y, w, h, fn};
  if(meta) Object.assign(o, meta);
  VR.hits.push(o);
}

  vrCanvas();
  const g = VR.ctx, W = VRC.W, H = VRC.H;
  VR.hits.length = 0;

  g.fillStyle = '#0b0d10'; g.fillRect(0,0,W,H);
  g.fillStyle = '#12161c'; g.fillRect(0,0,W,64);
  g.fillStyle = '#1d242e'; g.fillRect(0,62,W,2);

  // which board is this

  g.fillStyle = '#ffb347';

     and SCRIPT would be drawn off the edge with a hit region nobody can reach.
     From 300 the last box ends at 1190.  Nothing pins a tab by pixel —
     tests/vr.js sets VR.page directly — so the strip is free to move and the
     rows below it, which ARE pinned by pixel, are not. */
  let tx = 300;
  for(const t of VR_TABS){
    const w = 112, on = (VR.page === t.id);
    g.fillStyle = on ? '#26313e' : '#161b22';
    g.fillRect(tx, 12, w - 6, 40);
    g.fillStyle = on ? '#ffd08a' : '#8b95a3';
    g.font = '600 17px Helvetica, Arial, sans-serif';
    g.textAlign = 'center';
    g.fillText(t.label, tx + (w-6)/2, 33);
    vrHit(tx, 12, w - 6, 40, ((id)=>()=>{ VR.page = id; vrDrawConsole(true); })(t.id));
    tx += w;
  }
  g.textAlign = 'left';

  const P = {x:22, y:86, w:W-44, h:H-108};
  if(VR.page === 'cues')   vrPageCues(g, P);
  if(VR.page === 'lights') vrPageLights(g, P);
  if(VR.page === 'fly')    vrPageFly(g, P);
  if(VR.page === 'sets')   vrPageSets(g, P);    // RULING CV
  if(VR.page === 'goods')  vrPageGoods(g, P);   // a sub-page of the fly rail
  if(VR.page === 'shows')  vrPageShows(g, P);
  if(VR.page === 'venues') vrPageVenues(g, P);
  if(VR.page === 'smoke')  vrPageSmoke(g, P);
  if(VR.page === 'script') vrPageScript(g, P);
  if(VR.page === 'lighting') vrPageLighting(g, P);   // RULING DP

  if(VR.tex) VR.tex.needsUpdate = true;
}

function vrBtnBox(g, x, y, w, h, label, fn, style, meta){

  if(fn) vrHit(x, y, w, h, fn, meta);
}

/* ---------------------------------------------------------------- THE SETS
   RULING CV — A MENU TO CONTROL WHAT SETS ARE ON.

   "Make a menu to control what sets are on..."

   THE DESK HAS HAD ONE ALL ALONG (#sceneList) AND THE HEADSET HAS NOT, which is
   the whole of this: VR_TABS was CUES, LIGHTS, FLY RAIL, SHOWS, VENUES, SMOKE,
   SCRIPT, and he works this show on a Quest.  "A control that exists only in the
   DOM does not exist in VR" is in TRAPS with four rounds of unreachable Arc
   doors behind it.

   It changes the set through sceneChangeTo, so the change is CHOREOGRAPHED
   exactly as a cue's would be — the sets travel, they do not pop (RULING AY).

   AND IT SAYS WHERE EACH SET IS, not just which one is on, because RULING BQ
   made that a real question: a struck set is standing in a wing, or upstage
   behind the backdrop, or up in the tower, and that is the thing this menu is
   actually for now. */
function vrSetWhere(sc){
  if(sc.on) return 'ON STAGE';
  if(!sc.parks) return 'struck';
  if(sceneTravelling(sc)) return 'travelling';
  const b = new T.Box3().setFromObject(sc.group);
  if(b.isEmpty()) return 'parked';
  if(b.min.y > 9.5) return 'flown';
  /* THE SIDE IS ASKED BEFORE THE DEPTH, and RULING DI is why.  These three tests
     are not exclusive — a set can be in a wing AND upstage of the backdrop at
     once, and the packed stage-right parks are exactly that: the bedroom stands
     at x -14.20 .. -22.82, z -12.90 .. -17.51.  Asking about z first answered
     "upstage" for it, which is true and useless — it sends a man in a headset
     behind the cloth to look for a set that is out in the wing beside him.
     "Which wing" is the answer that walks you to it, so it goes first, and
     upstage keeps what it was always for: the house wagon, which parks within
     the width of the picture and really is only findable behind the backdrop. */
  if(b.min.x > 6.8) return 'stage left';
  if(b.max.x < -6.8) return 'stage right';
  if(b.max.z < -10.9) return 'upstage';
  return 'parked';
}

/* ---------------------------------------------------------- THE GOODS

   from the fly page's goods cell, never from the tab strip: it is about

/* ============================================================================
   RULING DP — THE LIGHTING PAGE IN THE HEADSET.
   A control that exists only in the DOM does not exist in VR (CV/CW), and the
   desk grew this page one ruling ago (DO).  This is the same page, same rows,
   same order.

   IT DRIVES THE DESK'S OWN HANDLERS.  Every press is lkSet(key, ...) — p7's
   one map, p7's one clamp.  A headset row that could push a constant past the
   desk's range would be two different controls wearing one name, and nothing
   would ever tell you which one you had used.  The step is LK[k].step, so the
   two surfaces move by the same amount as well as to the same limits.

   FOUND BY META, NEVER BY PIXEL.  vrHit copies the meta flat onto the record,
   so a test asks for {lk:'sat', d:+1} rather than for a coordinate.  vr.js
   pins some older rows at literal y (86+312, +366, +448, +502) and VR.md
   records that this has been a trap twice; adding a tab cannot move them
   because tabs lay out left to right and this one is appended last.
   ========================================================================== */

/* ============================================================================
   THE DESKS
   One in each house you can run a show from.  They all show the same board,
   because there is only one board — it is patched to whichever theatre you
   are standing in.
   ========================================================================== */
function vrMakeDesk(o){
  const g = new T.Group();
  g.position.set(o.x, o.y, o.z);
  g.rotation.y = o.yaw || 0;
  (o.parent || world).add(g);

  const shell = new T.MeshStandardMaterial({color:0x14181e, roughness:.7, metalness:.2});
  const trimM = new T.MeshStandardMaterial({color:0x2a323c, roughness:.5, metalness:.5});

  // the carcass: a desk you stand at, top at about a metre
  const top = new T.Mesh(new T.BoxGeometry(1.5, 0.06, 0.62), shell);
  top.position.set(0, 0.98, 0); g.add(top);
  const front = new T.Mesh(new T.BoxGeometry(1.5, 0.94, 0.06), shell);
  front.position.set(0, 0.5, 0.3); g.add(front);
  for(const sx of [-1, 1]){
    const leg = new T.Mesh(new T.BoxGeometry(0.07, 0.96, 0.55), trimM);
    leg.position.set(sx*0.7, 0.48, 0); g.add(leg);
  }
  // the face, raked back like a console
  const faceGeo = new T.PlaneGeometry(1.42, 0.9);
  const faceMat = new T.MeshBasicMaterial({map:VR.tex});
  const face = new T.Mesh(faceGeo, faceMat);
  face.position.set(0, 1.36, -0.24);
  face.rotation.x = -0.42;
  face.userData.deskFace = true;
  g.add(face);
  const bezel = new T.Mesh(new T.BoxGeometry(1.5, 0.98, 0.04), trimM);
  bezel.position.set(0, 1.36, -0.27);
  bezel.rotation.x = -0.42;
  g.add(bezel);

  // a little desk lamp so you can find it in a blackout
  const lamp = new T.Mesh(new T.SphereGeometry(0.03, 8, 6),
    new T.MeshBasicMaterial({color:0xff9a3c}));
  lamp.position.set(-0.62, 1.06, -0.1); g.add(lamp);

  // and the GO button, on the desk, to be pressed with a hand
  const goBtn = new T.Mesh(new T.CylinderGeometry(0.055, 0.055, 0.028, 16),
    new T.MeshStandardMaterial({color:0x2fbf4a, roughness:.5}));
  goBtn.position.set(0.52, 1.015, 0.02);
  goBtn.userData.goButton = true;
  g.add(goBtn);
  VR.goButtons.push({mesh:goBtn, y0:goBtn.position.y, cool:0});

  const d = {group:g, face, label:o.label, stage:o.stage};
  VR.desks.push(d);
  VR.deskMeshes.push(face);
  return d;
}
function vrBuildDesks(){
  if(VR.desks.length) return;
  vrCanvas();

  /* THE PALACE — on the first balcony rail, where a control position sits */
  vrMakeDesk({
    label:'THE PALACE', stage:'palace',
    /* yaw 0: the raked face looks back up the house (+z), so the operator
       stands BEHIND the desk and reads it looking downstage at the stage */
    x:0, y:D.mezzY, z:D.mezzZ + 1.5, yaw:0,
    parent:(typeof ROOM_GROUP !== 'undefined' && ROOM_GROUP.house) ? ROOM_GROUP.house : world
  });

  /* THE ARC — a tech table out in the stalls and a desk at the control room */
  if(typeof ARC !== 'undefined' && ARC.built){
    for(const key of ['main','studio']){
      const H = ARC.houses[key];
      if(!H || !H.rake) continue;
      const room = ARC.rooms[H.room];
      /* yaw 0, same ruling as the Palace desk: the Arc houses share the
         Palace's orientation (stage −z, house +z), so a readable desk
         points its raked face UP the house and the operator stands behind
         it looking downstage.  These two were built at yaw π — every desk
         in the building turned its screen to the stage and its back to
         the person running the show. */
      const midRow = Math.floor(H.rake.zFirst + (H.rake.zLast - H.rake.zFirst)*0.55);
      const rows = Math.round((midRow - H.rake.zFirst)/H.rake.RUN);
      vrMakeDesk({
        label:H.label + ' — TECH TABLE', stage:H.stage,
        x:H.cx, y:H.rake.Y0 + rows*H.rake.RISE + 0.2,
        z:H.rake.zFirst + rows*H.rake.RUN, yaw:0, parent:room
      });
      // the control room, at the back under its window
      vrMakeDesk({
        label:H.label + ' — CONTROL ROOM', stage:H.stage,
        x:H.cx, y:AS.BACKY, z:H.zHouse - 1.5, yaw:0, parent:room
      });
    }
  }
  vrBuildOrderScreens();
}

/* ============================================================================
   THE ORDER SCREENS (spec §6) — a wall panel in each shed, the console
   pattern at one-canvas-per-screen scale: its own texture, its own pixel
   hit list, the same ray + trigger and the same v = 1 - uv.y flip (the
   vrSelect branch).  State and delivery live in p2m (ORDERS/orderPlace);
   this is only the glass.
   ========================================================================== */
const ORDER_KINDS = ['profile','fresnel','par','cyc','mover','speaker'];
const ORDER_TABS = ['GEAR','WOOD','HDWE','PAINT'];
/* the rows of each tab: a key into sc.counts, a label, and the order UNIT
   a count of it turns into (p2m orderPlace) */
function orderRows(tab){
  if(tab === 0) return ORDER_KINDS.map(k=>({key:k, label:BODY_LABEL[k], unit:{kind:k}}));
  if(tab === 1) return [
    {key:'sheet', label:'4x8 SHEET 3/4', unit:{kind:'wood', prof:'sheet'}},
    {key:'s2x4',  label:'2x4 - 8FT',     unit:{kind:'wood', prof:'s2x4'}},
    {key:'s4x4',  label:'4x4 - 8FT',     unit:{kind:'wood', prof:'s4x4'}},
    {key:'s2x8',  label:'2x8 - 8FT',     unit:{kind:'wood', prof:'s2x8'}}];
  if(tab === 2) return [
    {key:'hinge',    label:'HINGE',       unit:{kind:'hinge'}},
    {key:'track',    label:'TRACK - 4FT', unit:{kind:'track'}},
    {key:'carriage', label:'CARRIAGE',    unit:{kind:'carriage'}},
    {key:'table',    label:'WORK TABLE',  unit:{kind:'table'}}];
  return PAINT_COLORS.map((pc, i)=>({key:'p'+i, label:pc.name, swatch:pc.c,
                                     unit:{kind:'paint', color:pc.c}}));
}
function orderTotal(sc){
  let n = 0; for(const k in sc.counts) n += sc.counts[k];
  return n;
}

function vrDrawOrder(sc){
  const g = sc.ctx, W = sc.canvas.width;
  sc.hits.length = 0;
  g.fillStyle = '#0b0e12'; g.fillRect(0, 0, W, sc.canvas.height);
  g.fillStyle = '#ffd08a'; g.font = 'bold 26px monospace';
  g.fillText('WAREHOUSE SUPPLY', 24, 40);
  /* the piece cap, on the glass where the refusal will happen (spec §3) */
  g.font = '18px monospace';
  g.fillText('PIECES ' + venueBuildCount(sc.which) + '/' + BUILD_CAP, 396, 40);
  const o = (typeof ORDERS !== 'undefined') ? ORDERS[sc.which] : null;
  ORDER_TABS.forEach((tl, i)=>{
    const x = 24 + i*130;
    g.fillStyle = sc.tab === i ? '#3a4657' : '#1a212b';
    g.fillRect(x, 56, 122, 40);
    g.fillStyle = sc.tab === i ? '#ffd08a' : '#8fa0b4';
    g.font = 'bold 20px monospace';
    g.fillText(tl, x + 24, 83);
    sc.hits.push({x, y:56, w:122, h:40, fn:()=>{ sc.tab = i; vrDrawOrder(sc); }});
  });
  const rows = orderRows(sc.tab);
  const paint = sc.tab === 3;              // ten rows only fit drawn small
  rows.forEach((r, i)=>{
    const y = 112 + i*(paint ? 28 : 48);
    const h = paint ? 24 : 40;
    const bx1 = paint ? 340 : 300, bx2 = paint ? 460 : 440, bw = paint ? 40 : 56;
    if(paint){
      g.fillStyle = '#' + ('00000' + r.swatch.toString(16)).slice(-6);
      g.fillRect(24, y + 2, 28, 20);
      g.fillStyle = '#cfd8e3'; g.font = '16px monospace';
      g.fillText(r.label, 60, y + 18);
    } else {
      g.fillStyle = '#cfd8e3'; g.font = '20px monospace';
      g.fillText(r.label, 24, y + 27);
    }
    g.fillStyle = '#232b35'; g.fillRect(bx1, y, bw, h); g.fillRect(bx2, y, bw, h);
    g.fillStyle = '#ffd08a'; g.font = paint ? 'bold 18px monospace' : 'bold 26px monospace';

    g.fillStyle = '#eef3f8';
    g.fillText('' + (sc.counts[r.key] || 0), (bx1 + bx2 + bw)/2 - 8, y + h - (paint ? 6 : 12));

      vrDrawOrder(sc);
    }});
    sc.hits.push({x:bx2, y, w:bw, h, fn:()=>{
      if(orderTotal(sc) < ORDER_MAX){ sc.counts[r.key] = (sc.counts[r.key] || 0) + 1; sc.status = ''; }

    }});
  });
  const oy = 404;
  g.fillStyle = '#2fbf4a'; g.fillRect(24, oy, 300, 56);
  g.fillStyle = '#08130a'; g.font = 'bold 30px monospace';
  g.fillText('ORDER', 118, oy + 39);

  } else g.fillText(line || 'THE SHELVES ARE YOURS', 24, oy + 92);
  sc.tex.needsUpdate = true;
}
function vrOrderPress(sc){
  const units = [];
  [0, 1, 2, 3].forEach(t=>{
    orderRows(t).forEach(r=>{
      for(let i = 0; i < (sc.counts[r.key] || 0); i++) units.push(r.unit);
    });
  });
  const r = (typeof orderPlace === 'function') ? orderPlace(sc.which, units) : 'BUSY';
  if(r === 'OK'){ sc.counts = {}; sc.status = ''; }
  else if(r === 'COUNT') sc.status = units.length ? 'TWELVE TO AN ORDER' : 'NOTHING ON THE SLIP';
  else if(r === 'BUSY') sc.status = 'THREE ORDERS OUT ALREADY';
  else sc.status = r;   // STOCK FULL / PIECES FULL
  vrDrawOrder(sc);
}

/* ============================================================================

   beside each shed's order screen, built in the order screen's mould:
   its own 560x520 canvas, its own hits[], its own userData tag and its
   own vrSelect branch.  The ORDER canvas is pixel-pinned by tests and
   does not move; every region HERE carries META on the hit record
   (carpKey / carpCall) so tests find it by meaning, never by pixel
   (VR.md, the goods-picker rule).  This is only the glass: the catalogue
   and the pure planner are p6c's, the run is the crew's (carpStart), and
   the refusals are worded here in spec order.
   ========================================================================== */

/* the CALL: refusals in spec order — NO MARK, MARK IS IN THE OTHER HOUSE,
   then carpStart's own contract worded on the glass (CREW BUSY, the NEED
   list, PIECES FULL).  Success gets the show crew's work light and toast:
   a build runs under work light like a get-in (crewLoadShow's shape). */
function vrCarpPress(sc){
  const mark = (typeof CARP !== 'undefined' && CARP) ? CARP.mark : null;
  /* the LIST (RULING AG): every row with a count on it, in catalogue
     order.  A count of nothing anywhere falls back to the selected row,
     so one press on a row and CALL still builds one of it. */
  sc.counts = sc.counts || {};
  let items = Object.keys(CARP_CAT).filter(k=>sc.counts[k] > 0)
                    .map(k=>({key:k, n:sc.counts[k]}));
  if(!items.length && sc.sel) items = [{key:sc.sel, n:1}];
  if(!items.length) sc.status = 'PICK A PIECE';
  else if(!mark) sc.status = 'NO MARK';
  else if(mark.venue !== sc.which) sc.status = 'MARK IS IN THE OTHER HOUSE';
  else {
    const r = (typeof carpStart === 'function')
      ? carpStart(items, mark, {skin:sc.skin}) : null;
    if(!r) sc.status = 'NO MARK';        // a dead mark record: cannot plan
    else if(r.busy) sc.status = 'CREW BUSY';
    else if(r.need) sc.status = carpNeedLine(r.need);
    else if(r.full) sc.status = 'PIECES FULL';
    else {
      sc.status = 'CALLED — ' + carpListLine(items);
      if(typeof crewWorkLight === 'function'){
        crewWorkLight(true);
        CREW.onDone = was=>{ if(was === 'carp') crewWorkLight(false); };
      }
      if(typeof toast === 'function')
        toast('the carpenters are called — ' + carpListLine(items) + ' at the mark');
      sc.counts = {};                    // the list is spoken for
    }
  }
  vrDrawCarp(sc);
}

/* the trigger of a tool hand: fire the tool */
function vrToolFire(hand, key){
  const c = VR.controllers[hand];
  if(!c) return;
  c.updateMatrixWorld(true);
  _rayO.setFromMatrixPosition(c.matrixWorld);
  if(key === 'nailgun'){
    /* a HINGE in the other hand: the shot installs it at the seam, its
       own pin axis the swing axis (p4c addHinge) */
    if(VR.held && VR.held.kind === 'body' && VR.held.body.kind === 'hinge'){
      const hb = VR.held.body;
      hb.mesh.getWorldPosition(_vecA);
      if(_rayO.distanceTo(_vecA) > 1.0){
        if(typeof toast === 'function') toast('closer to the hinge'); return;
      }
      _run.set(0, 1, 0).applyQuaternion(hb.mesh.getWorldQuaternion(_tq1));
      const hn = addHinge(hb, _vecA.clone(), _run.clone());
      if(hn) VR.held = null;
      if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
      if(typeof toast === 'function')
        toast(hn ? 'hinged — it swings now' : 'a hinge wants a loose piece against another');
      return;
    }
    const s = VR.snap;
    if(!s){
      /* no held offer: the nail goes where the gun POINTS (RULING L) —
         the ray picks the spot; the piece touching it there is the
         partner.  Only a ray that hits no wood at all falls back to the
         muzzle-proximity seam (#49), which still covers pointing in the
         general direction of a lying-together pair. */
      _m.identity().extractRotation(c.matrixWorld);
      _rayD.set(0, 0, -1).applyMatrix4(_m).normalize();
      const nr = (typeof nailRay === 'function') ? nailRay(_rayO, _rayD) : null;
      if(nr && nr.b){
        const n = addNail(nr.a, {body:nr.b}, nr.point, nr.axis);
        if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
        if(typeof toast === 'function') toast(n ? 'nailed' : 'that will not take a nail');
        return;
      }
      if(nr && nr.miss){
        if(typeof toast === 'function') toast('nothing behind it to bite');
        return;
      }
      const sm = (typeof seamSeek === 'function') ? seamSeek(_rayO) : null;
      if(sm){
        const n = addNail(sm.a, {body:sm.b}, sm.point, sm.axis);
        if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
        if(typeof toast === 'function') toast(n ? 'nailed' : 'that will not take a nail');
      } else if(typeof toast === 'function') toast('nothing lined up to nail');
      return;
    }
    if(s.point && _rayO.distanceTo(s.point) > 1.0){
      if(typeof toast === 'function') toast('closer to the joint'); return;
    }
    let n = null;
    if(s.lay !== undefined){
      /* a snapped track section: lay it into the run (or start one) */
      n = (typeof layTrack === 'function') ? layTrack(s.body, s.lay.asm) : null;
      if(n && VR.held && VR.held.kind === 'body' && VR.held.body === s.body) VR.held = null;
    } else if(s.target && s.target.carriage){
      n = (typeof nailToCarriage === 'function') ? nailToCarriage(s.body, s.target.carriage) : null;
      if(n && VR.held && VR.held.kind === 'body' && VR.held.body === s.body) VR.held = null;
    } else if(s.target && s.target.table){
      /* the table is a work surface, never a joint (build-usability §5) */
      if(typeof toast === 'function') toast('the table holds it — no nail needed');
      VR.snap = null;
      vrLabel(null);
      return;
    } else if(s.body){
      n = addNail(s.body, s.target, s.point, s.axis);
      if(n && VR.held && VR.held.kind === 'body' && VR.held.body === s.body) VR.held = null;
    } else if(s.asm){
      n = addNail(s.piece, s.target, s.point, s.axis);
      if(n && VR.held && VR.held.kind === 'asm' && VR.held.asm === s.asm) VR.held = null;
    }
    VR.snap = null;
    vrLabel(null);
    if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
    if(typeof toast === 'function') toast(n ? 'nailed' : 'that will not take a nail');
  } else if(key === 'hammer'){
    let best = null, bd = 0.25;
    if(typeof ASSEMBLIES !== 'undefined')
      for(const a of ASSEMBLIES)
        for(const n of a.nails){
          n.mesh.getWorldPosition(_vecA);
          const d = _vecA.distanceTo(_rayO);
          if(d < bd){ bd = d; best = n; }
        }
    if(best){
      removeNail(best);
      if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
      if(typeof toast === 'function') toast(best.hinge ? 'hinge off' : 'nail out');
      return;
    }
    /* no nail: an empty carriage under the hammer pops off its run */
    let car = null, crd = 0.25;
    for(const b of BODIES){
      if(b.kind !== 'carriage' || !b.slider) continue;
      b.mesh.getWorldPosition(_vecA);
      const d = _vecA.distanceTo(_rayO);
      if(d < crd){ crd = d; car = b; }
    }
    if(car){
      if(unrideTrack(car)){
        if(typeof toast === 'function') toast('carriage off the run');
      } else if(typeof toast === 'function') toast('take its load off first');
    } else if(typeof toast === 'function') toast('no nail under the hammer');
  } else if(key === 'tape'){
    if(!(VR.held && VR.held.kind === 'tapetab')) return;
    const oc = VR.controllers[VR.held.hand];
    if(!oc) return;
    oc.updateMatrixWorld(true);
    _vecA.setFromMatrixPosition(oc.matrixWorld);
    let best = null, bd = 1.6;
    for(const b of BODIES){
      if(b.kind !== 'wood') continue;
      b.mesh.getWorldPosition(_run);
      const d = _run.distanceTo(_vecA);
      if(d < bd){ bd = d; best = b; }
    }
    if(best){
      best.tick = {p: best.mesh.worldToLocal(_vecA.clone())};
      if(typeof toast === 'function')
        toast('marked at ' + ftIn(VR.held.len || 0));
    }
  } else if(key === 'crayon'){
    /* the build mark goes where the crayon points (RULING Z): a forward
       ray at the WALKABLE floors, ankle height only — the deck is y = 0
       everywhere, and the gallery at y ~8 is a floor but not a deck. */
    _m.identity().extractRotation(c.matrixWorld);
    _rayD.set(0, 0, -1).applyMatrix4(_m).normalize();
    _crayRay.set(_rayO, _rayD);
    const hits = _crayRay.intersectObjects(WALKABLE, false);
    const h = hits.length ? hits[0] : null;
    if(!h){
      if(typeof toast === 'function') toast('point the crayon at a floor');
      return;
    }
    if(h.point.y >= 0.3){
      if(typeof toast === 'function') toast('not up there — mark a deck');
      return;
    }
    const st = (typeof stageAt === 'function') ? stageAt(h.point.x, h.point.z) : 'palace';
    if(!st){
      if(typeof toast === 'function') toast('that floor belongs to no stage');
      return;
    }
    const v = (typeof venueAt === 'function') ? venueAt(h.point.x) : 'palace';
    carpSetMark(v, st, h.point.x, h.point.z, Math.atan2(_rayD.x, _rayD.z));
    if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
    if(typeof toast === 'function') toast('marked — the carpenters build here');
  }
}

/* ============================================================================
   THE ROPES
   A full operating line at the pin rail for every lineset that is hung: a
   floor block just off the deck, a head block up under the grid, and the line
   running all the way round the pair of them — two runs, the way a rope goes
   when it comes back to where it started.  Take hold of either run ANYWHERE
   along it and haul: on the front run, down brings the batten in and up sends
   it out; the back run is the other half of the same loop, so it works the
   other way round.  The blocks turn while the line travels.

   In front of the lines stands the LOCKING RAIL — a steel rail with a number
   plate for every lineset and a rope lock on top of each, its red handle
   facing the flyman.  The handle IS ls.locked, and only a hand that closes on
   it moves it: push it IN (upright) and the line is locked off, pull it OUT
   (toward you) and the line is released.  Release a line with nothing holding
   it — no lock, no hand — and there is nothing on the load at all: it runs
   away downwards (updateFly, p3).

   One rope on the stage is NOT a fly line: the traveler's hand line, a small
   loop of its own at the side of the proscenium.  Hauling it slides the house
   curtain open and shut instead of flying the pipe — front run down opens it
   out to both sides, the back run closes.  A hand line has no counterweight,
   so there is no lock and nothing to run away: let go and the curtain simply
   stays where the hand left it.
   ========================================================================== */
const ROPE = {
  headDrop: 0.6,      // the head block hangs this far under the grid
  floorY:   0.25,     // the floor block sits just off the deck
  gap:      0.06,     // half the distance between the two runs of the loop
  wheel:    0.10,     // sheave radius — also what the spin is worked out from
  handY:    1.55,     // where the whipped grab section sits at rest
  railY:    0.96,     // the top surface of the locking rail
  railH:    0.14,     // the rail beam, top to bottom
  railD:    0.12,     // and front to back
  railOff:  0.14,     // beam centre, this far in FRONT of the line of ropes
  lockH:    0.15,     // the rope-lock housing on top of the rail
  grabR:    0.32,     // a hand within this of a run has hold of the rope
  knobR:    0.12,     // a hand within this of a handle has hold of the LEVER
  throw_:   0.95,     // how far the handle lies over when it is pulled OUT
  travTop:  4.5,      // the traveler hand line's little head block
  travFloor:0.15,     // and its floor block, just off the deck
  travOff:  0.9,      // the hand line hangs this far offstage of the arch
  travScale:2.5       // metres of hand line for the curtain's full 0..1 travel
};
/* the number plates are drawn once and kept: vrClearRopes disposes geometry,
   and eleven little canvases a stage-walk would otherwise pile up as leaks */
const ROPE_PLATES = {};
function ropePlateTex(n){
  if(ROPE_PLATES[n]) return ROPE_PLATES[n];
  const c = cvs(64, 48), g = c.getContext('2d');
  g.fillStyle = '#ddd8cc'; g.fillRect(0, 0, 64, 48);
  g.strokeStyle = '#2a2d33'; g.lineWidth = 4; g.strokeRect(0, 0, 64, 48);
  g.fillStyle = '#15181d';
  g.font = '700 30px Helvetica, Arial, sans-serif';
  g.textAlign = 'center'; g.textBaseline = 'middle';
  g.fillText(String(n), 32, 26);
  ROPE_PLATES[n] = tex(c, 1, 1);
  return ROPE_PLATES[n];
}
function vrBuildRopes(){
  vrClearRopes();
  const st = (typeof STAGES !== 'undefined') ? STAGES[STAGE] : null;
  if(!st) return;
  const fr = (typeof crewFrame === 'function') ? crewFrame() : null;
  const railX = fr ? fr.rail : -D.stageW/2 + 2.8;
  const zOff = fr ? fr.zOff : 0;
  const parent = (typeof ROOM_GROUP !== 'undefined' && ROOM_GROUP.stage)
    ? ROOM_GROUP.stage : world;
  const mat  = new T.MeshStandardMaterial({color:0xb9a071, roughness:1});
  const grab = new T.MeshStandardMaterial({color:0xe0cb9c, roughness:1});
  const iron = new T.MeshStandardMaterial({color:0x3a3f48, roughness:.5, metalness:.6});
  const red  = new T.MeshStandardMaterial({color:0xc0392b, roughness:.45});

  const holder = new T.Group();
  holder.name = 'vr:ropes';
  /* the ropes are placed in WORLD coordinates, so the group holding them has
     to sit at the world origin whichever building we are in */
  if(st.venue === 'arc') holder.position.x = -ARC.X;
  (st.venue === 'arc' ? ARC.rooms[st.room] : parent).add(holder);
  VR.ropeRoot = holder;

  /* which side of the rail the stage is on.  The Palace's rail is stage right
     with the deck to its +x; the Arc's is stage left with the deck to its −x.
     The flyman stands on the deck side, so the run he takes hold of and the
     lever he throws both face that way. */
  const org = (typeof stageOrigin === 'function') ? stageOrigin() : {x:0, z:0};
  const inw = (org.x - railX) >= 0 ? 1 : -1;

  const headY = D.gridY - ROPE.headDrop;
  const runH  = headY - ROPE.floorY;
  /* one geometry of each, shared by every line on the rail */
  const gWheel = new T.CylinderGeometry(ROPE.wheel, ROPE.wheel, 0.03, 8);
  gWheel.rotateX(Math.PI/2);          // lay the sheave in the x-y plane: it now
                                      // spins about its own local z
  const gRun   = new T.CylinderGeometry(0.014, 0.014, runH, 6);
  const gGrab  = new T.CylinderGeometry(0.034, 0.034, 0.9, 6);
  const gLock  = new T.BoxGeometry(0.11, ROPE.lockH, 0.09);
  const gLever = new T.BoxGeometry(0.045, 0.26, 0.045);
  gLever.translate(0, 0.13, 0);       // pivot at the foot, so a rotation throws it
  const gKnob  = new T.SphereGeometry(0.045, 6, 5);
  const gPlate = new T.PlaneGeometry(0.085, 0.065);

  const hung = FLY.filter(l => l.goodsKey !== 'none');

  /* ---- the locking rail itself: one beam the length of the hung lines, a
     kick rail below it and a post every couple of metres, all in front of
     the ropes so the locks sit over the run the flyman hauls on ---------- */
  if(hung.length){
    const z0 = Math.min(...hung.map(l => l.z)) + zOff - 0.45;
    const z1 = Math.max(...hung.map(l => l.z)) + zOff + 0.45;
    const railC = railX + inw*ROPE.railOff;
    const beam = new T.Mesh(new T.BoxGeometry(ROPE.railD, ROPE.railH, z1 - z0), iron);
    beam.name = 'vr:rail';
    beam.position.set(railC, ROPE.railY - ROPE.railH/2, (z0 + z1)/2);
    holder.add(beam);
    const kick = new T.Mesh(new T.BoxGeometry(ROPE.railD*0.7, 0.06, z1 - z0), iron);
    kick.position.set(railC, 0.18, (z0 + z1)/2);
    holder.add(kick);
    const gPost = new T.BoxGeometry(0.06, ROPE.railY, 0.06);
    for(let z = z0 + 0.2; z < z1; z += 2.0){
      const post = new T.Mesh(gPost, iron);
      post.position.set(railC, ROPE.railY/2, z);
      holder.add(post);
    }
  }

  for(const ls of hung){
    const asm = new T.Group();
    asm.name = 'vr:rope' + ls.id;
    asm.position.set(railX, 0, ls.z + zOff);
    holder.add(asm);

    const head = new T.Mesh(gWheel, iron); head.position.y = headY;       asm.add(head);
    const foot = new T.Mesh(gWheel, iron); foot.position.y = ROPE.floorY; asm.add(foot);
    const runs = [];
    for(const s of [1, -1]){
      const run = new T.Mesh(gRun, mat);
      run.position.set(s*inw*ROPE.gap, ROPE.floorY + runH/2, 0);
      asm.add(run); runs.push(run);
    }
    /* the grab section — a whipped length of rope that slides to wherever a
       hand takes hold.  At rest it marks the front run at hand height, and
       it is the mesh the tests know a rope by. */
    const m = new T.Mesh(gGrab, grab);
    m.position.set(inw*ROPE.gap, ROPE.handY, 0);
    m.userData.rope = ls;
    asm.add(m);

    /* the rope lock: a housing on top of the rail that the front run passes
       through, its red handle on the operator's side.  IN (upright) is
       locked, OUT (lying toward the flyman) is released. */
    const lock = new T.Mesh(gLock, iron);
    lock.position.set(inw*0.11, ROPE.railY + ROPE.lockH/2, 0);
    asm.add(lock);
    const lever = new T.Mesh(gLever, red);
    lever.position.set(inw*0.11, ROPE.railY + ROPE.lockH, 0);
    lever.rotation.z = ls.locked ? 0 : -inw*ROPE.throw_;
    const knob = new T.Mesh(gKnob, red);
    knob.position.y = 0.26;
    lever.add(knob);
    asm.add(lever);

    /* the number plate on the rail's face, the way a real rail is indexed */
    const plate = new T.Mesh(gPlate,
      new T.MeshBasicMaterial({map:ropePlateTex(ls.id)}));
    plate.position.set(inw*(ROPE.railOff + ROPE.railD/2 + 0.004),
                       ROPE.railY - ROPE.railH/2, 0);
    plate.rotation.y = inw > 0 ? Math.PI/2 : -Math.PI/2;
    asm.add(plate);

    VR.ropes.push({mesh:m, ls, home:m.position.clone(), asm, inw,
                   head, foot, runs, lock, lever, knob, plate,
                   pos0:ls.pos, last:ls.pos});
  }

  /* ---- the traveler hand line ------------------------------------------
     The house curtain does not only fly: it TRAVELS.  Its hand line hangs
     at the side of the proscenium — stage right, just offstage of the arch,
     a step downstage of the curtain so the pleats slide past it — a small
     loop the flyman works standing on the deck.  Kept out of the wall: the
     proscenium starts at z = 0, so the loop never comes further downstage
     than a hand's breadth short of it. */
  const trav = hung.find(l => GOODS[l.goodsKey] && GOODS[l.goodsKey].traveler);
  if(trav){
    const asm = new T.Group();
    asm.name = 'vr:traveler' + trav.id;
    asm.position.set(org.x - (D.procW/2 + ROPE.travOff), 0,
                     Math.min(trav.z + 0.5, -0.35) + zOff);
    /* the hover label needs to find it: the asm is the one rope assembly in
       the pick list, and describe (p7) reads this tag off it */
    asm.userData.travRope = trav;
    holder.add(asm);
    INTERACT.push(asm);
    const runH2 = ROPE.travTop - ROPE.travFloor;
    const gWheel2 = new T.CylinderGeometry(ROPE.wheel*0.6, ROPE.wheel*0.6, 0.025, 8);
    gWheel2.rotateX(Math.PI/2);
    const gRun2 = new T.CylinderGeometry(0.012, 0.012, runH2, 6);
    const head = new T.Mesh(gWheel2, iron); head.position.y = ROPE.travTop;   asm.add(head);
    const foot = new T.Mesh(gWheel2, iron); foot.position.y = ROPE.travFloor; asm.add(foot);
    const runs = [];
    /* the front run faces centre stage, where the flyman stands: runs[0] */
    for(const s of [1, -1]){
      const run = new T.Mesh(gRun2, mat);
      run.position.set(s*ROPE.gap, ROPE.travFloor + runH2/2, 0);
      asm.add(run); runs.push(run);
    }
    const m = new T.Mesh(gGrab, grab);
    m.position.set(ROPE.gap, ROPE.handY, 0);
    m.userData.rope = trav;
    asm.add(m);
    VR.ropes.push({mesh:m, ls:trav, home:m.position.clone(), asm, inw:1,
                   head, foot, runs, traveler:true,
                   open0:trav.open, last:trav.open});
  }
}
function vrClearRopes(){
  /* let go of any held ROPE first — it references a rope and a lineset that
     are about to be disposed and parked, and vrUpdateHold would keep hauling
     them every frame until the squeeze was released.  A held BODY or CART is
     venue-level state and walks between stages in the hand (§3) — that hold
     survives the swap untouched. */
  if(VR.held && VR.held.rope) VR.held = null;
  VR.heldLever = null;
  /* the traveler's hand line sits in the pick list for its hover label —
     take it back out before its meshes are disposed */
  for(let i = INTERACT.length - 1; i >= 0; i--)
    if(INTERACT[i].userData && INTERACT[i].userData.travRope) INTERACT.splice(i, 1);
  if(VR.ropeRoot && VR.ropeRoot.parent) VR.ropeRoot.parent.remove(VR.ropeRoot);
  if(VR.ropeRoot && typeof disposeTree === 'function') disposeTree(VR.ropeRoot);
  VR.ropeRoot = null;
  VR.ropes.length = 0;
}

/* throw the lock on a line, or take it off.  ls.locked IS the rail's rope
   lock — the LOCK button on the fly page is the same flag, and the lever
   simply shows and works it from inside the headset.  Pulling the lever OUT
   with nothing holding the rope releases the load to gravity: nothing at all
   is on it, and it runs away exactly as a let-go line does. */
function vrRopeLock(r, on){
  const ls = r.ls;
  on = !!on;
  if(ls.locked === on) return;
  ls.locked = on;
  if(on){
    /* a lever thrown on a line that is running away stops it dead */
    ls.runaway = false; ls.flyVel = 0; ls.target = ls.pos;
  } else if(!(VR.held && VR.held.rope === r)){
    ls.runaway = true; ls.flyVel = 0; ls.target = ls.pos;
  }
  if(typeof syncFlyRow === 'function') syncFlyRow(ls);
  if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
  if(typeof toast === 'function')
    toast('lineset ' + ls.id + (on ? ' — locked off at the rail'
                                   : ' — lock off, mind the load'));
}
/* every frame: a held lever follows the hand that has it, and the loops turn
   with the line.  NOTHING else moves a lever — not a hand walking past, not
   a knuckle brushing the knob mid-haul.  Deliberate grabs only (vrSqueeze). */
const _lvp = new T.Vector3();
function vrUpdateRopes(dt){
  for(const r of VR.ropes){
    const ls = r.ls;
    if(r.traveler){
      /* the hand line turns with the CURTAIN, not the pipe: travScale metres
         of rope is the whole 0..1, and there is no lever to mind */
      const d = (ls.open - r.last) * ROPE.travScale;
      if(d){
        r.last = ls.open;
        const spin = d / (ROPE.wheel * 0.6);
        r.head.rotation.z += spin;
        r.foot.rotation.z += spin;
        if(!(VR.held && VR.held.rope === r))
          r.mesh.position.y = r.home.y - clamp((ls.open - r.open0)*ROPE.travScale, -0.5, 0.5);
      }
      continue;
    }
    if(VR.heldLever && VR.heldLever.rope === r){
      const c = VR.controllers[VR.heldLever.hand];
      if(c){
        c.updateMatrixWorld(true);
        _lvp.setFromMatrixPosition(c.matrixWorld);
        r.asm.worldToLocal(_lvp);
        /* the handle leans to meet the hand: + is out, toward the flyman */
        const dx = (_lvp.x - r.lever.position.x) * r.inw;
        const dy = Math.max(0.02, _lvp.y - r.lever.position.y);
        const a = clamp(Math.atan2(dx, dy), 0, ROPE.throw_);
        r.lever.rotation.z = -r.inw * a;
        if(a > ROPE.throw_*0.75 && ls.locked)  vrRopeLock(r, false);
        if(a < ROPE.throw_*0.25 && !ls.locked) vrRopeLock(r, true);
      }
    } else {
      /* at rest the handle sits in its detent: upright when the lock is on,
         lying toward the operator when it is off */
      const want = ls.locked ? 0 : -r.inw*ROPE.throw_;
      r.lever.rotation.z += clamp(want - r.lever.rotation.z, -dt*7, dt*7);
    }
    /* the blocks turn and the grab section rides with the load — cosmetic */
    const d = ls.pos - r.last;
    if(d){
      r.last = ls.pos;
      const spin = d / ROPE.wheel;
      r.head.rotation.z += spin;
      r.foot.rotation.z += spin;
      if(!(VR.held && VR.held.rope === r))
        r.mesh.position.y = r.home.y + clamp((ls.pos - r.pos0)/5.5, -0.5, 0.5);
    }
  }
}

const _rayO = new T.Vector3(), _rayD = new T.Vector3(), _m = new T.Matrix4();
const _vrRay = new T.Raycaster();
function vrPointAt(){
  if(!VR.controllers[1]) return null;
  const c = VR.controllers[1];
  c.updateMatrixWorld(true);
  _m.identity().extractRotation(c.matrixWorld);
  _rayO.setFromMatrixPosition(c.matrixWorld);
  _rayD.set(0, 0, -1).applyMatrix4(_m).normalize();
  _vrRay.set(_rayO, _rayD);
  _vrRay.near = 0; _vrRay.far = 6;
  const hits = _vrRay.intersectObjects(VR.deskMeshes, false);
  if(!hits.length){ if(VR.cursor) VR.cursor.visible = false; return null; }
  const h = hits[0];
  if(VR.cursor){
    VR.cursor.visible = true;
    VR.cursor.position.copy(VR.rig.worldToLocal(h.point.clone()));
  }
  return {u:h.uv.x, v:1 - h.uv.y, obj:h.object};
}
/* point a hand's ray into the ROOM — doors, linesets, fixtures, scenic,
   stations: the same curated pick list the desktop crosshair uses */
function vrCastWorld(hand){
  const c = VR.controllers[hand];
  if(!c) return null;
  c.updateMatrixWorld(true);
  _m.identity().extractRotation(c.matrixWorld);
  _rayO.setFromMatrixPosition(c.matrixWorld);
  _rayD.set(0, 0, -1).applyMatrix4(_m).normalize();
  _vrRay.set(_rayO, _rayD);
  _vrRay.near = 0; _vrRay.far = 22;            // the desktop reach
  const hit = (typeof pickAll === 'function') ? pickAll(_vrRay) : null;
  if(!hit) return null;
  const info = (typeof describe === 'function') ? describe(hit.object) : null;
  if(!info) return null;
  return {hit, info};
}
function vrSelect(hand, down){
  if(!down) return;
  /* a hand holding a tool: the trigger IS the tool (build spec §5) */
  if(VR.tools && VR.tools[hand]){ vrToolFire(hand, VR.tools[hand]); return; }
  /* a hand with the roller: the trigger paints the wood it touches */
  if(VR.held && VR.held.kind === 'roller' && VR.held.hand === hand){
    const ro = VR.held.rack.roller;
    if(ro.color === null){
      if(typeof toast === 'function') toast('dip it in a can first');
      return;
    }
    ro.head.getWorldPosition(_vecA);
    let bw = null, bwd = 1e9;
    for(const b of BODIES){
      if(b.kind !== 'wood') continue;
      b.mesh.getWorldPosition(_run);
      const reach = Math.max(b.mesh.scale.x, b.mesh.scale.y, b.mesh.scale.z)/2 + 0.3;
      const d = _run.distanceTo(_vecA);
      if(d < reach && d < bwd){ bwd = d; bw = b; }
    }
    if(bw && paintWood(bw, _vecA, ro.color)){
      if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
      if(typeof toast === 'function')
        toast(bw.prof === 'sheet' ? 'painted the face' : 'painted');
      return;
    }
    /* no wood under it: cloth (RULING T).  Goods paint whole — the pipe's
       whole hang takes the colour, both halves of a curtain together. */
    const gls = (typeof goodsAt === 'function') ? goodsAt(_vecA) : null;
    if(gls && paintGoods(gls, ro.color)){
      if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
      if(typeof toast === 'function')
        toast('painted the ' + ((GOODS[gls.goodsKey] && GOODS[gls.goodsKey].label) || 'goods'));
      return;
    }
    if(typeof toast === 'function') toast('nothing under the roller to paint');
    return;
  }
  /* a hand on a saw cutter: the trigger is the blade (build spec §6) */
  if(VR.held && VR.held.kind === 'saw' && VR.held.hand === hand){
    const r = (typeof sawCut === 'function') ? sawCut(VR.held.st) : null;
    if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
    if(typeof toast === 'function') toast(!r ? 'nothing under the blade'
      : r.off ? 'two pieces' : 'trimmed — the scrap end is gone');
    return;
  }
  /* the desks keep priority for the pointing hand: on a desk, the trigger
     presses the console and never reaches through it into the room */
  if(hand === 1){
    const p = vrPointAt();
    if(p){
      /* the warehouse order screens carry their own canvas and hit list —
         a press on one never falls through to the main console's regions */
      if(p.obj && p.obj.userData.orderScreen && VR.orders){
        const sc = VR.orders[p.obj.userData.orderScreen];
        if(sc){
          const sx = p.u * sc.canvas.width, sy = p.v * sc.canvas.height;
          for(let i = sc.hits.length - 1; i >= 0; i--){
            const h = sc.hits[i];
            if(sx >= h.x && sx <= h.x + h.w && sy >= h.y && sy <= h.y + h.h){
              h.fn(sx, sy);
              if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
              return;
            }
          }
        }
        return;
      }
      /* the carpenters screen: same shape, its own registry (RULING X) */
      if(p.obj && p.obj.userData.carpScreen && VR.carps){
        const sc = VR.carps[p.obj.userData.carpScreen];
        if(sc){
          const sx = p.u * sc.canvas.width, sy = p.v * sc.canvas.height;
          for(let i = sc.hits.length - 1; i >= 0; i--){
            const h = sc.hits[i];
            if(sx >= h.x && sx <= h.x + h.w && sy >= h.y && sy <= h.y + h.h){
              h.fn(sx, sy);
              if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
              return;
            }
          }
        }
        return;
      }
      const px = p.u * VRC.W, py = p.v * VRC.H;
      for(let i = VR.hits.length - 1; i >= 0; i--){
        const h = VR.hits[i];
        if(px >= h.x && px <= h.x + h.w && py >= h.y && py <= h.y + h.h){
          h.fn(px, py);
          if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
          return;
        }
      }
      return;
    }
  }
  /* off the desk, either trigger is the desktop E key */
  const w = vrCastWorld(hand);
  if(!w) return;
  useInfo(w.info);
  if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
}

/* ---- the floating label: what a trigger pull would do, drawn at the ray ---- */
function vrLabel(text, at, dist){
  if(!text){
    if(VR.labelSpr) VR.labelSpr.visible = false;
    VR.labelTxt = null;
    return;
  }
  if(!VR.labelSpr){
    const c = cvs(1024, 96);
    VR.labelCvs = c;
    VR.labelCtx = c.getContext('2d');
    VR.labelTex = tex(c, 1, 1);
    const m = new T.SpriteMaterial({map:VR.labelTex, transparent:true, depthTest:false});
    VR.labelSpr = new T.Sprite(m);
    VR.labelSpr.renderOrder = 999;
    scene.add(VR.labelSpr);
  }
  if(text !== VR.labelTxt){
    VR.labelTxt = text;
    const g = VR.labelCtx;
    g.clearRect(0, 0, 1024, 96);
    g.fillStyle = 'rgba(10,12,16,0.82)';
    g.fillRect(0, 0, 1024, 96);
    g.font = '600 40px Helvetica, Arial, sans-serif';
    g.textAlign = 'center'; g.textBaseline = 'middle';
    g.fillStyle = '#ffd08a';
    g.fillText(text.slice(0, 48), 512, 48);
    VR.labelTex.needsUpdate = true;
  }
  /* constant angular size: scale with distance so it reads across the room */
  const h = clamp(0.045 * Math.max(1.2, dist || 2), 0.06, 0.45);
  VR.labelSpr.scale.set(h * 1024/96, h, 1);
  VR.labelSpr.position.copy(at);
  VR.labelSpr.position.y += h * 1.1;
  VR.labelSpr.visible = true;
}
/* each frame: if the pointing hand is not on a desk, hover the room */

/* letting go of a rope.  With the lever thrown the line is held.  Without it
   the flyman has just taken his hands off an unbalanced load: it keeps
   whatever way it was going and gravity takes it from there — p3's updateFly
   does the falling.  Only a squeeze RELEASE gets here; a stage swap and the
   end of a session open the hand without letting the line go (vrClearRopes). */
function vrLetGo(h){
  const ls = h.rope.ls;
  if(ls.locked) return;
  ls.runaway = true;
  /* a hand can be flung about; cap what it can throw at the load */
  ls.flyVel = clamp(h.vel || 0, -4, 4);
  ls.target = ls.pos;
  if(typeof toast === 'function')
    toast('lineset ' + ls.id + ' — let go with the lock off, it is running away');
}
/* what releasing a held body will do: the nearest empty LIVE point within
   reach of the BODY, else the nearest empty slot (rack, cart shelf, pallet).
   Shared by the release itself and the floating label, so the label can
   never promise one thing and the hand do another.  Points first: hanging
   is the deliberate act, stowing the fallback. */
const _bodT = new T.Vector3(), _bodP = new T.Vector3();
function vrBodyTarget(b){
  b.mesh.getWorldPosition(_bodP);
  let pt = null, pd = 0.4;
  if(b.kind === 'speaker'){
    if(typeof SPKBARS !== 'undefined' && SPKBARS)
      for(const k in SPKBARS){
        const bar = SPKBARS[k];
        if(!bar.points) continue;
        for(const p of bar.points){
          if(p.body) continue;
          _bodT.set(0, p.y, 0);
          bar.group.localToWorld(_bodT);
          const d = _bodT.distanceTo(_bodP);
          if(d < pd){ pd = d; pt = p; }
        }
      }
  } else {
    for(const f of FIXTURES){
      if(f.body) continue;
      f.group.getWorldPosition(_bodT);
      const d = _bodT.distanceTo(_bodP);
      if(d < pd){ pd = d; pt = f; }
    }
  }
  if(pt) return {point:pt};
  let sl = null, sd = 0.4;
  const scan = (arr)=>{
    if(!arr) return;
    for(const a of arr){
      if(a.userData.body) continue;
      a.getWorldPosition(_bodT);
      const d = _bodT.distanceTo(_bodP);
      if(d < sd){ sd = d; sl = a; }
    }
  };
  if(typeof SHEDS !== 'undefined'){
    scan(SHEDS.palace && SHEDS.palace.slots);
    scan(SHEDS.arc && SHEDS.arc.slots);
  }
  if(typeof CARTS !== 'undefined'){
    scan(CARTS.palace && CARTS.palace.slots);
    scan(CARTS.arc && CARTS.arc.slots);
  }
  if(typeof ORDERS !== 'undefined')
    for(const k in ORDERS){ const o = ORDERS[k]; if(o && o.pallet) scan(o.pallet.slots); }
  return sl ? {slot:sl} : null;
}
const _run = new T.Vector3(), _vecA = new T.Vector3();
/* wood is grabbed by its SURFACE, not its centre — an 8ft stick is a
   handful anywhere along it (build-usability spec §2).  Wood meshes are
   the shared unit box scaled, so local-space clamp is the exact distance. */
const GRAB_WOOD = 0.15;
const _wristE = new T.Euler();   // the 45° carry grid (RULING M)
function vrSqueeze(hand, down){
  const c = VR.controllers[hand];
  if(!c) return;
  if(!down){
    /* an open hand holsters its tool before anything else */
    if(VR.tools && VR.tools[hand]){ vrHolster(hand); return; }
    if(VR.heldLever && VR.heldLever.hand === hand) VR.heldLever = null;
    if(VR.held && VR.held.hand === hand){
      const h = VR.held;
      VR.held = null;
      if(h.kind === 'tapetab'){
        if(VR.tapeLn) VR.tapeLn.visible = false;
        vrLabel(null);
        return;
      }
      /* an open hand re-racks the roller, wherever it is */
      if(h.kind === 'roller'){ vrReRack(h.rack); return; }
      /* a let-go assembly (or swinging, or sliding piece) stays as it lies */
      if(h.kind === 'asm' || h.kind === 'swing' || h.kind === 'slide'){
        VR.snap = null;
        vrLabel(null);
        return;
      }
      /* a released BODY snaps to what the label promised: an empty point,
         an empty slot, or the floor (updateBodies settles it) */
      if(h.kind === 'body'){
        const b = h.body;
        VR.snap = null;
        /* the drum FIRST — it outranks every snap: holding a thing over
           the bin is the one unambiguous act of intent (RULING P) */
        if(typeof TRASH !== 'undefined' && BUILD_KINDS && BUILD_KINDS[b.kind] &&
           TRASH[b.venue]){
          const tr = TRASH[b.venue];
          tr.group.updateMatrixWorld(true);
          b.mesh.getWorldPosition(_vecA);
          const lp = tr.group.worldToLocal(_vecA.clone());
          if(Math.sqrt(lp.x*lp.x + lp.z*lp.z) < tr.r && lp.y < 1.4 &&
             removeBody(b)){
            if(typeof Snd !== 'undefined' && Snd.play) Snd.play('click');
            if(typeof toast === 'function') toast(BODY_LABEL[b.kind] + ' — binned');
            vrLabel(null);
            return;
          }
        }
        /* a carriage let go over a track run RIDES it (p4c rideTrack) */
        if(b.kind === 'carriage' && typeof ASSEMBLIES !== 'undefined'){
          b.mesh.getWorldPosition(_vecA);
          for(const a2 of ASSEMBLIES){
            if(!a2.track || a2.venue !== b.venue) continue;
            a2.root.updateMatrixWorld(true);
            const lp = a2.root.worldToLocal(_vecA.clone());
            if(lp.x > -0.4 && lp.x < (a2.track.n - 1)*1.2192 + 0.4 &&
               Math.abs(lp.z) < 0.35 && lp.y > -0.3 && lp.y < 0.7 &&
               rideTrack(b, a2)){
              if(typeof toast === 'function') toast('carriage on the run');
              vrLabel(null);
              return;
            }
          }
        }
        /* a paint can let go over its shed's rack: color in, can gone */
        if(b.kind === 'paint' && typeof RACKS !== 'undefined' && RACKS[b.venue]){
          const rack = RACKS[b.venue];
          rack.group.getWorldPosition(_vecA);
          b.mesh.getWorldPosition(_run);
          if(_run.distanceTo(_vecA) < 1.8 && rackTakeCan(rack, b)){
            if(typeof toast === 'function') toast('a new color on the rack');
            vrLabel(null);
            return;
          }
        }
        /* wood let go over a cut station SEATS on it (p4c seatWood) */
        if(b.kind === 'wood' && typeof SAWS !== 'undefined' && SAWS[b.venue]){
          const set = SAWS[b.venue];
          for(const kk in set){
            const st = set[kk];
            st.group.updateMatrixWorld(true);
            b.mesh.getWorldPosition(_vecA);
            const lp = st.group.worldToLocal(_vecA.clone());
            if(Math.abs(lp.x) < st.span/2 + 0.3 && Math.abs(lp.z) < 1.0 &&
               lp.y > 0.4 && lp.y < 1.9 && seatWood(st, b)){
              if(typeof toast === 'function')
                toast('on the ' + (st.kind === 'track' ? 'table' : 'bench'));
              vrLabel(null);
              return;
            }
          }
        }
        const t = vrBodyTarget(b);
        if(t && t.point && hangBody(b, t.point)){
          if(typeof toast === 'function') toast(BODY_LABEL[b.kind] + ' — hung');
        } else if(t && t.slot && slotBody(b, t.slot)){
          if(typeof toast === 'function') toast(BODY_LABEL[b.kind] + ' — stowed');
        } else {
          b.state = 'loose';
          if(b.kind === 'wood' && typeof woodRestH === 'function') b.restH = woodRestH(b);
          /* a let-go table lands square: upright, yaw to the nearest 45 */
          if(b.kind === 'table'){
            b.restH = 0;
            b.mesh.getWorldQuaternion(_tq1);
            const e = new T.Euler().setFromQuaternion(_tq1, 'YXZ');
            e.y = Math.round(e.y/(Math.PI/4))*(Math.PI/4); e.x = 0; e.z = 0;
            _tq1.setFromEuler(e);
            if(b.mesh.parent){
              b.mesh.parent.getWorldQuaternion(_tq2).invert();
              _tq1.premultiply(_tq2);
            }
            b.mesh.quaternion.copy(_tq1);
          }
        }
        vrLabel(null);
        return;
      }
      /* a let-go CART simply stands where it is; only a rope has a whipped
         section to park and a load to run away */
      if(h.rope){
        /* the whipped section goes back to marking the front run at rest */
        h.rope.mesh.position.copy(h.rope.home);
        /* a let-go traveler just stays where it is: no weight on a hand line,
           so no runaway — vrLetGo is for counterweight lines only */
        if(!h.rope.traveler) vrLetGo(h);
      }
    }
    return;
  }
  c.updateMatrixWorld(true);
  _rayO.setFromMatrixPosition(c.matrixWorld);
  /* the belt first: a hand that closes at a holster draws the tool; a hand
     at the tape's tab (worn by the OTHER hand) takes the tab to stretch */
  if(VR.tools && !VR.tools[hand] && VR.holsters){
    for(const key in VR.holsters){
      if(VR.tools[0] === key || VR.tools[1] === key) continue;
      VR.holsters[key].getWorldPosition(_vecA);
      if(_vecA.distanceTo(_rayO) < 0.22){ vrDraw(hand, key); return; }
    }
    const oh = 1 - hand;
    if(VR.tools[oh] === 'tape'){
      VR.toolMesh.tape.getWorldPosition(_vecA);
      if(_vecA.distanceTo(_rayO) < 0.2){
        VR.held = {hand, kind:'tapetab', toolHand:oh, len:0};
        return;
      }
    }
  }
  if(VR.ropeRoot) VR.ropeRoot.updateMatrixWorld(true);   // the loops nest a group deep

  /* what is under the hand: the nearest lever knob, and the nearest point on
     any RUN of any loop — the whole rope is a handful, top to bottom */
  let lv = null, lvd = ROPE.knobR;
  let best = null, bd = ROPE.grabR, bestRun = 0, bestY = 0;
  for(const r of VR.ropes){
    if(r.knob){          // the traveler's hand line has no lever to grab
      const kd = _rayD.setFromMatrixPosition(r.knob.matrixWorld).distanceTo(_rayO);
      if(kd < lvd){ lvd = kd; lv = r; }
    }
    for(let i = 0; i < r.runs.length; i++){
      const run = r.runs[i];
      run.getWorldPosition(_run);
      const half = run.geometry.parameters.height/2;
      const y = clamp(_rayO.y, _run.y - half, _run.y + half);
      const dx = _rayO.x - _run.x, dy = _rayO.y - y, dz = _rayO.z - _run.z;
      const d = Math.sqrt(dx*dx + dy*dy + dz*dz);
      if(d < bd){ bd = d; best = r; bestRun = i; bestY = y; }
    }
  }
  /* the warehouse carts (p2m): a hand on the handle bar — the nearest point
     of the bar treated as a segment, so anywhere along it is a hold */
  let cart = null, cd = Infinity;
  if(typeof CARTS !== 'undefined'){
    for(const ck in CARTS){
      const ct = CARTS[ck];
      if(!ct) continue;
      ct.group.updateMatrixWorld(true);
      _run.set(-ct.handleHalf, ct.handleH, ct.handleZ);
      ct.group.localToWorld(_run);
      _vecA.set(ct.handleHalf, ct.handleH, ct.handleZ);
      ct.group.localToWorld(_vecA);
      const sx = _vecA.x - _run.x, sy = _vecA.y - _run.y, sz = _vecA.z - _run.z;
      const L2 = sx*sx + sy*sy + sz*sz;
      let t = L2 > 0 ? ((_rayO.x - _run.x)*sx + (_rayO.y - _run.y)*sy +
                        (_rayO.z - _run.z)*sz)/L2 : 0;
      t = clamp(t, 0, 1);
      const px = _run.x + sx*t - _rayO.x, py = _run.y + sy*t - _rayO.y,
            pz = _run.z + sz*t - _rayO.z;
      const d = Math.sqrt(px*px + py*py + pz*pz);
      if(d < ct.grabR && d < cd){ cd = d; cart = ct; }
    }
  }
  /* the saw cutters (p4c): a hand on the track rail or the chop arm */
  let saw = null, sd = 0.28;
  if(typeof SAWS !== 'undefined'){
    for(const vk in SAWS){
      const set = SAWS[vk];
      if(!set) continue;
      for(const kk in set){
        const st = set[kk];
        st.cutter.getWorldPosition(_vecA);
        _vecA.y += 0.15;
        const d = _vecA.distanceTo(_rayO);
        if(d < sd){ sd = d; saw = st; }
      }
    }
  }
  /* the paint rollers (p4c RACKS): one per shed, homed to its rack */
  let rol = null, rd = 0.22;
  if(typeof RACKS !== 'undefined'){
    for(const rk in RACKS){
      const rack = RACKS[rk];
      if(!rack || (VR.held && VR.held.kind === 'roller' && VR.held.rack === rack)) continue;
      rack.roller.mesh.getWorldPosition(_vecA);
      const d = _vecA.distanceTo(_rayO);
      if(d < rd){ rd = d; rol = rack; }
    }
  }
  /* loose gear (the p4 detach system): hung bodies on the LIVE rig, and
     anything slotted or standing loose — a body within reach of the closing
     hand.  A parked stage's rig is across town; its bodies are not takeable. */
  let bod = null, bdd = 0.35;
  if(typeof BODIES !== 'undefined'){
    for(const b of BODIES){
      if(b.state === 'held') continue;
      if(b.state === 'carried') continue;   // riding a carpenter's arms (p6b): not takeable
      if(b.state === 'hung'){
        const p = b.point;
        const live = p && (p.spk
          ? (SPKBARS && (SPKBARS.L === p.bar || SPKBARS.R === p.bar))
          : FIXTURES.indexOf(p) >= 0);
        if(!live) continue;
      }
      let d;
      if(b.kind === 'wood'){
        /* hand into the unit-box local frame; per-axis overshoot times the
           scale is the true metric gap to the nearest point on the piece */
        b.mesh.updateWorldMatrix(true, false);
        _vecA.copy(_rayO);
        b.mesh.worldToLocal(_vecA);
        const gx = Math.max(0, Math.abs(_vecA.x) - 0.5) * b.mesh.scale.x,
              gy = Math.max(0, Math.abs(_vecA.y) - 0.5) * b.mesh.scale.y,
              gz = Math.max(0, Math.abs(_vecA.z) - 0.5) * b.mesh.scale.z;
        d = Math.sqrt(gx*gx + gy*gy + gz*gz);
        if(d >= GRAB_WOOD) continue;
      } else if(b.kind === 'table'){
        /* the table's origin is its FEET — a point on the floor under the
           middle of the top, which no hand ever came within 0.35 of.  Grab
           it by its body instead: the gap to its frame box (RULING Q).
           The mesh is a Group at scale 1, so local IS metric. */
        b.mesh.updateWorldMatrix(true, false);
        _vecA.copy(_rayO);
        b.mesh.worldToLocal(_vecA);
        const gx = Math.max(0, Math.abs(_vecA.x) - 0.8),
              gy = Math.max(0, Math.abs(_vecA.y - TABLE_TOP/2) - TABLE_TOP/2),
              gz = Math.max(0, Math.abs(_vecA.z) - 0.4);
        d = Math.sqrt(gx*gx + gy*gy + gz*gz);
        if(d >= GRAB_WOOD) continue;
      } else {
        b.mesh.getWorldPosition(_vecA);
        d = _vecA.distanceTo(_rayO);
      }
      if(d < bdd){ bdd = d; bod = b; }
    }
  }
  /* nearest wins across all six classes — never fork this arbitration */
  if(rol && (!saw || rd < sd) && (!bod || rd <= bdd) && (!cart || rd < cd) &&
     (!lv || rd < lvd) && (!best || rd < bd)){
    const c2 = VR.controllers[hand];
    c2.add(rol.roller.mesh);
    rol.roller.mesh.position.set(0, 0, -0.02);
    rol.roller.mesh.rotation.set(-0.9, 0, 0);
    VR.held = {hand, kind:'roller', rack:rol};
    if(typeof toast === 'function')
      toast(rol.roller.color === null ? 'roller — dip it in a can first'
                                      : 'roller — trigger paints the wood it touches');
    return;
  }
  if(saw && (!rol || sd < rd) && (!bod || sd <= bdd) && (!cart || sd < cd) && (!lv || sd < lvd) && (!best || sd < bd)){
    saw.group.updateMatrixWorld(true);
    const lx = saw.group.worldToLocal(_rayO.clone()).x;
    VR.held = {hand, kind:'saw', st:saw, off:saw.cut - lx};
    if(typeof toast === 'function') toast(saw.kind === 'track'
      ? 'track saw — slide it, trigger cuts'
      : 'chop saw — slide it, trigger cuts');
    return;
  }
  if(bod && (!rol || bdd < rd) && (!saw || bdd < sd) && (!cart || bdd < cd) && (!lv || bdd < lvd) && (!best || bdd < bd)){
    /* a piece already NAILED (p4c): swing it on its one nail, carry the
       whole assembly by any rigid piece, or bounce off anchored work */
    /* anything riding a slider — the carriage, or wood nailed to it —
       slides the run under the hand (p4c, PR 6) */
    if(bod.slider){
      const a2 = bod.tAsm || bod.asm;
      a2.root.updateMatrixWorld(true);
      const lx2 = a2.root.worldToLocal(_rayO.clone()).x;
      VR.held = {hand, kind:'slide', asm:a2, slider:bod.slider,
                 off:bod.slider.position.x - lx2};
      if(typeof toast === 'function') toast('it runs on the track');
      return;
    }
    if(bod.asm){
      if(bod.pivot){
        const piv = bod.pivot;
        piv.updateMatrixWorld(true);
        const P = new T.Vector3().setFromMatrixPosition(piv.matrixWorld);
        const ax = piv.userData.axis.clone();
        bod.asm.root.getWorldQuaternion(_tq1);
        ax.applyQuaternion(_tq1).normalize();
        const ref = _rayO.clone().sub(P);
        ref.addScaledVector(ax, -ref.dot(ax));
        if(ref.lengthSq() < 1e-6) return;
        ref.normalize();
        VR.held = {hand, kind:'swing', body:bod, P, ax, ref0:ref,
                   q0:piv.quaternion.clone(),
                   range:bod.pivotN && bod.pivotN.range};
        if(typeof toast === 'function')
          toast(bod.pivotN && bod.pivotN.hinge ? 'it swings on its hinge'
                                               : 'it swings on its nail');
      } else if(!bod.asm.anchor){
        const a = bod.asm;
        a.root.updateMatrixWorld(true);
        a.root.getWorldPosition(_vecA);
        VR.held = {hand, kind:'asm', asm:a,
                   offX:_vecA.x - _rayO.x, offY:_vecA.y - _rayO.y, offZ:_vecA.z - _rayO.z};
        /* it turns in the hand exactly as a plank does (RULING AD): the
           pose is kept relative to the CONTROLLER and the point you
           grabbed stays in the palm.  An assembly root is an unscaled
           Group, so its grab vector is already metric — the wood branch's
           .multiply(scale) is only there for a scaled unit box. */
        c.getWorldQuaternion(_tq1);
        a.root.getWorldQuaternion(_tq2);
        VR.held.relQ = _tq1.invert().multiply(_tq2).clone();
        VR.held.grabV = a.root.worldToLocal(_rayO.clone()).clone();
        if(typeof toast === 'function')
          toast('the assembly comes whole — ' + a.pieces.length + ' pieces');
      } else if(typeof toast === 'function')
        toast('nailed down — the hammer first');
      return;
    }
    grabBody(bod);
    bod.mesh.getWorldPosition(_vecA);
    VR.held = {hand, kind:'body', body:bod,
               offX:_vecA.x - _rayO.x, offY:_vecA.y - _rayO.y, offZ:_vecA.z - _rayO.z};
    /* wood turns with the wrist (lanterns keep their old fixed carry) — and
       it rides IN the hand: the pose is kept relative to the CONTROLLER, so
       the grabbed point stays in the palm and the wrist pivots about it,
       not about a centre that may be four feet up the stick (RULING R) */
    if(bod.kind === 'wood' || bod.kind === 'table'){
      c.getWorldQuaternion(_tq1);
      bod.mesh.getWorldQuaternion(_tq2);
      VR.held.relQ = _tq1.invert().multiply(_tq2).clone();
      /* wood keeps the GRAB POINT itself (metric, piece frame): the 45°
         quantize (RULING M) re-orients about the palm, which a rigid
         grip-space offset cannot express.  The table carries rigid. */
      if(bod.kind === 'wood'){
        bod.mesh.updateWorldMatrix(true, false);
        VR.held.grabV = bod.mesh.worldToLocal(_rayO.clone())
          .multiply(bod.mesh.scale).clone();
      } else VR.held.relP = c.worldToLocal(_vecA.clone()).clone();
    }
    if(typeof toast === 'function') toast(BODY_LABEL[bod.kind] + ' — in hand');
    return;
  }
  if(cart && (!saw || cd < sd) && (!best || cd < bd) && (!lv || cd < lvd)){
    VR.held = {hand, kind:'cart', cart, offX:cart.x - _rayO.x, offZ:cart.z - _rayO.z};
    if(typeof toast === 'function') toast(cart.lift
      ? 'forklift — walk it; the right stick runs the forks'
      : 'warehouse cart — push it where you want it');
    return;
  }
  /* nearest wins: with the lock in, the handle stands right against the
     front run, and the hand should get whichever it actually closed on */
  if(lv && (!saw || lvd < sd) && (!best || lvd < bd)){
    VR.heldLever = {hand, rope:lv};
    return;
  }
  if(best){
    const ls = best.ls;
    if(best.traveler){
      /* the traveler's hand line: nothing on it to arrest and nothing that
         can run away — the hand simply takes the curtain.  The rope lock
         means nothing here, so ls.locked is left entirely alone. */
      const run = best.runs[bestRun];
      best.mesh.position.x = run.position.x;
      best.mesh.position.y = clamp(bestY, ROPE.travFloor + 0.35, ROPE.travTop - 0.35);
      VR.held = {hand, rope:best, y0:_rayO.y, open0:ls.open, vel:0,
                 dir: bestRun === 0 ? 1 : -1, grabY: best.mesh.position.y};
      if(typeof toast === 'function')
        toast('traveler — front run opens, back run closes');
      return;
    }
    /* a hand on the rope arrests a runaway — you have the weight now — but
       it does NOT touch the lock: only the lever does that */
    ls.runaway = false;
    ls.flyVel = 0;
    ls.relock = false;   // a board move in flight hands over: nothing relocks under a live hand
    ls.target = ls.pos;
    if(typeof syncFlyRow === 'function') syncFlyRow(ls);
    /* the whipped section slides to meet the hand, whichever run, whatever
       height it was taken at */
    const run = best.runs[bestRun];
    best.mesh.position.x = run.position.x;
    best.mesh.position.y = clamp(bestY, ROPE.floorY + 0.5, D.gridY - ROPE.headDrop - 0.5);
    VR.held = {hand, rope:best, y0:_rayO.y, trim0:ls.pos, vel:0,
               dir: bestRun === 0 ? 1 : -1, grabY: best.mesh.position.y};
    if(typeof toast === 'function')
      toast('lineset ' + ls.id + (ls.locked
        ? ' — locked off at the rail, pull the lever' : ' — haul away'));
  }
}
function vrUpdateHold(dt){
  if(!VR.held) return;
  const c = VR.controllers[VR.held.hand];
  if(!c) return;
  c.updateMatrixWorld(true);
  _rayO.setFromMatrixPosition(c.matrixWorld);
  /* a held CART chases the hand across the floor, through the same wall
     rules the player obeys (p2m), easing its nose round to the direction
     of travel — release just leaves it standing */
  if(VR.held.kind === 'cart'){
    const ct = VR.held.cart;
    const wx = _rayO.x + VR.held.offX, wz = _rayO.z + VR.held.offZ;
    const px = ct.x, pz = ct.z;
    if(typeof cartMoveTo === 'function') cartMoveTo(ct, wx, wz);
    const dx = ct.x - px, dz = ct.z - pz;
    if(dx*dx + dz*dz > 1e-8){
      const want = Math.atan2(dx, dz);
      let diff = want - ct.yaw;
      while(diff >  Math.PI) diff -= Math.PI*2;
      while(diff < -Math.PI) diff += Math.PI*2;
      const step = 2.5 * (dt || 0.016);
      ct.yaw += clamp(diff, -step, step);
      cartPose(ct);
    }
    /* the forklift only: the right stick runs the forks (push away = up).
       rx is smooth turn and the left stick walks; ry was the free axis,
       whichever hand took the handle. */
    if(ct.lift && VR.axes && VR.axes.ry)
      ct.forkY = clamp(ct.forkY - VR.axes.ry * LIFT_FORK_RATE * (dt || 0.016),
                       0, LIFT_FORK_TOP);
    return;
  }
  /* the roller rides the hand (parented — release always re-racks); the
     hold only watches for a DIP: the head near an open can takes its color */
  if(VR.held.kind === 'roller'){
    const rack = VR.held.rack, ro = rack.roller;
    ro.head.getWorldPosition(_vecA);
    for(const cm of rack.canMeshes){
      cm.getWorldPosition(_run);
      if(_run.distanceTo(_vecA) < 0.16){
        if(ro.color !== cm.userData.paintColor){
          ro.color = cm.userData.paintColor;
          ro.head.material = woodMat(ro.color);
          if(typeof toast === 'function')
            toast('dipped — ' + (PAINT_COLORS.find(p=>p.c === ro.color) || {name:'?'}).name);
        }
        break;
      }
    }
    return;
  }
  /* a hand on a saw cutter slides it along the table — snapped to the
     inch (a pencil tick first), read out against the piece under it */
  if(VR.held.kind === 'saw'){
    const st = VR.held.st;
    st.group.updateMatrixWorld(true);
    const lx = st.group.worldToLocal(_rayO.clone()).x;
    sawSetCut(st, lx + VR.held.off);
    const p = st.pieces.find(q=>{
      const half = seatLen(q)/2;
      return st.cut > q.mesh.position.x - half && st.cut < q.mesh.position.x + half;
    });
    if(p){
      st.cutter.getWorldPosition(_vecA);
      _vecA.y += 0.3;
      vrLabel('CUT AT ' + ftIn(st.cut - (p.mesh.position.x - seatLen(p)/2)), _vecA, 1);
    } else vrLabel(null);
    return;
  }
  /* the tape's tab: a line from the case to the hand, read out in ft-in */
  if(VR.held.kind === 'tapetab'){
    const tm = VR.toolMesh && VR.toolMesh.tape;
    if(!tm || VR.tools[VR.held.toolHand] !== 'tape'){ VR.held = null; return; }
    tm.getWorldPosition(_vecA);
    vrTapeLine(_vecA, _rayO);
    VR.held.len = _vecA.distanceTo(_rayO);
    vrLabel('TAPE  ' + ftIn(VR.held.len), _rayO, 1);
    return;
  }
  /* a held ASSEMBLY: the whole root follows the grip, kinematic like a
     body; the deck or a pipe under/over it makes the same offer a single
     piece gets (snapAsm), and the gun confirms it */
  if(VR.held.kind === 'asm'){
    const a = VR.held.asm;
    /* orientation first (RULING AD): the wrist turns the whole root, and
       with no X down it squares to the nearest 45° on every axis the way
       held wood does (RULING M) — which is how a flat built lying on the
       deck gets stood up.  The snap below still offers only the drop. */
    if(VR.held.relQ){
      c.getWorldQuaternion(_tq1).multiply(VR.held.relQ);
      if(!VR.btnX){
        _wristE.setFromQuaternion(_tq1, 'YXZ');
        _wristE.x = Math.round(_wristE.x/(Math.PI/4))*(Math.PI/4);
        _wristE.y = Math.round(_wristE.y/(Math.PI/4))*(Math.PI/4);
        _wristE.z = Math.round(_wristE.z/(Math.PI/4))*(Math.PI/4);
        _tq1.setFromEuler(_wristE);
      }
      a.root.quaternion.copy(_tq1);
    }
    /* then the grabbed point keeps the palm; a record with no grabV (one
       restored from anywhere that did not set it) keeps the old carry */
    if(VR.held.grabV)
      _vecA.copy(VR.held.grabV).applyQuaternion(a.root.quaternion).negate().add(_rayO);
    else _vecA.set(_rayO.x + VR.held.offX, _rayO.y + VR.held.offY, _rayO.z + VR.held.offZ);
    if(a.root.parent) a.root.parent.worldToLocal(_vecA);
    a.root.position.copy(_vecA);
    a.root.updateMatrixWorld(true);
    const s = (typeof snapAsm === 'function') ? snapAsm(a) : null;
    if(s){
      a.root.position.y += s.dy;
      a.root.updateMatrixWorld(true);
      VR.snap = {asm:a, target:s.target, point:s.point, axis:s.axis, piece:s.piece};
      vrLabel(s.target.deck ? 'ON THE DECK — nail it down' : 'UNDER THE PIPE — nail it on',
              s.point, 1);
    } else { VR.snap = null; }
    return;
  }
  /* a piece on ONE nail swings: the hand drives the angle about the axis */
  if(VR.held.kind === 'swing'){
    const H = VR.held, piv = H.body.pivot;
    if(!piv){ VR.held = null; return; }
    const ref = _rayO.clone().sub(H.P);
    ref.addScaledVector(H.ax, -ref.dot(H.ax));
    if(ref.lengthSq() > 1e-6){
      ref.normalize();
      let th = Math.acos(clamp(H.ref0.dot(ref), -1, 1));
      _vecA.crossVectors(H.ref0, ref);
      if(_vecA.dot(H.ax) < 0) th = -th;
      if(H.range) th = clamp(th, -H.range, H.range);   // a hinge has stops
      piv.quaternion.setFromAxisAngle(piv.userData.axis, th).multiply(H.q0);
    }
    return;
  }
  /* a slide hold: the hand drives the slider along the run, hard stops */
  if(VR.held.kind === 'slide'){
    const H = VR.held;
    H.asm.root.updateMatrixWorld(true);
    const lx = H.asm.root.worldToLocal(_rayO.clone()).x;
    slideTo(H.asm, H.slider, lx + H.off);
    return;
  }
  /* a held BODY follows the grip kinematically — never re-parented to the
     hand, so a hold record dropped without a release (session end, stage
     swap: vrOnEnd and vrClearRopes null VR.held cold) leaves a correctly
     parented mesh right where it was; updateBodies (p4) then demotes it to
     loose and settles it to the floor */
  if(VR.held.kind === 'body'){
    const b = VR.held.body;
    /* orientation first: wood and the table turn with the wrist, and held
       wood with no X down squares to the nearest 45° on every axis
       (RULING M).  The world quat lands directly on the mesh: both venue
       roots stand unrotated. */
    if(VR.held.relQ){
      c.getWorldQuaternion(_tq1).multiply(VR.held.relQ);
      if(b.kind === 'wood' && !VR.btnX){
        _wristE.setFromQuaternion(_tq1, 'YXZ');
        _wristE.x = Math.round(_wristE.x/(Math.PI/4))*(Math.PI/4);
        _wristE.y = Math.round(_wristE.y/(Math.PI/4))*(Math.PI/4);
        _wristE.z = Math.round(_wristE.z/(Math.PI/4))*(Math.PI/4);
        _tq1.setFromEuler(_wristE);
      }
      b.mesh.quaternion.copy(_tq1);
    }
    /* then position: the grabbed point stays in the palm (RULING R) —
       wood pivots about the palm through the quantize; the table rides a
       rigid grip offset; everything else keeps the world-offset carry. */
    if(VR.held.grabV)
      _vecA.copy(VR.held.grabV).applyQuaternion(b.mesh.quaternion).negate().add(_rayO);
    else if(VR.held.relP) _vecA.copy(VR.held.relP).applyMatrix4(c.matrixWorld);
    else _vecA.set(_rayO.x + VR.held.offX, _rayO.y + VR.held.offY, _rayO.z + VR.held.offZ);
    if(b.mesh.parent) b.mesh.parent.worldToLocal(_vecA);
    b.mesh.position.copy(_vecA);
    if(b.kind === 'wood'){
      b.mesh.updateMatrixWorld(true);
      const s = (typeof snapWood === 'function') ? snapWood(b) : null;
      if(s){
        _vecA.copy(s.pos);
        if(b.mesh.parent) b.mesh.parent.worldToLocal(_vecA);
        b.mesh.position.copy(_vecA);
        b.mesh.quaternion.copy(s.quat);
        VR.snap = {body:b, target:s.target, point:s.point, axis:s.axis};
      } else VR.snap = null;
    }
    /* a held TRACK section offers itself to a run's end, or the deck */
    else if(b.kind === 'track' && typeof snapTrackHeld === 'function'){
      b.mesh.updateMatrixWorld(true);
      const s = snapTrackHeld(b);
      if(s){
        _vecA.copy(s.pos);
        if(b.mesh.parent) b.mesh.parent.worldToLocal(_vecA);
        b.mesh.position.copy(_vecA);
        b.mesh.quaternion.copy(s.quat);
        VR.snap = {body:b, lay:s.lay, point:s.point};
      } else VR.snap = null;
    }
    const t = vrBodyTarget(b);
    b.mesh.getWorldPosition(_bodP);
    const camD = camera.getWorldPosition(_bodT).distanceTo(_bodP);
    vrLabel(BODY_LABEL[b.kind] + (t ? (t.point ? ' — release to HANG IT'
                                              : ' — release to STOW IT')
                                    : ' — release to set it down'), _bodP, camD);
    return;
  }
  const ls = VR.held.rope.ls;
  /* the traveler's hand line: hand travel IS curtain travel.  The front run
     pulled down slides the curtain OPEN, out to both sides; the back run is
     the other half of the loop and closes it.  Both open and travTarget are
     written, or p3's animator would haul the curtain back against the hand
     at its own 0.42/s.  The rope lock means nothing on a hand line. */
  if(VR.held.rope.traveler){
    const pulled = VR.held.y0 - _rayO.y;
    const want = VR.held.open0 + (pulled / ROPE.travScale) * VR.held.dir;
    ls.open = clamp(want, 0, 1);
    ls.travTarget = ls.open;
    /* against a dead end the rope cannot pay out: re-base, so the return
       pull takes up at once instead of paying back the overshoot first */
    if(want !== ls.open){ VR.held.y0 = _rayO.y; VR.held.open0 = ls.open; }
    VR.held.rope.mesh.position.y = VR.held.grabY - clamp(pulled, -0.5, 0.5);
    return;
  }
  /* with the lock in the rail has the weight: the hand rides the rope and
     nothing moves.  Keep re-basing so pulling the lever out mid-hold picks
     up smoothly from here rather than jumping the accumulated pull. */
  if(ls.locked){
    VR.held.y0 = _rayO.y; VR.held.trim0 = ls.pos; VR.held.vel = 0;
    return;
  }
  /* pull DOWN on the front run and the batten comes IN, the way a
     counterweight line works; the back run is the other half of the loop,
     so it carries the opposite sign */
  const pulled = VR.held.y0 - _rayO.y;
  const want = VR.held.trim0 - pulled * 5.5 * VR.held.dir;
  const was = ls.pos;
  /* the hand respects the same deck floor as everything else: goods hang
     below the pipe, and hauling by hand must not drag them through the
     stage.  Guarded because minTrimOf arrives on a parallel branch. */
  const lo = (typeof minTrimOf === 'function') ? minTrimOf(ls) : 0.6;
  ls.target = clamp(want, lo, OUT_TRIM); ls.pos = ls.target; ls.group.position.y = ls.pos;
  /* how fast the line is travelling in the hand — half-and-half against the
     last frame, so one jittery pose cannot throw the load across the grid.
     This is what the line carries away if the rope is let go of. */
  if(dt > 0) VR.held.vel = VR.held.vel*0.5 + ((ls.pos - was)/dt)*0.5;
  VR.held.rope.mesh.position.y = VR.held.grabY - clamp(pulled, -0.5, 0.5);
}

/* the gun talks: while a nail gun is in hand and no held offer stands, a
   nailable seam near the muzzle labels itself.  The seek is throttled —
   a pair walk over a dense build is not a per-frame cost — and the cached
   seam re-labels every frame so the hover never wins the flicker war. */
const _gunV = new T.Vector3();
function vrGunLabel(dt){
  VR.seamT = (VR.seamT || 0) + dt;
  if(VR.snap || (VR.held && VR.held.kind === 'body')){ VR.seam = null; return; }
  const gh = !VR.tools ? -1 : VR.tools[0] === 'nailgun' ? 0 : VR.tools[1] === 'nailgun' ? 1 : -1;
  if(gh < 0){ VR.seam = null; return; }
  if(VR.seamT > 0.12){
    VR.seamT = 0;
    const c = VR.controllers[gh];
    if(!c){ VR.seam = null; return; }
    c.updateMatrixWorld(true);
    _gunV.setFromMatrixPosition(c.matrixWorld);
    /* the SAME cast the trigger runs (RULING L), so the label can never
       promise a nail the shot would put somewhere else */
    let sm = null;
    if(typeof nailRay === 'function'){
      _m.identity().extractRotation(c.matrixWorld);
      _rayD.set(0, 0, -1).applyMatrix4(_m).normalize();
      const nr = nailRay(_gunV, _rayD);
      if(nr && nr.b) sm = nr;
      else if(nr && nr.miss){ VR.seam = null; return; }
    }
    if(!sm && typeof seamSeek === 'function') sm = seamSeek(_gunV);
    VR.seam = sm;
  }
  if(VR.seam){
    const d = camera.getWorldPosition(_gunV).distanceTo(VR.seam.point);
    vrLabel('TRIGGER TO NAIL', VR.seam.point, d);
  }
}
/* paint says how it works (build-feel, ask 4).  The mechanic is a scene
   shop's — rack, roller, dip, roll — and it stays; what failed on hardware
   is that nothing in the room SAID any of it.  Three labels, one per step,
   on the gun label's throttle discipline: an empty hand near the racked
   roller, the dry roller near its cans, the dipped roller against wood. */
const _pnV = new T.Vector3(), _pnV2 = new T.Vector3(), _pnAt = new T.Vector3();
function vrPaintLabel(dt){
  VR.paintT = (VR.paintT || 0) + dt;
  if(typeof RACKS === 'undefined') return;
  if(VR.paintT > 0.12){
    VR.paintT = 0;
    VR.paintTxt = null;
    if(VR.held && VR.held.kind === 'roller'){
      const rack = VR.held.rack, ro = rack.roller;
      ro.head.getWorldPosition(_pnV);
      if(ro.color === null){
        let cm = null, cd = 1e9;
        for(const m of rack.canMeshes){
          m.getWorldPosition(_pnV2);
          const d = _pnV2.distanceTo(_pnV);
          if(d < cd){ cd = d; cm = m; }
        }
        if(cm){
          VR.paintTxt = 'DIP THE HEAD IN A CAN';
          _pnAt.copy(cm.getWorldPosition(_pnV2));
        }
      } else {
        for(const b of BODIES){
          if(b.kind !== 'wood') continue;
          b.mesh.getWorldPosition(_pnV2);
          const reach = Math.max(b.mesh.scale.x, b.mesh.scale.y, b.mesh.scale.z)/2 + 0.3;
          if(_pnV2.distanceTo(_pnV) < reach){
            VR.paintTxt = 'TRIGGER TO PAINT';
            _pnAt.copy(_pnV);
            break;
          }
        }
        /* cloth answers the sign too, or it would lie at a curtain */
        if(!VR.paintTxt && typeof goodsAt === 'function' && goodsAt(_pnV)){
          VR.paintTxt = 'TRIGGER TO PAINT';
          _pnAt.copy(_pnV);
        }
      }
    } else if(!VR.held){
      for(const rk in RACKS){
        const rack = RACKS[rk];
        if(!rack) continue;
        rack.roller.mesh.getWorldPosition(_pnV);
        for(let i = 0; i < VR.controllers.length; i++){
          if(VR.tools && VR.tools[i]) continue;      // that hand is a tool hand
          VR.controllers[i].getWorldPosition(_pnV2);
          if(_pnV2.distanceTo(_pnV) < 0.6){
            VR.paintTxt = 'SQUEEZE TO TAKE THE ROLLER';
            _pnAt.copy(_pnV);
            break;
          }
        }
        if(VR.paintTxt) break;
      }
    }
  }
  if(VR.paintTxt){
    const d = camera.getWorldPosition(_pnV2).distanceTo(_pnAt);
    vrLabel(VR.paintTxt, _pnAt, d);
  }
}

/* ============================================================================
   RULING DV — THE CONSOLE TEXTURE UPLOADS ONLY WHEN SOMEONE COULD SEE IT
   ONE 1200x760 canvas serves ALL FIVE desk faces (vrMakeDesk maps the same
   VR.tex on every one), and the cadence at the foot of vrUpdate redrew it and
   re-uploaded ~3.6MB of RGBA five times a second for the whole session — on
   the fly floor, in the shed, in a venue with no desk in it.  It is the
   largest recurring texture upload a session pays and it showed up in the
   frame-time peaks.  So the CADENCE asks first whether a console is plausibly
   being looked at.  INTERACTION does not ask: every vrDrawConsole(true) call
   site — a tab, a row, a fly button, the GO cylinder — stays immediate and
   unconditional, which is why the gate lives here at the cadence and not
   inside vrDrawConsole.

   THE NUMBER: 12m, head to the nearest console face, and deliberately loose.
   The glass is 1.42 x 0.90m carrying 1200 x 760 pixels — 1.2mm a pixel — so
   the smallest type on it (17px, ~2cm tall) has stopped resolving on a Quest 3
   (~20 pixels a degree) by about 4m, and the biggest (the 26px board name,
   the GO lettering, ~5cm) by about 9m; at 12m the whole 1.42m width of it is
   6.8 degrees of view.  What actually set the number is not legibility but the
   PALACE: its desk is on the balcony rail looking down the house, and the
   middle of the stalls measures 7.7m from the glass, so a man standing in his
   own auditorium watching the board must be inside the gate with room to
   spare.  A gate that freezes a screen someone is reading is far worse than
   one that pays for an upload nobody needed.

   STALENESS ON RETURN is covered by leaving drawT RUNNING while out of range:
   it is already well past 0.2 when the head comes back, so the FIRST frame in
   range repaints and every frame after it keeps the old 5Hz.  Nothing has to
   know whether the board changed while nobody was looking — coming back is
   itself the trigger.

   In a session only: vrUpdate returns on !VR.active, so the desktop is
   untouched (and on the desktop the desk faces are not drawn at all). */
const VR_SEE_CONSOLE = 12;
const _dvHead = new T.Vector3(), _dvFace = new T.Vector3();
function vrConsoleSeen(){
  if(!VR.desks.length) return false;         // nothing carries the texture yet
  /* the head, not the rig: once the camera is inside the rig its own position
     stops being world (TRAPS), and it is the head the distance is about */
  camera.getWorldPosition(_dvHead);
  for(const d of VR.desks){
    d.face.getWorldPosition(_dvFace);
    if(_dvFace.distanceToSquared(_dvHead) < VR_SEE_CONSOLE*VR_SEE_CONSOLE)
      return true;
  }
  return false;
}

function vrUpdate(dt, rawDt){
  if(!VR || !VR.active) return;      // may be called before this file has run
  vrReadSticks();
  vrButtonJump(dt);
  vrButtonFreeze();
  vrUpdateBelt();
  Player.yaw -= VR.axes.rx * VR.turn * dt;      // smooth turn, as asked for
  if(vrPointAt()) vrLabel(null);                // on a desk, the desk is the label
  else vrHoverWorld();
  vrUpdateHold(dt);
  vrGunLabel(dt);
  vrPaintLabel(dt);
  vrUpdateRopes(dt);
  vrUpdateButtons(dt);
  vrCapBeams();
  vrPerf(dt, rawDt);
  VR.drawT += dt;
  /* RULING DV: the 5Hz repaint only happens where it could be seen.  drawT is
     deliberately NOT reset when it could not — it stays past the window, so
     walking back to a desk repaints on the first frame rather than up to 0.2s
     later, and a board that changed while nobody was there is never stale. */
  if(VR.drawT > 0.2 && vrConsoleSeen()){ VR.drawT = 0; vrDrawConsole(); }
}

     pixels.  It MUST be asked for here, before any session exists: WebXR
     reads it at session start and ignores it after. */
  if(renderer.xr.setFramebufferScaleFactor) renderer.xr.setFramebufferScaleFactor(0.85);
  if(renderer.xr.addEventListener){
    renderer.xr.addEventListener('sessionstart', vrOnStart);
    renderer.xr.addEventListener('sessionend', vrOnEnd);
  }
}
