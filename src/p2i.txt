
/* ============================================================================
   ROOMS — portal culling

   The building is five rooms in a line: the scene shop, the backstage crossover
   and dock, the stage, the auditorium, and the front of house.  You can only
   ever see the room you are standing in and the one either side of it, so
   everything else is switched off before the renderer ever looks at it.

   Nothing here moves geometry.  Each room is an empty Group sitting at the
   origin, so a mesh keeps exactly the position it was built with when it is
   handed over to its room.  Anything that straddles a boundary — the walls, the
   proscenium, the doors — goes in SHARED and is never hidden, and so does
   anything carrying a light, because hiding a light would change the look of a
   room you can still see into.
   ========================================================================== */
const ROOM_ORDER = ['shed','stage','house','lobby'];
const ROOMS = {
  /* Four rooms.  The warehouse shed (p2m) lives behind the stage house wall;
     the loading dock is off the stage-right wing — same z as the stage, so
     it is part of the stage and culls with it.                            */
  shed:  {label:'the warehouse',      z0:-Infinity,          z1:PAL_BACK - 0.7},
  stage: {label:'the stage',          z0:PAL_BACK - 0.7,     z1:1.3},
  house: {label:'the auditorium',     z0:1.3,                z1:D.houseBack + 0.5},
  lobby: {label:'the front of house', z0:D.houseBack + 0.5,  z1:Infinity}
};
/* what you can see from where.  A room always sees itself and its neighbours,
   so walking through a door never pops anything in. */
const ROOM_SEES = {
  shed:  ['shed','stage'],
  stage: ['stage','house','shed'],
  house: ['house','stage','lobby'],
  lobby: ['lobby','house']
};
const SHARED = new T.Group();
SHARED.name = 'shared';
const ROOM_GROUP = {};
let ROOMS_BUILT = false;
let curRoom = 'house';

/* which room is a point standing in */
function roomAt(z){
  for(const n of ROOM_ORDER){ const r = ROOMS[n]; if(z >= r.z0 && z < r.z1) return n; }
  return 'stage';
}

/* A world-space box that understands instanced meshes.  r128 builds an
   instanced bounding box from the base geometry alone, which would put all
   1392 seats at the origin and file the whole auditorium under "the stage". */
const _rb = new T.Box3(), _rm = new T.Matrix4();
function worldBox(obj){
  const box = new T.Box3();
  obj.traverse(c=>{
    const g = c.geometry;
    if(!g || !(c.isMesh || c.isLine || c.isPoints)) return;
    if(!g.boundingBox) g.computeBoundingBox();
    if(!g.boundingBox) return;
    if(c.isInstancedMesh){
      for(let i=0;i<c.count;i++){
        c.getMatrixAt(i, _rm);
        if(_rm.elements[0] === 0 && _rm.elements[5] === 0 && _rm.elements[10] === 0) continue;
        _rm.premultiply(c.matrixWorld);
        _rb.copy(g.boundingBox).applyMatrix4(_rm);
        if(!isFinite(_rb.min.x) || !isFinite(_rb.max.x)) continue;
        box.union(_rb);
      }
    } else {
      _rb.copy(g.boundingBox).applyMatrix4(c.matrixWorld);
      box.union(_rb);
    }
  });
  return box.isEmpty() ? null : box;
}
function carriesLight(obj){
  let found = false;
  obj.traverse(c=>{ if(c.isLight) found = true; });
  return found;
}
/* If I pin a whole room by hand and it happens to contain lights, the lights
   have to come out of it — a hidden light is a dark room.  They are lifted into
   SHARED at their exact world transform, so nothing moves.                   */
const _lp = new T.Vector3(), _lq = new T.Quaternion(), _ls = new T.Vector3();
function liftLights(obj){
  const found = [];
  obj.traverse(c=>{ if(c.isLight) found.push(c); });
  for(const l of found){
    l.updateMatrixWorld(true);
    l.matrixWorld.decompose(_lp, _lq, _ls);
    if(l.target && l.target.isObject3D && l.target.parent){
      l.target.updateMatrixWorld(true);
      const tp = new T.Vector3().setFromMatrixPosition(l.target.matrixWorld);
      SHARED.add(l.target);
      l.target.position.copy(tp);
    }
    SHARED.add(l);
    l.position.copy(_lp); l.quaternion.copy(_lq); l.scale.copy(_ls);
    l.updateMatrix();
  }
  return found.length;
}
/* which rooms does this thing overlap?  More than one and it is shared. */
function roomsOver(box){
  const hit = [];
  for(const n of ROOM_ORDER){
    const r = ROOMS[n];
    if(box.max.z > r.z0 + 0.02 && box.min.z < r.z1 - 0.02) hit.push(n);
  }
  return hit;
}

/* ---- sort the building into its rooms.  Run once, after it is all built. ---- */
const ROOM_FORCE = new Map();          // things I want kept together by hand
function roomForce(obj, room){ if(obj) ROOM_FORCE.set(obj, room); }

/* The warehouse (p2m) built before this part ran, so ROOM_FORCE did not
   exist yet for it to register with (a const is dead until its own line
   executes, however early the function around it hoists).  Pin it here:
   the shed group to its own room, and the door's [E] station to the stage,
   where you stand when you use it.                                        */
if(typeof SHEDS !== 'undefined' && SHEDS.palace){
  roomForce(SHEDS.palace.group, 'shed');
  roomForce(SHEDS.palace.ctl, 'stage');
  /* the cart is pushed about the whole building — file it with the walls
     and the doors, never with a room that can shut behind it */
  if(typeof CARTS !== 'undefined' && CARTS.palace)
    roomForce(CARTS.palace.group, 'shared');
}

function buildRooms(){
  if(ROOMS_BUILT) return;
  world.updateMatrixWorld(true);
  for(const n of ROOM_ORDER){
    const g = new T.Group(); g.name = n;
    ROOM_GROUP[n] = g;
  }
  const kids = world.children.slice();
  const tally = {shared:0};
  for(const n of ROOM_ORDER) tally[n] = 0;

  for(const o of kids){
    let room = ROOM_FORCE.get(o) || null;
    if(room && room !== 'shared' && !ROOM_GROUP[room]) room = 'shared';   // a room that no longer exists
    if(!room && carriesLight(o)) room = 'shared';
    if(!room){
      const box = worldBox(o);
      if(!box) room = 'shared';
      else {
        const over = roomsOver(box);
        room = (over.length === 1) ? over[0] : 'shared';
      }
    }
    (room === 'shared' ? SHARED : ROOM_GROUP[room]).add(o);
    if(room !== 'shared') liftLights(o);
    tally[room]++;
  }
  for(const n of ROOM_ORDER) world.add(ROOM_GROUP[n]);
  world.add(SHARED);
  world.updateMatrixWorld(true);
  ROOMS_BUILT = true;
  ROOM_TALLY = tally;
  return tally;
}
let ROOM_TALLY = null;

/* ---- the doors are real portals ----
   The only way between the auditorium and the front of house is through the
   pass doors.  With every one of them shut there is nothing to see, so the
   room on the far side comes off the list entirely.  Everything else — the
   crossover openings, the proscenium arch — is a permanent hole in a wall
   and is always treated as open.                                            */
function doorsShut(){
  for(const dw of DOORWAYS){
    if(dw.plane !== 'foh') continue;
    for(const l of dw.leaves) if(l.open > 0.01 || l.target > 0.01) return false;
  }
  return true;
}
/* The house curtain is a portal too, and the one you spend most of a working
   day behind.  With it in and closed there is no auditorium to see from
   upstage of it — so from the stage the whole house comes off the list.  Only
   in that direction: from the auditorium the curtain itself is part of the
   stage, so the stage is never culled from out front.                       */
function houseCurtainShut(){
  for(const ls of FLY){
    const spec = GOODS[ls.goodsKey];
    // any traveler is a front curtain — a show can hang its own
    if(!spec || !spec.traveler) continue;
    const trim = TRIMS[ls.goodsKey] !== undefined ? TRIMS[ls.goodsKey] : TRIMS.house;
    if(ls.pos > trim + 0.4) continue;                     // flown out
    if(ls.open > 0.02 || ls.travTarget > 0.02) continue;  // drawn open
    return ls;
  }
  return null;
}
function portalShut(a, b){
  if((a === 'house' && b === 'lobby') || (a === 'lobby' && b === 'house'))
    return doorsShut();
  /* the warehouse roller door is the only way between the stage and the
     shed: down and staying down, there is nothing of the other to see */
  if((a === 'stage' && b === 'shed') || (a === 'shed' && b === 'stage'))
    return !!(typeof SHEDS !== 'undefined' && SHEDS.palace &&
              SHEDS.palace.door.open < 0.01 && SHEDS.palace.door.target < 0.01);
  if(a === 'stage' && b === 'house'){
    const c = houseCurtainShut();
    if(!c) return false;
    // and only if you are actually standing behind it
    const z = (Player.mode === 'walk') ? Player.pos.z : camera.position.z;
    return z < c.z - 0.4;
  }
  return false;
}

/* ---- switch the rooms on and off as you move ---- */
let roomsShown = '';
function updateRooms(force){
  if(!ROOMS_BUILT) return curRoom;
  /* Two buildings.  Whichever one you are standing in, the other is switched
     off at the root — geometry, lights and all — so the far venue costs
     nothing while you are not in it.

     RULING DW put a number and a citation on "lights and all", because the
     2026-08-14 audit read the census as thirteen point lights from BOTH
     buildings at once and proposed a venue gate that is already right here.
     The mechanism: r128 walks the graph in `projectObject` and its very first
     line is `if (object.visible === false) return;` — ABOVE the `pushLight`
     and above the recursion into children (three.js r128:17954-17974).  So one
     false flag on a root prunes every light beneath it, and these three lines
     are that flag for a whole venue: `ARC.group.visible` for the Arc's, and
     `SHARED.visible` for the Palace's, since `buildRooms` files every
     light-carrying child of `world` into SHARED and `liftLights` pulls the
     lights out of the ones it does not.  Measured, both ways round, over the
     real `goToView` walk, BEFORE this same ruling's p2c gate landed: standing
     in the Palace 25 of 32 lights were gathered — 13 point lights, all of them
     the Palace's — and the Arc's 5 point lights plus its ambient and
     hemisphere were pruned; standing in the Arc it was 19 of 32, and the
     Palace's 13 were the ones pruned.  Neither building has ever paid for the
     other.  (Re-measure on this build and the Palace reads 24 of 32 and 12
     points — the yard light p2c gates out is the difference.  Both numbers
     are right; mind which build you are counting.)

     What is NOT gated, and is the honest remaining bite: within the Palace the
     house's own points live in SHARED and are never room-culled, deliberately
     (the comment at the top of this file — a hidden light is a dark room),
     while the Arc's own lamps were built into their room groups and therefore
     DO cull with them, so the Arc pays for 3 point lights on a stage where the
     Palace pays 12 (13 before p2c took the yard light).
     Closing that means deciding what a light may spill through, room by room —
     the stage work lights carry 40m of range across the whole auditorium —
     and it changes the picture, so it is not a free gate and it is not this
     ruling's.                                                              */
  if(typeof ARC !== 'undefined' && ARC.built){
    const px = (Player.mode === 'walk') ? Player.pos.x : camera.position.x;
    const ven = venueAt(px);
    if(ven !== VENUE){ VENUE = ven; force = true; roomsShown = ''; shadowsDirty = true; }
    ARC.group.visible = (ven === 'arc');
    /* the third building, switched off at the group the same way — and it is
       switched off HERE rather than inside its own branch, so that walking out
       of it cannot leave it drawn behind you */
    if(STU.built) STU.group.visible = (ven === 'studio');
    if(ven === 'arc'){
      for(const n of ROOM_ORDER) ROOM_GROUP[n].visible = false;
      SHARED.visible = false;
      updateArcRooms(force);
      return curRoom;
    }
    if(ven === 'studio'){
      for(const n of ROOM_ORDER) ROOM_GROUP[n].visible = false;
      SHARED.visible = false;
      updateStuRooms(force);
      return curRoom;
    }
    SHARED.visible = true;
  }
  const z = (Player.mode === 'walk') ? Player.pos.z : camera.position.z;
  const now = roomAt(z);
  let on;
  if(QUALITY.rooms){
    on = ROOM_SEES[now].filter(n => n === now || !portalShut(now, n));
  } else on = ROOM_ORDER;
  const key = QUALITY.rooms ? on.join(',') : 'all';
  if(key === roomsShown && !force){ curRoom = now; return now; }
  roomsShown = key;
  for(const n of ROOM_ORDER) ROOM_GROUP[n].visible = on.indexOf(n) !== -1;
  if(curRoom !== now && QUALITY.rooms) shadowsDirty = true;
  curRoom = now;
  return now;
}
/* what is actually being drawn right now */
function roomsLive(){
  if(typeof VENUE !== 'undefined' && VENUE === 'arc')
    return ARC.order.filter(n=>ARC.rooms[n].visible).map(n=>'arc:'+n);
  if(typeof VENUE !== 'undefined' && VENUE === 'studio' && STU.built)
    return STU.order.filter(n=>STU.rooms[n].visible).map(n=>'stu:'+n);
  if(!ROOMS_BUILT) return ROOM_ORDER.concat(['shared']);
  const out = ['shared'];
  for(const n of ROOM_ORDER) if(ROOM_GROUP[n].visible) out.push(n);
  return out;
}
/* what a room costs, for the readout on the settings card */
function roomCost(name){
  const g = (name === 'shared') ? SHARED : ROOM_GROUP[name];
  if(!g) return {meshes:0, tris:0};
  let meshes = 0, tris = 0;
  g.traverse(c=>{
    if(!c.isMesh || !c.geometry) return;
    meshes++;
    const idx = c.geometry.index;
    const n = idx ? idx.count/3 : (c.geometry.attributes.position ?
      c.geometry.attributes.position.count/3 : 0);
    tris += n * (c.isInstancedMesh ? c.count : 1);
  });
  return {meshes, tris:Math.round(tris)};
}
