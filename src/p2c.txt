/* ============================================================================
   DOORS — hinged, and they really do block you
   ========================================================================== */
const DOORS = [];
const DOORWAYS = [];
const doorGroup = new T.Group(); world.add(doorGroup);

function makeDoorway(o){
  // o: {x, y, z, w (clear width), h, plane}
  const dw = {x:o.x, y:o.y, z:o.z, halfW:o.w/2, h:o.h, plane:o.plane, leaves:[]};
  [-1, 1].forEach(side=>{
    const g = new T.Group();
    g.position.set(o.x + side*o.w/2, o.y, o.z);
    doorGroup.add(g);
    // the leaves overlap on the centre line, like a real pair with an astragal,
    // so there is no slot of daylight down the middle when they are shut
    const leafW = o.w/2 + 0.06;
    const leaf = new T.Mesh(new T.BoxGeometry(leafW, o.h, 0.09), M.mahog);
    leaf.position.set(-side*leafW/2, o.h/2, side < 0 ? 0 : 0.02);
    leaf.castShadow = leaf.receiveShadow = true;
    g.add(leaf);
    // a brass push bar and a small vision panel
    const bar = new T.Mesh(new T.CylinderGeometry(.035,.035,leafW*0.7,8), M.bronze);
    bar.rotation.z = Math.PI/2;
    bar.position.set(-side*leafW/2, o.h*0.45, 0.075); g.add(bar);
    const pane = new T.Mesh(new T.PlaneGeometry(leafW*0.5, 0.5),
      new T.MeshPhysicalMaterial({color:0x2a3442, transparent:true, opacity:.3, roughness:.1}));
    pane.position.set(-side*leafW/2, o.h*0.74, 0.05); g.add(pane);
    const d = {group:g, side:side, open:0, target:0, dw:dw};
    DOORS.push(d); dw.leaves.push(d);
  });
  DOORWAYS.push(dw);
  return dw;
}
function updateDoors(dt){
  for(const d of DOORS){
    if(Math.abs(d.target - d.open) < 0.002){ d.open = d.target; }
    else {
      const was = d.open;
      d.open += clamp(d.target - d.open, -dt*1.5, dt*1.5);
      if(was > 0.06 && d.open <= 0.06) Snd.play('doorSlam');
    }
    d.group.rotation.y = d.side * d.open * 1.75;    // ~100 degrees
  }
}
function toggleDoorway(dw){
  const closed = dw.leaves[0].target < 0.5;
  dw.leaves.forEach(l=>l.target = closed ? 1 : 0);
  toast('House doors ' + (closed ? 'open' : 'closed'));
  return closed;
}
function setAllDoors(open){
  DOORS.forEach(d=>d.target = open ? 1 : 0);
  toast(open ? 'All house doors open' : 'All house doors closed');
}
function doorwayAt(x, y){
  for(const dw of DOORWAYS){
    if(Math.abs(x - dw.x) < dw.halfW - 0.25 && y > dw.y - 0.6 && y < dw.y + dw.h) return dw;
  }
  return null;
}
/* true if this step would take you through the lobby wall illegally */
function doorBlocks(nx, nz, oz, py){
  const plane = FOH.z0 - 0.5;
  const crossing = (oz < plane) !== (nz < plane);
  if(!crossing) return false;
  const dw = doorwayAt(nx, py);
  if(!dw) return true;                       // solid wall
  return dw.leaves[0].open < 0.55;           // shut, or barely cracked
}

/* ============================================================================
   THE REAR WALL OF THE AUDITORIUM, with its door openings
   ========================================================================== */
(function rearWall(){
  const wallH = D.coveY + 2, halfW = D.houseW/2 + 1;
  const sh = new T.Shape();
  sh.moveTo(-halfW, 0); sh.lineTo(halfW, 0);
  sh.lineTo(halfW, wallH); sh.lineTo(-halfW, wallH); sh.lineTo(-halfW, 0);

  const cut = (cx, cy, w, h)=>{
    const p = new T.Path();
    p.moveTo(cx-w/2, cy); p.lineTo(cx+w/2, cy);
    p.lineTo(cx+w/2, cy+h); p.lineTo(cx-w/2, cy+h); p.lineTo(cx-w/2, cy);
    sh.holes.push(p);
  };
  const stallsY = houseFloorY(D.houseBack), mezzY = mezzFloorY(D.houseBack);
  [-9.5, 0, 9.5].forEach(x=>cut(x, stallsY, 3.2, 2.9));
  [-6.5, 6.5].forEach(x=>cut(x, mezzY, 3.2, 2.9));

  const geo = new T.ExtrudeGeometry(sh, {depth:1.0, bevelEnabled:false});
  const m = new T.Mesh(geo, M.ivoryD);
  m.position.set(0, 0, D.houseBack);
  m.castShadow = m.receiveShadow = true;
  world.add(m); STATIC.push(m);

  // wainscot panels between the doorways, at both levels
  const panelRun = (fy, gaps)=>{
    let x = -D.houseW/2 + 0.4;
    const edges = [];
    gaps.forEach(g=>{ edges.push([x, g-1.8]); x = g + 1.8; });
    edges.push([x, D.houseW/2 - 0.4]);
    edges.forEach(([a,b])=>{
      const w = b-a; if(w < 0.4) return;
      box(w, 1.36, .16, M.mahog, (a+b)/2, fy+0.68, D.houseBack-0.09);
      box(w, .18,  .30, M.gold,  (a+b)/2, fy+1.42, D.houseBack-0.14);
    });
  };
  panelRun(stallsY, [-9.5, 0, 9.5]);
  panelRun(mezzY,   [-6.5, 6.5]);

  // the doors themselves, and the architraves round them
  [-9.5, 0, 9.5].forEach(x=>{
    makeDoorway({x:x, y:stallsY, z:D.houseBack+0.06, w:3.2, h:2.9, plane:'foh'});
    box(3.8, .26, .34, M.gold, x, stallsY+3.0, D.houseBack+0.1);
  });
  [-6.5, 6.5].forEach(x=>{
    makeDoorway({x:x, y:mezzY, z:D.houseBack+0.06, w:3.2, h:2.9, plane:'foh'});
    box(3.8, .26, .34, M.gold, x, mezzY+3.0, D.houseBack+0.1);
  });
})();

/* ============================================================================
   THE LOADING DOCK — off the stage-right wing, with roller doors to the street
   Nothing sits behind the stage any more: the stage house is one big room with
   deep wings, and the get-in comes in from the side.
   ========================================================================== */
const DOCKDOORS = [];
(function dock(){
  const cm = new T.MeshStandardMaterial({color:0x3a3d42, roughness:.95});
  const bm = new T.MeshStandardMaterial({color:0x2f3237, roughness:.94});

  /* Stage right is negative x — the performer's right facing the audience —
     so the whole bay is built outward from XR with dir = -1.               */
  const dir = -1;
  DOCK.dir   = dir;
  DOCK.inner = XR;                       // the stage-house wall
  DOCK.outer = XR + dir*12;              // the street wall
  DOCK.x0 = Math.min(DOCK.inner, DOCK.outer);
  DOCK.x1 = Math.max(DOCK.inner, DOCK.outer);
  DOCK.z0 = -1.5;                        // downstage end of the bay
  DOCK.z1 = -13.5;                       // upstage end
  /* A loading dock IS stage height — that is the point of one.  The road
     outside drops away to a lorry bed below, so there is no ramp cluttering
     up the wing.                                                          */
  DOCK.y    = 0;
  DOCK.road = -1.25;
  DOCK.h    = 7.4;
  const cz = (DOCK.z0 + DOCK.z1)/2, L = DOCK.z0 - DOCK.z1;
  const W = Math.abs(DOCK.outer - DOCK.inner), mid = (DOCK.inner + DOCK.outer)/2;

  /* the dock floor, flush with the stage, lapping under the wall so the
     threshold is not a hole you fall through */
  WALKABLE.push(box(W + 1.6, .3, L, cm, mid + dir*-0.8, DOCK.y - 0.15, cz));
  // the dock lip, and a chequer plate edge along it
  box(0.3, 0.5, L, new T.MeshStandardMaterial({color:0x53585f, roughness:.6, metalness:.5}),
      DOCK.outer + dir*0.1, -0.25, cz);

  /* the shell of the bay — roof, both ends, and the street wall */
  box(W + 1, .5, L + 1, bm, mid, DOCK.y + DOCK.h, cz);
  box(W, DOCK.h, .5, bm, mid, DOCK.y + DOCK.h/2, DOCK.z1 - 0.25);
  box(W, DOCK.h, .5, bm, mid, DOCK.y + DOCK.h/2, DOCK.z0 + 0.25);
  /* The street wall, built from boxes.  It was an ExtrudeGeometry rotated
     into place, and rotateY(90) maps shape-u to world -z — so the whole wall
     came out mirrored about the centre line and ended up 15 m the other side
     of the building, leaving the bay with no street wall at all.          */
  {
    const wt = 0.5, DH = 4.6;
    const doorZ = [cz - 3.4, cz + 3.4], hw = 2.1;
    DOCK.doorZ = doorZ; DOCK.doorHalfW = hw; DOCK.doorH = DH;
    // header over the openings, lapped 0.1 down over the head so there is no
    // hairline seam where it meets the piers
    box(wt, DOCK.h - DH + 0.1, L, bm, DOCK.outer, DH - 0.05 + (DOCK.h - DH + 0.1)/2, cz);
    const piers = [
      [DOCK.z1, doorZ[0] - hw],
      [doorZ[0] + hw, doorZ[1] - hw],
      [doorZ[1] + hw, DOCK.z0]
    ];
    for(const [a2, b2] of piers){
      const w2 = b2 - a2;
      if(w2 > 0.02) box(wt, DH, w2, bm, DOCK.outer, DH/2, (a2 + b2)/2);
    }
  }

  /* ---- the roller doors ----
     Twelve slats each.  Rolling up, the slats travel and then stack into the
     drum above the opening, so it reads as a real shutter and not a plane
     that shrinks.                                                          */
  [-3.4, 3.4].forEach((off, i)=>{
    const cz2 = cz + off, DW = 4.2, H = 4.55, N = 12;
    const g = new T.Group();
    g.position.set(DOCK.outer - dir*0.34, DOCK.y, cz2);
    world.add(g);
    const slatMat = new T.MeshStandardMaterial({color:0x6a6f76, roughness:.55, metalness:.6});
    const slats = [];
    for(let k=0;k<N;k++){
      const sl = new T.Mesh(new T.BoxGeometry(0.09, H/N - 0.015, DW), slatMat);
      sl.position.set(0, (k + 0.5)*(H/N), 0);
      sl.castShadow = sl.receiveShadow = true;
      g.add(sl); slats.push(sl);
    }
    [-1,1].forEach(sd=>box(0.16, H + 0.5, 0.16, M.steel,
      DOCK.outer - dir*0.34, DOCK.y + H/2, cz2 + sd*(DW/2 + 0.1)));
    const drum = new T.Mesh(new T.CylinderGeometry(0.34, 0.34, DW + 0.3, 12), M.steel);
    drum.rotation.x = Math.PI/2;
    drum.position.set(DOCK.outer - dir*0.34, DOCK.y + H + 0.36, cz2);
    world.add(drum); STATIC.push(drum);

    const d = {group:g, slats, H, N, open:0, target:0, id:i+1, z:cz2,
               halfW: DW/2, drumY: H + 0.36, plane: DOCK.outer - dir*0.34};
    DOCKDOORS.push(d);

      DOCK.outer - dir*0.95, DOCK.y + 1.5, cz2 + DW/2 + 0.6);

  });

  /* ---- the big shutter into the stage house ----
     The stage-right wall was solid, so there was no way into the bay at all.
     It has an opening in it now, filled by one large door at scenery scale —
     wide enough and tall enough to walk a flat straight through.          */
  {
    const OZ0 = -12.5, OZ1 = -3.0, OH = 6.6;
    DOCK.openZ0 = OZ0; DOCK.openZ1 = OZ1; DOCK.openH = OH;
    const ocz = (OZ0 + OZ1)/2, OW = OZ1 - OZ0;

    for(let k=0;k<N;k++){
      const sl = new T.Mesh(new T.BoxGeometry(0.11, OH/N - 0.02, OW - 0.1), slatMat);
      sl.position.set(0, (k + 0.5)*(OH/N), 0);
      sl.castShadow = sl.receiveShadow = true;
      g.add(sl); slats.push(sl);
    }
    [-1,1].forEach(sd=>box(0.2, OH + 0.6, 0.2, M.steel, plane, OH/2, ocz + sd*(OW/2 + 0.12)));
    const drum = new T.Mesh(new T.CylinderGeometry(0.44, 0.44, OW + 0.4, 12), M.steel);
    drum.rotation.x = Math.PI/2;
    drum.position.set(plane, OH + 0.48, ocz);
    world.add(drum); STATIC.push(drum);

    DOCKDOORS.push({group:g, slats, H:OH, N, open:0, target:0, id:3, z:ocz,
                    halfW: OW/2, drumY: OH + 0.48, plane, big:true});
    const ctl = box(0.5, 0.7, 0.24, M.fixture, plane + dir*-0.55, 1.5, ocz + OW/2 + 0.9);
    station(ctl, 'dock3', 'THE BIG DOOR  —  [E] roll it up or down');
  }

  /* every wall of the bay, and where the holes in it are */
  DOCK.walls = [
    {plane: DOCK.inner - dir*0.3, gaps:[{z:(DOCK.openZ0 + DOCK.openZ1)/2,
                                         halfW:(DOCK.openZ1 - DOCK.openZ0)/2}]},
    {plane: DOCK.outer,           gaps: DOCK.doorZ.map(z2=>({z:z2, halfW:DOCK.doorHalfW}))}
  ];

  /* what you can see out of them: the road a lorry-bed below, and a ramp at
     the upstage end to walk back up */
  {
    WALKABLE.push(box(18, .2, L + 10, new T.MeshStandardMaterial({color:0x24262a, roughness:1}),
                      DOCK.outer + dir*9, DOCK.road - 0.1, cz));
    {
      const rl = 5.0;
      const ramp = new T.Mesh(new T.BoxGeometry(3.2, .26, rl),
        new T.MeshStandardMaterial({color:0x2e3136, roughness:.95}));
      ramp.position.set(DOCK.outer + dir*1.7, (DOCK.y + DOCK.road)/2, DOCK.z1 - 3.0);
      ramp.rotation.x = Math.atan2(DOCK.y - DOCK.road, rl);
      ramp.receiveShadow = true;
      world.add(ramp); STATIC.push(ramp); WALKABLE.push(ramp);
    }
    const lorry = new T.Group();
    lorry.position.set(DOCK.outer + dir*4.8, 0, cz - 4.4);
    lorry.rotation.y = 0.06;
    world.add(lorry);
    const bodyM = new T.MeshStandardMaterial({color:0x4a4f57, roughness:.7, metalness:.25});
    const bx = new T.Mesh(new T.BoxGeometry(2.6, 2.9, 8.4), bodyM);
    bx.position.set(0, 2.5, 0); bx.castShadow = bx.receiveShadow = true; lorry.add(bx);
    const cab = new T.Mesh(new T.BoxGeometry(2.5, 2.2, 2.4),
      new T.MeshStandardMaterial({color:0x6d2b24, roughness:.6, metalness:.3}));
    cab.position.set(0, 2.1, 5.6); cab.castShadow = true; lorry.add(cab);
    for(const z2 of [-3.0, -1.4, 4.6]) for(const sd of [-1,1]){
      const wh = new T.Mesh(new T.CylinderGeometry(0.62,0.62,0.34,12),
        new T.MeshStandardMaterial({color:0x141519, roughness:.98}));
      wh.rotation.z = Math.PI/2;
      wh.position.set(sd*1.3, 0.62, z2); lorry.add(wh);
    }
    lorry.traverse(o=>{ if(o.isMesh) STATIC.push(o); });
  }

  /* dock furniture */
  const cases = [];
  for(let i=0;i<7;i++)
    cases.push({x:DOCK.inner + dir*(2.4 + (i%3)*1.5), y:DOCK.y + 0.55,
                z:DOCK.z1 + 2.2 + Math.floor(i/3)*1.4, ry:rnd(-0.08,0.08)});
  instanced(new T.BoxGeometry(1.3,1.1,0.9),
    new T.MeshStandardMaterial({color:0x1b1d22, roughness:.85}), cases, {cast:true});

  BOH.light  = new T.PointLight(0xbfd0e0, 0, 30, 1);
  BOH.light.position.set(DOCK.inner + dir*4, DOCK.y + DOCK.h - 1.2, cz + 3); world.add(BOH.light);
  BOH.light2 = new T.PointLight(0xbfd0e0, 0, 30, 1);
  BOH.light2.position.set(DOCK.inner + dir*4, DOCK.y + DOCK.h - 1.2, cz - 4); world.add(BOH.light2);
  /* RULING DW — the yard light out over the road, and the one light in this
     building that serves NO ONE.  It was minted at intensity 0 and nothing has
     ever written it: `updateRig` (p4) drives BOH.light and BOH.light2 off
     HOUSE.backstage and stops there, and nothing anywhere writes light3's
     intensity.  Intensity 0 is NOT out of the loop — r128
     gathers every visible light in `projectObject` (three.js r128:17954: the
     `visible === false` return sits ABOVE the pushLight) and then every
     standard material's fragment runs a full point-light iteration for it, both
     eyes, every frame, for a lamp nobody has ever seen.  So it leaves the loop.
     If the yard should light one day the fix is to drive it off `bo` beside the
     other two and delete this flag — and the price of that picture is one
     point-light slot in every lit fragment in the Palace, which is worth
     stating rather than discovering. */
  BOH.light3 = new T.PointLight(0xd8e4f0, 0, 26, 1);
  BOH.light3.position.set(DOCK.outer + dir*4, 6.0, cz);
  BOH.light3.visible = false;
  world.add(BOH.light3);
})();

/* the shutters run like the pass doors do */
function updateDockDoors(dt){
  for(const d of DOCKDOORS){
    if(Math.abs(d.target - d.open) < 0.002){ d.open = d.target; }
    else d.open += clamp(d.target - d.open, -dt*0.55, dt*0.55);
    const lift = d.open * d.H;
    for(let k=0;k<d.N;k++){
      const base = (k + 0.5)*(d.H/d.N);
      const y = base + lift;
      const sl = d.slats[k];
      if(y > d.H + 0.05){
        // it has reached the head: from here it rolls into the drum
        const over = Math.min(1, (y - d.H)/(d.H/d.N));
        sl.position.y = d.drumY || (d.H + 0.36);
        sl.position.x = -0.02 * (k % 3);
        sl.scale.set(1, 1 - over*0.92, 1);
      } else {
        sl.position.y = y; sl.position.x = 0; sl.scale.set(1,1,1);
      }
    }
  }
}
function toggleDockDoor(d){
  if(!d) return false;
  const shut = d.target < 0.5;
  d.target = shut ? 1 : 0;
  toast('Dock door ' + d.id + (shut ? ' rolling up' : ' coming down'));
  Snd.play(shut ? 'whoosh' : 'doorSlam');
  return shut;
}
function setAllDockDoors(open){
  DOCKDOORS.forEach(d=>d.target = open ? 1 : 0);
  toast(open ? 'Both dock doors up' : 'Dock doors down');
}
