
/* ============================================================================
   BEETLEJUICE — the fifth production, and the first taken off a recording

   Two different kinds of work wear one name here, and the line between them
   matters more than anything else in this file:

     MEASURED     cue times, fade durations, blackouts, the act break.  They
                  came off a recording of a real performance with
                  tools/video.js and they are quoted in the cue comments as
                  the times they actually are.
     INTERPRETED  every channel level, every focus, and all of this scenery.
                  We are rendering lit AREAS onto our own 25-channel rig,
                  which is not the rig they used.

   RULING AV (2026-08-10) REPEALS RULING AO for this show.  The owner:
   "remove the cluase entirely that it has to be our onw shapes. make the
   sets look as close to the real stuff as you can get it".  So this one is
   MODELLED ON THE PRODUCTION — proportions, colours, lettering and all —
   where the other four remain interpretations in their own vocabulary.  AO
   is left standing in its own spec, marked superseded: a ruling that got
   reversed is worth more on the record than one quietly deleted.

   The set changes are SCENES (p5c): one stage, and the sets take turns on
   it.  That machinery was built for this show, removed with it, and this is
   the first production to use it.  A scene that is off has its layers
   disabled, so it costs no draw call and no raycast while it waits.
   ========================================================================== */

/* THE SIGN'S LAMPS ON A CUE.  `signCol:'#ff1e10'` throws every lamp on the

/* ============================================================================
   RULING CU — THE MARQUEE GOES DARK AS IT FLIES

   "Also make it so the lights on the bj sign turn off when it goes up."

   The sign is an ALWAYS scene on its own y mover (RULINGS AS, AT).  It goes out
   at BJ_SIGN_OUT at the end of the opening sequence and its lamps have burned
   all night up in the grid ever since it was built.

   A GATE, not a repaint.  What a cue asked for is REMEMBERED (SHOW.signWant) and
   the gate multiplies it, so the signCol state survives underneath: a cue that
   turned the sign red keeps it red, the sign simply is not lit while it is up,
   and it comes back red if it comes back in.  Writing final values into the
   materials — which is what this function used to do — cannot express "red, but
   currently switched off" at all.

   IT RIDES THE SIGN'S OWN MOVER AND THE FRAME dt, never a timer (CLAUDE.md).
   The lamps start going out the moment the sign starts to rise, over about half
   a second, which is a marquee being switched off rather than a lamp dimming.

   AND AN IMPORTED MATERIAL MAY HAVE NO EMISSIVE AT ALL — RULING CF guards for
   it, because his file could land as a MeshBasic — so the gate falls back to
   driving the COLOUR toward black, which is the one thing such a material can
   do.  Both halves are exercised: the stand-in is emissive, his is not
   guaranteed to be. */
const BJ_SIGN_LAMP_FADE = 2.5;   // per second: out in ~0.4s
const BJ_SIGN_LAMP_CUT  = 0.05;  // metres of travel that count as "it went up"

function bjSignApply(){
  if(!SHOW.signLamps) return;
  const hex = SHOW.signWant;
  const g = (SHOW.signGate === undefined) ? 1 : SHOW.signGate;
  SHOW.signLamps.forEach(r=>{
    if(hex){
      r.mat.color.set(hex);
      if(r.mat.emissive){
        r.mat.emissive.set(hex);
        r.mat.emissiveIntensity = BJ_SIGN_LIT * g;
      } else r.mat.color.multiplyScalar(g);
    } else {
      r.mat.color.copy(r.c);
      if(r.mat.emissive && r.e){
        r.mat.emissive.copy(r.e);
        r.mat.emissiveIntensity = r.ei * g;
      } else r.mat.color.multiplyScalar(g);
    }
  });
}
function setSignLamps(hex){
  SHOW.signWant = hex || null;
  bjSignApply();
}
function updateSignLamps(dt){
  if(!SHOW.signLamps) return;
  const sc = (typeof sceneFind === 'function') ? sceneFind('bjSign') : null;
  /* NO MOVER IS NOT "IT IS UP".  A production whose sign cannot fly, or a sign
     built before its travel is wired, must leave the lamps alone rather than
     switching them off for ever. */
  const up = !!(sc && sc.mv && sc.mv.off > BJ_SIGN_LAMP_CUT);
  const was = (SHOW.signGate === undefined) ? 1 : SHOW.signGate;
  const want = up ? 0 : 1;
  const g = clamp(was + clamp(want - was, -dt*BJ_SIGN_LAMP_FADE, dt*BJ_SIGN_LAMP_FADE), 0, 1);
  if(Math.abs(g - was) < 1e-5) return;
  SHOW.signGate = g;
  bjSignApply();
}
function bjStripeTex(vertical){
  const W = 512, H = 512, c = cvs(W, H), g = c.getContext('2d');
  g.fillStyle = '#efe9dd'; g.fillRect(0, 0, W, H);
  g.fillStyle = '#141018';
  const band = 44;
  for(let i = 0; i < W/band; i += 2){
    if(vertical) g.fillRect(i*band, 0, band, H);
    else         g.fillRect(0, i*band, W, band);
  }
  /* they are painted by hand, so the edges wobble */
  g.globalAlpha = 0.5;
  for(let i = 0; i < 90; i++){
    g.fillStyle = Math.random() < 0.5 ? '#141018' : '#efe9dd';
    if(vertical) g.fillRect(rnd(0, W), rnd(0, H), 3, rnd(20, 90));
    else         g.fillRect(rnd(0, W), rnd(0, H), rnd(20, 90), 3);
  }
  g.globalAlpha = 1;
  const t = tex(c, 1, 1);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  return t;
}
function bjDressing(sc, key){
  const g = new T.Group();
  g.name = 'dress:' + key;
  sc.group.add(g);
  (sc.dress || (sc.dress = {}))[key] = g;
  return {group:g};
}
function bjDress(scOrName, key){
  const sc = (typeof scOrName === 'string') ? sceneFind(scOrName) : scOrName;
  if(!sc || !sc.dress || !sc.dress[key]) return null;
  sc.dressOn = key;
  bjRedress(sc);
  return key;
}
/* sceneShow enables the layers of every descendant when a set comes on, which
   would light all three dressings at once.  sceneApply calls this after. */
function bjRedress(sc){
  if(!sc || !sc.dress) return;
  for(const k of Object.keys(sc.dress)){
    const on = (k === sc.dressOn), g = sc.dress[k];
    /* RULING CN — ONE HOUSE IN THE WORLD.  "Make it so only one house exist in
       the world at a time and it switches between them when needed."

       The dressing that is up is a child of the scene; the other two are held
       out of the graph entirely (showHoldOffstage, p5c) and come back whole when
       they are called.  That is the ruling: three complete houses of his standing
       in the same 13.6m of stage, two of them switched off inside the third, is
       what he watched and what he has ruled on.

       THE LAYER WORK STAYS, and it is not redundant.  It is what makes a dressing
       inert for the frame it IS attached — the one between these two calls — and
       it is the guard that catches a detach that failed: a dressing that somehow
       stayed in the graph is still dark and still unpickable rather than standing
       inside its neighbour. */
    g.userData.sceneOff = !on;
    if(typeof setPieceVisible === 'function') setPieceVisible(g);
    g.traverse(o=>{ if(on) o.layers.set(0); else o.layers.disableAll(); });
    if(typeof showHoldOffstage !== 'function') continue;
    if(on) showBringOnstage(g, sc.group);
    else showHoldOffstage(g);
  }
}
/* an enclosed set is a flown piece (RULING AY): the whole of it goes out
   through the header on one named part, 'all'.  Called AFTER the scene is
   built: every existing child — meshes and dressing groups alike — moves
   into the wrapper without its transform changing, because both the scene
   group and the wrapper sit at the origin. */
/* RULING CE — AS FEW SETS FLY AS POSSIBLE.

   "make it so as little sets as posible are flown just like roof and the bedroom
   and closet should eb flown the otheres should come on from the sides or back"

   So the wrapper travels on whatever AXIS the set arrives along, and flying is
   just the y case of it.  Nothing else about the choreography changes: the
   changeover already drives `all` to its `out` on strike and back to `home` on
   entry, so a set that tracks on from stage left is the same machinery pointed
   sideways.

   THE LIST OF WHICH SET GOES WHERE HAS SINCE BEEN RE-CUT BY RULINGS CQ, CR AND
   CS — "make the roof nethereworld and house exterior flown ... make the bedroom
   and the closet come in from one side" — so it is deliberately no longer written
   out here.  It lives at each scene's own call, where the measurement that
   chooses the number lives too, and in the traffic-plan block at the top of this
   file.  This function is the part of CE that did not move. */
function bjTrackWhole(sc, axis, out, speed){
  const g = new T.Group();
  g.name = 'bj:' + sc.name + ':all';
  while(sc.group.children.length) g.add(sc.group.children[0]);
  sc.group.add(g);
  sceneTravelPart(sc, 'all', g, axis, 0, out, speed);
  /* RULING BQ — AND WHERE IT TRAVELS TO IS ALREADY A PARK.  A set sent out on
     `all` is standing in the grid or in a wing, which is exactly where a real set
     lives between its scenes, so it only has to be DECLARED: no second mover, no
     new number.  Measured (tools/parked.js): the flown ones land at y 10.50 with
     their tops between 15.63 and 19.70 against a grid at 25, clear of the brick
     and both wing walls, and NOT ONE RAY of 1025 from a stalls eye reaches any of
     them. */
  scenePark(sc);
  return g;
}
/* the three he named, plus the netherworld: out through the header on y */

/* RULING CT — squash a built set in DEPTH about its own downstage face.

   The wrapper does the whole job: a point at the front face maps to itself
   (front*f + front*(1-f) = front) and a point d metres upstage of it maps to
   d*f upstage.  So the picture starts where it started, exactly as BJ_AFT_DEEP
   does for the netherworld's authored z values — the difference is that this
   one squashes the PIECES too, which is the point.  A 5.2m roof deck has to
   become a 1.5m one or the set is still 5.2m deep however close together its
   parts stand.

   CALLED BEFORE bjFlyWhole, like everything else that wraps: it takes the
   children that exist when it runs.

   The box is read in the scene group's own space — the group is added to
   showRoot(), which is at the origin for the Palace and need not be for another
   stage, and a front face measured in world coordinates would put the squash
   plane in the wrong building. */
function bjThinWhole(sc, f){
  if(!sc || !(f > 0) || f === 1) return null;
  const g = new T.Group();
  g.name = 'bj:' + sc.name + ':thin';
  sc.group.updateMatrixWorld(true);
  const b = new T.Box3().setFromObject(sc.group);
  if(b.isEmpty() || !isFinite(b.max.z)) return null;
  const front = b.max.z - sc.group.getWorldPosition(new T.Vector3()).z;
  while(sc.group.children.length) g.add(sc.group.children[0]);
  g.scale.z = f;
  g.position.z = front * (1 - f);
  sc.group.add(g);
  return g;
}

/* ---- the graveyard backdrop, repainted to the production photograph
   (RULING AW): a deep teal-to-indigo night sky, an ENORMOUS pale moon
   upper-stage-left-of-centre with painted grey craters and the halo it
   throws, and banks of blue-grey cloud streaked horizontally around and
   behind it, lit pale from below.  No painted ground row any more — the
   hills, stones and trees are CUT pieces downstage of this cloth now, so
   painting them here would double every silhouette. ---- */
function bjBackdropTex(){
  const W = 2048, H = 1024, c = cvs(W, H), g = c.getContext('2d');
  /* the cloth hangs (D.procW + 6) x 11.5, the canvas is 2:1 — anything meant
     to read ROUND on the cloth is drawn wider by this much in canvas space */
  const SQX = (W * 11.5) / (H * (D.procW + 6));
  /* the sky: teal at the horizon climbing into indigo-black */
  const sky = g.createLinearGradient(0, 0, 0, H);
  sky.addColorStop(0,    '#050e18');
  sky.addColorStop(0.40, '#0d2436');
  sky.addColorStop(0.72, '#1a2a52');
  sky.addColorStop(1,    '#283462');
  g.fillStyle = sky; g.fillRect(0, 0, W, H);
  /* the field darkens into both top corners, the way the photo has it */
  for(const cx of [0, W]){
    const v = g.createRadialGradient(cx, 0, 40, cx, 0, W*0.55);
    v.addColorStop(0, 'rgba(2,6,12,.62)');
    v.addColorStop(1, 'rgba(2,6,12,0)');
    g.fillStyle = v; g.fillRect(0, 0, W, H);
  }
  /* one cloud bank: a run of flat overlapping lumps with a pale underside —
     the banks are LIT FROM BELOW by the moon, so the light edge hangs under
     the dark body, not over it */
  const bank = (bx, by, bw, lump, body, under)=>{
    for(let i = 0; i < lump; i++){
      const x = bx + (i/(lump-1) - 0.5)*bw + rnd(-30, 30);
      const y = by + rnd(-14, 14);
      const rx = rnd(90, 170)*SQX, ry = rnd(22, 42);
      g.fillStyle = body;
      g.save(); g.translate(x, y); g.scale(rx, ry);
      g.beginPath(); g.arc(0, 0, 1, 0, 6.283); g.fill(); g.restore();
      g.fillStyle = under;
      g.save(); g.translate(x, y + ry*0.72); g.scale(rx*0.86, ry*0.34);
      g.beginPath(); g.arc(0, 0, 1, 0, 6.283); g.fill(); g.restore();
    }
  };
  /* the banks behind the moon, working down the cloth */
  bank(W*0.30, H*0.22, W*0.52, 7, 'rgba(38,52,72,.55)',  'rgba(150,172,196,.20)');
  bank(W*0.72, H*0.34, W*0.50, 7, 'rgba(44,60,82,.58)',  'rgba(168,190,212,.26)');
  bank(W*0.24, H*0.48, W*0.60, 8, 'rgba(50,66,88,.60)',  'rgba(176,198,218,.30)');
  bank(W*0.66, H*0.62, W*0.58, 8, 'rgba(42,56,76,.55)',  'rgba(160,182,204,.26)');
  bank(W*0.32, H*0.76, W*0.72, 9, 'rgba(34,46,64,.50)',  'rgba(140,160,182,.20)');
  /* horizontal streaking through the banks — it is scenic paint, laid
     sideways with a wide brush */
  for(let i = 0; i < 70; i++){
    const y = rnd(H*0.14, H*0.86);
    g.fillStyle = Math.random() < 0.5 ? 'rgba(180,200,220,.05)' : 'rgba(8,14,24,.07)';
    g.fillRect(rnd(-60, W*0.7), y, rnd(W*0.2, W*0.55), rnd(2, 6));
  }
  /* THE MOON — enormous, a third the height of the cloth, hung upper-stage-
     left-of-centre (stage left is canvas right: u=1 lands at world +x). */
  const mx = W*0.60, my = H*0.30, mr = H*0.167;      // ry: diameter ~ H/3
  const halo = g.createRadialGradient(mx, my, mr*0.6, mx, my, mr*2.8);
  halo.addColorStop(0,    'rgba(214,228,238,.42)');
  halo.addColorStop(0.45, 'rgba(190,208,224,.16)');
  halo.addColorStop(1,    'rgba(190,208,224,0)');
  g.save(); g.translate(mx, my); g.scale(SQX, 1); g.translate(-mx, -my);
  g.fillStyle = halo;
  g.beginPath(); g.arc(mx, my, mr*2.8, 0, 6.283); g.fill();
  /* the disc, pale and bone-coloured */
  g.fillStyle = '#e7e3d3';
  g.beginPath(); g.arc(mx, my, mr, 0, 6.283); g.fill();
  /* the craters, painted grey and clearly meant — each a soft-edged pool
     with a darker rim on its lower side, the way a scenic artist shades one */
  const craters = [[-0.34,-0.30,0.20],[0.22,-0.42,0.13],[0.44,0.06,0.17],
                   [-0.08,0.16,0.24],[-0.52,0.28,0.12],[0.18,0.44,0.15],
                   [-0.24,-0.02,0.10],[0.50,-0.28,0.09]];
  for(const cr of craters){
    const x = mx + cr[0]*mr, y = my + cr[1]*mr, r = cr[2]*mr;
    g.fillStyle = 'rgba(148,146,132,.42)';
    g.beginPath(); g.arc(x, y, r, 0, 6.283); g.fill();
    g.fillStyle = 'rgba(112,110,98,.30)';
    g.beginPath(); g.arc(x, y + r*0.28, r*0.72, 0, 6.283); g.fill();
  }
  /* the shading that keeps the disc a body, not a plate */
  const sh = g.createRadialGradient(mx - mr*0.3, my - mr*0.3, mr*0.2, mx, my, mr);
  sh.addColorStop(0, 'rgba(255,255,246,.16)');
  sh.addColorStop(0.75, 'rgba(120,124,120,0)');
  sh.addColorStop(1, 'rgba(96,100,104,.28)');
  g.fillStyle = sh;
  g.beginPath(); g.arc(mx, my, mr, 0, 6.283); g.fill();
  g.restore();
  /* and two thin streaks crossing the lower quarter of the disc — the banks
     run AROUND the moon, and one of them runs in front */
  for(const st of [[my + mr*0.55, 0.42], [my + mr*0.85, 0.34]]){
    g.fillStyle = 'rgba(40,54,74,' + st[1] + ')';
    g.fillRect(mx - mr*SQX*1.7, st[0] - 12, mr*SQX*3.4, 24);
    g.fillStyle = 'rgba(170,192,214,.18)';
    g.fillRect(mx - mr*SQX*1.5, st[0] + 10, mr*SQX*3.0, 6);
  }
  return tex(c, 1, 1);
}

        gr.addColorStop(0, 'rgba(104,58,26,.30)'); gr.addColorStop(1, 'rgba(104,58,26,0)');

}

/* ---- the show curtain: a hard vertical stripe, our own colours ---- */
function bjCurtainCanvas(){
  const W = 1024, H = 512, c = cvs(W, H), g = c.getContext('2d');
  /* the ground: deep violet, going bluer stage left and hotter stage right,
     which is how the cloth reads in the photograph */
  const base = g.createLinearGradient(0, 0, W, 0);
  base.addColorStop(0,    '#2a1a63');
  base.addColorStop(0.34, '#3b1f7a');
  base.addColorStop(0.62, '#57208a');
  base.addColorStop(1,    '#3a2170');
  g.fillStyle = base; g.fillRect(0, 0, W, H);
  /* it is a DRAPE, so the colour lives in the pleats — a bright edge and a
     shadowed fold every band, not a flat wash */
  const band = W / 26;
  for(let i = 0; i < 26; i++){
    const gr = g.createLinearGradient(i*band, 0, (i+1)*band, 0);
    gr.addColorStop(0,    'rgba(0,0,0,.46)');
    gr.addColorStop(0.30, 'rgba(150,110,225,.20)');
    gr.addColorStop(0.52, 'rgba(190,150,255,.28)');
    gr.addColorStop(0.74, 'rgba(90,60,170,.10)');
    gr.addColorStop(1,    'rgba(0,0,0,.46)');
    g.fillStyle = gr; g.fillRect(i*band, 0, band, H);
  }
  /* THE SCROLLWORK.  The cloth is covered in loose black curls — big lazy
     spirals with hooks coming off them, drawn the size of a person and
     scattered over the whole drop. */
  g.lineCap = 'round'; g.lineJoin = 'round';
  const curl = (cx, cy, s, turns, dir)=>{
    g.beginPath();
    let a = 0, r = 2 * s;
    g.moveTo(cx + r, cy);
    const steps = Math.round(turns * 14);
    for(let i = 1; i <= steps; i++){
      a = dir * (i / 14) * 6.283;
      r = (2 + i * 0.30) * s;
      g.lineTo(cx + Math.cos(a) * r, cy + Math.sin(a) * r * 0.86);
    }
    g.stroke();
  };
  const sprig = (x, y, s, dir)=>{
    g.beginPath();
    g.moveTo(x, y);
    g.bezierCurveTo(x + dir*18*s, y - 26*s, x + dir*40*s, y - 30*s, x + dir*46*s, y - 6*s);
    g.bezierCurveTo(x + dir*50*s, y + 14*s, x + dir*32*s, y + 20*s, x + dir*26*s, y + 6*s);
    g.stroke();
  };
  for(let i = 0; i < 46; i++){
    const x = rnd(-40, W + 40), y = rnd(-30, H + 30), s = 0.5 + Math.random()*1.25;
    g.strokeStyle = 'rgba(8,4,16,' + (0.55 + Math.random()*0.35).toFixed(2) + ')';
    g.lineWidth = 4.5 * s;
    curl(x, y, s * 2.1, 1.2 + Math.random()*0.9, Math.random() < 0.5 ? 1 : -1);
    sprig(x + rnd(-30, 30), y + rnd(-40, 40), s, Math.random() < 0.5 ? 1 : -1);
  }
  /* the seams between the widths of fabric, and the soft floor bounce */
  for(let i = 1; i < 6; i++){
    g.strokeStyle = 'rgba(0,0,0,.30)'; g.lineWidth = 2;
    g.beginPath(); g.moveTo(i*W/6, 0); g.lineTo(i*W/6, H); g.stroke();
  }
  const gr2 = g.createLinearGradient(0, H, 0, H*0.55);
  gr2.addColorStop(0, 'rgba(120,60,190,.30)');
  gr2.addColorStop(1, 'rgba(120,60,190,0)');
  g.fillStyle = gr2; g.fillRect(0, 0, W, H);
  const t = tex(c, 1, 1);
  return {tex: t};
}

/* ---- the marquee: two decks of BETELGEUSE in red neon inside a bulb

    g.strokeStyle = '#ff3d18';             g.lineWidth = 8;  draw();

/* ============================================================================
   THE PROSCENIUM NEON'S COLOUR AND LEVEL, CUE-DRIVEN (RULING CH).

   A cue carries `portal:{col:'#hex', lvl:N}`.  MISSING FIELD = DARK:
   showCueExtras calls setPortal on EVERY cue, so a cue that says nothing about
   the frame fades it to nothing, which is his "For the rest it is off" exactly.

   The fade rides the frame dt like the netherworld neon and everything else in
   this file — never a timer (CLAUDE.md).

   THE COLOUR SNAPS WHILE THE BAR IS DARK, and only then.  Lerping it would be
   invisible at zero anyway, but a cue that lights the frame red straight out of
   a dark stretch that was last blue would spend its whole rise crossfading, and
   the first thing seen would be the wrong colour coming up.  Lit, it crossfades,
   because a neon transformer changing colour under the audience's eyes is a
   thing that should take a moment.
   ========================================================================== */
function setPortal(p){
  const b = SHOW.bjPortal; if(!b) return;
  b.tLvl = (p && p.lvl !== undefined) ? p.lvl : 0;
  if(p && p.col !== undefined){
    b.tCol.set(p.col);
    if(b.lvl <= 0.02) b.col.copy(b.tCol);      // dark: come up in the new colour
  }
}
function updatePortal(dt){
  const b = SHOW.bjPortal; if(!b) return;
  b.lvl += clamp(b.tLvl - b.lvl, -dt*BJ_NEON_FADE, dt*BJ_NEON_FADE);  // clamps AT the target
  b.col.lerp(b.tCol, Math.min(1, dt*BJ_NEON_FADE));
  b.mat.emissiveIntensity = b.lvl;
  b.mat.emissive.copy(b.col);
  /* the bar's own face: unlit glass when dark, lifted toward the emissive when
     lit, so it reads as a lit thing rather than a black rod with a halo */
  b.mat.color.setHex(BJ_NEON_DARK);
  if(b.lvl > 0.02) b.mat.color.lerp(b.col, clamp(b.lvl, 0, 1)*0.8);
}

/* ============================================================================
   THE PLOT — the opening sequence.

   Every TIME in these comments is measured off the recording; every LEVEL and
   every FOCUS is interpreted onto our 25 channels.  Fade durations are the
   measured ones where a slope was clean inside a single held shot.  The rest
   of the show's cues arrive with the full measured list.
