function goToView(n){
  const v = VIEWS[n]; if(!v) return;
  curView = n;
  Player.pos.copy(v.pos); Player.yaw = v.yaw; Player.pitch = v.pitch;
  Player.vel.set(0,0,0);
  /* Crossing town has to happen BEFORE the ground probe: the floor you are
     about to land on is in the building that is currently switched off, and
     a raycast into a hidden room still finds it but the venue bookkeeping
     would be a frame behind. */
  VENUE = venueAt(v.pos.x);
  if(ARC.built) ARC.group.visible = (VENUE === 'arc');
  updateArcRooms(true);
  if(typeof updateStageFromPosition === 'function') updateStageFromPosition();
  // land on whatever floor is actually there, rather than falling through it
  const g = groundAt(v.pos.x, v.pos.z, v.pos.y + 2.5);
  if(g !== null && Math.abs(g - v.pos.y) < 6){ Player.pos.y = g; Player.onGround = true; }
  Orbit.target.copy(v.orbit[0]); Orbit.dist=v.orbit[1]; Orbit.yaw=v.orbit[2]; Orbit.pitch=v.orbit[3];
  updateRooms(true);          // a teleport can cross the whole building
  $('#posChip').textContent = v.name;
  if(typeof refreshVenueUI === 'function') refreshVenueUI();
}

/* ---- ground probe ---- */
const downRay = new T.Raycaster(new T.Vector3(), new T.Vector3(0,-1,0), 0, 30);
function groundAt(x,z,fromY){
  downRay.set(new T.Vector3(x, fromY+1.4, z), new T.Vector3(0,-1,0));
  downRay.far = 28;
  const hits = downRay.intersectObjects(WALKABLE, true);
  return hits.length ? hits[0].point.y : null;
}
/* Every rule below describes a wall of the Palace, in the Palace's own
   coordinates.  None of them mean anything in the other building — and left
   ungated they would follow you across town and stop you walking through
   thin air at the Arc.                                                     */
function inPalace(){ return VENUE !== 'arc'; }
/* the upstage wall — solid except for the warehouse doorway at x = 0 (p2m),
   and there the roller door has to be most of the way up to pass */
function backWallBlocks(x, z, oz){
  if(!inPalace()) return false;
  const plane = D.backWall - 0.3;
  if((oz < plane) === (z < plane)) return false;   // not crossing it
  const S = (typeof SHEDS !== 'undefined') ? SHEDS.palace : null;
  if(S && Math.abs(x - S.door.x) < S.door.halfW - 0.2)
    return S.door.open < 0.72;
  return true;
}
/* the shed's own shell: its side walls, and the thickness of the theatre's
   back wall anywhere but the doorway (position checks, not crossings, so a
   step can never tunnel in from the side) */
function shedWallBlocks(nx, nz){
  if(!inPalace()) return false;
  const S = (typeof SHEDS !== 'undefined') ? SHEDS.palace : null;
  if(!S) return false;
  if(nz <= S.z1)                                    // inside the shed proper
    return nx < S.x0 + 0.4 || nx > S.x1 - 0.4;
  if(nz < D.backWall - 0.05)                        // inside the wall's thickness
    return Math.abs(nx - S.door.x) > S.door.halfW - 0.2;
  return false;
}
/* the stage-right wall: the dock opening is the only way out to the get-in */
/* Every wall of the dock bay knows where its own openings are, so one rule
   covers the stage-house wall and the street wall both.                    */
function dockWallBlocks(x, z, ox){
  if(!inPalace() || !DOCK.walls) return false;
  for(const w of DOCK.walls){
    if((ox < w.plane) === (x < w.plane)) continue;     // not crossing this one
    if(!w.gaps.some(g=>Math.abs(z - g.z) < g.halfW - 0.3)) return true;
  }
  return false;
}
/* and the shutters themselves stop you while they are down */
function dockDoorBlocks(x, z, ox, y){
  for(const d of DOCKDOORS){
    if((ox < d.plane) === (x < d.plane)) continue;
    if(Math.abs(z - d.z) > d.halfW) continue;
    if(y > d.H + 0.15) continue;                       // stepped over the top of it
    if(d.open < 0.72) return true;                     // most of the way up to pass
  }
  return false;
}
/* the proscenium walls: you can't walk through the arch off-centre */
function throughWall(x,z,oz){
  const inArch = (a,b)=>(a<1.3 && b>1.3)||(a>1.3 && b<1.3);
  return Math.abs(x) > D.procW/2-0.5 && inArch(z,oz);
}

/* ---- hauling a lineset by hand ----

let machDrag = null;
function grabMachine(m){
  if(!m) return false;
  if(m.powered){ toast(m.label+' #'+m.id+' is powered — drive it from the MACHINES page'); return false; }
  machDrag = m;
  toast('Pushing '+m.label+' #'+m.id);
  return true;
}
let flyDrag = null, flyDragged = false;
function grabLineset(ls){
  if(!ls) return false;
  if(ls.locked){ toast('Lineset '+ls.id+' is locked off — unlock it on the rail first'); return false; }
  /* a board move in flight hands over, exactly as at the VR rail (p9):
     nothing relocks itself under a live hand */
  ls.relock = false;
  flyDrag = ls; flyDragged = false;
  toast('Hauling lineset '+ls.id+' — pull to fly, let go to set it');
  return true;
}
function haul(dy){
  if(!flyDrag) return;
  const ls = flyDrag;
  const feel = 0.028 * (1.7/(1 + ls.weight/520));     // weight in the hand
  ls.target = clamp(ls.target - dy*feel, minTrimOf(ls), OUT_TRIM+0.4);
  if(Math.abs(dy) > 0.5) flyDragged = true;
  syncFlyRow(ls);
}
function releaseLineset(){
  if(flyDrag){ toast('Lineset '+flyDrag.id+' set at '+flyDrag.target.toFixed(1)+' m'); flyDrag = null; }
}

    const tag = (e.target.tagName||'').toLowerCase();
    if(tag!=='input' && tag!=='textarea' && tag!=='select') toggleShiftLock();
  }

/* ---- shift lock: TAP Shift and the cursor locks to the centre of the
   screen with the mouse steering the camera; tap again and it lets go.
   HOLD Shift and it is still run — the toggle only fires on a tap shorter
   than 250ms, so sprinting never drops the mouse.  `shiftLock` is not a
   second state: the chip simply mirrors the real pointer lock, however it
   changed (tap, canvas click, or the browser's Esc). ---- */
let shiftLock = false, shiftTapAt = -1e9;
function setShiftLock(v){
  if(v === shiftLock) return;
  shiftLock = v;
  const c = $('#slChip');
  c.classList.toggle('on', v);
  c.querySelector('b').textContent = v ? 'ON' : 'OFF';
  toast(v ? 'Cursor locked — mouse steers the camera, tap Shift to release'
          : 'Cursor released — tap Shift to lock it again');
}
function toggleShiftLock(){
  if(Player.mode !== 'walk') return;
  if(typeof VR !== 'undefined' && VR && VR.active) return;
  if(pointerLocked) unlock(); else lock();
}

/* ---- walking ---- */
function updatePlayer(dt){
  if(BENCH.open) return;              // the bench has the keyboard
  if(Player.mode!=='walk') return;
  const running = keys['ShiftLeft'] || keys['ShiftRight'];
  const spd = running ? 6.4 : 3.1;
  let fx=0, fz=0;
  if(keys['KeyW']) fz+=1; if(keys['KeyS']) fz-=1;
  if(keys['KeyA']) fx-=1; if(keys['KeyD']) fx+=1;
  if(typeof VR !== 'undefined' && VR && VR.active){
    fx += VR.axes.lx; fz -= VR.axes.ly;     // stick forward reads as -1
  }
  const len = Math.hypot(fx,fz);
  /* a stick half over should walk at half pace, so keep the magnitude
     rather than normalising every input to a full stride */
  const mag = Math.min(1, len);
  if(len>0){ fx = fx/len*mag; fz = fz/len*mag; }
  // the camera looks down -Z, so forward = (-sin yaw, -cos yaw), right = (cos yaw, -sin yaw)
  const sin=Math.sin(Player.yaw), cos=Math.cos(Player.yaw);
  const wx = ( fx*cos - fz*sin)*spd;
  const wz = (-fx*sin - fz*cos)*spd;

  /* flying is VR-only: double-tap A (p9) sets VR.fly, landing clears it */
  const flying = typeof VR !== 'undefined' && VR && VR.active && VR.fly;

  // gravity — unless you are flying
  let ny;
  if(flying){ Player.vel.y = 0; ny = Player.pos.y; }
  else {
    Player.vel.y -= 13.5*dt;
    ny = Player.pos.y + Player.vel.y*dt;
  }

  // horizontal with step check
  const tryMove=(ax,az)=>{
    const nx = Player.pos.x + ax*dt, nz = Player.pos.z + az*dt;
    if(!inPalace()){
      // the other building has its own edges, walls and doors
      if(!arcBounds(nx, nz)) return;
      if(arcWallBlocks(nx, nz, Player.pos.x, Player.pos.z, Player.pos.y)) return;
    } else {
      // the world runs from the street beyond the dock across to the stage-left wall
      if(nx < DOCK.x0 - 16 || nx > XL + 0.6) return;
      // upstage it now ends at the warehouse shed's rear wall, not the stage's
      const zFloor = (typeof SHEDS !== 'undefined' && SHEDS.palace)
        ? SHEDS.palace.z0 + 0.5 : D.backWall + 0.6;
      if(nz < zFloor || nz > FOH.z1 - 0.9) return;
      if(Player.pos.y < 5 && throughWall(nx, nz, Player.pos.z)) return;
      if(doorBlocks(nx, nz, Player.pos.z, Player.pos.y)) return;
      if(backWallBlocks(nx, nz, Player.pos.z)) return;
      if(shedWallBlocks(nx, nz)) return;
      if(dockWallBlocks(nx, nz, Player.pos.x)) return;
      if(dockDoorBlocks(nx, nz, Player.pos.x, Player.pos.y)) return;
    }
    const g = groundAt(nx, nz, Player.pos.y);
    if(g===null) return;
    if(g - Player.pos.y > 0.62) return;          // too tall to step up
    Player.pos.x = nx; Player.pos.z = nz;
  };
  if(flying){
    /* FLYING — the stick moves you along the way the HEAD looks: look up and
       push forward to climb.  Strafe stays on the flat, walls still block. */
    camera.getWorldDirection(_flyDir);
    const fs = 8;                                  // a notch over sprint
    const vx = (_flyDir.x*fz + fx*cos)*fs;
    const vz = (_flyDir.z*fz - fx*sin)*fs;
    tryMove(vx,0); tryMove(0,vz);
    const vy = _flyDir.y*fz*fs;
    ny = clamp(Player.pos.y + vy*dt, 0, 60);
    const g = groundAt(Player.pos.x, Player.pos.z, Player.pos.y);
    if(vy < -0.01 && g!==null && ny <= g + 0.01){
      /* feet met a floor on the way down: land, and walk off */
      ny = g; VR.fly = false; Player.onGround = true;
      toast('Landed — walking');
    } else Player.onGround = false;
    Player.pos.y = ny;
  } else {
    tryMove(wx,0); tryMove(0,wz);

    const g = groundAt(Player.pos.x, Player.pos.z, Player.pos.y);
    if(g!==null && ny <= g + 0.01){ ny = g; Player.vel.y=0; Player.onGround=true; }
    else Player.onGround=false;
    Player.pos.y = ny;
  }

  // head bob
  const moving = len>0 && Player.onGround;
  Player.bob = (Player.bob||0) + (moving? dt*(spd>5?13:8) : 0);
  const bob = moving ? Math.sin(Player.bob)*0.035 : 0;

  if(typeof VR !== 'undefined' && VR && VR.active){
    /* in a session the headset writes the camera every frame — all we may do
       is move the rig it is sitting in */
    vrPlaceRig();
  } else {
    camera.position.set(Player.pos.x, Player.pos.y + Player.eye + bob, Player.pos.z);
    camera.rotation.set(0,0,0);
    camera.rotateY(Player.yaw); camera.rotateX(Player.pitch);
  }

  // where am I?
  const zone = !inPalace() ? arcZone(Player.pos.x, Player.pos.y, Player.pos.z) :
      Player.pos.z < D.apron+0.4 ?
      (Player.pos.y > 15 ? 'THE GRID' : Player.pos.y > 5 ? 'FLY RAIL' :
       Math.abs(Player.pos.x) > D.procW/2 ? 'THE WINGS' : 'ON DECK') :
      (Player.pos.y > D.balcY-1 ? 'SECOND BALCONY' :
       Player.pos.y > D.mezzY-1 ? 'FIRST BALCONY' : 'ORCHESTRA');
  $('#posChip').textContent = zone;
}
function updateOrbit(){
  if(Player.mode!=='orbit') return;
  const cp = Math.cos(Orbit.pitch);
  camera.position.set(
    Orbit.target.x + Math.sin(Orbit.yaw)*cp*Orbit.dist,
    Orbit.target.y - Math.sin(Orbit.pitch)*Orbit.dist,
    Orbit.target.z + Math.cos(Orbit.yaw)*cp*Orbit.dist);
  camera.lookAt(Orbit.target);
}

const mouseNDC = new T.Vector2();
let hoverInfo = null, focusMode=false, dragPiece=null, dragPlane=new T.Plane(new T.Vector3(0,1,0),0);

function castFromCamera(){
  ray.setFromCamera(new T.Vector2(0,0), camera);
  return ray;
}
function castFromMouse(e){
  mouseNDC.set((e.clientX/innerWidth)*2-1, -(e.clientY/innerHeight)*2+1);
  ray.setFromCamera(mouseNDC, camera);
  return ray;
}
const _pickList = [];
function pickAll(r){
  /* a short, curated list — raycasting the whole building every frame was the
     single most expensive thing the game did                                */
  _pickList.length = 0;
  _pickList.push(rigGroup, flyGroup, setGroup, doorGroup);
  for(const o of INTERACT) _pickList.push(o);
  const hits = r.intersectObjects(_pickList, true);
  return hits.length?hits[0]:null;
}
function describe(obj){
  let o=obj;
  while(o){
    if(o.userData.piece) return {kind:'scenic', piece:o.userData.piece, label:'SCENIC — '+o.userData.piece.label+'  [E] select'};
    const f = FIXTURES.find(x=>x.group===o);
    if(f) return {kind:'fixture', f, label:'CH '+f.ch+'  '+f.name+'   [E] select / bump'};
    const ls = FLY.find(x=>x.group===o);
    if(ls) return {kind:'lineset', ls, label:'LINESET '+ls.id+' — '+GOODS[ls.goodsKey].label+
      (ls.locked ? '   [LOCKED OFF]' : '   [E] fly in / out  ·  hold LMB and pull to haul')};
    if(o.userData && o.userData.station)
      return {kind:'station', st:o.userData.station, obj:o, label:o.userData.station.label};
    if(o.userData && o.userData.machinePart){
      const m = o.userData.machinePart;
      return {kind:'machine', m, label:(m.powered ? 'POWERED ' : '') + m.label.toUpperCase() +
        ' #' + m.id + (m.powered ? '   [driven from the MACHINES page]'
                                 : '   hold the left button and pull')};
    }
    /* the traveler's hand line (p9): the VR hover label reads this, and it
       says which run does what rather than offering an E nobody has */
    if(o.userData && o.userData.travRope)
      return {kind:'traveler', ls:o.userData.travRope,
        label:'TRAVELER — front run opens · back run closes'};
    /* the warehouse cart (p2m): VR-only, a squeeze on the handle takes it */
    if(o.userData && o.userData.cartInfo)
      return {kind:'cart', label:'WAREHOUSE CART — squeeze the handle and push'};
    const dr = DOORS.find(x=>x.group===o);
    if(dr) return {kind:'door', dw:dr.dw,
      label:'HOUSE DOOR   [E] '+(dr.target>0.5?'close':'open')};
    o = o.parent;
  }
  return null;
}
function updateHover(){
  if(Player.mode!=='walk' || !pointerLocked){ $('#target').classList.remove('on'); hoverInfo=null; return; }
  const hit = pickAll(castFromCamera());
  if(hit && hit.distance < 22){
    const info = describe(hit.object);
    if(info){ hoverInfo = info; $('#target').textContent = info.label; $('#target').classList.add('on'); return; }
  }
  hoverInfo=null; $('#target').classList.remove('on');
}
/* act on a described target — the E key feeds it hoverInfo, and the VR
   triggers (p9) feed it whatever the controller ray lands on */
function useInfo(info){
  if(!info) return;
  if(info.kind==='fixture'){
    const i = sectionOfChannel(info.f.ch);
    selectSection(i);
    setSection(SECTIONS[i], SECTIONS[i].level > 0.02 ? 0 : 1, 0.6);
    showPage('lights');
  } else if(info.kind==='lineset'){
    const ls=info.ls;
    if(ls.locked){ toast('Lineset '+ls.id+' is locked off'); return; }
    (ls.pos > 15 ? flyIn : flyOut)(ls);
    toast('Lineset '+ls.id+' '+(ls.pos>15?'coming in':'going out'));
  } else if(info.kind==='scenic'){
    selPiece = info.piece; refreshSetUI(); showPage('sets'); toast('Selected '+selPiece.label);
  } else if(info.kind==='station'){
    useStation(info.st, info.obj);
  } else if(info.kind==='door'){
    toggleDoorway(info.dw);
  }
}
function useTarget(){ useInfo(hoverInfo); }
function toggleFollow(){
  const f = FIXTURES.find(x=>x.follow); if(!f) return;
  if(followTarget){ followTarget=null; toast('Followspot released'); }
  else { followTarget = new T.Vector3(); toast('Followspot picked you up'); setLevel(f.ch, 1, 1.5); }
}

/* placing & dragging scenic */
dom.addEventListener('click', e=>{
  const r = Player.mode==='walk' ? castFromCamera() : castFromMouse(e);
  if(placingKey){
    const dk = (typeof stageDeck === 'function') ? stageDeck() : deck;
    const hits = dk ? r.intersectObject(dk, false) : [];
    if(hits.length){
      const p = hits[0].point;
      const piece = placeScenic(placingKey, Math.round(p.x*4)/4, Math.round(p.z*4)/4);
      selPiece = piece; toast(SCENIC[placingKey].label+' placed');
      if(!keys['ShiftLeft']) { placingKey=null; $$('#setPal .b').forEach(x=>x.classList.remove('on')); }
      refreshSetUI();
    }
    return;
  }
  if(focusMode){
    const dk = (typeof stageDeck === 'function') ? stageDeck() : deck;
    const hits = r.intersectObjects([dk, setGroup].filter(Boolean), true);
    if(hits.length){
      const sec = SECTIONS[selSec], p = hits[0].point;
      if(sec.chans) sec.chans.forEach((c,i)=>{
        const f = chan(c); if(!f) return;
        const spread = (i - (sec.chans.length-1)/2) * 1.6;
        const pt = p.clone(); pt.x += spread;
        if(f.mover) aimMoverAt(f, pt); else f.aim.copy(pt);
      });
      toast(sec.name.replace('<br>',' ')+' refocused');
      focusMode=false; $('#btnFocus').classList.remove('on');
    }
    return;
  }
  if(Player.mode==='orbit'){
    const hit = pickAll(r);
    if(hit){ const info=describe(hit.object);
      if(info && info.kind==='scenic'){ selPiece=info.piece; refreshSetUI(); }
      else if(info && info.kind==='fixture'){ selectSection(sectionOfChannel(info.f.ch)); showPage('lights'); }
      else if(info && info.kind==='lineset'){ selLineset=info.ls; refreshFlyUI(); showPage('fly'); }
    }
  }
});
function tryGrabScenic(e){
  const r = castFromMouse(e);
  const hits = r.intersectObject(setGroup, true);
  if(!hits.length) return false;
  let o=hits[0].object; while(o && !o.userData.piece) o=o.parent;
  if(!o) return false;
  dragPiece = o.userData.piece; selPiece = dragPiece; refreshSetUI();
  return true;
}
function dragScenic(e){
  const r = castFromMouse(e);
  if(keys['ShiftLeft']||keys['ShiftRight']){
    dragPiece.group.position.y = clamp(dragPiece.group.position.y - e.movementY*0.01, 0, 6);
  } else {
    dragPlane.set(new T.Vector3(0,1,0), -dragPiece.group.position.y);
    const p = new T.Vector3();
    if(r.ray.intersectPlane(dragPlane, p)){
      /* the clamps are the stage box, wherever the live stage's box is */
      const o = (typeof stageOrigin === 'function') ? stageOrigin() : {x:0, z:0};
      dragPiece.group.position.x = clamp(p.x, o.x - 13, o.x + 13);
      dragPiece.group.position.z = clamp(p.z, o.z + D.backWall + 1, o.z + D.apron);
    }
  }
  refreshSetUI();
}

function buildSmokeUI(){
  const rack = $('#smokeRack'); if(!rack) return;
  // rebuild when a show rigs or strikes machines, or the board patches to
  // another stage — the count alone collides across stages (§5's DOM trap:
  // faders would keep driving the parked rack's units)
  const rackKey = (typeof STAGE !== 'undefined' ? STAGE : '') + ':' + SMOKE.units.length;
  if(rack.dataset.built === rackKey) return;
  rack.dataset.built = rackKey;
  rack.innerHTML = SMOKE.units.map((u,i)=>
    '<div class="smk" data-u="'+i+'">'+
      '<div class="nm">'+u.name+'</div>'+
      '<div class="kd">'+(u.kind === 'fog' ? 'low fog' : 'haze')+
        (u.show ? ' &middot; <span style="color:var(--amber)">show</span>' : '')+'</div>'+
      '<input type="range" class="smkFader" min="0" max="100" value="0" style="width:100%">'+
      '<div class="rd"><span class="led"></span>'+
        '<span class="tk"><i></i></span><span class="out">0%</span></div>'+
      '<button class="b" style="width:100%;margin-top:6px;letter-spacing:2px">BURST</button>'+
    '</div>').join('');
  Array.from(rack.querySelectorAll('.smk')).forEach(el=>{
    const u = SMOKE.units[+el.dataset.u];
    el.querySelector('.smkFader').value = Math.round(u.target*100);
    el.querySelector('.smkFader').oninput = e=>{ setSmoke(u, (+e.target.value)/100); };
    el.querySelector('button').onclick = ()=>smokeBurst(u);
  });
  syncSmokeUI();
}
function syncSmokeUI(){
  const rack = $('#smokeRack'); if(!rack || !rack.dataset.built) return;
  Array.from(rack.querySelectorAll('.smk')).forEach(el=>{
    const u = SMOKE.units[+el.dataset.u];
    const led = el.querySelector('.led');
    const able = u.tank > 0.01 && u.heat > 0.25;
    led.className = 'led' + (u.tank <= 0.01 ? ' dry' : able ? ' on' : '');
    el.querySelector('.tk i').style.width = Math.round(u.tank*100)+'%';
    el.querySelector('.out').textContent = Math.round(u.out*100)+'%';
  });
  const st = $('#smokeStat');
  if(st) st.innerHTML =
    'in the air: <b style="color:var(--amber)">'+Math.round(SMOKE.haze*100)+'%</b><br>'+
    (SMOKE.live || 0)+' puffs drifting<br>'+
    'effective haze '+Math.round(clamp(RIG.haze + SMOKE.haze*0.7, 0, 1)*100)+'%';
}

function refreshHangUI(){

  const loaded = !!SHOW.key;
  const saved  = loaded && SHOW_HANGS[SHOW.key];

  if(!loaded){ st.innerHTML = 'No show loaded. Load one on the SHOWS page and the '+

let last = performance.now(), fpsT=0, fpsN=0, showClock=0, uiT=0, frameErrs=0;
let LOOP_XR = false;      // set by the VR module: the headset drives the clock
function frame(now){
  if(!LOOP_XR) requestAnimationFrame(frame);
  try{ tick(now); }
  catch(err){
    if(++frameErrs === 1){ console.error(err); FATAL('The show stopped', err.message+'\n\n'+err.stack); }
  }
}
let hoverT = 0, frameNo = 0;
function tick(now){
  const dt = Math.min(0.05, (now-last)/1000); last = now;
  const t = now/1000;
  frameNo++;

  updatePlayer(dt);
  updateOrbit();
  updateRooms(false);         // switch off the rooms you cannot see into
  updateFades(dt);
  stepProgram(dt);
  updateFly(dt);
  updateDoors(dt);
  updateDockDoors(dt);
  if(typeof updateSheds === 'function') updateSheds(dt);
  updateMachines(dt);
  updateStorm(dt);            // rain, lightning and firelight
  updateCrew(dt);             // the stagehands, if they have been called
  updateSmoke(dt);            // the fog machines
  updateArc(dt);              // the other building's foyer
  updateStageFromPosition();  // the board is patched to the room you are in
  updateStageHouseLights();   // and every auditorium answers its own master
  if(typeof vrUpdate === 'function') vrUpdate(dt);
  if(VENUE !== lastVenue){ lastVenue = VENUE; refreshVenueUI(); }
  if(followTarget){
    if(Player.mode==='walk') followTarget.set(Player.pos.x, Player.pos.y+1.2, Player.pos.z);
    else followTarget.copy(Orbit.target);
  }
  updateRig(dt, t);

  // picking is only needed a few times a second, not every frame
  hoverT += dt;
  if(hoverT > 0.12){ hoverT = 0; updateHover(); }

  // shadow maps refresh on a cadence rather than every single frame
  renderer.shadowMap.needsUpdate = QUALITY.shadows && (shadowsDirty || (frameNo & 1) === 0);
  shadowsDirty = false;

  showClock += dt;
  uiT += dt;
  if(uiT > 0.1){
    uiT = 0;
    refreshChannelStrip();
    if($('#p-fly').classList.contains('on')){ syncFlyUI(); syncRailGroups(); refreshHangUI(); }
    const m = Math.floor(showClock/60), s = Math.floor(showClock%60);
    $('#clockTxt').textContent = String(m).padStart(2,'0')+':'+String(s).padStart(2,'0');
    const live = HOUSE.house < 0.15;
    const lc = $('#liveChip');
    lc.textContent = live ? 'SHOW IN PROGRESS' : 'HOUSE OPEN';
    lc.classList.toggle('live', live);
    if($('#p-lights').classList.contains('on')) showRoomCost();
    if($('#p-show').classList.contains('on')) refreshCrewUI();
    if($('#p-smoke').classList.contains('on')) syncSmokeUI();
  }
  fpsN++; fpsT+=dt;
  if(fpsT>0.6){
    const fps = Math.round(fpsN/fpsT);
    $('#fpsTxt').textContent = fps;
    fpsN=0; fpsT=0;
    autoTune(fps);
  }

  /* with the bench up, the theatre is hidden behind it — draw the grid instead */
  if(BENCH.open){
    if(BENCH.tab === 'build') edRender();
    return;
  }
  renderer.render(scene, camera);
}

/* if the machine is struggling, quietly step the renderer down a gear */
let tuneStreak = 0;
function autoTune(fps){
  if(QUALITY.level !== 'auto') return;
  if(fps < 34){ tuneStreak++; } else if(fps > 52){ tuneStreak--; } else { tuneStreak = 0; }
  if(tuneStreak >= 3 && QUALITY.scale > 0.63){
    QUALITY.scale = Math.max(0.62, QUALITY.scale - 0.16);
    if(QUALITY.scale <= 0.7 && QUALITY.shadows){ QUALITY.shadows=false; renderer.shadowMap.enabled=false; }
    resize(); tuneStreak = 0;
    toast('Eased the render quality back to hold the frame rate');
  } else if(tuneStreak <= -6 && QUALITY.scale < 1){
    QUALITY.scale = Math.min(1, QUALITY.scale + 0.12); resize(); tuneStreak = 0;
  }
}

  function set(list,v,c){ list.forEach(n=>{ const f=chan(n); if(f){ f.level=v; if(c) f.color.set(c); } }); }
  const savedLevels = snapshotLX();
  seed.forEach((s,i)=>{
    looks[i]();
    CUES.push({n:s.n, label:s.label, fade:s.fade, follow:null, lx:snapshotLX(), fly:null, sfx:null,
      house:(s.house!==undefined?s.house:0), work:(s.work!==undefined?s.work:0),
      practical:0, haze:RIG.haze});
  });
  // restore the entry look
  savedLevels.forEach((r,i)=>{ const f=chan(i+1); f.level=r.lvl; f.color.set(r.col); });
  CUES[2].fly = snapshotFly();
  nextCue = 0;
  refreshCues();

  refreshShowUI();
  refreshCrewUI();
  showPage('lights');
  toast('Welcome to the house. Press H for the manual.');
}

(function boot(){
  const bar = $('#bar i'), txt = $('#loadTxt');
  const steps = ['RAISING THE IRON…','HANGING THE PLOT…','LOADING THE ARBORS…',
                 'PATCHING THE BOARD…','FOCUS SESSION…','HOUSE OPEN'];
  let i=0;
  const iv = setInterval(()=>{
    i++; bar.style.width = (i/steps.length*100)+'%';
    txt.textContent = steps[Math.min(i,steps.length-1)];
    if(i>=steps.length){
      clearInterval(iv);
      $('#startBtn').style.display='block';
    }
  }, 260);
  $('#startBtn').onclick = ()=>{
    Snd.init();
    $('#veil').style.opacity=0;
    setTimeout(()=>$('#veil').style.display='none', 800);
    if(Player.mode==='walk') lock();
  };
})();
