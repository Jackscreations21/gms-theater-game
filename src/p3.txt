/* drapery geometry with fullness (sine-wave pleats) */
function drape(w,h,mat,pleats,amp){
  const g = new T.PlaneGeometry(w,h, Math.max(24,pleats*4), 10);
  const p = g.attributes.position;
  for(let i=0;i<p.count;i++){
    const x=p.getX(i), y=p.getY(i);
    const sag = (1 - Math.abs(y/(h/2)))*0.35;
    p.setZ(i, Math.sin(x*pleats*0.6)*(amp)*(0.5+sag));
  }
  g.computeVertexNormals();
  const m = new T.Mesh(g, mat);
  m.castShadow = true; m.receiveShadow = true;
  return m;
}

function dropGoods(kind){
  const g=new T.Group();
  const m=new T.MeshStandardMaterial({map:TX.drops[kind],roughness:1,side:T.DoubleSide});
  const d=drape(D.procW+6,11.5,m,2,.05); d.position.y=-5.75; g.add(d); return g;

function makeLineset(i){
  const z = -0.50 - i*0.80;    // every lineset lives UPSTAGE of the plaster line
  const ls = {
    /* a counterweight rail at rest is locked off — every line wakes with its
       lock thrown.  The lock is the HAND's interlock; the board is the
       flyman, and flyTo works the lock itself. */
    id:i+1, z:z, pos:OUT_TRIM, target:OUT_TRIM, locked:true, goodsKey:'none',
    speed:1.2, weight:0, moving:false, open:1, travTarget:1,
    /* a line let go of at the rail with no lock thrown on it — see updateFly */
    runaway:false, flyVel:0,
    /* a board move that took the lock off itself owes it back on arrival */
    relock:false
  };
  const grp = new T.Group(); grp.position.set(0, ls.pos, z); flyGroup.add(grp);
  const batten = new T.Mesh(new T.CylinderGeometry(.055,.055,D.procW+6,8), M.pipe);
  batten.rotation.z = Math.PI/2; grp.add(batten);
  // lift lines up to the grid
  const lines = new T.Group(); grp.add(lines);
  [-6.2,-2.1,2.1,6.2].forEach(x=>{
    const l = new T.Mesh(new T.CylinderGeometry(.012,.012,1,4), M.rope);
    l.position.x = x; lines.add(l);
  });
  ls.group = grp; ls.lines = lines; ls.goods = null;
  FLY.push(ls);
  return ls;
}
for(let i=0;i<14;i++) makeLineset(i);

function hangGoods(ls, key){
  if(ls.goods){ ls.group.remove(ls.goods); disposeTree(ls.goods); ls.goods=null; }
  ls.goodsKey = key;
  const spec = GOODS[key];
  ls.weight = spec.wt;
  ls.speed = 1.55/(1+spec.wt/420);
  ls.h = spec.h;
  if(key!=='none'){ const g = spec.build(); ls.goods = g; ls.group.add(g); }
  if(key==='electric') buildElectric(ls);
  else stripElectric(ls);
  ls.practical = ls.goods && ls.goods.userData.practical || null;
  /* RULING V — a pipe lifts to make room for what you just hung.
     minTrimOf (#15) is enforced by everything that MOVES a pipe, and
     hanging is not a move: a 13m house curtain hung on a pipe standing at
     2m put ten metres of cloth straight through the stage.  A pipe with
     room to spare is left exactly where it stands. */
  const lo = minTrimOf(ls);
  if(ls.pos < lo){ ls.pos = ls.target = lo; ls.group.position.y = ls.pos; }
  refreshFlyUI();
}
function disposeTree(o){ o.traverse(n=>{ if(n.geometry) n.geometry.dispose(); }); }
/* working trim heights — where a piece of goods lives when it's "in" */
const TRIMS = {none:9.6, pipe:9.6, electric:9.6, truss:10.0, border:12.2, valance:12.7,
               chand:8.6, cyc:13.7, house:12.6};
function inTrimOf(ls){
  if(TRIMS[ls.goodsKey] !== undefined) return TRIMS[ls.goodsKey];
  const spec = GOODS[ls.goodsKey];
  if(!spec || !spec.h) return 8.0;
  return clamp(spec.h + 0.05, 0.6, OUT_TRIM);
}
/* the lowest a lineset may be flown: goods hang BELOW the pipe, so a pipe
   brought in far enough puts its cloth through the deck (y = 0) */
function minTrimOf(ls){
  // the pipe may only come down until what hangs on it kisses the deck —
  // but never above the goods' own working trim: the house curtain is CUT
  // to puddle (h 13.0, trim 12.6) and must still make its trim.  ls.asmH
  // is anything BUILT and nailed to this pipe (p4c asmHangDepth): a flown
  // flat gets the same floor the cloth gets.
  const goods = Math.min((ls.h || 0) + 0.05, inTrimOf(ls));
  const built = ls.asmH ? ls.asmH + 0.05 : 0;
  return clamp(Math.max(goods, built), 0.6, OUT_TRIM);
}
function flyTo(ls, y, instant){
  /* the board is the flyman: a thrown lock does not refuse it.  It takes the
     lock off itself, runs the line, and locks off again on arrival
     (updateFly).  Only the hand is interlocked — the desktop haul (p7) and
     the VR rope (p9) get nothing while the lever is in. */
  if(ls.locked){ ls.locked = false; ls.relock = true; }
  ls.target = clamp(y, minTrimOf(ls), OUT_TRIM+0.4);
  if(instant) ls.pos = ls.target;
  return true;
}
const flyIn  = ls => flyTo(ls, inTrimOf(ls));
const flyOut = ls => flyTo(ls, OUT_TRIM);

function updateFly(dt){
  for(const ls of FLY){
    /* ---- A LINE THAT HAS BEEN LET GO OF -----------------------------------
       Take your hand off a rope at the pin rail without throwing the little
       red lever and there is nothing at all holding the load: it runs away
       downwards until it is on the deck.  A line let go of while it was
       flying OUT slows, stops and comes back down the way it went up; one
       already coming in just keeps coming.  Anything that writes ls.target
       — the board, a cue, a group call — has taken the line back, and the
       runaway is over.  Only the VR rail starts one (p9); the desktop haul
       still sets a line where you leave it.                              */
    if(ls.runaway){
      if(Math.abs(ls.target - ls.pos) > 0.02) ls.runaway = false;
      else {
        const lo = (typeof minTrimOf === 'function') ? minTrimOf(ls) : 0.6;
        const hi = OUT_TRIM + 0.4;
        ls.flyVel = Math.max(ls.flyVel - (1.2 + ls.weight/400)*dt,
                            -(1.5 + ls.weight/500));
        ls.pos += ls.flyVel*dt;
        if(ls.pos <= lo){                       // on the deck, and that is that
          ls.pos = lo; ls.flyVel = 0; ls.runaway = false;
          if(ls.moving){
            ls.moving = false;
            if(typeof Snd !== 'undefined' && Snd.railStop) Snd.railStop(ls);
          }
        } else {
          if(ls.pos >= hi){ ls.pos = hi; if(ls.flyVel > 0) ls.flyVel = 0; }
          if(!ls.moving){
            ls.moving = true;
            if(typeof Snd !== 'undefined' && Snd.railStart) Snd.railStart(ls);
          }
        }
        ls.target = ls.pos;
      }
    }
    if(!ls.runaway){
      const d = ls.target - ls.pos;
      const step = ls.speed*dt;
      if(Math.abs(d) < 0.004){
        if(ls.moving){ ls.moving=false; Snd.railStop(ls); }
        ls.pos = ls.target;
        /* the flyman locks off on arrival — but only the move he began:
           runaways never carry relock, so a fall can never lock itself */
        if(ls.relock){ ls.relock = false; ls.locked = true; }
      }
      else {
        if(!ls.moving){ ls.moving=true; Snd.railStart(ls); }
        ls.pos += clamp(d, -step, step);
      }
    }
    ls.group.position.y = ls.pos;
    // lift lines stretch to grid
    const len = Math.max(.05, D.gridY - ls.pos);
    ls.lines.children.forEach(l=>{ l.scale.y = len; l.position.y = len/2; });
    // arbor counterweights ride opposite
    const arb = ARBORS[ls.id-1];
    if(arb) arb.position.y = -(ls.pos-20.5)*0.62;
    // traveler (house curtain) open/close
    if(ls.goods && GOODS[ls.goodsKey].traveler){
      ls.open += clamp(ls.travTarget-ls.open, -dt*0.42, dt*0.42);
      ls.goods.children.forEach(c=>{
        if(c.userData.side!==undefined){
          c.position.x = c.userData.side*(D.procW/4-0.4) + c.userData.side*ls.open*(D.procW/1.75);
          c.scale.x = 1 - ls.open*0.32;
        }
      });
    }
  }
}

/* default hang — run from init() once the rig exists */
const DEFAULT_HANG = ['valance','house','border','legs','electric','border','legs','scrim',
                      'electric','sky','border','legs','electric','cyc'];
function initHang(){
  DEFAULT_HANG.forEach((k,i)=>{ hangGoods(FLY[i],k); });
  FLY[0].pos=FLY[0].target=TRIMS.valance;   // valance in
  FLY[1].pos=FLY[1].target=TRIMS.house;     // house curtain in, closed
  FLY[1].open=FLY[1].travTarget=0;
  [2,3,5,6,10,11].forEach(i=>{ FLY[i].pos=FLY[i].target=inTrimOf(FLY[i]); });
  FLY[13].pos=FLY[13].target=TRIMS.cyc; // cyc
  FLY[9].pos=FLY[9].target=OUT_TRIM;   // sky drop out
  FLY[7].pos=FLY[7].target=OUT_TRIM;   // scrim out
  FLY.forEach(ls=>{ ls.group.position.y = ls.pos; });
}
