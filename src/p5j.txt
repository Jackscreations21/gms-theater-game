
/* ============================================================================
   THE SHOW'S SOUND, AND THE AUDIENCE RIG'S PATTERNS      (rulings BA, BB, BD)

   Spec: docs/superpowers/specs/2026-08-11-beetlejuice-sound-and-house-design.md

   Three things live here, and they are together because the middle one joins
   the other two:

     1. RECORDED TRACKS (BA, amended by BI).  THREE of them now, read out of
        assets/audio/ by exact filename: the pre-show music, and his show
        recording cut in half at the act break.  This is the model-import
        pipeline over again (RULING AZ): a manifest, a doc whose File column is
        the contract, a test that pins the two together both ways, and a
        fallback so quiet that nothing downstream can tell.

        THE FILES ARE COMMITTED NOW, which BA said they never would be.  BA's
        first reason was that they COULD not be — one 134MB file against
        GitHub's 100MB hard limit — and splitting the show removed it: the
        halves are 69.4MB and 70.1MB.  Its other two reasons (off a video; a
        commercial recording on a repo with Pages on) were the owner's to
        overrule and he did, with all three in front of him.  See docs/AUDIO.md.

        A MISSING TRACK IS STILL A NORMAL STATE and every line below still
        assumes it.  That is not vestigial: every suite in this repo runs
        silent, jsdom never fetches media, and the no-audio path is still the
        only one a test can walk.

     2. THE TRANSPORT (BB).  Every timestamp the owner wrote is a position in
        the show track, so when that track is genuinely playing it IS the cue
        clock: a cue fires when its `at` has passed.  Absolute timecode, so it
        cannot drift, and a seek drags the lights to the music.  With no audio
        the `follow` chain drives exactly as it always has — untouched, and
        still the only path a suite exercises.

     3. THE PATTERN ENGINE (BD).  "moving slowly doing random patterns",
        "slowly fading between green and purple", "sweep up", "a random
        pattern fast", "flash white as bright as posible" — none of which the
        cue system can say, because a cue is a static look plus a fade time.
        So a cue may ARM a named effect and this steps it off the frame `dt`.
        Never setTimeout: the standing rule, and the reason a suite can watch
        ten seconds of a pattern in 600 stepped calls.

   Why this is not folded into `Snd` (p5): `Snd` is synthesised WebAudio with
   no assets, and `Snd.play(name)` TOGGLES looping names — a cue needs
   set-to-a-state, never toggle.  A track also needs `currentTime`, which is
   the whole of (2).  It does route through Snd's `music` bus level, so the
   existing fader still means something.

   RULING B is untouched: the flown PA boxes are still rigging with no audio
   wired to them.  This comes out of the browser, not out of the modelled
   speakers.
   ========================================================================== */

/* --------------------------------------------------------------------------
   THE MANIFEST — docs/AUDIO.md's File column IS this table.  A test pins them
   to each other in BOTH directions, because the fallback is silent by design:
   a misnamed file would otherwise just never play and never say why.
   -------------------------------------------------------------------------- */
const AUDIO_DIR = 'assets/audio/';

/* OFFSET is what makes the show track splittable (RULING BI).  His recording is
   one 2h23m file and every timestamp he wrote is a position in it — that is
   RULING BB and it is not up for revision.  So the file may be cut into pieces
   as long as each piece SAYS WHERE IT BEGINS, and the plot never learns:

     the clock  = offset + el.currentTime      (so a cue at 5000 fires 708s
                                                into act2, not 5000s in)
     a seek to `at` = at - offset              (so {play:'act2', at:4292} asks
                                                for exactly the number it
                                                always asked for, and gets 0.0)

   CLOCK is which tracks are allowed to drive the cue stack.  It replaced a
   hardcoded reach for AUD.tracks.show, and it is the whole reason two files
   cost no new machinery: whichever clock track is really playing IS the clock.
   The pre-show music can never be one, which is what makes it underscore. */
const BJ_AUDIO = {
  /* "Play pre show music i have provided" — and again through the interval,
     because act two says "Play audio number one again till go is pressed". */
  preshow: {file:'preshow.mp3', loop:true,  vol:0.85, offset:0, clock:false,
            label:'the pre-show music (his audio one)'},
  /* the show, in two halves — "split the bugger audio in half one for each
     half".  Cut at 4292.000 exactly, which is FREE: the act-break cue stops
     the track at 4269 and act two resumes it at 4292, so nothing between those
     two numbers is ever heard.  Not one `at` in the plot changed. */
  act1:    {file:'act1.m4a',    loop:false, vol:1.0,  offset:0,    clock:true,
            label:'act one (his audio two, up to the act break)'},
  act2:    {file:'act2.m4a',    loop:false, vol:1.0,  offset:4292, clock:true,
            label:'act two (his audio two, from 1:11:32)'}
};

var AUD = {
  tracks: {},          // key -> the runtime record
  armed:  false,       // has a user gesture happened yet?
  pending:null,        // a play the browser refused before that gesture
  clock:  false,       // is the show track driving the cue stack right now?
  fx:     {},          // target name -> the running effect (RULING BE)
  notes:  {}           // one console line per subject, said once
};

/* say a thing once.  A missing track is normal, so it must not scroll. */
function audNote(key, msg){
  if(AUD.notes[key]) return;
  AUD.notes[key] = true;
  if(typeof console !== 'undefined' && console.log) console.log('AUDIO: ' + msg);
}

/* An entry may be a bare filename (assets/audio/) or an absolute URL.  The URL
   form is the only way the owner gets sound on the headset over Pages without
   the recordings entering the repo, and it costs one line. */
function audioSrc(file){
  return /^(https?:)?\/\//.test(file) ? file : AUDIO_DIR + file;
}

function audTrack(key){
  if(AUD.tracks[key]) return AUD.tracks[key];
  const spec = BJ_AUDIO[key];
  if(!spec) return null;
  const tr = {key, file:spec.file, loop:!!spec.loop, vol:spec.vol === undefined ? 1 : spec.vol,
              offset:spec.offset || 0, clock:!!spec.clock,
              el:null, blocked:false, want:false, seek:null, fade:null};
  /* jsdom has an <audio> element but never fetches media, so `el` exists and
     stays at readyState 0 for the whole of every suite — which is exactly the
     no-audio path, and why the suites never hear anything and never break. */
  try{
    if(typeof document !== 'undefined' && document.createElement){
      const el = document.createElement('audio');
      el.preload = 'auto';
      el.loop = tr.loop;
      el.src = audioSrc(spec.file);
      el.addEventListener('error', ()=>{
        audNote('miss'+key, spec.label + ' is not in ' + AUDIO_DIR +
                ' (wanted "' + spec.file + '") — the show runs silent, which is fine');
      });
      tr.el = el;
    }
  }catch(e){ tr.el = null; }
  AUD.tracks[key] = tr;
  return tr;
}

/* the master this track answers to: Snd's `music` bus, so one fader moves both
   the synthesised world and the recordings */
function audBusVol(){
  let m = 1;
  try{ if(typeof Snd !== 'undefined' && Snd.level) m = Snd.level('music') * Snd.level('master'); }
  catch(e){ m = 1; }
  return (isFinite(m) ? m : 1);
}

/* IS THIS TRACK REALLY RUNNING?  Not "did we ask it to" — really running, with
   data.  Everything downstream keys off this, which is what makes the silent
   fallback total: with no file, readyState never leaves 0, `live` is never
   true, the transport never takes the clock and the follow chain drives. */
function audLive(tr){
  return !!(tr && tr.el && !tr.el.paused && tr.el.readyState >= 2);
}

/* A CUE ONLY EVER STATES AN INTENT.  It does not call play(), and that is
   deliberate rather than tidy: play() on a track the browser has not found yet
   is meaningless, and in jsdom — which never fetches media at all — it raises
   "Not implemented: HTMLMediaElement.prototype.play" straight onto the console
   of every suite that fires a cue.  So the wanting is recorded here and
   audPump() does the playing, once the media is actually there.

   It also gets slow loading right for free: a track wanted before it has
   buffered simply starts the moment it can. */
function audPlay(key, at){
  const tr = audTrack(key); if(!tr) return false;
  tr.want = true;
  /* `at` is a position in HIS WHOLE RECORDING, always — the file it happens to
     live in is this function's problem and nobody else's (RULING BI) */
  tr.seek = (at === undefined ? null : Math.max(0, at - tr.offset));
  return true;
}

function audStop(key){
  const tr = AUD.tracks[key]; if(!tr) return;
  tr.want = false; tr.seek = null;
  if(!tr.el) return;
  try{ if(!tr.el.paused) tr.el.pause(); }catch(e){}
}

/* the frame's worth of actually talking to the browser.  readyState >= 1 is
   HAVE_METADATA: the file exists and has been opened.  With no file it never
   gets there, so nothing is ever asked to play and the silence is total. */
function audPump(dt){
  for(const key in AUD.tracks){
    const tr = AUD.tracks[key];
    if(!tr.el) continue;
    /* "Audio fades out" — his last line of act two.  A ramp on the element's
       own volume, off the frame dt like everything else, and it pauses itself
       at the bottom rather than cutting. */
    if(tr.fade){
      tr.fade.t += (dt || 0);
      const u = Math.min(1, tr.fade.t / Math.max(0.05, tr.fade.secs));
      try{ tr.el.volume = Math.max(0, Math.min(1, tr.vol * audBusVol() * (1 - u))); }catch(e){}
      if(u >= 1){ tr.fade = null; audStop(key); }
      continue;
    }
    if(!tr.want){ continue; }
    let rs = 0;
    try{ rs = tr.el.readyState || 0; }catch(e){ rs = 0; }
    if(rs < 1) continue;                       // not found yet — try again next frame
    if(tr.seek !== null && tr.seek !== undefined){
      try{ tr.el.currentTime = tr.seek; }catch(e){}
      tr.seek = null;
    }
    try{ tr.el.volume = Math.max(0, Math.min(1, tr.vol * audBusVol())); }catch(e){}
    if(!tr.el.paused || tr.blocked) continue;
    let p = null;
    try{ p = tr.el.play(); }catch(e){ p = null; tr.blocked = true; }
    /* AUTOPLAY.  There is no gesture between loading the page and the pre-show
       cue, so the first play may well be refused.  Remember it and start it on
       the first gesture that reaches the game — the same bargain Snd.init()
       already makes with the browser. */
    if(p && p.catch) p.catch(()=>{
      tr.blocked = true;
      AUD.pending = {key, at:null};
      audNote('block', 'the browser refused to start ' + key +
              ' before a click — it will start on the first one');
    });
  }
}

/* the first gesture anywhere in the game.  p7 calls this beside Snd.init(). */
function audArm(){
  AUD.armed = true;
  AUD.pending = null;
  /* clear every refusal: the pump will try again next frame, and anything
     still `want`ed picks up where the cue asked for it */
  for(const key in AUD.tracks) AUD.tracks[key].blocked = false;
}

/* WHAT A CUE SAYS ABOUT SOUND.  `audio:{play:'show', at:35, stop:'preshow'}` */
function showCueAudio(a){
  if(!a) return;
  if(a.stop) (Array.isArray(a.stop) ? a.stop : [a.stop]).forEach(audStop);
  if(a.stopAll) Object.keys(AUD.tracks).forEach(audStop);
  if(a.play) audPlay(a.play, a.at);
  /* and a fade-out, which is a stop that takes its time */
  if(a.fade){
    const tr = audTrack(a.fade);
    if(tr) tr.fade = {secs:(a.secs === undefined ? 6 : a.secs), t:0};
  }
}

/* --------------------------------------------------------------------------
   THE TRANSPORT (RULING BB)

   Called every frame.  If the show track is really running it owns the cue
   stack: fire everything whose `at` has gone by, and cancel any follow the
   firing armed — two clocks must never both drive.

   A cue carrying `hold` stops the sweep dead and waits for GO, which is what
   makes the interval work with the music still the clock (RULING AU).
   -------------------------------------------------------------------------- */
const AUD_CATCHUP = 40;        // cues per frame, a runaway backstop only

/* WHICH FILE IS THE CLOCK RIGHT NOW (RULING BI).  Not "which did we ask for" —
   which is really running, with data.  At most one ever is: act one is stopped
   by the act-break cue before act two is asked for.  If somehow two were, the
   manifest order decides, and the console would say so on the next seek. */
function audClockTrack(){
  for(const key in BJ_AUDIO){
    if(!BJ_AUDIO[key].clock) continue;
    const tr = AUD.tracks[key];
    if(audLive(tr)) return tr;
  }
  return null;
}

function showAudioTick(dt){
  audFxStep(dt);
  audPump(dt);
  const tr = audClockTrack();
  if(!tr){ AUD.clock = false; return; }
  try{ tr.el.volume = Math.max(0, Math.min(1, tr.vol * audBusVol())); }catch(e){}
  AUD.clock = true;
  if(typeof CUES === 'undefined' || !CUES.length) return;
  /* the position in the WHOLE recording, which is what every `at` means */
  const t = tr.offset + tr.el.currentTime;
  let n = 0;
  while(nextCue < CUES.length && n++ < AUD_CATCHUP){
    const c = CUES[nextCue];
    if(c.at === undefined || c.at === null) break;   // an unplotted stack: not ours
    if(c.hold) break;                                // waits for GO
    if(c.at > t) break;
    fireCue(nextCue);
    if(typeof cancelFollow === 'function') cancelFollow();
  }
}

/* the stage swap, and anything else that ends a performance.  TRAPS lists
   audio among the things that have driven the WRONG stage's rig after a walk,
   so the sound stops with the show the same way cancelFollow kills the chain. */
function showSoundStop(){
  Object.keys(AUD.tracks).forEach(audStop);
  AUD.pending = null;
  AUD.clock = false;
  audFxClear();
}

/* ==========================================================================
   THE PATTERN ENGINE (RULING BD)

   A cue arms an effect per target; this steps it off dt.  Targets are the two
   audience groups and nothing else — the stage is cued by hand, as it should
   be.  An effect writes ONLY the properties it must, so everything it leaves
   alone stays under the cue's own control:

     wander   pan, tilt, colour     the pre-show: slow drift, slow green<->purple
     sweep    level, tilt           "purple and sweep up"
     random   pan, tilt, level      "a random pattern fast"
     flash    level, colour         "flash white as bright as posible"

   Levels written here overwrite the fade for those channels while the effect
   runs — the firelight precedent (SHOW.flicker does the same).  It heals
   itself on the way out: setLevel captures lvlFrom from the LIVE level, so
   the cue that clears an effect fades from wherever the effect left it.

   Motion is a sum of sines on an accumulating phase, offset per fixture, so
   there is no per-fixture state to park, nothing to reseed, and a suite can
   step ten seconds and watch pan actually change.
   ========================================================================== */
const AUD_WANDER_RATE = 1.0;   // pre-show drift.  Owner: "slowly" — headset tune
const AUD_RANDOM_RATE = 1.0;   // the 1:16 chaos.  Owner: "fast"  — headset tune
/* WHERE A SWEEP STARTS AND ENDS (RULING BK).  A head sits at tilt + 90 degrees
   below horizontal, so -15 is 75 deg down — the stalls floor — and -98 is 8 deg
   UP, out over the balcony heads.  Both ends are headset tunes; the sign of the
   travel is not, because "sweep up" is his word for it. */
const AUD_SWEEP_LO = -15, AUD_SWEEP_HI = -98;
const AUD_STROBE_HZ   = 9.0;   // "really fast".  9Hz reads fast and is not a
                               // seizure risk the way 15-20Hz would be — the
                               // one constant here with a reason not to raise
const _afC = new T.Color(), _afD = new T.Color();

/* THE TARGETS (RULING BE).  This started out as the audience rig and nothing
   else, on the reasoning that the stage is cued by hand.  Act two asked for
   more and it was right to: "just a purple light is on the curtain an moving
   around a bunch" is the STAGE movers, and "lights flash green really fast" is
   the stage.  So an effect names a group, and only these groups exist.

   An unrecognised name moves NOTHING and says so on one console line.  That is
   deliberate: aiming at the wrong lamps is the failure this engine invites —
   the 9:01 flash in act one was very nearly armed at 'all', which would have
   flashed eight blinders and left the stage dark — and a silent typo that
   moves the wrong fourteen channels is worse than one that moves none. */
const AUD_FX_TARGETS = {
  blind:  ()=>GROUPS.blind,     // the 8 round the arch
  aud:    ()=>GROUPS.aud,       // the 6 over the seating
  house:  ()=>GROUPS.house,     // both audience groups together
  movers: ()=>GROUPS.movers,    // the 2 STAGE moving heads
  stage:  ()=>GROUPS.stage,     // the stage wash and backlight
  front:  ()=>GROUPS.front      // front of house and booms
};
function audFxChans(on){
  const g = AUD_FX_TARGETS[on];
  if(!g){
    audNote('fxtgt:' + on, 'no pattern target called "' + on + '" — nothing will move');
    return [];
  }
  return g();
}

/* what a cue's `fx` field means.  One record per target; a cue that says
   nothing CLEARS both, because an effect that outlived the cue that wanted it
   is exactly the pop RULING AY forbids. */
function showCueFx(fx){
  audFxClear();
  if(!fx) return;
  /* One effect per TARGET, keyed by the target's own name — so a cue can run a
     blinder strobe and a mover drift at once, which is exactly what act two
     opens with: "audiance blinders flash a bunch for a second then just a
     purple light is on the curtain an moving around a bunch". */
  (Array.isArray(fx) ? fx : [fx]).forEach(e=>{
    if(!e || !e.kind) return;
    const on = e.on || 'aud';
    AUD.fx[on] = {
      kind: e.kind,
      on:   on,
      rate: e.rate === undefined ? 1 : e.rate,
      secs: e.secs === undefined ? 2 : e.secs,
      cols: (e.cols || (e.col ? [e.col] : null)),
      t:    0
    };
    /* A SWEEP STARTS AT ITS START (RULING BK).  Every other effect is a
       continuous function of t and may pick the rig up wherever it finds it,
       but a sweep is a journey with two ends and it is the only one whose
       FIRST frame is part of the point.  Tilt slews at dt*3.2 — a ~0.3s time
       constant — so without this the head spends the first third of a
       two-second sweep still travelling to the bottom, and the stalls never
       get the beginning of it.  Measured: the stalls went from purple 62% of
       the cue to never, purely from the slew.  So snap to the low end once,
       here, and let the slew smooth everything after that. */
    if(e.kind === 'sweep') audFxChans(on).forEach(c=>{
      const f = chan(c);
      if(f && f.mover) f.tilt = f.tiltT = AUD_SWEEP_LO;
    });
  });
}

function audFxClear(){ for(const k in AUD.fx) delete AUD.fx[k]; }

function audFxStep(dt){
  for(const slot in AUD.fx){
    const e = AUD.fx[slot];
    if(!e) continue;
    e.t += dt;
    const chans = audFxChans(e.on);
    for(let i = 0; i < chans.length; i++){
      const f = chan(chans[i]); if(!f) continue;
      audFxApply(e, f, i);
    }
  }
}

function audFxApply(e, f, i){
  const t = e.t;
  if(e.kind === 'wander'){
    const r = AUD_WANDER_RATE * e.rate;
    if(f.mover){
      /* PAN STAYS IN THE ROOM, AND THAT IS THE BALCONY FIX (RULING BK).  At
         ±62° a third of the drift was aimed through the brick — the probe
         caught a beam tip at x = 22.0 with the side wall at 15, and another at
         z = 37.8 with the back wall at 30.  Keeping it inside the auditorium
         is what puts it round the house, balcony included: measured, the back
         of the balcony went from 7% of a cycle to 56%.                      */
      f.panT  =  34 * Math.sin(t*0.31*r + i*1.7) * Math.cos(t*0.17*r + i*0.9);
      /* AND THE TILT RANGE IS WHAT KEEPS THE STALLS.  -62 ± 24 is 4°..52°
         below horizontal (a head sits at tilt + 90) — a narrow band that,
         once the pan was reined in, left the drift working the upper house
         and skipping the lower: the stalls fell to 20% of a cycle.  -55 ± 42
         runs 7° above horizontal down to 77° below and puts them back at 80%.

         The two do DIFFERENT jobs and it took a negative check to find that
         out — the first version of this comment credited the tilt with the
         balcony, and the tilt mutation sailed through the balcony test.    */
      f.tiltT = -55 + 42 * Math.sin(t*0.23*r + i*2.3);
    }
    if(e.cols && e.cols.length > 1){
      const m = 0.5 + 0.5*Math.sin(t*0.42*r + i*0.4);
      _afC.set(e.cols[0]); _afD.set(e.cols[1]);
      f.color.copy(_afC).lerp(_afD, m);
      f.colDur = 0;                       // the effect owns the colour now
    }
    return;
  }
  if(e.kind === 'sweep'){
    /* "purple and sweep up": levels and tilt run up together over the cue's
       own window, then hold at the top until the next cue takes them out.

       AND IT REALLY GOES UP NOW (RULING BK).  It used to read -84 -> -10, and
       because the head sits at tilt + 90 that is 6° below horizontal down to
       80° below: the number went up and the light went DOWN, ending on the
       stalls floor.  It now starts steeply down on the stalls and rises — the
       rake, the mezzanine front, the balcony front, and out over the balcony
       heads, which is the first time anything in this rig has been aimed at
       them.  -98 is 8° ABOVE horizontal, and a balcony head at z=26 sits 6°
       above the upstage bar's lens; real moving heads tilt well past level. */
    const u = Math.min(1, t / Math.max(0.05, e.secs));
    f.level = u;
    if(f.mover) f.tiltT = AUD_SWEEP_LO + (AUD_SWEEP_HI - AUD_SWEEP_LO)*u;
    if(e.cols && e.cols.length){ f.color.set(e.cols[0]); f.colDur = 0; }
    f.lvlDur = 0;
    return;
  }
  if(e.kind === 'random'){
    /* fast, and every unit on its own beat.  Stepping the phase rather than
       rolling dice per frame keeps it frame-rate independent and steppable. */
    const r = AUD_RANDOM_RATE * e.rate;
    const s = Math.sin(t*7.3*r + i*2.9), c = Math.cos(t*5.1*r + i*1.3);
    if(f.mover){ f.panT = 150*s; f.tiltT = -70 + 55*c; }
    f.level = 0.35 + 0.65 * (0.5 + 0.5*Math.sin(t*11.0*r + i*4.1));
    f.lvlDur = 0;
    if(e.cols && e.cols.length){ f.color.set(e.cols[0]); f.colDur = 0; }
    return;
  }
  if(e.kind === 'flash'){
    /* full for a beat, then out and stay out.  "as bright as posible". */
    f.level = (t <= e.secs) ? 1 : 0;
    f.lvlDur = 0;
    if(e.cols && e.cols.length){ f.color.set(e.cols[0]); f.colDur = 0; }
    return;
  }
  if(e.kind === 'strobe'){
    /* hard on, hard off, at a rate: "flash a bunch for a second", "lights
       flash green really fast".  A strobe with secs runs out and stays out; the
       square wave comes off the phase, so it is frame-rate independent. */
    const done = e.secs > 0 && t > e.secs;
    f.level = done ? 0
      : (Math.sin(t * AUD_STROBE_HZ * e.rate * Math.PI * 2) > 0 ? 1 : 0);
    f.lvlDur = 0;
    if(e.cols && e.cols.length){ f.color.set(e.cols[0]); f.colDur = 0; }
    return;
  }
}
