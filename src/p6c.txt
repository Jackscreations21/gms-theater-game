
/* ============================================================================
   p6c — THE CARPENTERS: the catalogue, the planner, and the run (PRs 2-4
   of the carpenters round, spec docs/superpowers/specs/2026-08-08-
   carpenters-design.md, RULINGS Y, AA, AB and AC).  The catalogue and
   planner are pure data and pure functions; the run entry at the bottom
   (carpStart/carpRun + CARP_RUN) is the one stateful thing here — the
   crew engine that executes the jobs (fetch/cut, and PR 4's haul/nail
   at the mark) lives in p6b.  PR 5 adds the warehouse screen.

   The catalogue (CARP_CAT) is a data table: adding a buildable is adding
   a row, not code.  Each row names the loose stock it consumes (real
   profiles from p4c's WOOD_PROF, full 8ft sticks), an executable cut
   schedule (chop bench for lumber, track table for sheets — the same
   stations sawCut runs on, so every length is on the saw's inch grid and
   every off side clears SAW_MIN or is honest scrap), and a blueprint:
   piece poses LOCAL TO THE MARK ORIGIN (deck y = 0, assembled lying flat
   per RULING AC) and a nail list with TWO nails per joint (RULING G: one
   nail is a pivot, two is rigid — carpenters build rigid).

   OWNERSHIP: `CARP` (the crayon's mark) is DECLARED AND OWNED by p9,
   which executes after this part — a second `var CARP = {...}` here
   would clobber it (the concatenated-file trap, TRAPS.md).  Nothing in
   this part declares or reads CARP: carpPlan takes the mark as an
   argument.
   ========================================================================== */

/* ---- the stock survey -----------------------------------------------------
   What a venue really has on hand: consumable loose wood by profile —
   floor pieces ('loose') and pallet stock ('slotted'), never anything
   seated on a saw, nailed into an assembly, in a hand or hung.  Each
   list comes back nearest-its-saw first (sheets to the track table,
   lumber to the chop bench — RULING Y: nearest first), so the planner
   and later the lead take the close piece, not the far one.  Own temps
   (the aliasing trap, TRAPS.md).                                          */
const _carpV = new T.Vector3(), _carpV2 = new T.Vector3();
function carpSurvey(venue){
  const out = {};
  for(const k in WOOD_PROF) out[k] = [];
  for(const b of BODIES){
    if(b.kind !== 'wood' || b.venue !== venue) continue;
    if(b.state !== 'loose' && b.state !== 'slotted') continue;
    if(out[b.prof]) out[b.prof].push(b);
  }
  const saws = (typeof SAWS !== 'undefined' && SAWS) ? SAWS[venue] : null;
  if(saws){
    for(const k in out){
      const st = (k === 'sheet') ? saws.track : saws.chop;
      if(!st || !st.group || out[k].length < 2) continue;
      st.group.getWorldPosition(_carpV);
      const dec = out[k].map(b=>(
        {b, d:b.mesh.getWorldPosition(_carpV2).distanceToSquared(_carpV)}));
      dec.sort((p, q)=>p.d - q.d);
      out[k] = dec.map(p=>p.b);
    }
  }
  return out;
}
/* the catalogue consumes whole sticks and whole sheets: an off-cut is
   stock for the PLAYER, but a cut schedule is written from 8ft */
function carpFullStick(b){
  if(!b.dims || b.dims.L < WOOD_LEN - 0.001) return false;
  if(b.prof === 'sheet' && b.dims.W < WOOD_PROF.sheet.W - 0.001) return false;
  return true;
}

/* ---- the glass wording (PR 5, RULING X) -----------------------------------
   The NEED list as the screen speaks it — 'NEED 1× SHEET · 3× 2x4' — the
   catalogue shorthand, never ftIn: it is a shopping list, and the order
   screen an arm's length away sells by the stick.  Lives here so the
   words sit with the data they describe; p9 only paints them. */
function carpProfShort(prof){
  return prof === 'sheet' ? 'SHEET' : prof.slice(1);
}
function carpStockLine(list){
  return list.map(s=>s.n + '× ' + carpProfShort(s.prof)).join(' · ');
}
function carpNeedLine(need){ return 'NEED ' + carpStockLine(need); }

/* ---- the planner ----------------------------------------------------------
   carpPlan(key, mark) is PURE: it reads the venue's bodies and the
   catalogue row and returns one of
     null                      — no such row, or no mark
     {need:[{prof, n}...]}     — the venue is short: what to order
     {full:true, pieces:n}     — the job would end past BUILD_CAP pieces
                                 (cutting registers off-cuts as new bodies,
                                 and nothing downstream of the order screen
                                 checks the cap — this is the enforcement
                                 point, RULING Y)
     {jobs:[...], piecesAfter} — the ordered queue: carpFetch/carpCut for
                                 every stick that meets a saw, a hold, one
                                 carpHaul per blueprint piece, a hold, one
                                 carpNail per joint (both nails ride it).
   Nothing is mutated, nothing global is written: the jobs are plain
   records for PR 3/4 to execute, poses and nail points mark-local.
     carpFetch {kind, venue, body, prof, saw, axis}  lead: stock -> saw seat
                 axis is which way a SHEET meets the table: 'L' — its
                 length along table X, a crosscut, the default — or 'W', a
                 rip off its width (RULING AE: an opening is skinned in
                 pieces AROUND it, which needs strips).  Lumber ignores it.
                 body null means RE-SEAT what the run has on that bench,
                 turned the other way: a schedule that crosscuts and then
                 rips the same sheet emits one of these between the two.
     carpCut   {kind, venue, saw, len, piece}        lead: blade at len off
                 the LOW end of the piece this run has seated there (the
                 fetched stick, then the registered off side after each
                 cut); piece is the blueprint index the kept side becomes
     carpHaul  {kind, venue, piece, body, prof, len, pos, rot}
                 hand: body for an uncut stick, null when the piece comes
                 off a saw (PR 3 records which body each cut kept)
     carpNail  {kind, venue, i, j, nails:[{p, ax} x2]}  hand: two nails
     hold      {kind, say}                           the p6b barrier        */
/* RULING AF — the skin is a switch on the CALL, not a row of its own.
   SKIN OFF drops the skin pieces, the nails that fasten them, the cuts
   that make them and the stock they came out of.  Piece INDICES never
   move: the jobs, CARP_RUN.made and CARP_RUN.placed are all keyed by
   blueprint index, so a dropped piece leaves a HOLE in the numbering
   rather than shifting everything after it.  All pure:
     live(pi)    — is blueprint piece pi being built
     liveCut(ci) — does cut entry ci still make something wanted
     wanted[ui]  — is stock unit ui still wanted
     stock       — the stock list to show on the glass and check against */
function carpParts(row, skin){
  const on = !(row.skinnable && skin === false);
  const pieces = row.blueprint.pieces;
  const live = pi => on || !pieces[pi].skin;
  const liveCut = ci => on ||
    pieces.some((p, pi)=>live(pi) && p.src.cut && p.src.cut[0] === ci);
  const wanted = {};
  pieces.forEach((p, pi)=>{ if(live(pi) && p.src.stock !== undefined) wanted[p.src.stock] = true; });
  row.cuts.forEach((c, ci)=>{ if(liveCut(ci)) wanted[c.stock] = true; });
  const stock = [];
  let ui = 0;
  for(const s of row.stock){
    let n = 0;
    for(let i = 0; i < s.n; i++, ui++) if(wanted[ui]) n++;
    if(n) stock.push({prof:s.prof, n});
  }
  return {live, liveCut, wanted, stock};
}
function carpStock(row, skin){ return row ? carpParts(row, skin).stock : []; }

function carpPlan(key, mark, opts){
  const row = CARP_CAT[key];
  if(!row || !mark || !mark.venue) return null;
  const venue = mark.venue;
  const P = carpParts(row, !(opts && opts.skin === false));
  const live = P.live, liveCut = P.liveCut, wanted = P.wanted;

  /* the stock check — full sticks of the right profile, or the NEED list */
  const survey = carpSurvey(venue);
  const pool = {}, need = [];
  /* opts.taken is READ-ONLY here: the sticks an earlier item of a build
     list already spoke for.  carpPlan mutates nothing — it reports what
     it claimed in plan.uses and lets the caller keep the tally. */
  const spoken = (opts && opts.taken) || null;
  for(const s of P.stock){
    if(!pool[s.prof]) pool[s.prof] = (survey[s.prof] || [])
      .filter(b=>carpFullStick(b) && !(spoken && spoken.indexOf(b) >= 0));
    const short = s.n - pool[s.prof].length;
    if(short > 0) need.push({prof:s.prof, n:short});
  }
  if(need.length) return {need};

  /* the cap — count the bodies the saws will register: every cut whose
     off side clears SAW_MIN mints one; a scrap end mints nothing */
  let mint = 0;
  const left = {};
  for(let ci = 0; ci < row.cuts.length; ci++){
    const c = row.cuts[ci];
    if(!liveCut(ci)) continue;
    /* what is left to cut, keyed by stock AND axis: two entries may name
       the same stick — a second 'L' entry carries on down the remainder,
       while a 'W' entry starts fresh across the sheet's full width,
       because crosscutting never touched it */
    const ax = c.axis || 'L', k = c.stock + ':' + ax;
    if(left[k] === undefined)
      left[k] = (ax === 'W') ? WOOD_PROF.sheet.W : WOOD_LEN;
    for(const len of c.make){
      left[k] -= len;
      if(left[k] >= SAW_MIN) mint++;
    }
  }
  /* opts.minted is what earlier items of a build list will already have
     cut by the time this one starts (RULING AH) */
  const piecesAfter = venueBuildCount(venue) + ((opts && opts.minted) || 0) + mint;
  if(piecesAfter > BUILD_CAP) return {full:true, pieces:piecesAfter};

  /* assign real bodies to the row's stock units, nearest first — a unit
     nothing wants any more (SKIN OFF) is left standing in the shed */
  const units = [], drawn = {}, uses = [];
  for(const s of row.stock){
    drawn[s.prof] = drawn[s.prof] || 0;
    for(let i = 0; i < s.n; i++){
      const body = wanted[units.length] ? pool[s.prof][drawn[s.prof]++] : null;
      if(body) uses.push(body);
      units.push({prof:s.prof, body});
    }
  }

  /* the queue: every stick to its saw and through the schedule, a hold
     while the benches finish, every piece to the mark, a hold while the
     deck fills, then the nails (RULING G: two per joint, rigid) */
  const jobs = [];
  const onBench = {};                   // stock index -> the axis it lies on
  row.cuts.forEach((c, ci)=>{
    if(!liveCut(ci)) return;
    const ax = c.axis || 'L';
    /* the first entry for a stick fetches it; a later entry only needs the
       lead if it wants the stick round the other way */
    if(onBench[c.stock] === undefined)
      jobs.push({kind:'carpFetch', venue, body:units[c.stock].body,
                 prof:units[c.stock].prof, saw:c.saw, axis:ax});
    else if(onBench[c.stock] !== ax)
      jobs.push({kind:'carpFetch', venue, body:null,
                 prof:units[c.stock].prof, saw:c.saw, axis:ax});
    onBench[c.stock] = ax;
    c.make.forEach((len, k)=>{
      let piece = -1;
      row.blueprint.pieces.forEach((p, pi)=>{
        if(p.src.cut && p.src.cut[0] === ci && p.src.cut[1] === k) piece = pi;
      });
      jobs.push({kind:'carpCut', venue, saw:c.saw, len, piece});
    });
  });
  jobs.push({kind:'hold', say:'cuts are done — haul it to the mark'});
  row.blueprint.pieces.forEach((p, pi)=>{
    if(!live(pi)) return;
    jobs.push({kind:'carpHaul', venue, piece:pi,
               body:(p.src.stock !== undefined) ? units[p.src.stock].body : null,
               prof:p.prof, len:p.len, w:p.w, pos:p.pos.slice(), rot:p.rot.slice()});
  });
  jobs.push({kind:'hold', say:'pieces are set — nail it up'});
  const joints = {};
  for(const n of row.blueprint.nails){
    if(!live(n.i) || !live(n.j)) continue;
    const jk = n.i + ':' + n.j;
    if(!joints[jk]){
      joints[jk] = {kind:'carpNail', venue, i:n.i, j:n.j, nails:[]};
      jobs.push(joints[jk]);
    }
    joints[jk].nails.push({p:n.p.slice(), ax:n.ax.slice()});
  }
  return {jobs, piecesAfter, uses, mint};
}

/* ---- the build LIST (RULINGS AG + AH) -------------------------------------
   One CALL takes a list of {key, n} and works it IN ORDER — one queue,
   not a concurrent one: each item's cuts, hauls and nails run to the end
   before the next item starts.  Items STACK on the one mark (RULING Z
   still stands, one mark), each lifted by the height of everything
   already standing there, the way a crew stacks flats.

   Pure, like carpPlan, and judged WHOLE before a single cut:
     null                    — an empty list, or a key that does not exist
     {need:[...]}            — summed across every item, reported once
     {full:true, pieces:n}   — the list would end past BUILD_CAP
     {jobs, piecesAfter}     — the whole queue, ready to run

   Piece indices are offset per item by the row's full blueprint length,
   so CARP_RUN.made and CARP_RUN.placed — both keyed by piece index —
   never collide between two items of the same list.                    */
const _stackM = new T.Mesh(WOODG), _stackB = new T.Box3();
function carpStackH(row, skin){
  if(!row) return 0;
  const P = carpParts(row, skin);
  let top = 0;
  row.blueprint.pieces.forEach((p, pi)=>{
    if(!P.live(pi)) return;
    const PR = WOOD_PROF[p.prof];
    if(p.prof === 'sheet') _stackM.scale.set(p.len, p.w || PR.W, PR.t);
    else _stackM.scale.set(PR.a, p.len, PR.b);
    _stackM.rotation.set(p.rot[0], p.rot[1], p.rot[2]);
    _stackM.position.set(0, p.pos[1], 0);
    _stackM.updateMatrixWorld(true);
    _stackB.setFromObject(_stackM);
    top = Math.max(top, _stackB.max.y);
  });
  return top;
}
function carpPlanList(items, mark, opts){
  if(!items || !items.length || !mark || !mark.venue) return null;
  const list = [];
  for(const it of items){
    const row = CARP_CAT[it.key];
    if(!row) return null;
    for(let i = 0; i < (it.n || 0); i++) list.push({key:it.key, row});
  }
  if(!list.length) return null;
  const skin = !(opts && opts.skin === false);

  /* the stock check for the WHOLE list, against one survey: summing the
     rows and asking once is exact, where planning item by item would
     count the same shortfall twice */
  const survey = carpSurvey(mark.venue), want = {}, need = [];
  for(const e of list)
    for(const s of carpStock(e.row, skin)) want[s.prof] = (want[s.prof] || 0) + s.n;
  for(const prof in want){
    const have = (survey[prof] || []).filter(carpFullStick).length;
    if(want[prof] > have) need.push({prof, n:want[prof] - have});
  }
  if(need.length) return {need};

  const jobs = [], taken = [];
  let minted = 0, lift = 0, base = 0, piecesAfter = 0;
  for(const e of list){
    const p = carpPlan(e.key, mark, {skin, taken, minted});
    if(!p) return null;
    if(p.need) return {need:p.need};      // the survey moved under us
    if(p.full) return p;                  // RULING AH: refused whole, nothing cut
    /* stack it on what is already there, and keep the piece indices apart */
    p.jobs.forEach(j=>{
      if(j.kind === 'carpHaul'){ j.pos[1] += lift; j.piece += base; }
      else if(j.kind === 'carpCut'){ if(j.piece >= 0) j.piece += base; }
      else if(j.kind === 'carpNail'){
        j.i += base; j.j += base;
        j.nails.forEach(n=>{ n.p[1] += lift; });
      }
    });
    jobs.push.apply(jobs, p.jobs);
    p.uses.forEach(b=>taken.push(b));
    minted += p.mint;
    piecesAfter = p.piecesAfter;
    lift += carpStackH(e.row, skin);
    base += e.row.blueprint.pieces.length;
  }
  return {jobs, piecesAfter};
}
function carpListLine(items){
  return items.filter(it=>it.n > 0)
    .map(it=>(it.n > 1 ? it.n + '× ' : '') + CARP_CAT[it.key].label).join(' · ');
}

/* ---- the run (PR 3, RULING AA) --------------------------------------------
   CARP_RUN is the ONE live carpenter run — the state the executing jobs
   write and PR 4 reads.  Shape, exactly:
     CARP_RUN = {
       key,        // catalogue key, or null when carpRun was fed raw jobs
       venue,      // the venue the run works
       mark,       // the mark the plan was made against, or null
       made:  {},  // blueprint piece index -> the KEPT body of the carpCut
                   //   that named it (PR 4: a carpHaul with body:null takes
                   //   made[job.piece] off the floor by the bench)
       bench: {},  // saw kind 'track'/'chop' -> the body this run currently
                   //   has seated there: the fetched stick, then the
                   //   registered off side after each cut, null when done
       placed: {}, // blueprint piece index -> the body a carpHaul has set
                   //   down at its pose (PR 4: carpNail joins placed[i] to
                   //   placed[j])
       aside: 0    // how many pieces carpAside has rowed out by the benches
     }
   Cleared by crewStop (a stood-down build never resumes — RULING AC: the
   wood and the cuts stay, honest workshop state; call them again) and
   replaced whole by the next carpStart.  A var, not const: p6b assigns it
   and executes earlier in the concatenated file (TRAPS.md).             */
var CARP_RUN = null;

/* push a job queue at the crew engine and wake the lead.  One queue with
   the show crew — if anything is running this refuses (CREW BUSY both
   ways, RULING AA; PR 5's screen words the refusal on the glass). */
function carpRun(jobs){
  if(!jobs || !jobs.length || CREW.running) return 0;
  if(!CARP_RUN){
    const j = jobs.find(x=>x.venue);
    CARP_RUN = {key:null, venue:j ? j.venue : VENUE, mark:null, made:{}, bench:{}, placed:{}, aside:0};
  }
  const lead = carpLead();
  crewAttach();
  CREW.onDone = null;
  CREW.running = 'carp';
  CREW.t = 0;
  CREW.log = [];
  CREW.jobs = jobs.slice();
  const off = crewOff();
  lead.state = 'idle'; lead.job = null; lead.wait = 0; lead.carryB = null;
  lead.x = off.x; lead.z = off.z; lead.yaw = Math.PI/2;
  lead.group.visible = true;
  /* the whole crew answers a carpenter call: the six hands wake HERE, at
     call time, not lazily at the first haul — an 'off' hand still counts
     as free in updateCrew's assign loop, and a haul handed to an invisible
     figure at a stale position is work done by a ghost.  crewStart's own
     spread at the dock in crewStart's own fields, so the show path is
     untouched, and the finish walk in updateCrew (which skips only 'off'
     people) sends everyone woken here home when the queue drains. */
  crewSpawn(6).forEach((h, i)=>{
    h.state = 'idle'; h.job = null; h.wait = 0;
    crewPutDown(h);
    h.x = off.x + rnd(-1.5, 1.5); h.z = off.z + rnd(-2.2, 2.2) + i*0.4;
    h.yaw = Math.PI/2;
    h.group.visible = true;
  });
  crewLog('the carpenter is called — ' + CREW.jobs.length + ' jobs');
  if(typeof refreshCrewUI === 'function') refreshCrewUI();
  return CREW.jobs.length;
}
/* the headless entry the screen wires up in PR 5: plan, then run.  Returns
   carpPlan's own answer ({need}/{full}/null) unchanged on refusal, or
   {busy:true} when the one queue is taken, so the caller can word it. */
function carpStart(key, mark, opts){
  if(CREW.running) return {busy:true};
  /* a catalogue key, or a build LIST of {key, n} (RULING AG) */
  const list = Array.isArray(key);
  const plan = list ? carpPlanList(key, mark, opts) : carpPlan(key, mark, opts);
  if(!plan || plan.need || plan.full) return plan;
  CARP_RUN = {key:list ? null : key, venue:mark.venue, mark,
              made:{}, bench:{}, placed:{}, aside:0};
  carpRun(plan.jobs);
  return plan;
}
