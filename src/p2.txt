/* ============================================================================
   RULING DL — THE ATMOSPHERE IS HEIGHT-BASED, WITH HAZE AND GLARE.

   A flat FogExp2 fogs the grid as heavily as the deck.  Roblox's Atmosphere is
   height-banded with a haze lift and a glare term, and a theatre wants exactly
   that: haze pools low over the deck and the fly tower stays clear.

   THE FogExp2 STAYS.  It is what makes three.js define USE_FOG and emit the fog
   code path at all; delete it and none of the chunks below are ever compiled.

   TWO THINGS THE PLAN GOT WRONG HERE, both checked against the r128 artifact
   rather than the changelog, and both would have produced a shader that fails
   to compile while every suite stayed green (jsdom stubs WebGLRenderer, so no
   assertion in this repo can see a shader body — TRAPS):

     1. r128's fog varying is `fogDepth`, NOT `vFogDepth`.  The renamed one
        arrived in a later release.
     2. Extending T.UniformsLib.fog reaches NOTHING.  ShaderLib merged and
        CLONED its uniforms at module load — ShaderLib.standard.uniforms.fogColor
        is not even the same object as UniformsLib.fog.fogColor — and
        WebGLPrograms.getUniforms clones again per material.  The only seam that
        can hand a material a SHARED uniform object is onBeforeCompile, which
        the renderer calls with `parameters` immediately before acquiring the
        program (three.js r128, getProgram).

   So the three atmosphere uniforms travel as ONE shared object each, injected
   per material by atmTrack, which rides RULING DK's registry walk — one
   traversal serving both rulings rather than a second one.  Because they are
   shared, the per-frame drive is THREE writes total, not one per material.

   atmMix EXISTS SO A MISS IS HARMLESS.  A fogged material that never reached
   atmTrack gets uniforms defaulting to 0, and at atmMix 0 the two lines below
   collapse to exactly the fog three.js already drew.  The failure mode of an
   incomplete registry is therefore "unchanged", not "quietly less fog".
   ========================================================================== */
const ATM = { density:0.0055, height:9.0, haze:1.8, glare:0.4, mix:0.75 };
scene.fog = new T.FogExp2(0x07080e, ATM.density);
ATM.u = { atmHeight:{value:ATM.height}, atmHaze:{value:ATM.haze},
          atmGlare:{value:ATM.glare},   atmMix:{value:ATM.mix} };

T.ShaderChunk.fog_pars_vertex += '\n#ifdef USE_FOG\n varying float vAtmY;\n#endif\n';
/* `position` and not `transformed`: the SPRITE shader includes fog_vertex but
   includes neither begin_vertex nor project_vertex, so `transformed` does not
   exist there and naming it would fail to compile the moment anything used a
   sprite.  `position` is declared for every non-raw shader.  Instancing is
   applied the same way project_vertex applies it. */
T.ShaderChunk.fog_vertex +=
  '\n#ifdef USE_FOG\n' +
  ' vec4 atmW = vec4( position, 1.0 );\n' +
  ' #ifdef USE_INSTANCING\n' +
  '  atmW = instanceMatrix * atmW;\n' +
  ' #endif\n' +
  ' vAtmY = ( modelMatrix * atmW ).y;\n' +
  '#endif\n';
T.ShaderChunk.fog_pars_fragment +=
  '\n#ifdef USE_FOG\n varying float vAtmY;\n uniform float atmHeight;\n' +
  ' uniform float atmHaze;\n uniform float atmGlare;\n uniform float atmMix;\n#endif\n';
T.ShaderChunk.fog_fragment =
  '#ifdef USE_FOG\n' +
  ' #ifdef FOG_EXP2\n' +
  '  float fogFactor = 1.0 - exp( - fogDensity * fogDensity * fogDepth * fogDepth );\n' +
  ' #else\n' +
  '  float fogFactor = smoothstep( fogNear, fogFar, fogDepth );\n' +
  ' #endif\n' +
  ' float atmBand = 1.0 - clamp( vAtmY / max( atmHeight, 0.001 ), 0.0, 1.0 );\n' +
  ' fogFactor *= mix( 1.0, atmBand * atmHaze, atmMix );\n' +
  ' fogFactor = clamp( fogFactor, 0.0, 1.0 );\n' +
  ' vec3 atmCol = fogColor * ( 1.0 + atmGlare * fogFactor );\n' +
  ' gl_FragColor.rgb = mix( gl_FragColor.rgb, atmCol, fogFactor );\n' +
  '#endif\n';

/* ============================================================================
   RULING DM — THE COLOUR GRADE RIDES THE TONEMAPPING STAGE.

   Roblox's ColorCorrectionEffect is a post-tonemap curve — brightness,
   contrast, saturation, tint — and ACES already sits at exactly that point in
   the pipeline.  So this is a chunk patch and not a composer pass: it costs
   nothing, it works in BOTH eyes, and it keeps driver fixed-foveated rendering,
   which a hand-rolled stereo composer would forfeit.  Foveation is the only
   governor vrPerf has mid-session (p9) — and the first headset readings show it
   already pegged at 1.00 with no travel left, so it cannot be spent on bloom.

   Same seam as RULING DL, for the same reason: there is no UniformsLib for
   tonemapping at all, and even where there is one it does not work (ShaderLib
   clones at module load).  ONE shared uniform object each, injected through
   onBeforeCompile, riding DK's registry walk.

   gradeMix EXISTS BECAUSE A MISS HERE WOULD BE CATASTROPHIC, NOT MERELY WRONG.
   tonemapping_pars_fragment is included unguarded, so a material that declared
   these uniforms and never had them supplied would read gradeMat as mat3(0)
   and RENDER BLACK.  At gradeMix 0 the grade is bypassed entirely and the pixel
   is exactly what ACES produced, so an unhooked material looks like it did
   before this ruling instead of like a hole in the picture.

   RULING DU — THE GRADE IS ONE MATRIX, AND IT IS COMPOSED ON THE CPU.
   Brightness, contrast, saturation and tint are all AFFINE in RGB, so the four
   of them are one 3x3 and one offset — worked out once per frame on the CPU
   instead of ~22 scalar ALU in every fragment of every graded surface in both
   eyes.  The derivation, in the order the old shader applied them (that order
   is the thing being preserved: matrices do not commute and tint LAST is what
   makes it scale the offset as well as the matrix):

     1 brightness  g = c + b                     M1 = I           o1 = b*1
     2 contrast    g = (g-0.5)*s + 0.5, s = 1+contrast
                                                 M2 = s*I         o2 = 0.5(1-s)*1
     3 saturation  g = mix(vec3(l), g, k), k = 1+sat, l = w.g
                   = k*g + (1-k)*(1 w^T) g       M3 = k*I + (1-k)*1 w^T   o3 = 0
                   w = (0.2126, 0.7152, 0.0722), the same weights as before
     4 tint        g *= tint                     M4 = diag(t)     o4 = 0

     M = M4 M3 M2 M1 = s * diag(t) * ( k*I + (1-k) 1 w^T )
       so  M[i][j] = s * t[i] * ( k*delta(i,j) + (1-k)*w[j] )
     o = M4 M3 ( M2 o1 + o2 ) ; and because sum(w) = 1, M3 maps any GREY vector
       u*1 to itself, so the luma mix drops out of the offset entirely:
       o[i] = t[i] * ( s*b + 0.5*(1-s) )

   Checked two ways rather than trusted: composed generically as three matrix
   products and compared with the closed form (agrees to 2.2e-16), and both
   compared against the old shader body evaluated numerically on sample colours
   (same, to 2.2e-16).  Identity GRADE — contrast 0, sat 0, tint white,
   brightness 0 — gives exactly I and a zero offset, which is asserted.
   ========================================================================== */
const GRADE = { bright:0.0, contrast:0.12, sat:-0.05, tint:0xffeedd, mix:1.0 };
GRADE.u = { gradeMat:{value:new T.Matrix3()},
            gradeOff:{value:new T.Vector3()},
            gradeMix:{value:GRADE.mix} };
/* Rec.709 luma, the weights the shader used to carry inline */
const GRADE_LUMA = [0.2126, 0.7152, 0.0722];
const _gradeCol = new T.Color();            // scratch: hex -> rgb, never a new one
const _gradeT = [0,0,0], _gradeM = [0,0,0,0,0,0,0,0,0]; // scratch: per-frame, never new
/* Composed HERE, at the declaration, and not only from updateRig: a material
   can compile before the first frame runs, and an all-zero mat3 with gradeMix
   at 1 would be a black frame rather than a late one.
   NOTE for the next grade step: the loop below is the CLOSED FORM of exactly
   these four steps in exactly this order — a fifth step prepends its matrix on
   the LEFT of M = M4·M3·M2·M1 and the closed form must be re-derived, not
   extended in place. */
function gradeCompose(){
  const s = 1.0 + GRADE.contrast, k = 1.0 + GRADE.sat, w = GRADE_LUMA;
  _gradeCol.set(GRADE.tint);
  const t = _gradeT; t[0] = _gradeCol.r; t[1] = _gradeCol.g; t[2] = _gradeCol.b;
  const m = _gradeM;
  for(let i=0;i<3;i++) for(let j=0;j<3;j++)
    m[i*3+j] = s * t[i] * ((i===j ? k : 0) + (1-k)*w[j]);
  /* Matrix3.set is ROW-major and row i is output channel i; three.js uploads
     the column-major elements array, which is what GLSL mat3 wants */
  GRADE.u.gradeMat.value.set(m[0],m[1],m[2], m[3],m[4],m[5], m[6],m[7],m[8]);
  const o = s*GRADE.bright + 0.5*(1.0 - s);
  GRADE.u.gradeOff.value.set(t[0]*o, t[1]*o, t[2]*o);
  GRADE.u.gradeMix.value = GRADE.mix;
}
gradeCompose();

T.ShaderChunk.tonemapping_pars_fragment +=
  '\nuniform mat3 gradeMat;\nuniform vec3 gradeOff;\n' +
  'uniform float gradeMix;\n' +
  'vec3 houseGrade( vec3 c ){\n' +
  '  vec3 g = gradeMat * c + gradeOff;\n' +
  '  return mix( c, max( g, vec3( 0.0 ) ), gradeMix );\n' +
  '}\n';
T.ShaderChunk.tonemapping_fragment =
  '#if defined( TONE_MAPPING )\n' +
  '  gl_FragColor.rgb = houseGrade( toneMapping( gl_FragColor.rgb ) );\n' +
  '#endif\n';

/* ONE hook per material, ever, carrying BOTH rulings.  needsUpdate because a
   material that has already compiled keeps its program until it is told
   otherwise.  DL and DM share the hook rather than taking one each: two
   onBeforeCompile wrappers per material is two closures and two chances for a
   later one to drop the earlier — and the chain below is asserted precisely
   because that is the failure it would cause. */
function atmTrack(m){
  if(!m || m.userData.atmHooked) return m;
  const prev = m.onBeforeCompile;
  m.onBeforeCompile = function(sh, rn){
    if(prev) prev.call(this, sh, rn);
    sh.uniforms.atmHeight = ATM.u.atmHeight;   // the SAME objects, never copies —
    sh.uniforms.atmHaze   = ATM.u.atmHaze;     // a copy would freeze at first
    sh.uniforms.atmGlare  = ATM.u.atmGlare;    // compile and never move again
    sh.uniforms.atmMix    = ATM.u.atmMix;
    sh.uniforms.gradeMat  = GRADE.u.gradeMat;  // RULING DM, same seam —
    sh.uniforms.gradeOff  = GRADE.u.gradeOff;  // DU folded four of these into
    sh.uniforms.gradeMix  = GRADE.u.gradeMix;  // one matrix and one offset
  };
  m.userData.atmHooked = true;
  m.needsUpdate = true;
  return m;
}
/* RULING DM — ADDITIVE LIGHT IS LEFT OUT OF THE GRADE ON PURPOSE, and this is
   the note that stops someone "fixing" it later as an omission.  A beam, a gobo
   flare or a glow plane is the LIGHT ITSELF; the surfaces it falls on are
   already graded, so grading the source as well applies the tint twice to the
   same photon.  toneMapped=false is what takes a material out, and it also
   drops the TONE_MAPPING define so the chunk above is never reached. */
function gradeExempt(m){
  if(m) m.toneMapped = false;
  return m;
}

/* ============================================================================
   RULING DT — THE ENVIRONMENT BELONGS TO THE METALS, NOT TO EVERY PIXEL.

   DK assigned scene.environment, and r128 hands scene.environment to EVERY
   MeshStandardMaterial: setProgram reads (material.envMap || environment)
   (three.js :18170), so a global environment puts USE_ENVMAP and
   ENVMAP_TYPE_CUBE_UV into the program of every standard material in the
   building.  That is PMREM cube-UV sampling in every one of those fragments —
   up to 16 dependent texture taps plus ~100 ALU, per pixel, PER EYE, and the
   Quest has no multiview.  envMapIntensity only scales the RESULT: THE TAPS RUN
   AT INTENSITY 0, so a blackout paid full price for a reflection of nothing.
   Measured on hardware either side of the lighting round, an EMPTY house went
   25ms -> 47ms against a 13.9ms budget; the audit of 2026-08-14 puts most of
   that here.

   So the texture stops being global and becomes material.envMap on the
   materials that visibly need a reflection, and on nothing else.  DK's PURPOSE
   survives whole — metal does not render near-black, and it matches in VR
   because the rule is ONE function that every registration path already runs
   through — while the deck, the walls, the seats, the brick and 99 of the
   building's 120 standard materials stop paying for a reflection nobody could
   see.

   THE MATERIAL PATH IS THE SAME PATH, VERIFIED IN THE ARTIFACT AND NOT IN A
   CHANGELOG.  r128 resolves the two identically — cubemaps.get(material.envMap
   || environment) in WebGLPrograms.getParameters (:13106) and in setProgram
   (:18170), and WebGLCubeMaps.get returns a CubeUV texture untouched (:11292) —
   so envMapMode is the texture's own mapping either way, PMREMGenerator stamps
   CubeUVReflectionMapping on its output (:33867), and ENVMAP_TYPE_CUBE_UV is
   emitted from that (:12840, :12916).  maxMipLevel comes off
   properties.get(envMap).__maxMipLevel (:16810) through the same
   materialProperties.envMap slot.  If anything, material.envMap is the
   better-supported of the two: it takes PRECEDENCE, and setProgram re-acquires
   a program on its own when materialProperties.envMap changes (:18197), which
   is why no needsUpdate is written anywhere below — every assignment here lands
   before that material's first render, and atmTrack's own one-time
   needsUpdate rides in the same call besides.

   THE RULE IS METALNESS, AND IT IS READ RATHER THAN LISTED.  r128's standard
   material does diffuseColor *= (1 - metalness) and
   specularColor = mix(vec3(0.04), diffuseColor, metalness), so metalness is
   exactly the axis DK is about: at 1 the diffuse is gone and an unlit specular
   lobe is all that is left, which IS the near-black DK fixed.  At
   ENV_METAL_MIN half the diffuse still survives and the 4% dielectric Fresnel
   underneath is what every non-metal in the world has anyway.  A THRESHOLD AND
   NOT A HAND-LIST, because a list written here could not see the metals p2b,
   p2c, p2g, p2j, p5c, p5d, p5f, p5g, p5h and p9 mint for themselves, nor a
   single material out of a .glb — and the four registration paths (the M table
   at init, the model importer, the two keyed caches, and the ~142 materials p9
   mints in a session) then share one rule instead of four that can drift apart.

   0.5 SITS IN A GAP, WHICH IS WHY IT IS NOT A KNIFE-EDGE.  Censused on the
   built file, the metalness values standing in the scene at boot are
   0, .02, .04, .05, .10, .15, .20, .25, .30, then .50, .55, .60, .70, .80,
   .85, .90 — NOTHING between .31 and .49.  A threshold of 0.4 selects the same
   21 materials, so this number is not being tuned against a cliff.

   AND A MISSING metalness MEANS YES, NOT NO.  The vendored GLTFLoader's
   KHR_materials_pbrSpecularGlossiness branch DELETES metalness off the material
   it makes (p5i, GLTFMeshStandardSGMaterial), so a chrome arriving that way has
   no number to read.  Err toward the environment there: a wrong NO is the
   near-black metal this ruling exists to keep a fix for, while a wrong YES
   costs pixels on at most BJ_MAT_BUDGET (8) materials per set — and his exports
   carry ONE material each.  A metalnessMap does not change the test either: the
   shader multiplies the map INTO the scalar, so the scalar is the ceiling and a
   ceiling under 0.5 can never render as metal.

   WHAT DOES *NOT* NARROW: the atmosphere and the colour grade.  envTrack calls
   atmTrack for EVERY material before it asks about metalness (RULINGS DL and
   DM, and the INVARIANT that names it) — a material that lost the environment
   keeps the fog and the grade exactly as it had them.  Only the envMap
   assignment and the ENV_MATS registry narrow, and ENV_MATS is envDrive's list,
   which is what ENV_INTENSITY and the LIGHTING page's env row reach.
   ========================================================================== */
/* `let`, not `const`, because RULING DO's LIGHTING panel assigns it — this is
   the row Roblox calls EnvironmentSpecularScale.  ONE knob and not two: r128
   scales environment diffuse and specular together (DK).  It drives the envMap
   CARRIERS now, which is every material that has one (DT). */
let ENV_INTENSITY = 0.55;       // one-line retune: how much room ambience there is
let ENV_LIVE = ENV_INTENSITY;   // what the room actually has right now
let ENV_SRC = null;             // the source box, kept so it can be measured
let ENV_TEX = null;             // the PMREM — handed to the carriers, never to the scene (DT)
const ENV_HALF = 10;            // the box is 2 x ENV_HALF every way, camera dead centre
/* RULING DT — the one number that decides who samples the room.  See the gap
   census above before moving it. */
const ENV_METAL_MIN = 0.5;
/* the box as data, so what is measured is what was built */
const ENV_FACES = [
  {col:0x0a0c12, pos:[0,  ENV_HALF, 0], rot:[ Math.PI/2, 0, 0], why:'ceiling, darkest'},
  {col:0x24262b, pos:[0, -ENV_HALF, 0], rot:[-Math.PI/2, 0, 0], why:'floor, mid'},
  {col:0x14171d, pos:[0, 0, -ENV_HALF], rot:[0, 0, 0],          why:'walls, dim'},
  {col:0x14171d, pos:[0, 0,  ENV_HALF], rot:[0, Math.PI, 0],    why:'walls, dim'},
  {col:0x11141a, pos:[-ENV_HALF, 0, 0], rot:[0,  Math.PI/2, 0], why:'walls, dimmer'},
  {col:0x11141a, pos:[ ENV_HALF, 0, 0], rot:[0, -Math.PI/2, 0], why:'walls, dimmer'}
];
function envBuild(){
  const es = new T.Scene();
  for(const f of ENV_FACES){
    const m = new T.Mesh(new T.PlaneGeometry(2*ENV_HALF, 2*ENV_HALF),
      new T.MeshBasicMaterial({color:f.col, side:T.DoubleSide}));
    m.position.set(f.pos[0], f.pos[1], f.pos[2]);
    m.rotation.set(f.rot[0], f.rot[1], f.rot[2]);
    es.add(m);
  }
  /* NO es.background, and be exact about why that is safe.  Leaving it off
     sends _sceneToCubeUV down its ELSE branch, which DOES read the renderer's
     clear colour, paints backgroundMaterial from it and draws backgroundBox on
     all six faces before the scene.  Two things make that harmless and BOTH are
     load-bearing: the box above is closed, so every one of those pixels is
     overdrawn; and the clear colour is black — in a browser by default, and in
     the suites only because the jsdom stub's getClearColor was fixed to MUTATE
     its target the way the real one does.  A stub that returned the argument
     left PMREM reading its own module-level Color, which is WHITE.  Take the
     closed box away and the harness would build a white environment while the
     browser built a black one, and every suite would report success. */
  const pm = new T.PMREMGenerator(renderer);
  const rt = pm.fromScene(es, 0.04);
  pm.dispose();
  ENV_SRC = es;
  /* RULING DT — kept HERE and not on the scene.  envBuild() runs at the top of
     this part, so ENV_TEX is standing by the time the M table below exists and
     long before any other part can call envTrack. */
  ENV_TEX = rt.texture;
}
/* RULING DT — WHO SAMPLES THE ROOM.  Two guards, and they are different
   questions.  envMapIntensity is the MEMBERSHIP test: MeshBasic and the raw
   ShaderMaterials do not have one, and writing to them would only hang a stray
   property.  metalness is the DT test, and a material that has no metalness at
   all is the spec-gloss import — see the header for why that answers yes. */
function envCarrier(m){
  if(!m || !('envMapIntensity' in m)) return false;
  if(typeof m.metalness !== 'number') return true;
  return m.metalness >= ENV_METAL_MIN;
}
/* THE one registry.  envTrack takes a single material and RETURNS it, so a
   keyed cache can mint straight through it (goodsMat, woodMat). */
const ENV_MATS = new Set();
function envTrack(m){
  if(!m) return m;
  /* RULINGS DL and DM ride this same walk and reach EVERY material, which is
     why this runs before either guard below: fog reaches MeshBasic, points and
     sprites too, and DT's metalness test turns away five materials in six. */
  atmTrack(m);
  if(!envCarrier(m)) return m;
  /* the environment itself, on the metals alone (DT).  Assigned here rather
     than left to the scene, so a deck and a drape compile with no USE_ENVMAP
     in them at all. */
  m.envMap = ENV_TEX;
  m.envMapIntensity = ENV_LIVE;
  ENV_MATS.add(m);
  return m;
}
function envRegister(root){
  if(!root) return ENV_MATS.size;
  root.traverse(o=>{
    if(!o.isMesh || !o.material) return;
    const mats = Array.isArray(o.material) ? o.material : [o.material];
    for(const m of mats) envTrack(m);
  });
  return ENV_MATS.size;
}
/* Rebuild from what is actually there.  MUST NOT RUN BEFORE p5c HAS EXECUTED:
   SHOW is a `const` of this one concatenated program, and a `const` in its
   temporal dead zone throws on a plain reference AND through `typeof` (TRAPS,
   twice) — so there is no guard that would make this safe to call from p2..p5c
   and it does not pretend to have one.  Every call site is post-boot.
   (`typeof WOODM` below is a different case and is sound: WOODM and GOODSM are
   both declared `var` in p4c, which hoists.) */
function envRecollect(){
  ENV_MATS.clear();
  envRegister(scene);
  /* RULING CN holds two of the three houses OUT of the graph at any moment, so
     a walk of the scene cannot see them */
  if(SHOW && SHOW.offstage)
    for(const g of SHOW.offstage) envRegister(g);
  /* the keyed caches INVARIANTS names: a wood paint and a dyed drape are minted
     on demand and outlive the mesh that asked for them, so a rebuild from the
     scene alone would strand one the moment nothing is currently wearing it.
     LENSM is not here on purpose — it holds MeshBasicMaterial, which has no
     envMapIntensity to drive. */
  if(typeof WOODM !== 'undefined')
    for(const k in WOODM) envTrack(WOODM[k]);
  if(typeof GOODSM !== 'undefined')
    for(const k in GOODSM) envTrack(GOODSM[k]);
  /* the shared M table, for RULINGS DL and DM rather than DK: a paint that is
     not on a mesh at this instant still compiles the moment something wears it,
     and an unhooked material would reach the fog and grade chunks with their
     uniforms unsupplied.  M is a const of p2, so it exists by the time any call
     site here can run — see the dead-zone note above. */
  for(const k in M) envTrack(M[k]);
  return ENV_MATS.size;
}
/* the per-frame walk — one Set, one pass, no traversal */
function envDrive(){
  for(const m of ENV_MATS) m.envMapIntensity = ENV_LIVE;
}

function cvs(w,h){ const c=document.createElement('canvas'); c.width=w; c.height=h; return c; }
function tex(c, rx, ry, srgb){
  const t = new T.CanvasTexture(c);
  t.wrapS = t.wrapT = T.RepeatWrapping;
  t.repeat.set(rx||1, ry||1);
  t.anisotropy = 4;
  if(srgb !== false) t.encoding = T.sRGBEncoding;
  return t;
}
function grain(x,w,h,amt){
  const img=x.getImageData(0,0,w,h), d=img.data;
  for(let i=0;i<d.length;i+=4){ const n=(Math.random()-0.5)*amt;
    d[i]=clamp(d[i]+n,0,255); d[i+1]=clamp(d[i+1]+n,0,255); d[i+2]=clamp(d[i+2]+n,0,255); }
  x.putImageData(img,0,0);
}

/* ----------------------------------------------------------------------------
   mergeParts — bake a cluster of static detail into ONE geometry.
   r128's core ships no BufferGeometryUtils, so this is the house version.
   Each part is {geo, pos:[x,y,z], rot:[x,y,z], scale:[x,y,z]}; the transform
   is baked into the vertices and the parts are concatenated, so a detailed
   object costs one draw call instead of one per part (RULING AK).
   EVERY part must be drawn with the material the merged mesh is given.
   Anything that moves, is grabbed, or is recoloured at runtime must stay its
   own mesh — see RULING AL for the list that must never come through here.
   No negative scale: mirroring reverses triangle winding relative to the
   shading normals, and a live mesh hides that with the matrix determinant,
   but baked vertices carry no determinant — a mirrored part vanishes under
   the default FrontSide. Mirrored parts must flip winding instead.
   De-indexing costs vertices — the right trade on a Quest, but visible: the
   planned nail-gun shell goes from 160 verts / 5 draw calls to 264 verts /
   1 draw call, 1.65x the vertices for the same triangles.
   -------------------------------------------------------------------------- */
const _mpM = new T.Matrix4(), _mpQ = new T.Quaternion(), _mpE = new T.Euler();
const _mpP = new T.Vector3(), _mpS = new T.Vector3();
function mergeParts(parts){
  const pos = [], nor = [], uvs = [];
  parts.forEach(p=>{
    const src = p.geo.index ? p.geo.toNonIndexed() : p.geo.clone();
    _mpP.set(p.pos ? p.pos[0] : 0, p.pos ? p.pos[1] : 0, p.pos ? p.pos[2] : 0);
    _mpE.set(p.rot ? p.rot[0] : 0, p.rot ? p.rot[1] : 0, p.rot ? p.rot[2] : 0);
    _mpS.set(p.scale ? p.scale[0] : 1, p.scale ? p.scale[1] : 1, p.scale ? p.scale[2] : 1);
    src.applyMatrix4(_mpM.compose(_mpP, _mpQ.setFromEuler(_mpE), _mpS));
    if(!src.attributes.normal) src.computeVertexNormals();   // per-part, so smooth normals elsewhere survive
    const pa = src.attributes.position, na = src.attributes.normal, ua = src.attributes.uv;
    for(let i=0;i<pa.count;i++){
      pos.push(pa.getX(i), pa.getY(i), pa.getZ(i));
      nor.push(na.getX(i), na.getY(i), na.getZ(i));
      ua ? uvs.push(ua.getX(i), ua.getY(i)) : uvs.push(0, 0);
    }
    src.dispose();
  });
  const out = new T.BufferGeometry();
  out.setAttribute('position', new T.Float32BufferAttribute(pos, 3));
  if(nor.length) out.setAttribute('normal', new T.Float32BufferAttribute(nor, 3));
  if(uvs.length) out.setAttribute('uv',     new T.Float32BufferAttribute(uvs, 2));
  out.computeBoundingSphere();
  return out;
}

  return STENCILS[key] = tex(c,1,1);
}

function box(w,h,d,mat,x,y,z,parent){
  const m = new T.Mesh(new T.BoxGeometry(w,h,d), mat);
  m.position.set(x,y,z); m.castShadow=true; m.receiveShadow=true;
  (parent||world).add(m); STATIC.push(m); return m;
}
const _dummy = new T.Object3D();
/* one draw call for hundreds of identical ornaments */
function instanced(geo, mat, list, opt){
  opt = opt || {};
  if(!list.length) return null;
  const m = new T.InstancedMesh(geo, mat, list.length);
  for(let i=0;i<list.length;i++){
    const t = list[i];
    _dummy.position.set(t.x||0, t.y||0, t.z||0);
    _dummy.rotation.set(t.rx||0, t.ry||0, t.rz||0);
    _dummy.scale.set(t.s||t.sx||1, t.s||t.sy||1, t.s||t.sz||1);
    _dummy.updateMatrix();
    m.setMatrixAt(i, _dummy.matrix);
  }
  m.instanceMatrix.needsUpdate = true;
  /* r128 sizes an InstancedMesh's bounding sphere from ONE instance at the
     origin, so a spread-out batch vanishes the moment the origin leaves the
     frustum. Give it a sphere that actually contains every instance.        */
  geo.computeBoundingSphere();
  const r0 = geo.boundingSphere ? geo.boundingSphere.radius : 1;
  let cx=0, cy=0, cz=0, sMax=1;
  for(const t of list){ cx+=t.x||0; cy+=t.y||0; cz+=t.z||0;
    sMax = Math.max(sMax, t.s||t.sx||1, t.s||t.sy||1, t.s||t.sz||1); }
  cx/=list.length; cy/=list.length; cz/=list.length;
  let rad = 0;
  for(const t of list) rad = Math.max(rad,
    Math.hypot((t.x||0)-cx, (t.y||0)-cy, (t.z||0)-cz));
  geo.boundingSphere = new T.Sphere(new T.Vector3(cx,cy,cz), rad + r0*sMax + 0.5);
  m.castShadow = opt.cast !== false;
  m.receiveShadow = opt.receive !== false;
  m.frustumCulled = opt.cull !== false;
  (opt.parent||world).add(m);
  STATIC.push(m);
  return m;
}
/* a run of ornament along a straight line */
function runAlong(x0,y0,z0, x1,y1,z1, n, extra){
  const out=[];
  for(let i=0;i<n;i++){
    const t = n===1?0.5:i/(n-1);
    out.push(Object.assign({x:lerp(x0,x1,t), y:lerp(y0,y1,t), z:lerp(z0,z1,t)}, extra||{}));
  }
  return out;
}
/* a run of ornament along the curved front of a balcony */
function runCurve(fn, n, extra){
  const out=[];
  for(let i=0;i<n;i++){ const p = fn(i/(n-1));
    out.push(Object.assign({x:p.x, y:p.y, z:p.z, ry:p.ry||0}, extra||{})); }
  return out;
}

/* the curved sweep both balconies share: a shallow arc bowing toward the stage */
const BOW = 3.4;                               // how far the centre reaches forward
function balconyFront(t, zFront, halfW){       // t: -1 (house left) .. 1 (house right)
  const x = t*halfW;
  const z = zFront + BOW*(t*t);
  return {x, z};
}
