
/* ============================================================================
   SHOWS — a whole production, loaded in one go

   A show preset does the load-in for you: it builds the scenery, hangs and
   trims the goods, sets the haze and the practicals, and plots a cue stack you
   can then sit at the board and run.  Everything it makes goes into SHOW.group
   so STRIKE takes it all out again in one move.

   These are interpretations, not reproductions.  Nobody's ground plan or
   light plot went into them — they are built from the vocabulary of the show:
   the shape of the room, the materials, the colours, the way the light sits.
   ========================================================================== */
const SHOWS = {};
/* THE one definition of SHOW's shape.  The live object below, the parked
   templates p2k gives the Arc stages, and the strike-reset all build from
   here — three hand-copies had drifted apart, and keys a show wrote ad hoc
   (wall, dropKey, …) leaked across stage swaps because Object.assign can
   overwrite but never remove.  Add new show state HERE, nowhere else. */

}

/* is anything, on any stage, still hanging this? */
function goodsInUse(key){
  const on = list => !!(list && list.some(ls=>ls.goodsKey === key));
  if(on(FLY)) return true;
  if(typeof STAGES !== 'undefined')
    for(const k in STAGES) if(on(STAGES[k].fly)) return true;
  return false;
}

/* everything a preset makes hangs off here, on the stage, so the room culling
   takes it with the rest of the stage */
function showRoot(){
  if(SHOW.group) return SHOW.group;
  const g = new T.Group(); g.name = 'show';
  /* a production goes onto the stage the board is patched to, not onto the
     Palace's stage by default — otherwise loading a show at the Arc builds
     it four hundred metres away in an empty building */
  const st = (typeof stageLive === 'function') ? stageLive() : null;
  if(st && st.showParent) st.showParent().add(g);
  else if(typeof ROOM_GROUP !== 'undefined' && ROOM_GROUP.stage) ROOM_GROUP.stage.add(g);
  else world.add(g);
  SHOW.group = g;
  return g;
}
function showWalk(o){ SHOW.walk.push(o); WALKABLE.push(o); return o; }

/* ============================================================================
   SCENES — more than one set, and a way to change between them

   A show with eight locations does not get eight stages.  It gets one, and the
   sets take turns on it.  Each scene is a group built at load-in and parked
   invisible until it is called.

   Two things can hide a piece of scenery, and they must not fight: the scene
   system, when this set is not the one playing, and the crew, when they have
   not carried it on yet.  Either one alone is enough to keep it off — so both
   are recorded and the visibility is worked out from the pair of them, rather
   than each writing over the other.
   ========================================================================== */
function setPieceVisible(o, crewHidden){
  if(crewHidden !== undefined) o.userData.crewHidden = crewHidden;
  const on = !o.userData.crewHidden && !o.userData.sceneOff;
  o.visible = on;
  /* §5: visible is only a drawing flag — a raycast still hits the piece, so
     groundAt would stand you mid-air on a gallery the crew have not carried
     in yet.  Clear the layers so a hidden piece is inert, and take its
     walkable surfaces out of WALKABLE while it is off (marked, so exactly
     those come back when it is). */
  o.traverse(c=>{
    if(on) c.layers.set(0); else c.layers.disableAll();
    if(typeof WALKABLE === 'undefined') return;
    const i = WALKABLE.indexOf(c);
    if(!on && i >= 0){ WALKABLE.splice(i, 1); c.userData.wasWalkable = true; }
    if(on && c.userData.wasWalkable && i < 0) WALKABLE.push(c);
  });
}
function sceneAdd(name, label){
  const g = new T.Group();
  g.name = 'scene:' + name;
  g.userData.scene = name;
  g.userData.sceneOff = true;
  g.visible = false;
  showRoot().add(g);
  const sc = {name, label: label || name.toUpperCase(), group:g, walk:[], on:false,
              mv:null, pmv:null, mvHide:false, always:false};
  SHOW.scenes.push(sc);
  return sc;
}
function sceneFind(name){ return SHOW.scenes.find(s=>s.name === name) || null; }
/* anything you can stand on inside a scene goes on the scene's own list, so it
   leaves WALKABLE the moment that set goes off — otherwise you would be
   walking on the staircase of a house that is not on the stage */
function sceneWalk(sc, o){ if(sc) sc.walk.push(o); return o; }
/* turning one scene on or off — the whole of it, so the mover can reuse it
   when a travelling set finally arrives and is allowed to disappear */
/* ============================================================================
   RULING BQ — A SET THAT COMES OFF IS NEVER GONE

   "make it so when i set comes off it is never gone it is always somewhere
   backstage."

   A scene may declare a PARK: somewhere backstage it stands, solid and visible,
   for the rest of the run.  The stage is empty because the set is somewhere else
   rather than because it has been switched off.

   DECLARED, NEVER ASSUMED.  A scene with no park keeps the old behaviour exactly
   — switched off on strike — which is what leaves the other four productions
   byte-identical, and what lets a set with nowhere to go say so.  Two of
   Beetlejuice's own cannot park and the reasons are in p5h.

   THE PARK IS A PART MOVER called `park`, over a wrapper group holding all the
   scene's children (the bjFlyWhole pattern, so nothing repositions at build).
   sceneChangeTo already drives every part mover to its `out` on strike and back
   to `home` on entry, so declaring one is all the travel this needs.  Scenes
   already flown out by a mover of their own just set the flag.

   AND A PARKED SET IS DRAWN BUT NOT PICKED, which is measured and not a
   preference.  layers.disableAll() was doing two jobs — not drawn AND not
   raycast — and keeping his models drawn puts 582,736 triangles in front of the
   crosshair: one pick went from 1.34ms to 7.70ms, 5.7x, 69% of a 90Hz frame
   (tools/parked.js).  That is RULING BY's shape exactly: three.js collects and
   sorts every intersection, so there is no early exit and a miss costs the same
   as a hit.  So a parked mesh gets `raycast = NOOP` — the trick TRAPS already
   records for decorative instanced batches — and its walkables still come off
   WALKABLE, so groundAt never sees it at all.
   ========================================================================== */
const SCENE_PARK_SPEED = 2.0;
function sceneParkNoop(){}
/* declare a park.  With an axis it builds the wrapper and its mover; with none
   it only marks the scene as having somewhere to go, for a set whose own mover
   already takes it off (the flown rooms). */
function scenePark(sc, axis, out, speed, opt){
  if(!sc) return null;
  sc.parks = true;
  /* a park may also drive the whole-group mover to a stated offset as it goes.
     The wagon is the case: its z is cue-authored and its upstage park no longer
     fits his 12.98m room, so the strike puts the depth back to the ACTING offset
     and lets the wing do the hiding.  Stated, not `home` — the wagon's home IS
     BJ_WAGON_BACK, the very number that stands the house in the brick.  Absent by
     default: a flown set keeps whatever the cue said about its own mover. */
  if(opt && opt.mvAt !== undefined) sc.parkMvAt = opt.mvAt;
  /* a scene whose OWN whole-group mover already points the right way parks on
     that, rather than growing a second one (the flown exterior) */
  if(opt && opt.mv !== undefined) sc.parkMv = opt.mv;
  if(axis === undefined) return sc;
  /* THE PARK MOVER DRIVES THE SCENE GROUP ITSELF, on an axis its whole-group
     mover is not using — NOT a wrapper group.

     Wrapping the children was the first attempt and it broke two things that
     read a scene's structure: the model importer strips the built-in shell out of
     sc.group before landing a whole house, and with a wrapper in the way the
     shell survived and his house landed on top of it.  sceneMvAdvance writes
     g.position[axis] and nothing else, so two movers over the same group are
     free as long as the axes differ — the wagon keeps z, the park takes x. */
  sceneTravelPart(sc, 'park', sc.group, axis, 0, out, speed || SCENE_PARK_SPEED);
  if(sc.mv && sc.mv.axis === axis)
    console.warn('scenePark: ' + sc.name + ' parks on ' + axis +
                 ', which its own mover already drives — they will fight');
  return sc;
}
/* drawn, but never in front of the crosshair.  Only ever undone for meshes this
   put a raycast ON: a decorative instanced batch carries its own `raycast` noop
   deliberately (TRAPS), and deleting that would put 1,400 seats back on every
   pick. */
function sceneParkPick(sc, off){
  sc.group.traverse(o=>{
    if(!o.isMesh) return;
    if(off){
      if(!Object.prototype.hasOwnProperty.call(o, 'raycast')){
        o.raycast = sceneParkNoop; o.userData.parkRc = true;
      }
    } else if(o.userData.parkRc){
      delete o.raycast; delete o.userData.parkRc;
    }
  });
}
function sceneApply(sc, on){
  /* a dress cue that arrived while this set was up has been waiting for this
     moment: the set is going off, so the swap is finally invisible (RULING
     AY).  It lands FIRST, so the layer sweep below leaves the whole scene
     inert, new dressing included. */
  if(!on && SHOW.pendDress && SHOW.pendDress.scene === sc.name &&
     typeof bjDress === 'function'){
    const pd = SHOW.pendDress; SHOW.pendDress = null; bjDress(pd.scene, pd.key);
  }
  /* RULING BQ — a set that declared a park is STANDING BACKSTAGE, not gone, so
     coming off leaves it drawn.  A scene with no park is switched off exactly as
     it always was, which is what keeps the other four productions unchanged. */
  const park = !on && !!sc.parks;
  sc.group.userData.sceneOff = park ? false : !on;
  setPieceVisible(sc.group);
  /* visible is only a drawing flag — a raycast goes straight through it and
     hits the set anyway.  A set that is off has to be inert as well as
     unseen, or you can click on a house that is not there and the picking
     lands on a wall in the wings.  Clearing the layers does both: nothing
     renders it and no ray tests it.                                     */
  sc.group.traverse(o=>{ if(on || park) o.layers.set(0); else o.layers.disableAll(); });
  /* which leaves the SECOND of those two jobs to be done by hand for a parked
     set: drawn, but never in front of the crosshair.  Measured, not preferred —
     582,736 triangles of his models took one pick from 1.34ms to 7.70ms. */
  sceneParkPick(sc, park);
  for(const o of sc.walk){
    const i = WALKABLE.indexOf(o);
    if(on && i < 0) WALKABLE.push(o);
    if(!on && i >= 0) WALKABLE.splice(i, 1);
  }
  /* a set that comes on has just had the layers of EVERY descendant enabled,
     which would light all three of a wagon dressings at once.  The show puts
     the right one back. (RULING AQ)  A PARKED set has just had exactly the same
     thing done to it, so it needs the same correction — or the house standing in
     the wing wears all three dressings at once. */
  if((on || park) && typeof bjRedress === 'function') bjRedress(sc);
}
/* a piece of scenery that is on the stage whatever set is up — the show's
   own sign, hanging downstage of the house curtain.  It is a scene so that
   it can carry a travel and be flown; it is ALWAYS so that sceneShow's
   sweep leaves it alone.  The portal does the same job by not being a
   scene at all, which works only because the portal never moves. */
function sceneAlways(sc){
  if(!sc) return null;
  sc.always = true;
  sc.on = true;
  sceneApply(sc, true);
  return sc;
}
function sceneShow(name){
  if(!SHOW.scenes.length) return null;
  const want = sceneFind(name);
  if(!want) return null;
  if(want.always) return null;          // it is never the set that is "on"
  for(const sc of SHOW.scenes){
    if(sc.always) continue;             // and it is never swept off by one
    const on = (sc === want);
    if(sc.on === on && SHOW.scene === name) continue;
    sc.on = on;
    /* an instant swap has no travel in it: a part parked at OUT comes home
       in the same frame, or the set would come on hollow, its walls still
       standing in the wings (RULING AY) */
    if(on && sc.pmv) for(const k in sc.pmv){
      const m = sc.pmv[k];
      m.off = m.target = m.home; m.group.position[m.axis] = m.home;
    }
    /* a set that is still TRAVELLING stays drawn until it arrives, whatever
       the cue says — otherwise the house winks out halfway across the deck.
       sceneMoveStep does the hiding when the move finishes.              */
    if(!on && sceneTravelling(sc)){ sc.mvHide = true; continue; }
    sc.mvHide = false;
    sceneApply(sc, on);
  }
  SHOW.scene = name;
  if(typeof refreshSceneUI === 'function') refreshSceneUI();
  return want;
}
function sceneNext(step){
  if(!SHOW.scenes.length) return null;
  const i = SHOW.scenes.findIndex(s=>s.name === SHOW.scene);
  const n = (i < 0 ? 0 : i + (step || 1) + SHOW.scenes.length) % SHOW.scenes.length;
  /* the prev/next buttons choreograph like everything else (RULING AY) */
  return sceneChangeTo(SHOW.scenes[n].name);
}

/* ============================================================================
   THE MOVER — scenery that travels, and you watch it do it (RULING AP)

   sceneShow is an instant swap, which is the right answer for a set that
   changes in the dark and the wrong one for a house that slides onto the
   deck in full view.  So a scene may carry a TRAVEL: one axis, where it is
   now, and where it is going.  The frame advances it off `dt` from
   updateStorm — NEVER off a timer, which is the standing rule of this
   codebase and the reason the `follow` chain is called out as a violation
   everywhere it appears.

   Three things fall out of "you watch it", and each one is a test:
     - a travelling set is ON for the whole move, going on or coming off;
     - its walkables ride the group, so the floor of a house goes with the
       house rather than staying where the house used to be;
     - a move fired mid-travel RETARGETS.  It does not queue, because a
       cue stack that banks moves would run the set to the wrong place
       several cues later.
   ========================================================================== */
const SCENE_MV_EPS = 1e-4;              // closer than this and it has arrived

function sceneTravelling(sc){
  if(!sc) return false;
  if(sc.mv && Math.abs(sc.mv.target - sc.mv.off) > SCENE_MV_EPS) return true;
  /* the deferred hide waits for the LAST of them — one part still crossing
     keeps the whole set drawn, exactly as the whole-group mover does */
  if(sc.pmv) for(const k in sc.pmv){
    if(Math.abs(sc.pmv[k].target - sc.pmv[k].off) > SCENE_MV_EPS) return true;
  }
  return false;
}
/* give a scene the ability to travel.  `home` is where it sits at load. */
function sceneTravel(sc, axis, home, speed){
  if(!sc) return null;
  const a = axis || 'x', h = home || 0;
  sc.mv = {axis:a, off:h, target:h, home:h, speed: speed || 2.2};
  /* both halves of this matter, because lockShowStatic may run either side
     of it: the flag makes the freeze skip this group, and the assignment
     undoes a freeze that already happened. */
  sc.group.userData.sceneTravels = true;
  sc.group.matrixAutoUpdate = true;
  sc.group.position[a] = h;
  return sc;
}
/* a scene may also carry several NAMED movers over child groups, alongside
   the whole-group travel (RULING AY): the hills part and run to the wings,
   a flown room drops in, each part its own record.  `home` is the on-stage
   pose the part is BUILT in; `out` is where it goes to clear the stage. */
function sceneTravelPart(sc, name, group, axis, home, out, speed){
  if(!sc || !group) return null;
  const a = axis || 'x', h = home || 0;
  const m = {group, axis:a, off:h, target:h, home:h,
             out:(out === undefined ? h : out), speed: speed || 2.2};
  (sc.pmv || (sc.pmv = {}))[name] = m;
  /* the same two-sided freeze guard sceneTravel carries, for the same
     reason: lockShowStatic may run either side of this call, and a frozen
     group takes the position write and stands still. */
  group.userData.sceneTravels = true;
  group.matrixAutoUpdate = true;
  group.position[a] = h;
  return m;
}
function sceneMoveTo(name, target, speed){
  const sc = (typeof name === 'string') ? sceneFind(name) : name;
  if(!sc || !sc.mv) return null;
  sc.mv.target = target || 0;
  if(speed) sc.mv.speed = speed;
  return sc;
}
/* a cue can also send ONE NAMED PART somewhere — scenery moving WITHIN a
   scene, with no scene change: the graveyard empties to its backdrop and the
   set stays on (RULING AY).  Same shape as sceneMoveTo, and the same manners:
   a scene or part that does not exist is a no-op, not a throw. */
function sceneMovePartTo(name, part, target, speed){
  const sc = (typeof name === 'string') ? sceneFind(name) : name;
  if(!sc || !sc.pmv || !sc.pmv[part]) return null;
  const m = sc.pmv[part];
  m.target = target || 0;
  if(speed) m.speed = speed;
  return m;
}
/* one mover record, one frame: the same eps and the same clamp whether it is
   the whole set on its wagon or one part of it on its own track */
function sceneMvAdvance(m, g, dt){
  const d = m.target - m.off;
  if(Math.abs(d) <= SCENE_MV_EPS){
    if(m.off !== m.target){ m.off = m.target; g.position[m.axis] = m.off; }
    return;
  }
  const step = m.speed * dt;
  m.off += Math.abs(d) <= step ? d : (d < 0 ? -step : step);
  g.position[m.axis] = m.off;
}
/* ============================================================================
   RULING CI — A MOVE MAY WAIT FOR THE FLY LINE IT NEEDS

   "make it so the house cant start sliding till the backdrop is fully up"

   Cue 7 flies the backdrop out and slides the Maitlands house on in the same
   breath.  The backdrop is line 14 and the wagon travels 10m; today they simply
   both start, and he has watched the house set off into a cloth still on its way
   up.

   A cue move may name a lineset it waits on: move:{scene:'interior', off:0,
   after:14}.  It is a GATE ON THE MOVE, not a delay on the cue — the cue fires,
   the changeover happens, the lights change, and the one thing that would
   collide holds until the pipe has arrived.

   It rides the frame, never a timer (CLAUDE.md), and it is the same shape as
   sc.mvHide, which already defers a hide until the last part has landed.

   DECLARED, NEVER ASSUMED.  A move with no `after` behaves exactly as it always
   has, which is what leaves the other four productions and the other eight moves
   in this plot untouched.

   AND A CUE NEVER BANKS MOVES.  sceneMoveTo's own rule is that a move fired
   mid-travel RETARGETS rather than queueing, "because a cue stack that banked
   moves would run the set to the wrong place several cues later" — so the
   pending list is cleared by the next cue that fires, whatever it says.
   ========================================================================== */
function flyArrived(id){
  const ls = FLY[id - 1];
  if(!ls) return true;               // no such line: never wait for nothing
  return Math.abs(ls.target - ls.pos) < 0.02;
}
function sceneMoveApply(m){
  if(!m) return;
  if(m.part) sceneMovePartTo(m.scene, m.part, m.off, m.speed);
  else sceneMoveTo(m.scene, m.off, m.speed);
}
function scenePendingStep(){
  const q = SHOW.pendMoves;
  if(!q || !q.length) return;
  for(let i = q.length - 1; i >= 0; i--){
    if(!flyArrived(q[i].after)) continue;
    const m = q[i];
    q.splice(i, 1);
    sceneMoveApply(m);
  }
}

/* ============================================================================
   RULING CJ — THE HOUSE LIGHTS WAIT FOR THE CURTAIN

   "the house lights dont come up to 15 until the main curtain is closed"
   "the lights dont come back up to 15 at the end till the curtain is fully
    closed"

   He says it twice, at the two places the house comes up, and both are the same
   fault: the front curtain is a traveler that FLIES, so it takes
   (OUT_TRIM - trim) / speed seconds to come in — while the interval cue fires
   seven seconds after the blackout and the final cue closes the curtain and
   raises the house in the same breath.  Either way the audience gets the house
   up on a stage that is still open.

   A cue may say houseAfterCurtain:true.  The master is held at 0 and goes to the
   cue's own level the frame the cloth is really in.  HOUSE.house is read live by
   updateRig every frame and has never faded, so holding and then setting is
   exactly what the cue did before, one condition later.

   FULLY CLOSED IS TWO THINGS, and it is judged on the LINESET rather than on a
   cue field: the pipe home at its in trim AND the halves together.  This show's
   curtain flies rather than draws, so `open` is 0 throughout and the trim is
   what moves — but a production that parts its curtain would need both, and a
   test that only checked the trim would pass a stage standing wide open.
   ========================================================================== */
function frontCurtainShut(){
  const ls = (typeof frontCurtainLineset === 'function') ? frontCurtainLineset() : null;
  if(!ls) return true;               // no traveler hung: nothing to wait for
  const trim = TRIMS[ls.goodsKey] !== undefined ? TRIMS[ls.goodsKey] : TRIMS.house;
  return Math.abs(ls.pos - trim) < 0.05 && (ls.open || 0) < 0.02;
}
function updateHouseWait(){
  if(SHOW.houseWait === null || SHOW.houseWait === undefined) return;
  if(!frontCurtainShut()) return;
  HOUSE.house = SHOW.houseWait;
  SHOW.houseWait = null;
  if(typeof syncMasters === 'function') syncMasters();
}

function sceneMoveStep(dt){
  scenePendingStep();
  if(!SHOW.scenes || !SHOW.scenes.length) return;
  for(const sc of SHOW.scenes){
    if(sc.mv) sceneMvAdvance(sc.mv, sc.group, dt);
    if(sc.pmv) for(const k in sc.pmv) sceneMvAdvance(sc.pmv[k], sc.pmv[k].group, dt);
    /* everything has arrived, so a hide the changeover deferred can happen —
       and it waits on the LAST part, not the first */
    if(sc.mvHide && !sceneTravelling(sc)){ sc.mvHide = false; sceneApply(sc, false); }
  }
}
/* ============================================================================
   THE CHANGEOVER — a scene change you watch happen (RULING AY)

   sceneShow stays the instant swap, and boot and preset paths still use it.
   Everything cue-driven or hand-driven comes through here instead: the
   outgoing set sends every part it carries to its OUT and hides only when
   the last one arrives; the incoming set comes on with its parts AT their
   OUT — positioned before a single layer is enabled, so no frame ever shows
   them at home — and they travel in, overlapped with the outgoing.

   The whole-group mover is deliberately NOT sent anywhere: the wagon's
   z-moves stay cue-authored through the `move:` field.  And a set with no
   part movers and nothing mid-travel swaps instantly, which keeps the other
   four productions behaving exactly as they always have.

   Fired mid-changeover it RETARGETS, never queues — a still-drawn set on
   its way out keeps its place and turns round.
   ========================================================================== */
function sceneChangeTo(name){
  if(!SHOW.scenes.length) return null;
  const want = sceneFind(name);
  if(!want) return null;
  if(want.always) return null;
  for(const sc of SHOW.scenes){
    if(sc.always) continue;
    const on = (sc === want);
    if(sc.on === on && SHOW.scene === name) continue;
    sc.on = on;
    if(on){
      /* a set that is still drawn — mid-way out of an interrupted changeover
         — keeps its place and turns round; a genuinely hidden one starts
         from OUT, so the first visible frame has the parts off and high */
      const drawn = !sc.group.userData.sceneOff;
      if(sc.pmv) for(const k in sc.pmv){
        const m = sc.pmv[k];
        if(!drawn){ m.off = m.target = m.out; m.group.position[m.axis] = m.off; }
        m.target = m.home;
      }
      /* a set parked ON its own mover has to be recalled by it too (RULING BQ).
         A cue that authors this scene's move still wins: the split rule applies
         an incoming scene's moves AFTER the changeover. */
      if(sc.parkMv !== undefined && sc.mv) sc.mv.target = sc.mv.home;
      sc.mvHide = false;
      sceneApply(sc, true);
    } else {
      if(sc.pmv) for(const k in sc.pmv) sc.pmv[k].target = sc.pmv[k].out;
      /* RULING BQ — a park that owns the depth as well (the wagon).  This runs
         AFTER the outgoing scene's cue-authored move, which the SPLIT RULE puts
         before the changeover, so it deliberately wins: "the house slides back"
         becomes "the house tracks off", because 12.98m of his room does not fit
         between the backdrop and the brick.  Only for a scene that asked. */
      if(sc.parkMvAt !== undefined && sc.mv) sc.mv.target = sc.parkMvAt;
      /* and the flown exterior parks ON its own mover, because up is where it
         goes and its mover is the thing that goes up */
      if(sc.parkMv !== undefined && sc.mv) sc.mv.target = sc.parkMv;
      if(sceneTravelling(sc)){ sc.mvHide = true; continue; }
      sc.mvHide = false;
      sceneApply(sc, false);
    }
  }
  SHOW.scene = name;
  if(typeof refreshSceneUI === 'function') refreshSceneUI();
  return want;
}
function showStrike(){
  if(typeof crewForgetLoads === 'function') crewForgetLoads();
  if(typeof CREW !== 'undefined' && CREW.running && typeof crewStop === 'function') crewStop(true);
  for(const o of SHOW.walk){ const i = WALKABLE.indexOf(o); if(i >= 0) WALKABLE.splice(i,1); }
  SHOW.walk = [];
  if(SHOW.rainSound && typeof Snd !== 'undefined'){ Snd.play('rain'); SHOW.rainSound = false; }
  if(SHOW.group){
    SHOW.group.parent && SHOW.group.parent.remove(SHOW.group);
    disposeTree(SHOW.group);
    SHOW.group = null;
  }
  LIGHTNING.intensity = 0; LIGHTNING_AMB.intensity = 0;
  SHOW.stormOn = false; SHOW.porch = null;
  // a show curtain gets the house's own curtain back, not a bare pipe
  FLY.forEach(ls=>{
    if(SHOW.goods.indexOf(ls.goodsKey) === -1) return;
    const wasTraveler = GOODS[ls.goodsKey] && GOODS[ls.goodsKey].traveler;
    hangGoods(ls, wasTraveler ? 'house' : 'none');
    if(wasTraveler){ ls.target = ls.pos = TRIMS.house; ls.travTarget = ls.open = 0; }
  });
  /* GOODS is one catalogue for the whole game, but a show's own cloths are
     registered into it — and the same production can be up on two stages at
     once.  Striking one of them must not take the other's house curtain out
     of the catalogue, or the far rail is left pointing at nothing.       */
  for(const k of SHOW.goods) if(!goodsInUse(k)){ delete GOODS[k]; delete TRIMS[k]; }
  /* one wipe from the single template — keys a show wrote ad hoc go with it,
     instead of surviving into the next production's SHOW */
  for(const k of Object.keys(SHOW)) delete SHOW[k];
  Object.assign(SHOW, showBlank());
  if(typeof removeShowSmoke === 'function') removeShowSmoke();
  if(typeof buildGoodsPalette === 'function') buildGoodsPalette();
  if(typeof refreshShowUI === 'function') refreshShowUI();
  if(typeof refreshFlyUI === 'function') refreshFlyUI();
}
/* ============================================================================
   THE SAVED HANG
   A preset arrives with the hang its designer intended.  The moment you move
   a lineset, re-hang a batten or lock something off, that is your hang, and
   you can write it back to the show so it loads in that way from then on.
   Kept per show and outside SHOW, so striking the set does not lose it.
   ========================================================================== */
const SHOW_HANGS = {};
function hangSnapshot(){
  return FLY.map(ls=>({
    id: ls.id,
    goods: ls.goodsKey,
    target: +ls.target.toFixed(3),
    open: +((ls.travTarget || 0).toFixed(3)),
    locked: !!ls.locked
  }));
}
function saveHang(){
  if(!SHOW.key){ toast('Load a show first — there is nothing to save the hang to'); return null; }
  SHOW_HANGS[SHOW.key] = hangSnapshot();
  const hung = SHOW_HANGS[SHOW.key].filter(r=>r.goods !== 'none').length;
  toast('Hang saved to ' + SHOWS[SHOW.key].name + ' — ' + hung +
        ' linesets. It will load in like this from now on.');
  if(typeof refreshHangUI === 'function') refreshHangUI();
  return SHOW_HANGS[SHOW.key];
}
function applyHang(key){
  const snap = SHOW_HANGS[key];
  if(!snap) return {set:0, missing:0};
  let set = 0, missing = 0;
  for(const r of snap){
    const ls = FLY[r.id - 1];
    if(!ls) continue;
    if(!GOODS[r.goods]){ missing++; continue; }   // a shop cloth that has since gone
    if(ls.goodsKey !== r.goods) hangGoods(ls, r.goods);
    ls.locked = false;                            // trim it before locking it off
    ls.target = ls.pos = r.target;
    ls.group.position.y = ls.pos;
    ls.travTarget = ls.open = r.open;
    ls.locked = r.locked;
    set++;
  }
  if(typeof refreshFlyUI === 'function') refreshFlyUI();
  if(typeof refreshHangUI === 'function') refreshHangUI();
  return {set, missing};
}
function forgetHang(){
  if(!SHOW.key) return false;
  if(!SHOW_HANGS[SHOW.key]){ toast('That show has no saved hang'); return false; }
  delete SHOW_HANGS[SHOW.key];
  toast('Saved hang discarded — back to the hang the show came with');
  if(typeof refreshHangUI === 'function') refreshHangUI();
  return true;
}
/* has the rail been touched since it was loaded or last saved? */
function hangIsDirty(){
  if(!SHOW.key) return false;
  const ref = SHOW_HANGS[SHOW.key] || SHOW.loadedHang;
  if(!ref) return false;
  const now = hangSnapshot();
  for(let i=0;i<now.length;i++){
    const a = now[i], b = ref[i];
    if(!b) return true;
    if(a.goods !== b.goods || a.locked !== b.locked) return true;
    if(Math.abs(a.target - b.target) > 0.05) return true;
    if(Math.abs(a.open - b.open) > 0.02) return true;
  }
  return false;
}

function showLoad(key){
  const spec = SHOWS[key]; if(!spec) return false;
  showStrike();
  strikeStageSet();                  // clear THIS deck — not the other building's
  SHOW.key = key;
  /* Clear the rail first.  A show's hang has to be deterministic — locks and
     leftovers from whatever you were doing before must not survive into it,
     or FORGET would never really take you back to the show as written. */
  FLY.forEach(ls=>{
    ls.locked = false;
    if(ls.goodsKey !== 'none') hangGoods(ls, 'none');
    ls.travTarget = ls.open = 0;
    ls.target = ls.pos = OUT_TRIM;
    ls.group.position.y = ls.pos;
  });
  showRoot();
  spec.load();
  // your hang, if you have saved one, wins over the one it shipped with
  if(SHOW_HANGS[key]) applyHang(key);
  SHOW.loadedHang = hangSnapshot();
  if(typeof lockShowStatic === 'function') lockShowStatic();
  /* the owner's delivered set models replace the stand-ins (RULING AZ) —
     fetched async per scene, applied when they arrive, silent when absent.
     p5i defines it and loads later in the build, so the typeof guard is the
     usual late-part pattern; hooked HERE, after the show's build registered
     its scenes, so loading the show on the other stage fetches there too. */
  if(typeof loadSetModels === 'function') loadSetModels();
  if(typeof crewForgetLoads === 'function') crewForgetLoads();
  if(typeof refreshShowUI === 'function') refreshShowUI();
  if(typeof refreshCues === 'function') refreshCues();
  if(typeof refreshFlyUI === 'function') refreshFlyUI();
  if(typeof refreshHangUI === 'function') refreshHangUI();
  showLog(spec.name + ' is loaded in — go to CUES and press GO');
  return true;
}
/* freeze the set: it is scenery, it does not move on its own */
function lockShowStatic(){
  if(!SHOW.group) return;
  SHOW.group.updateMatrixWorld(true);
  SHOW.group.traverse(o=>{
    if(o === SHOW.group) return;
    if(o.userData.moves) return;
    /* a scene that can TRAVEL keeps a live matrix of its own.  Freeze the
       wagon's group and writing position.x updates the record and moves
       nothing on the stage — and a test that reads position.x back reads its
       own write and passes.  Only the GROUP is spared; everything inside it
       is still static relative to the group, so it stays frozen and rides
       along.  (RULING AP) */
    if(o.userData.sceneTravels) return;
    o.updateMatrix();
    o.matrixAutoUpdate = false;
  });
}

function showBox(w,h,d, mat, x,y,z, parent, rx,ry,rz){
  const m = new T.Mesh(new T.BoxGeometry(w,h,d), mat);
  m.position.set(x,y,z);
  if(rx) m.rotation.x = rx; if(ry) m.rotation.y = ry; if(rz) m.rotation.z = rz;
  m.castShadow = m.receiveShadow = true;
  (parent || showRoot()).add(m);
  return m;
}

  const m = new T.Mesh(new T.CylinderGeometry(rt,rb,h,seg), mat);
  m.position.set(x,y,z);
  m.castShadow = m.receiveShadow = true;
  (parent || showRoot()).add(m);
  return m;
}
/* A batch of identical parts in one draw call.

   The bounding sphere is a trap here.  r128 uses geometry.boundingSphere for
   BOTH frustum culling and the first rejection test in a raycast — but the
   raycast applies it per instance, on top of that instance's own transform.
   Widening the sphere to cover every instance fixes the culling and breaks
   the raycast, which is how the scaffold ended up un-standable-on.  So:
   anything you have to be able to walk on keeps its honest local sphere and
   is simply never culled; everything else gets the wide sphere and gives up
   raycasting, which it never needed.                                        */
function showBatch(geo, mat, list, opt){
  opt = opt || {};
  const parent = opt.parent;
  const im = new T.InstancedMesh(geo, mat, list.length);
  const d = new T.Object3D();
  const box = new T.Box3();
  if(!geo.boundingBox) geo.computeBoundingBox();
  const tmp = new T.Box3();
  list.forEach((p,i)=>{
    d.position.set(p.x, p.y, p.z);
    d.rotation.set(p.rx||0, p.ry||0, p.rz||0);
    d.scale.set(p.sx||1, p.sy||1, p.sz||1);
    d.updateMatrix();
    im.setMatrixAt(i, d.matrix);
    tmp.copy(geo.boundingBox).applyMatrix4(d.matrix);
    box.union(tmp);
  });
  im.instanceMatrix.needsUpdate = true;
  if(opt.walk){
    geo.computeBoundingSphere();       // honest and local, so raycasts land
    im.frustumCulled = false;
  } else {
    const s = new T.Sphere(); box.getBoundingSphere(s);
    im.geometry.boundingSphere = s;    // wide, so it is not culled at a glance
    im.raycast = ()=>{};
  }
  im.userData.spread = box.clone();
  im.castShadow = im.receiveShadow = true;
  (parent || showRoot()).add(im);
  return im;
}

/* ============================================================================
   RAIN, LIGHTNING AND FIRELIGHT
   Effects a cue can call for.  The rain is one instanced batch of streaks
   falling in a slab in front of the wall; lightning and firelight are cheap
   modulations laid over whatever the board is already doing.
   ========================================================================== */
function makeRain(o){
  const n = o.count || 900;
  const geo = new T.PlaneGeometry(0.018, 0.42);
  const mat = new T.MeshBasicMaterial({color:0xbcd2e8, transparent:true, opacity:0.34,
    depthWrite:false, side:T.DoubleSide, blending:T.AdditiveBlending});
  const im = new T.InstancedMesh(geo, mat, n);
  im.frustumCulled = false;
  im.raycast = ()=>{};
  im.visible = false;
  const drops = [];
  for(let i=0;i<n;i++)
    drops.push({x:rnd(o.x0,o.x1), y:rnd(o.y0,o.y1), z:rnd(o.z0,o.z1), v:rnd(9,15), len:rnd(0.7,1.6)});
  const d0 = new T.Object3D();
  drops.forEach((p,i)=>{
    d0.position.set(p.x, p.y, p.z);
    d0.scale.set(1, p.len, 1);
    d0.updateMatrix();
    im.setMatrixAt(i, d0.matrix);
  });
  im.instanceMatrix.needsUpdate = true;
  showRoot().add(im);
  const r = {mesh:im, drops, o, level:0, target:0, _d:new T.Object3D()};
  r.mesh.userData.moves = true;
  r.mesh.userData.effect = true;          // not the crew's to carry
  SHOW.rain = r;
  return r;
}
function updateRain(dt){
  const r = SHOW.rain; if(!r) return;
  r.level += clamp(r.target - r.level, -dt*0.8, dt*0.8);
  r.mesh.visible = r.level > 0.01;
  if(!r.mesh.visible) return;
  r.mesh.material.opacity = 0.34 * r.level;
  const d = r._d, o = r.o;
  const live = Math.max(1, Math.floor(r.drops.length * r.level));
  for(let i=0;i<r.drops.length;i++){
    const p = r.drops[i];
    if(i >= live){ d.scale.set(0,0,0); d.updateMatrix(); r.mesh.setMatrixAt(i, d.matrix); continue; }
    p.y -= p.v * dt;
    if(p.y < o.y0){ p.y = o.y1; p.x = rnd(o.x0,o.x1); p.z = rnd(o.z0,o.z1); }
    d.position.set(p.x, p.y, p.z);
    d.rotation.set(0,0,0);
    d.scale.set(1, p.len, 1);
    d.updateMatrix();
    r.mesh.setMatrixAt(i, d.matrix);
  }
  r.mesh.instanceMatrix.needsUpdate = true;
}
/* a lightning hit: a hard white flash that decays over about a third of a second */
function strikeLightning(power){
  SHOW.lightning = power === undefined ? 1 : power;
  SHOW.lightningT = 0;
  if(typeof Snd !== 'undefined' && Snd.play) Snd.play('thunder');
}
function updateStorm(dt){
  updateRain(dt);
  updateFire(dt);
  if(typeof updateNeon === 'function') updateNeon(dt);
  if(typeof updatePortal === 'function') updatePortal(dt);   // RULING CH
  updateHouseWait();                                          // RULING CJ
  if(typeof bjUpdateSetFill === 'function') bjUpdateSetFill(dt);   // RULING CC
  updateShowStorm(dt);
  updateConfetti(dt);
  sceneMoveStep(dt);
  if(typeof updateRevolve === 'function') updateRevolve(dt);
  if(typeof updateWrong === 'function') updateWrong(dt);
  if(SHOW.lightning > 0.001){
    SHOW.lightningT += dt;
    SHOW.lightning = Math.max(0, SHOW.lightning - dt*3.4);
  } else SHOW.lightning = 0;
  // a flash reads as a hard raking key plus a lift in the ambient
  LIGHTNING.intensity     = SHOW.lightning * 2.6;
  LIGHTNING_AMB.intensity = SHOW.lightning * 0.5;
  // the porch light is a practical, so it rides the practical master
  if(SHOW.porch){
    const v = (typeof HOUSE !== 'undefined' ? HOUSE.practical : 0);
    SHOW.porch.lamp.intensity = v * 1.7;
    SHOW.porch.bulb.material = v > 0.02 ? M.bulb : M.fixture;
  }
  // the rain has a sound, and it comes in and out with the water
  if(typeof Snd !== 'undefined' && SHOW.rain){
    const want = SHOW.rain.target > 0.05;
    if(want !== SHOW.rainSound){ Snd.play('rain'); SHOW.rainSound = want; }
  }
  if(SHOW.flicker){
    const f = SHOW.flicker;
    f.t += dt;
    // two detuned waves plus a little noise reads as firelight, not a sine
    f.v = 0.72 + 0.18*Math.sin(f.t*7.3) + 0.10*Math.sin(f.t*17.1 + 1.3) + rnd(-0.05,0.05);
    for(const ch of f.chans){
      const fx = chan(ch); if(!fx) continue;
      fx.level = clamp(f.base * f.v, 0, 1);
    }
  }
}

  for(const g of SHOW.gyms) g.group.position.x = g.home - g.side * SHOW.gymIn;
  return SHOW.gymIn;
}

  SHOW.flicker = chans ? {chans, base:base===undefined?0.8:base, t:0, v:1} : null;
}


/* ============================================================================
   FIRE — the church goes up
   Flame tongues are instanced quads that rise, lean, shrink and die, turned
   to face the camera each frame.  Embers ride up on top of them.  Two point
   lights sit inside it so the timber around it actually gets lit, and the
   rig channels flicker underneath the whole thing.
   ========================================================================== */
let FIRE_TEX = null, EMBER_TEX = null;
function fireTextures(){
  if(FIRE_TEX) return;
  // a flame tongue: white hot at the base, orange through the middle, gone at the tip
  const c = cvs(128, 256), g = c.getContext('2d');
  g.clearRect(0,0,128,256);
  for(let i=0;i<44;i++){
    const t = i/43;
    const w = 58 * Math.sin(Math.PI * Math.pow(t, 0.62)) * (1 - t*0.25);
    const y = 250 - t*244;
    const a = (1 - t) * 0.30 + 0.04;
    const col = t < 0.28 ? '255,246,206' : t < 0.6 ? '255,186,64' : '226,92,20';
    g.fillStyle = 'rgba('+col+','+a.toFixed(3)+')';
    g.beginPath(); g.ellipse(64, y, Math.max(2,w), Math.max(4, 26*(1-t*0.5)), 0, 0, 6.283); g.fill();
  }
  FIRE_TEX = tex(c,1,1);
  const e = cvs(32,32), ge = e.getContext('2d');
  const gr = ge.createRadialGradient(16,16,0,16,16,16);
  gr.addColorStop(0,'rgba(255,236,180,1)'); gr.addColorStop(0.35,'rgba(255,150,50,.75)');
  gr.addColorStop(1,'rgba(255,90,20,0)');
  ge.fillStyle = gr; ge.fillRect(0,0,32,32);
  EMBER_TEX = tex(e,1,1);
}
function makeFire(o){
  fireTextures();
  const n = o.count || 260, nE = o.embers || 130;
  const mat = new T.MeshBasicMaterial({map:FIRE_TEX, transparent:true, opacity:1,
    blending:T.AdditiveBlending, depthWrite:false, side:T.DoubleSide});
  const im = new T.InstancedMesh(new T.PlaneGeometry(1, 1), mat, n);
  im.frustumCulled = false; im.raycast = ()=>{}; im.visible = false;
  im.userData.moves = true;
  const eMat = new T.MeshBasicMaterial({map:EMBER_TEX, transparent:true, opacity:1,
    blending:T.AdditiveBlending, depthWrite:false, side:T.DoubleSide});
  const em = new T.InstancedMesh(new T.PlaneGeometry(1, 1), eMat, nE);
  em.frustumCulled = false; em.raycast = ()=>{}; em.visible = false;
  em.userData.moves = true;

  const seed = ()=>({
    x:rnd(o.x0,o.x1), y:o.y + rnd(-0.1,0.3), z:rnd(o.z0,o.z1),
    t:Math.random(), life:rnd(0.9,2.1), v:rnd(2.2,4.4),
    w:rnd(0.9,2.2), h:rnd(2.0,4.6), sway:rnd(-0.5,0.5), ph:Math.random()*6.283
  });
  const seedE = ()=>({
    x:rnd(o.x0,o.x1), y:o.y + rnd(0,1.5), z:rnd(o.z0,o.z1),
    t:Math.random(), life:rnd(1.6,3.6), v:rnd(1.4,3.6),
    s:rnd(0.08,0.26), drift:rnd(-0.8,0.8), ph:Math.random()*6.283
  });
  const flames = [], embers = [];
  for(let i=0;i<n;i++)  flames.push(seed());
  for(let i=0;i<nE;i++) embers.push(seedE());

  const g = new T.Group();
  g.userData.effect = true;               // not the crew's to carry
  g.add(im); g.add(em);
  const l1 = new T.PointLight(0xff7a22, 0, 22, 1.5);
  const l2 = new T.PointLight(0xffb040, 0, 16, 1.6);
  l1.position.set((o.x0+o.x1)/2 - 2.4, o.y + 2.0, (o.z0+o.z1)/2);
  l2.position.set((o.x0+o.x1)/2 + 2.4, o.y + 3.4, (o.z0+o.z1)/2);
  l1.userData.moves = l2.userData.moves = true;
  g.add(l1); g.add(l2);
  showRoot().add(g);

  const f = {group:g, mesh:im, embers:em, flames, ember:embers, o,
             level:0, target:0, t:0, lights:[l1,l2], _d:new T.Object3D()};
  SHOW.fire = f;
  return f;
}
const _fireCam = new T.Vector3();
function updateFire(dt){
  const f = SHOW.fire; if(!f) return;
  f.level += clamp(f.target - f.level, -dt*0.55, dt*0.7);
  const on = f.level > 0.01;
  f.mesh.visible = f.embers.visible = on;
  if(!on){ f.lights.forEach(l=>l.intensity = 0); return; }
  f.t += dt;
  const o = f.o, d = f._d;
  // flames face the camera: yaw only, so they stay upright.  The fire lives
  // in showRoot's stage space, which at the Arc sits 420m out — bring the
  // camera's WORLD pose (VR moves it out of camera.position) into that space
  // or every quad faces the stage-left wall.
  const camL = f.group.parent
    ? f.group.parent.worldToLocal(camera.getWorldPosition(_fireCam))
    : camera.position;
  const yaw = Math.atan2(camL.x - (o.x0+o.x1)/2,
                         camL.z - (o.z0+o.z1)/2);
  f.mesh.material.opacity = f.level;
  f.embers.material.opacity = f.level;
  const live = Math.max(1, Math.floor(f.flames.length * f.level));
  for(let i=0;i<f.flames.length;i++){
    const p = f.flames[i];
    if(i >= live){ d.scale.set(0,0,0); d.updateMatrix(); f.mesh.setMatrixAt(i, d.matrix); continue; }
    p.t += dt / p.life;
    if(p.t >= 1){ Object.assign(p, seedFlame(o)); p.t = 0; }
    const k = p.t;                                   // 0 new, 1 spent
    const rise = p.v * k * p.life;
    const grow = Math.sin(Math.PI * Math.min(1, k*1.25));   // fat in the middle
    d.position.set(p.x + Math.sin(f.t*2.2 + p.ph)*0.25*k + p.sway*k,
                   p.y + rise + p.h*0.35,
                   p.z + Math.cos(f.t*1.7 + p.ph)*0.18*k);
    d.rotation.set(0, yaw, Math.sin(f.t*2.6 + p.ph)*0.12);
    d.scale.set(p.w * grow * (1 - k*0.35), p.h * grow, 1);
    d.updateMatrix();
    f.mesh.setMatrixAt(i, d.matrix);
  }
  f.mesh.instanceMatrix.needsUpdate = true;
  const liveE = Math.max(1, Math.floor(f.ember.length * f.level));
  for(let i=0;i<f.ember.length;i++){
    const p = f.ember[i];
    if(i >= liveE){ d.scale.set(0,0,0); d.updateMatrix(); f.embers.setMatrixAt(i, d.matrix); continue; }
    p.t += dt / p.life;
    if(p.t >= 1){ Object.assign(p, seedEmber(o)); p.t = 0; }
    const k = p.t;
    d.position.set(p.x + p.drift*k*3.2 + Math.sin(f.t*3 + p.ph)*0.4*k,
                   p.y + p.v*k*p.life*2.1,
                   p.z + Math.cos(f.t*2.4 + p.ph)*0.5*k);
    d.rotation.set(0, yaw, 0);
    const sc = p.s * (1 - k*0.7);
    d.scale.set(sc, sc, 1);
    d.updateMatrix();
    f.embers.setMatrixAt(i, d.matrix);
  }
  f.embers.instanceMatrix.needsUpdate = true;
  // the light inside it never sits still
  const a = 0.74 + 0.2*Math.sin(f.t*9.1) + 0.12*Math.sin(f.t*21.3 + 2.1) + rnd(-0.06,0.06);
  const b = 0.74 + 0.2*Math.sin(f.t*7.4 + 1.1) + 0.12*Math.sin(f.t*18.2) + rnd(-0.06,0.06);
  f.lights[0].intensity = f.level * 3.4 * a;
  f.lights[1].intensity = f.level * 2.2 * b;
}
function seedFlame(o){
  return {x:rnd(o.x0,o.x1), y:o.y + rnd(-0.1,0.3), z:rnd(o.z0,o.z1),
          t:0, life:rnd(0.9,2.1), v:rnd(2.2,4.4),
          w:rnd(0.9,2.2), h:rnd(2.0,4.6), sway:rnd(-0.5,0.5), ph:Math.random()*6.283};
}

          t:0, life:rnd(1.6,3.6), v:rnd(1.4,3.6),

}

   Tulsa, 1967.  A lot behind the houses: a rough plank wall with the daylight
   coming through the gaps in it, a scaffold you can climb, a dead car the
   boys sit on and in, tyres, cut timber, an oil drum.  One drop upstage that

/* The wall behind the frame is not a picture.  It is a flat field of aged
   ochre board, seamed vertically, that the cyc wash turns to gold — so the
   frame reads as a black skeleton standing against a lit surface.          */
function outsidersCanvas(kind){
  if(kind === 'curtain') return outsidersCurtainCanvas();
  const c = cvs(1024, 576), g = c.getContext('2d');
  const W = 1024, H = 576;
  g.fillStyle = '#8a6524'; g.fillRect(0,0,W,H);
  // a slow vertical shade, brighter through the middle
  const gr = g.createLinearGradient(0,0,0,H);
  gr.addColorStop(0,'rgba(60,40,12,.55)'); gr.addColorStop(0.34,'rgba(255,206,110,.20)');
  gr.addColorStop(0.62,'rgba(255,190,86,.12)'); gr.addColorStop(1,'rgba(28,18,8,.62)');
  g.fillStyle = gr; g.fillRect(0,0,W,H);
  // board seams
  g.strokeStyle = 'rgba(40,26,10,.55)'; g.lineWidth = 3;
  for(let x=0; x<W; x+=64){
    g.beginPath(); g.moveTo(x, 0); g.lineTo(x, H); g.stroke();
  }
  g.strokeStyle = 'rgba(40,26,10,.3)'; g.lineWidth = 2;
  for(let y=0; y<H; y+=96){
    g.beginPath(); g.moveTo(0, y); g.lineTo(W, y); g.stroke();
  }
  // grain and grime, so it is not a flat fill
  for(let i=0;i<2600;i++){
    const x = Math.random()*W, y = Math.random()*H;
    g.fillStyle = 'rgba('+(120+Math.random()*90|0)+','+(84+Math.random()*60|0)+',30,'+
      (0.03+Math.random()*0.09).toFixed(2)+')';
    g.fillRect(x, y, 1 + Math.random()*46, 1 + Math.random()*2);
  }
  for(let i=0;i<40;i++){
    g.fillStyle = 'rgba(28,18,8,'+(0.05+Math.random()*0.12).toFixed(2)+')';
    g.beginPath();
    g.ellipse(Math.random()*W, Math.random()*H, rnd(20,90), rnd(14,60), 0, 0, 6.283);
    g.fill();
  }
  return {canvas:c, tex:tex(c,1,1)};
}

/* The show curtain: a painted cloth in the same weathered ochre as the wall
   behind the frame, boarded and distressed, with a low sun bleeding through
   it and a dark ground band along the bottom.  It is a traveler, so it draws
   like any house curtain and answers the CURTAINS group call.              */
function outsidersCurtainCanvas(){
  const W = 1024, H = 1024, c = cvs(W, H), g = c.getContext('2d');
  // the ground of the cloth: deep rust up into amber
  const base = g.createLinearGradient(0, 0, 0, H);
  base.addColorStop(0.00, '#2a1608');
  base.addColorStop(0.26, '#7a3d10');
  base.addColorStop(0.50, '#b8701e');
  base.addColorStop(0.68, '#d99a34');
  base.addColorStop(0.84, '#5e3413');
  base.addColorStop(1.00, '#180e06');
  g.fillStyle = base; g.fillRect(0, 0, W, H);
  // a low sun burning through from behind
  const sun = g.createRadialGradient(W*0.5, H*0.60, 10, W*0.5, H*0.60, W*0.44);
  sun.addColorStop(0, 'rgba(255,232,158,.62)');
  sun.addColorStop(0.32, 'rgba(255,186,72,.24)');
  sun.addColorStop(1, 'rgba(255,150,40,0)');
  g.fillStyle = sun; g.fillRect(0, 0, W, H);
  // boarded, like the wall it hides
  g.strokeStyle = 'rgba(30,18,7,.5)'; g.lineWidth = 4;
  for(let x = 0; x <= W; x += 58){
    g.beginPath(); g.moveTo(x, 0); g.lineTo(x + rnd(-6,6), H); g.stroke();
  }
  g.strokeStyle = 'rgba(255,214,140,.10)'; g.lineWidth = 2;
  for(let x = 29; x <= W; x += 58){
    g.beginPath(); g.moveTo(x, 0); g.lineTo(x + rnd(-5,5), H); g.stroke();
  }
  // grain
  for(let i=0;i<5200;i++){
    const y = Math.random()*H;
    g.fillStyle = 'rgba('+(150+Math.random()*90|0)+','+(96+Math.random()*70|0)+',36,'+
      (0.03 + Math.random()*0.08).toFixed(2)+')';
    g.fillRect(Math.random()*W, y, 2 + Math.random()*70, 1 + Math.random()*2);
  }
  // a flat horizon and a dark treeline low down, barely there
  g.fillStyle = 'rgba(22,13,6,.82)';
  g.beginPath(); g.moveTo(0, H*0.80);
  for(let x=0; x<=W; x+=12)
    g.lineTo(x, H*0.80 - Math.sin(x*0.011)*13 - Math.sin(x*0.037)*7);
  g.lineTo(W, H); g.lineTo(0, H); g.closePath(); g.fill();
  // wear: scuffs, water marks, and a rubbed patch through the middle
  for(let i=0;i<70;i++){
    g.fillStyle = 'rgba(24,14,6,'+(0.04 + Math.random()*0.10).toFixed(2)+')';
    g.beginPath();
    g.ellipse(Math.random()*W, Math.random()*H, rnd(20,120), rnd(10,70), 0, 0, 6.283);
    g.fill();
  }
  for(let i=0;i<26;i++){
    g.fillStyle = 'rgba(255,206,132,'+(0.03 + Math.random()*0.06).toFixed(2)+')';
    g.fillRect(Math.random()*W, Math.random()*H, rnd(6,30), rnd(60,300));
  }
  // a heavier band top and bottom, where a cloth takes the most punishment
  const top = g.createLinearGradient(0,0,0,H*0.12);
  top.addColorStop(0,'rgba(12,7,3,.85)'); top.addColorStop(1,'rgba(12,7,3,0)');
  g.fillStyle = top; g.fillRect(0,0,W,H*0.12);
  return {canvas:c, tex:tex(c,1,1)};
}

SHOWS.outsiders = {
  name: 'THE OUTSIDERS',
  blurb: 'Tulsa, 1967 — a plank wall, a scaffold, a dead car and a rumble in the rain.',
  note: 'An interpretation in the show’s own vocabulary, not a copy of anyone’s design.',
  load(){
    const G = showRoot();
    const HW = D.stageW/2;

    /* ---- materials: everything weathered, nothing new ---- */
    const timber   = new T.MeshStandardMaterial({map:TX.deck, color:0x6c5540, roughness:.95});
    const timberDk = new T.MeshStandardMaterial({map:TX.deck, color:0x4a3a2c, roughness:.97});
    const timberLt = new T.MeshStandardMaterial({map:TX.deck, color:0x8a7255, roughness:.94});
    const dirt     = new T.MeshStandardMaterial({map:TX.deck, color:0x4b4238, roughness:1});
    const rust     = new T.MeshStandardMaterial({color:0x6d4326, roughness:.92, metalness:.25});
    const rustDk   = new T.MeshStandardMaterial({color:0x4a2e1e, roughness:.95, metalness:.2});
    const chrome   = new T.MeshStandardMaterial({color:0x8a9099, roughness:.34, metalness:.9});
    const glassMat = new T.MeshStandardMaterial({color:0x1b2630, roughness:.18, metalness:.2,
                                                 transparent:true, opacity:.42});
    const rubber   = new T.MeshStandardMaterial({color:0x14151a, roughness:.98});
    const scaff    = new T.MeshStandardMaterial({color:0x6a6257, roughness:.62, metalness:.55});

    /* ---- the deck: dark boards, not dirt ---- */
    {
      /* the shape is drawn in plan and then laid down: after rotateX(-90) a
         shape y of -k becomes a world z of +k, so every edge is written as the
         negative of the stage coordinate it belongs to.                      */
      const sh = new T.Shape();
      const usZ = -(D.backWall + 0.4);
      sh.moveTo(-HW+0.4, usZ);
      sh.lineTo( HW-0.4, usZ);
      sh.lineTo( HW-0.4, -D.apron);
      sh.quadraticCurveTo(0, -(D.apron + 2*D.thrust*0.9), -HW+0.4, -D.apron);
      sh.lineTo(-HW+0.4, usZ);
      const geo = new T.ExtrudeGeometry(sh, {depth:0.1, bevelEnabled:false, curveSegments:36});
      geo.rotateX(-Math.PI/2);          // the deck now sits 0 to 0.1 high
      const lot = new T.Mesh(geo, dirt);
      lot.receiveShadow = true;
      G.add(lot);
      showWalk(lot);
    }

    /* ================================================================
       THE FRAME
       The whole set is one thing: the skeleton of a house, two storeys
       of stud and rail with the bracing left showing, a gable over it
       and a round window in the gable.  You see straight through it to
       the wall behind, and every backlight cue comes through the gaps
       between the members.

    const FR = {
      z:      D.backWall + 5.4,      // the frame's plane
      halfW:  7.2,                   // just inside the proscenium
      bay:    1.2,                   // stud centres
      sill:   0.15,
      gallery:3.5,                   // first floor
      plate:  7.0,                   // top of the wall
      peak:   10.4,                  // the gable apex
      thick:  0.21
    };
    {
      const t = FR.thick, HWF = FR.halfW, Z = FR.z;
      const studs = [], rails = [], braces = [];

      /* studs, floor to plate, at bay centres */
      for(let x = -HWF; x <= HWF + 0.01; x += FR.bay)
        studs.push({x:+x.toFixed(3), y:(FR.plate + FR.sill)/2, z:Z,
                    sy:(FR.plate - FR.sill)/FR.plate});
      showBatch(new T.BoxGeometry(t, FR.plate, t*1.15), timber, studs);

      /* the horizontal runs: sill, first floor, plate, and a nogging band */
      [FR.sill, FR.gallery, FR.gallery + 0.42, FR.plate].forEach(y=>{
        rails.push({x:0, y, z:Z});
      });
      showBatch(new T.BoxGeometry(HWF*2 + t, t*1.4, t*1.3), timberDk, rails);

      /* the X-bracing.  Not every bay — that would read as a lattice.  Pairs
         of crossed diagonals in the outer bays and one wide X either side of
         centre, which is what gives the thing its shoulders.               */
      const brace = (x0,y0,x1,y1)=>{
        const dx = x1-x0, dy = y1-y0;
        const len = Math.hypot(dx,dy);
        braces.push({x:(x0+x1)/2, y:(y0+y1)/2, z:Z - 0.06,
                     rz:Math.atan2(dy,dx) - Math.PI/2, sy:len/4});

      const bayX = i => -HWF + i*FR.bay;

      });
      // upper storey
      [0, 2,3, 6, 9,10, 11].forEach(i=>{
        brace(bayX(i), FR.gallery + 0.42, bayX(i+1), FR.plate);
        if(i % 3 === 0) brace(bayX(i+1), FR.gallery + 0.42, bayX(i), FR.plate);
      });
      showBatch(new T.BoxGeometry(t*0.85, 4, t), timber, braces);

         of the plate at (sd*run, plate) up to the apex at (0, peak) the
         direction has to be (-sd*run, rise)/len, so sin a = sd*run/len and
         cos a = rise/len — that is a = sd*atan2(run, rise).  I had the sign
         inverted, which is why the two rafters leaned the same way, crossed
         over each other and shot out past the peak.                        */
      const rise = FR.peak - FR.plate, run = HWF;

          sd*run/2, FR.plate + rise/2, Z);

      showBox(t*1.6, t*1.7, t*1.6, timberDk, 0, FR.peak - t*0.6, Z);

         The gable is braced instead by two raking struts off the plate.    */

        const st = showBox(t*0.9, rise*0.66, t, timber,
          sd*HWF*0.34, FR.plate + rise*0.26, Z);

      showBox(t, rise*0.24, t, timber, 0, FR.peak - rise*0.12, Z);            // king post

        for(let i=0;i<8;i++){

      }

      /* ---- the balcony ----
         It was a thin slab tucked in behind the frame with steel posts on
         it, which from out front read as a few white sticks floating in mid
         air.  It is a proper timber gallery now: a deck with a fascia board
         along the front edge, joists and knee braces under it, and a heavy

      const gallery = (cy, x0, x1, dp)=>{
        const w = x1 - x0, cx = (x0 + x1)/2;
        /* it hangs off the DOWNSTAGE face and projects towards the house — it
           was tucked in behind the frame, so you were looking at the back of
           it through the studs.  z grows towards the audience.              */
        const back  = Z + 0.06;              // where it meets the frame
        const front = back + dp;             // the edge you look at
        const zc    = back + dp/2;
        showWalk(showBox(w, 0.16, dp, timberLt, cx, cy + 0.08, zc));
        // the fascia is what you actually see from the house
        showBox(w, 0.44, 0.1, timberDk, cx, cy - 0.14, front);
        const joists = [], knees = [];
        for(let x = x0 + 0.5; x <= x1 - 0.49; x += 1.2){
          joists.push({x:+x.toFixed(2), y:cy - 0.18, z:zc});
          // the knee braces rake back up to the frame behind it
          knees.push({x:+x.toFixed(2), y:cy - 0.62, z:back + 0.34, rx:0.72, sy:0.24});
        }
        showBatch(new T.BoxGeometry(0.12, 0.24, dp - 0.1), timberDk, joists);
        showBatch(new T.BoxGeometry(0.1, 4, 0.1), timber, knees);
        const posts = [];
        for(let x = x0 + 0.25; x <= x1 - 0.24; x += 1.15)
          posts.push({x:+x.toFixed(2), y:cy + 0.68, z:front});
        showBatch(new T.BoxGeometry(0.12, 1.1, 0.12), timberDk, posts);
        showBox(w + 0.24, 0.13, 0.15, timberLt, cx, cy + 1.26, front);   // top rail
        showBox(w, 0.09, 0.11, timber, cx, cy + 0.74, front);            // mid rail
        // and a rail returning down each end, so it is not open at the sides

          showBox(0.12, 1.1, 0.12, timberDk, x, cy + 0.68, back + 0.1);
          showBox(0.12, 0.13, dp, timberLt, x, cy + 1.26, zc);

       belongs, so the whole frame can be tracked in and out on a slider
       without rebuilding anything.                                          */
    {
      const gym = (bx, bz, bays, levels, sd)=>{
        const g = new T.Group();
        g.position.set(bx, 0, bz);
        g.userData.moves = true;            // it tracks, so do not freeze it
        G.add(g);
        const W = 2.6, DP = 1.7, lift = 1.45, H = levels*lift;
        const posts = [], rungs = [], braces = [];
        for(let b2=0;b2<=bays;b2++) for(const s2 of [-1,1])
          posts.push({x:s2*W/2, y:H/2, z:-b2*DP, sy:H/4});
        for(let l=1;l<=levels;l++){
          for(let b2=0;b2<=bays;b2++)
            rungs.push({x:0, y:l*lift, z:-b2*DP});                          // across
          for(const s2 of [-1,1]) for(let b2=0;b2<bays;b2++)
            rungs.push({x:s2*W/2, y:l*lift, z:-b2*DP - DP/2,
                        ry:Math.PI/2, sx:DP/W});                            // fore and aft
        }
        for(let b2=0;b2<bays;b2++){
          if(b2 % 2) continue;
          braces.push({x:sd*W/2, y:H/2, z:-b2*DP - DP/2,
                       rx:Math.atan2(DP, H), sy:Math.hypot(DP, H)/4});
        }
        showBatch(new T.CylinderGeometry(0.05,0.05,4,7), scaff, posts, {parent:g});
        showBatch(new T.BoxGeometry(W, 0.075, 0.075), scaff, rungs, {parent:g});
        if(braces.length)
          showBatch(new T.CylinderGeometry(0.038,0.038,4,6), scaff, braces, {parent:g});
        const planks = [];
        for(let b2=0;b2<bays;b2++) planks.push({x:0, y:H + 0.07, z:-b2*DP - DP/2});
        const deck = showWalk(showBatch(new T.BoxGeometry(W - 0.12, 0.1, DP - 0.1),
                              timberLt, planks, {walk:true, parent:g}));

/* ============================================================================
   THE FRONT CURTAIN IN THE CUE STACK
   Every cue that carries a fly state carries the curtain's state with it, so
   "shut" has to be declared per cue.  It defaulted to whatever the curtain
   happened to be at plotting time — which was IN and SHUT — so every cue in
   the stack pulled a black cloth across the stage.  That is what looked like
   a blackout on Q2.
   ========================================================================== */
function frontCurtainLineset(){
  return FLY.find(l=>GOODS[l.goodsKey] && GOODS[l.goodsKey].traveler) || null;
}
/* Point a set of channels at a spot — used to throw the front of house up
   onto a show cloth instead of down at the deck.  Non-movers only; the movers
   have pan and tilt of their own.                                          */
/* The rig's own focus, not whatever the last cue fired happened to leave it
   pointing at — otherwise plotting a show straight after running one bakes
   that cue's aim into every look.                                         */
function captureAims(){
  return FIXTURES.map(f=>{
    if(f.mover) return null;
    const a = f.homeAim || f.aim;
    return [a.x, a.y, a.z];
  });
}
function restoreAims(a){
  if(!a) return;
  FIXTURES.forEach((f,i)=>{
    if(!a[i]) return;
    f.aim.set(a[i][0], a[i][1], a[i][2]);
    f.group.lookAt(f.aim);
  });
}
function aimFrontAt(chans, x, y, z){
  chans.forEach(n=>{
    const f = chan(n); if(!f || f.mover) return;
    // fan them a little so it reads as a wash, not six hard hits
    const spread = (n - (chans[0] + chans[chans.length-1])/2) * 1.1;
    stageToWorld(x + spread, y, z, f.aim);
    f.group.lookAt(f.aim);
  });
}
function setCurtainForCue(state){
  const ls = frontCurtainLineset();
  if(!ls) return null;
  const trim = TRIMS[ls.goodsKey] !== undefined ? TRIMS[ls.goodsKey] : TRIMS.house;
  /* It flies, it does not draw.  The halves stay closed to each other the
     whole time and the whole cloth goes up out of sight — which is what a
     painted show curtain does, and it keeps the title readable on the way. */
  ls.travTarget = 0;
  ls.target = (state === 'shut') ? trim : OUT_TRIM;
  return ls;
}

/* Put the board where a show starts: preset up, next GO is the first real cue. */
function standByAtTheTop(){
  if(!CUES.length) return;
  fireCue(0);
  CUES[0].lx.forEach((r,i)=>{
    const f = chan(i+1); if(!f) return;
    f.level = r.lvl; f.color.set(r.col); f.lvlDur = 0; f.colDur = 0;
  });
  nextCue = 1;
  if(typeof refreshCues === 'function') refreshCues();
}

/* the extra business a show cue can carry, run after the normal cue fires */
function showCueExtras(c){
  if(!c) return;
  if(SHOW.rain) SHOW.rain.target = c.rain || 0;
  if(SHOW.fire) SHOW.fire.target = c.fire ? (c.fire.size === undefined ? 1 : c.fire.size) : 0;
  if(c.neon !== undefined && typeof setNeon === 'function') setNeon(c.neon);
  /* THE PROSCENIUM NEON, ON EVERY CUE AND NOT JUST THE ONES THAT CARRY IT
     (RULING CH).  "For the rest it is off" — so a cue saying nothing about the
     frame DARKENS it, which is the opposite bargain from `signCol` two lines
     down and from `neon` one line up, and it is the owner's own words that draw
     the distinction.  It is also what makes the plot readable: four cues in
     ninety-five say `portal:` and the frame is out for the other ninety-one
     without any of them having to say so. */
  if(typeof setPortal === 'function') setPortal(c.portal);
  /* RULING CJ — the house master waits for the cloth.  fireCue has already
     written HOUSE.house from the cue; this takes it back to 0 and hands it over
     once the curtain is really in.  Declared per cue, so every cue that says
     nothing behaves exactly as it always did. */
  if(c.houseAfterCurtain && !frontCurtainShut()){
    SHOW.houseWait = c.house;
    HOUSE.house = 0;
    if(typeof syncMasters === 'function') syncMasters();
  } else SHOW.houseWait = null;
  /* the marquee's own lamps.  `undefined` means LEAVE IT — one cue turns the
     sign red and the cues after it keep it red without saying so again. */
  if(c.signCol !== undefined && typeof setSignLamps === 'function') setSignLamps(c.signCol);
  /* THE WHOLE SET IS LIT UNLESS THIS CUE SAYS OTHERWISE (RULING CC).  Read per
     cue and NOT remembered, which is the opposite of `signCol` two lines up —
     the owner's words draw that distinction: "unless i said otherwise IN THE
     CUE".  What it scales is the stage rig's own output, so a blackout cue still
     lights nothing (bjUpdateSetFill). */
  SHOW.bjFillCue = (c.fill === undefined) ? 1 : c.fill;
  setFirelight(c.fire ? c.fire.chans : null, c.fire ? c.fire.base : 0.8);
  /* which dressing the room wears (RULING AQ) — judged BEFORE the changeover
     below, against where the set stands as the cue FIRES.  A swap on a set
     the audience cannot see is honest and lands now: the parked wagon is
     dressed in the wings, then called on already wearing it.  A swap on a
     set they CAN see waits on SHOW.pendDress until the set goes off
     (RULING AY: nothing ever pops), which sceneApply applies.  Seen was
     judged on DRAWN-ness, not on sc.on: a set mid-way out of a changeover
     has already been marked off, but its layers are on and the audience is
     watching it leave — a swap then is exactly the pop the ruling forbids.

     RULING BQ BROKE THAT EQUIVALENCE AND THE SUITE CAUGHT IT.  Drawn no longer
     means seen: a parked set is drawn precisely so that it goes on existing
     backstage.  Judged on drawn-ness alone, a dress cue on a parked wagon
     deferred FOR EVER — and the room was then called on still wearing the old
     dressing, which is the opposite of what AY wants.  So the test is drawn AND
     (on, or still travelling): on is seen, mid-exit is seen, parked is not.  The
     original reasoning above is intact — it is the third state that is new. */
  if(c.dress && typeof bjDress === 'function' && SHOW.scenes.length){
    const dn = c.scene || SHOW.scene, dsc = sceneFind(dn);
    const seen = dsc && !dsc.group.userData.sceneOff &&
                 (dsc.on || (typeof sceneTravelling === 'function' && sceneTravelling(dsc)));
    if(seen) SHOW.pendDress = {scene:dn, key:c.dress};
    else bjDress(dn, c.dress);
  }
  /* a cue can send travelling scenery somewhere.  One cue may move more than
     one thing — act two opens with the exterior flying out WHILE the house
     slides on — so this takes a list as readily as a single move.  A move
     that names a PART retargets that part's own mover instead of the whole
     group: scenery moving within a scene, no scene change (RULING AY).

     THE SPLIT RULE.  Where a move lands relative to the changeover is
     load-bearing in BOTH directions, so each move is judged on whose it is:

       - a move on the scene the cue is LEAVING lands BEFORE sceneChangeTo.
         The changeover hides an outgoing set instantly unless
         sceneTravelling says it is mid-move, and a whole-group mover only
         reads as travelling once its target is set — judged the other way
         round, act two's exterior flew out DARK, an instant hide dressed
         up as a travel.

       - every other move lands AFTER sceneChangeTo.  The incoming branch
         SNAPS a hidden set's part movers to OUT and retargets them home —
         moves applied first are overwritten by that snap: a JUMP to the
         9:45 cue from another scene sent the hills home to centre, a full
         graveyard on a cue whose whole point is emptying it.  Same-scene
         moves (9:45 fired in sequence) are safe on this side too: the
         changeover no-ops for a scene that is already on.

     The partition is taken against SHOW.scene as the cue FIRES, before the
     changeover rewrites it.

     A PART move on the leaving scene lands on the before side too, and the
     changeover then retargets every part of the leaving scene to OUT — the
     exit choreography wins over the cue's part move, by design; no current
     cue carries that shape. */
  const mvList = (c.move && SHOW.scenes.length)
    ? (Array.isArray(c.move) ? c.move : [c.move]) : [];
  /* THIS CUE SUPERSEDES ANY MOVE STILL WAITING ON A LINE (RULING CI).  Cleared
     before the list below is built, so a cue may gate a fresh one. */
  if(SHOW.pendMoves) SHOW.pendMoves.length = 0;
  const mvApply = m => {
    /* a move that names a line waits for it, unless the line is already there */
    if(m.after !== undefined && !flyArrived(m.after)){
      (SHOW.pendMoves || (SHOW.pendMoves = [])).push(m);
      return;
    }
    sceneMoveApply(m); };
  const mvBefore = [], mvAfter = [];
  for(const m of mvList){
    if(!m) continue;
    if(c.scene && m.scene === SHOW.scene && m.scene !== c.scene) mvBefore.push(m);
    else mvAfter.push(m);
  }
  mvBefore.forEach(mvApply);
  if(c.scene && SHOW.scenes.length) sceneChangeTo(c.scene);
  mvAfter.forEach(mvApply);
  if(c.confetti && typeof fireConfetti === 'function') fireConfetti();
  /* the sound (BA) and the audience rig's patterns (BD).  Both are p5j, which
     loads after this part, so both are typeof-guarded the way showCueExtras
     itself is guarded from p6.  `fx` is called even when the cue has none —
     that is what CLEARS a running effect. */
  if(typeof showCueAudio === 'function') showCueAudio(c.audio || null);
  if(typeof showCueFx === 'function') showCueFx(c.fx || null);
  if(c.rev && SHOW.revolve && typeof setRevolve === 'function'){
    setRevolve('outer', c.rev.outer || 0);
    setRevolve('inner', c.rev.inner || 0);
  }
  /* the scenery that is meant to fail.  A cue can stand it all back up —
     which is what the top of the show does — or push a named piece over. */
  if(c.wrongReset && typeof wrongReset === 'function') wrongReset();
  if(c.wrong && typeof wrongTrigger === 'function') c.wrong.forEach(k=>wrongTrigger(k));
  if(c.storm){
    SHOW.stormOn = true; SHOW.stormT = rnd(0.4, 1.4);
  } else SHOW.stormOn = false;
}
/* ----------------------------------------------------------------------
   CONFETTI — a burst, not a state.  "2:15:00. Confetti goes off".  One
   instanced batch fired once, falling under gravity off the frame dt and
   clearing itself when the last piece is down.  It is never walkable and it
   never raycasts.
   -------------------------------------------------------------------- */
const CONFETTI_N = 260;
function fireConfetti(){
  if(!SHOW.group) return null;
  if(SHOW.confetti) clearConfetti();
  const g = new T.PlaneGeometry(0.13, 0.09);
  const m = new T.MeshBasicMaterial({color:0xffffff, side:T.DoubleSide,
                                     vertexColors:true});
  const mesh = new T.InstancedMesh(g, m, CONFETTI_N);
  mesh.name = 'bj:confetti';
  mesh.frustumCulled = false;              // r128 sizes the sphere off ONE plane
  mesh.raycast = ()=>{};                   // decorative: never in a pick
  const col = new T.Color(), bits = [];
  const PAL = [0xff3d18, 0xbfe08a, 0xffd24a, 0x6ad0ff, 0xffffff, 0xb266ff];
  for(let i = 0; i < CONFETTI_N; i++){
    bits.push({x:rnd(-7.2, 7.2), y:rnd(11.5, 16.5), z:rnd(-8.5, 1.0),
               vy:-rnd(1.1, 2.6), sp:rnd(-3.2, 3.2), ph:rnd(0, 6.283),
               dx:rnd(-0.5, 0.5)});
    col.setHex(PAL[(Math.random()*PAL.length)|0]);
    mesh.setColorAt(i, col);
  }
  if(mesh.instanceColor) mesh.instanceColor.needsUpdate = true;
  showRoot().add(mesh);
  SHOW.confetti = {mesh, bits, t:0};
  return SHOW.confetti;
}
function clearConfetti(){
  const c = SHOW.confetti;
  if(!c) return;
  if(c.mesh.parent) c.mesh.parent.remove(c.mesh);
  if(typeof disposeTree === 'function') disposeTree(c.mesh);
  SHOW.confetti = null;
}
function updateConfetti(dt){
  const c = SHOW.confetti;
  if(!c) return;
  c.t += dt;
  const d = new T.Object3D();
  let live = 0;
  for(let i = 0; i < c.bits.length; i++){
    const b = c.bits[i];
    if(b.y > 0.02){
      b.vy -= dt * 1.4;                     // it does not fall like a stone
      if(b.vy < -2.8) b.vy = -2.8;
      b.y += b.vy * dt;
      b.x += Math.sin(c.t * 1.7 + b.ph) * b.dx * dt;
      if(b.y < 0.02) b.y = 0.02;
      live++;
    }
    d.position.set(b.x, b.y, b.z);
    d.rotation.set(c.t * b.sp * 0.6, c.t * b.sp, b.ph);
    d.updateMatrix();
    c.mesh.setMatrixAt(i, d.matrix);
  }
  c.mesh.instanceMatrix.needsUpdate = true;
  if(!live && c.t > 2) clearConfetti();     // it clears itself once it is down
}
function updateShowStorm(dt){
  if(!SHOW.stormOn) return;
  SHOW.stormT -= dt;
  if(SHOW.stormT <= 0){
    strikeLightning(rnd(0.6, 1));
    SHOW.stormT = rnd(1.8, 5.2);
  }
}
